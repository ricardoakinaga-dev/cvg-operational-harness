from pathlib import Path
import datetime
import hashlib
import json
import os
import secrets
import signal
import socket
import subprocess
import time

ROOT = Path('/home/ricardo/Área de trabalho/cvg-operational-harness')
E = ROOT / 'docs/04_audit/evidence/CODEX-INCORPORATION-20261101/cleanup-global-qualification'
task = json.loads((E / 'task.json').read_text())
setup = json.loads((E / 'setup.json').read_text())
W = Path(setup['candidate'])
TMP = Path(setup['TMPDIR'])


def now():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()


def write(name, value):
    (E / name).write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')


def command(args, **kwargs):
    return subprocess.check_output(args, text=True, timeout=30, **kwargs).strip()


def pins():
    return {name: hashlib.sha256((W / name).read_bytes()).hexdigest()
            for name in setup['source_pins']}


assert json.loads((E / 'format-preflight.json').read_text())['exit'] == 0
assert command(['git', '-C', str(W), 'rev-parse', 'HEAD']) == task['source_commit']
assert command(['git', '-C', str(W), 'status', '--porcelain']) == ''
assert pins() == setup['source_pins']
assert not (E / 'certify-live.json').exists()
for port in task['e2e_ports'].values():
    with socket.socket() as probe:
        probe.bind(('127.0.0.1', port))
command(['docker', 'image', 'inspect', task['pg_image']])
containers = command(['docker', 'ps', '-aq', '--filter', 'name=^/' + task['pg_name'] + '$'])
assert containers == '', 'Existing name is not our new resource'
credential_file = TMP / 'postgres.env'
password = secrets.token_hex(24)
credential_file.write_text('POSTGRES_PASSWORD=' + password + '\nPOSTGRES_DB=cvg_cleanup_cert\n')
credential_file.chmod(0o600)
cid = None
child = None
started = time.monotonic()
before = pins()
outcome = {'at': now(), 'status': 'STARTING', 'source_commit': task['source_commit']}
write('controller-state.json', outcome)

try:
    args = ['docker', 'run', '-d', '--name', task['pg_name']]
    for key, value in task['pg_labels'].items():
        args.extend(['--label', key + '=' + value])
    args.extend(['--env-file', str(credential_file), '--memory', '2g', '--cpus', '2',
                 '--pids-limit', '256', '--tmpfs', '/var/lib/postgresql/data:rw,size=2147483648',
                 '-p', '127.0.0.1::5432', task['pg_image']])
    cid = command(args)
    metadata = json.loads(command(['docker', 'inspect', cid]))[0]
    assert metadata['Id'] == cid
    assert all(metadata['Config']['Labels'].get(k) == v for k, v in task['pg_labels'].items())
    write('pg-created.json', {'at': now(), 'cid': cid, 'name': task['pg_name'],
                             'image': metadata['Image'], 'labels': task['pg_labels']})
    ready_until = time.monotonic() + 60
    while True:
        ready = subprocess.run(['docker', 'exec', cid, 'pg_isready', '-h', '127.0.0.1',
                                '-U', 'postgres', '-d', 'cvg_cleanup_cert'], capture_output=True)
        if ready.returncode == 0:
            break
        if time.monotonic() >= ready_until:
            raise RuntimeError('Own disposable PostgreSQL TCP readiness deadline')
        time.sleep(0.5)
    def sql(query):
        return command(['docker', 'exec', cid, 'psql', '-h', '127.0.0.1', '-U', 'postgres',
                        '-d', 'cvg_cleanup_cert', '-At', '-v', 'ON_ERROR_STOP=1', '-c', query])
    assert sql("SELECT count(*) FROM pg_stat_activity WHERE datname='cvg_cleanup_cert' AND pid<>pg_backend_pid()") == '0'
    assert sql('SHOW fsync') == 'on'
    metadata = json.loads(command(['docker', 'inspect', cid]))[0]
    bindings = metadata['NetworkSettings']['Ports']['5432/tcp']
    assert len(bindings) == 1 and bindings[0]['HostIp'] == '127.0.0.1'
    pg_port = int(bindings[0]['HostPort'])
    write('pg-ready.json', {'at': now(), 'cid': cid, 'port': pg_port, 'clients': 0,
                           'fsync': 'on', 'default_role': 'postgres', 'synthetic': True})
    env = {k: os.environ[k] for k in ['HOME', 'USER', 'LOGNAME', 'LANG', 'TZ'] if k in os.environ}
    env.update(PATH='/home/ricardo/.nvm/versions/node/v22.23.2/bin:/usr/local/bin:/usr/bin:/bin',
               TMPDIR=str(TMP), CI='true', NO_COLOR='1', CVG_TEST_SCOPE='core',
               npm_config_cache=str(TMP / 'npm-cache'), npm_config_update_notifier='false',
               TEST_DATABASE_URL=f'postgres://postgres:{password}@127.0.0.1:{pg_port}/cvg_cleanup_cert',
               PHASE4A_PG_REQUIRED='1', PHASE4A_DISPOSABLE_PG='1',
               CVG_API_PORT=str(task['e2e_ports']['api']), CVG_WEB_PORT=str(task['e2e_ports']['web']))
    env['DATABASE_URL'] = env['TEST_DATABASE_URL']
    with (E / 'certify-native.log').open('x') as stream:
        child = subprocess.Popen(['npm', 'run', 'certify'], cwd=W, env=env, stdout=stream,
                                 stderr=subprocess.STDOUT, start_new_session=True)
        live = {'at': now(), 'pid': child.pid, 'pgid': child.pid, 'runner_pid': os.getpid(),
                'pid_start_ticks': Path(f'/proc/{child.pid}/stat').read_text().rsplit(')', 1)[1].split()[19],
                'argv': ['npm', 'run', 'certify'], 'candidate': str(W), 'cid': cid,
                'limit_seconds': task['certification_outer_deadline_seconds'],
                'environment': {k: ('<OWN_SYNTHETIC_PG_URL>' if k.endswith('DATABASE_URL') else v)
                                for k, v in env.items()}}
        write('certify-live.json', live)
        outcome.update(status='RUNNING', pid=child.pid, runner_pid=os.getpid(), cid=cid)
        write('controller-state.json', outcome)
        print(json.dumps({'status': 'CERTIFY_RUNNING', 'pid': child.pid,
                          'runner_pid': os.getpid(), 'cid': cid, 'pg_port': pg_port}), flush=True)
        try:
            child.wait(timeout=task['certification_outer_deadline_seconds'] - 8)
            outcome.update(status='TERMINAL', exit=child.returncode, timed_out=False)
        except subprocess.TimeoutExpired:
            os.killpg(child.pid, signal.SIGTERM)
            try:
                child.wait(timeout=5)
            except subprocess.TimeoutExpired:
                os.killpg(child.pid, signal.SIGKILL)
                child.wait(timeout=3)
            outcome.update(status='TERMINAL', exit=child.returncode, timed_out=True)
except BaseException as error:
    outcome.update(status='FAILED_CONTROLLER', exception=type(error).__name__, reason=str(error))
    if child is not None and child.poll() is None:
        os.killpg(child.pid, signal.SIGTERM)
        try:
            child.wait(timeout=5)
        except subprocess.TimeoutExpired:
            os.killpg(child.pid, signal.SIGKILL)
            child.wait(timeout=3)
    raise
finally:
    outcome.update(completed_at=now(), elapsed_seconds=round(time.monotonic() - started, 3),
                   source_pre=before, source_post=pins())
    outcome['source_unchanged'] = outcome['source_pre'] == outcome['source_post']
    write('controller-state.json', outcome)
    credential_file.unlink(missing_ok=True)
    if cid is not None:
        metadata = json.loads(command(['docker', 'inspect', cid]))[0]
        assert metadata['Id'] == cid
        assert all(metadata['Config']['Labels'].get(k) == v for k, v in task['pg_labels'].items())
        clients = 'UNKNOWN'
        try:
            clients = command(['docker', 'exec', cid, 'psql', '-h', '127.0.0.1', '-U', 'postgres',
                               '-d', 'cvg_cleanup_cert', '-At', '-v', 'ON_ERROR_STOP=1', '-c',
                               "SELECT count(*) FROM pg_stat_activity WHERE datname='cvg_cleanup_cert' AND pid<>pg_backend_pid()"])
        except subprocess.CalledProcessError:
            pass
        cleanup = {'at': now(), 'cid': cid, 'clients_before_cleanup': clients, 'removed': False}
        if clients == '0':
            command(['docker', 'rm', '-f', cid])
            cleanup['removed'] = True
        else:
            cleanup['pending'] = 'Own clients remain; do not remove peer or live resources'
        write('pg-cleanup.json', cleanup)
    print(json.dumps(outcome, ensure_ascii=False), flush=True)
