const read = name => {
  assert.equal(path.isAbsolute(name), false);
  const p = path.resolve(W, name);
  assert.ok(p.startsWith(W + path.sep));
  if (!fs.existsSync(p)) return undefined;
  assert.ok(fs.lstatSync(p).isFile());
  assert.equal(fs.lstatSync(p).isSymbolicLink(), false);
  return fs.readFileSync(p);
};

function recount(file) {
  const r = json(file);
  const counts = {};
  for (const suite of r.testResults) for (const a of suite.assertionResults) counts[a.status] = (counts[a.status] ?? 0) + 1;
  assert.equal(r.success, true);
  assert.equal(counts.passed, r.numPassedTests);
  assert.equal(r.numTotalTests, r.numPassedTests);
  assert.equal(r.numFailedTests, 0);
  assert.equal(r.numPendingTests, 0);
  assert.ok(Object.keys(counts).every(k => k === 'passed'));
  return {files:r.testResults.length, passed:counts.passed, failed:0, pending:0, paths:r.testResults.map(f => path.relative(W,f.name)).sort()};
}
