import {
  constants,
  openSync,
  readSync,
  closeSync,
  lstatSync,
  statfsSync,
  accessSync
} from 'node:fs'
import { isAbsolute, normalize, join, dirname, basename, sep } from 'node:path'
import { ProcessServiceError } from './errors.ts'
import type { ProcessAdmissionReason } from './containment.ts'

const MOUNT = '/sys/fs/cgroup'
const CGROUP2 = 0x63677270
const TMPFS = 0x01021994
const CONTROLLERS = ['cpu', 'memory', 'pids']
const MAX_READ = 8192
export interface ProcessDelegationFacts {
  readonly linux: boolean
  readonly identity: boolean
  readonly cgroupFilesystem: boolean
  readonly rootOwned: boolean
  readonly rootEmpty: boolean
  readonly ownerInSupervisor: boolean
  readonly controllers: boolean
  readonly kernelFiles: boolean
  readonly runtimePrivate: boolean
}
export interface ProcessDelegationReport {
  readonly state: 'candidate' | 'refused'
  /** Diagnostic only. No caller may use this report to authorize a workload. */
  readonly qualified: false
  readonly reason: ProcessAdmissionReason
  readonly checks: readonly {
    readonly name: keyof ProcessDelegationFacts
    readonly passed: boolean
  }[]
}
const CHECKS: readonly [
  keyof ProcessDelegationFacts,
  ProcessAdmissionReason
][] = [
  ['linux', 'architecture_unsupported'],
  ['identity', 'delegation_invalid'],
  ['cgroupFilesystem', 'cgroup_unavailable'],
  ['rootOwned', 'delegation_invalid'],
  ['rootEmpty', 'delegation_invalid'],
  ['ownerInSupervisor', 'delegation_invalid'],
  ['controllers', 'controller_missing'],
  ['kernelFiles', 'kernel_feature_missing'],
  ['runtimePrivate', 'delegation_invalid']
]
/** Pure test/diagnostic seam: even fabricated facts can never yield qualified:true. */
export function assessProcessDelegation(
  facts: ProcessDelegationFacts
): ProcessDelegationReport {
  const failed = CHECKS.find(([name]) => facts[name] !== true)
  return Object.freeze({
    state: failed ? 'refused' : 'candidate',
    qualified: false,
    reason: failed?.[1] ?? 'qualification_missing',
    checks: Object.freeze(
      CHECKS.map(([name]) =>
        Object.freeze({ name, passed: facts[name] === true })
      )
    )
  })
}
export interface ProcessDelegationInput {
  readonly root: string
  readonly runtimeDirectory: string
  readonly instance: string
  /** Operator-composed ControlGroup relative to /sys/fs/cgroup, not a tool parameter. */
  readonly expectedControlGroup: string
  readonly signal?: AbortSignal
}
const absolute = (p: unknown): p is string =>
  typeof p === 'string' &&
  !p.includes('\0') &&
  isAbsolute(p) &&
  normalize(p) === p
function noSymlinks(path: string): boolean {
  let at: string = sep
  try {
    for (const part of path.split(sep).filter(Boolean)) {
      at = join(at, part)
      if (!lstatSync(at).isDirectory()) return false
    }
    return true
  } catch {
    return false
  }
}
/** Fixed-size read, including pseudo-files whose stat size is zero; never follow final symlinks. */
function small(path: string): string | undefined {
  let fd: number | undefined
  try {
    fd = openSync(
      path,
      constants.O_RDONLY | constants.O_NOFOLLOW | constants.O_NONBLOCK
    )
    const buffer = Buffer.alloc(MAX_READ + 1)
    const size = readSync(fd, buffer, 0, buffer.length, 0)
    return size > MAX_READ
      ? undefined
      : buffer.subarray(0, size).toString('utf8')
  } catch {
    return undefined
  } finally {
    if (fd !== undefined) closeSync(fd)
  }
}
function ownedDirectory(path: string, privateMode: boolean): boolean {
  try {
    const stat = lstatSync(path)
    return (
      stat.isDirectory() &&
      stat.uid === process.geteuid!() &&
      (!privateMode || (stat.mode & 0o777) === 0o700)
    )
  } catch {
    return false
  }
}
function filesystem(path: string, magic: number): boolean {
  try {
    return statfsSync(path).type === magic
  } catch {
    return false
  }
}
function requiredFiles(root: string): boolean {
  const readable = [
    'cpu.max',
    'cpu.stat',
    'memory.max',
    'memory.events',
    'memory.oom.group',
    'memory.swap.max',
    'pids.max',
    'pids.events'
  ]
  try {
    const kill = join(root, 'cgroup.kill')
    if (!lstatSync(kill).isFile()) return false
    accessSync(kill, constants.W_OK)
    return (
      small(join(root, 'cgroup.type'))?.trim() === 'domain' &&
      readable.every((n) => small(join(root, n)) !== undefined)
    )
  } catch {
    return false
  }
}
function controllers(root: string): boolean {
  let path = root
  for (;;) {
    const available =
      small(join(path, 'cgroup.controllers'))?.trim().split(/\s+/) ?? []
    const enabled =
      small(join(path, 'cgroup.subtree_control'))?.trim().split(/\s+/) ?? []
    if (!CONTROLLERS.every((n) => available.includes(n) && enabled.includes(n)))
      return false
    if (path === MOUNT) return true
    // No internal processes in distributing domains, except the kernel root domain.
    if (small(join(path, 'cgroup.procs'))?.trim() !== '') return false
    path = dirname(path)
  }
}
/** Read-only observation, without mkdir/write/exec/systemctl. Native admission must revalidate. */
export async function inspectProcessDelegation(
  input: ProcessDelegationInput
): Promise<ProcessDelegationReport> {
  if (
    !input ||
    typeof input !== 'object' ||
    typeof input.instance !== 'string' ||
    !/^[a-z0-9-]{1,64}$/.test(input.instance) ||
    !absolute(input.root) ||
    !absolute(input.runtimeDirectory) ||
    !absolute(input.expectedControlGroup) ||
    !input.root.startsWith(MOUNT + '/') ||
    (input.signal !== undefined && !(input.signal instanceof AbortSignal))
  ) {
    throw new ProcessServiceError('invalid_request', 'invalid delegation input')
  }
  input.signal?.throwIfAborted()
  await Promise.resolve()
  input.signal?.throwIfAborted()
  const identity =
    input.root === MOUNT + input.expectedControlGroup &&
    basename(input.root) ===
      `cvg-process-qualification@${input.instance}.service` &&
    basename(input.runtimeDirectory) === input.instance &&
    basename(dirname(input.runtimeDirectory)) === 'cvg-process-qualification'
  const rootSafe = identity && noSymlinks(input.root)
  const runtimeSafe = identity && noSymlinks(input.runtimeDirectory)
  const current = small('/proc/self/cgroup')
    ?.split('\n')
    .find((n) => n.startsWith('0::'))
    ?.slice(3)
  const facts: ProcessDelegationFacts = {
    linux: process.platform === 'linux',
    identity,
    cgroupFilesystem: rootSafe && filesystem(input.root, CGROUP2),
    rootOwned: rootSafe && ownedDirectory(input.root, false),
    rootEmpty:
      rootSafe && small(join(input.root, 'cgroup.procs'))?.trim() === '',
    ownerInSupervisor: current === input.expectedControlGroup + '/supervisor',
    controllers: rootSafe && controllers(input.root),
    kernelFiles: rootSafe && requiredFiles(input.root),
    runtimePrivate:
      runtimeSafe &&
      ownedDirectory(input.runtimeDirectory, true) &&
      filesystem(input.runtimeDirectory, TMPFS)
  }
  input.signal?.throwIfAborted()
  return assessProcessDelegation(facts)
}
