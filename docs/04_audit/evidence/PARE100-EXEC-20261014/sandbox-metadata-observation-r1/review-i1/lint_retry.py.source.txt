import json
from pathlib import Path
import subprocess
import sys

D = Path(__file__).resolve().parent
P = D.parent
W = P.parent / 'worktree'
config = D / 'eslint.i1-r2.config.cjs'
config.write_text("const { createRequire } = require('node:module');\nconst request = createRequire(" + json.dumps(str(W / 'eslint.config.js')) + ");\nconst tseslint = request('typescript-eslint');\nmodule.exports = tseslint.config(...tseslint.configs.recommended.map(c => ({...c, basePath: " + json.dumps(str(P)) + "})));\n")
env = json.loads((D / 'quality.json').read_text())[0]['env']
argv = ['/home/ricardo/.nvm/versions/node/v22.23.2/bin/node', str(W / 'node_modules/eslint/bin/eslint.js'), '--no-cache', '--max-warnings', '0', '--config', str(config), str(P / 'observe.mts')]
result = subprocess.run(argv, cwd=D, env=env, stdout=subprocess.PIPE, stderr=subprocess.PIPE, timeout=30)
(D / 'lint-r2.stdout').write_bytes(result.stdout)
(D / 'lint-r2.stderr').write_bytes(result.stderr)
record = {'argv': argv, 'cwd': str(D), 'env': env, 'exitCode': result.returncode, 'stdout': str(D / 'lint-r2.stdout'), 'stderr': str(D / 'lint-r2.stderr'), 'stdoutBytes': len(result.stdout), 'stderrBytes': len(result.stderr), 'truncated': False, 'reason': 'Correct reviewer private config module resolution; retain first failure.'}
(D / 'lint-r2.command.json').write_text(json.dumps(record, indent=2) + '\n')
print(json.dumps(record)); print(result.stdout.decode()); print(result.stderr.decode(), file=sys.stderr)
sys.exit(result.returncode)
