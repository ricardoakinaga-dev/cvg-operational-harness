def run(args,cwd,e,log,bound=90):
 log.parent.mkdir(parents=True,exist_ok=True)
 start=time.monotonic(); seen={}; reason=None
 with log.open('wb') as out:
  p=subprocess.Popen(args,cwd=cwd,env=e,stdout=out,stderr=subprocess.STDOUT,start_new_session=True)
  def scan():
   for q in Path('/proc').iterdir():
    if not q.name.isdigit():continue
    try:
     fields=(q/'stat').read_text().rsplit(')',1)[1].split()
     if int(fields[3])==p.pid:
      seen[q.name]={'pid':int(q.name),'ppid':int(fields[1]),'cmdline':(q/'cmdline').read_bytes().replace(b'\0',b' ').decode(errors='replace')}
    except (OSError,ValueError,IndexError): pass
  while p.poll() is None:
   scan()
   if time.monotonic()-start>bound:reason='deadline';break
   if log.stat().st_size>2097152:reason='output_limit';break
   time.sleep(.05)
  scan()
  if reason:
   os.killpg(p.pid,signal.SIGTERM)
   try:p.wait(timeout=2)
   except subprocess.TimeoutExpired:os.killpg(p.pid,signal.SIGKILL);p.wait(timeout=3)
  p.wait(timeout=3)
  # terminate lingering owned group descendants, if any, and preserve observation.
  live=[]
  for pid in seen:
   try:
    stat=(Path('/proc')/pid/'stat').read_text().rsplit(')',1)[1].split()
    if int(stat[3])==p.pid and stat[0]!='Z':live.append(int(pid))
   except OSError:pass
  if live:
   os.killpg(p.pid,signal.SIGTERM);time.sleep(.15)
   try:os.killpg(p.pid,signal.SIGKILL)
   except ProcessLookupError:pass
 result={'argv':args,'cwd':str(cwd),'environment':e,'environment_sha256':hashlib.sha256(json.dumps(e,sort_keys=True,separators=(',',':')).encode()).hexdigest(),'pid':p.pid,'process_group':p.pid,'children_observed':list(seen.values()),'elapsed_seconds':round(time.monotonic()-start,3),'bound_seconds':bound,'output_bound_bytes':2097152,'exit':p.returncode,'termination_reason':reason,'live_descendants_after_exit_before_cleanup':live,'log':str(log.relative_to(P)),'log_sha256':sha(log)}
 dump(log.with_suffix('.command.json'),result);return result
