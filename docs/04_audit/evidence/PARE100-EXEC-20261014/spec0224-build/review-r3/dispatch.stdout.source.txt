   108 export function validateDecision(
   109   ctx: IterativeDispatchContext,
   110   run: LoopRun,
   111   decision: LoopDecision
   112 ): DecisionValidationResult {
   113   const allowed = ctx.allowedDecisionTypes(run)
   114   if (!allowed.includes(decision.decisionType)) {
   115     return {
   116       valid: false,
   117       errors: [
   118         `capability: decision ${decision.decisionType} is not available in this state`
   119       ]
   120     }
   121   }
   122   const validation = validateLoopDecision(decision)
   123   if (!validation.valid) return validation
   124   if (
   125     decision.decisionType === 'CALL_TOOL' ||
   126     decision.decisionType === 'REQUEST_APPROVAL'
   127   ) {
   128     if (!decision.toolId) {
   129       return {
   130         valid: false,
   131         errors: ['capability: tool selection requires a toolId']
   132       }
   133     }
   134     const tool = ctx.options.tools.resolve(
   135       decision.toolId,
   136       decision.toolVersion
   137     )
   138     if (!tool) {
   139       return {
   140         valid: false,
   141         errors: [`capability: tool "${decision.toolId}" is not in the catalog`]
   142       }
   143     }
   144     const profileTools = run.input.agent.tools
   145     if (!agentExposesTool(profileTools, tool.id)) {
   146       return {
   147         valid: false,
   148         errors: [
   149           `capability: tool "${tool.id}" is not exposed to this agent profile`
   150         ]
   151       }
   152     }
   153   }
   154   if (decision.decisionType === 'SEARCH_KNOWLEDGE' && !ctx.options.knowledge) {
   155     return {
   156       valid: false,
   157       errors: ['capability: no knowledge provider is configured']
   158     }
   159   }
   160   return { valid: true, errors: [] }
   161 }

   163 export function applyResume(
   164   ctx: IterativeDispatchContext,
   165   run: LoopRun,
   166   checkpoint: ExecutionCheckpoint | null
   167 ): Extract<DispatchOutcome, { kind: 'stop' }> | null {
   168   const resume = run.input.resume
   169   if (!resume) return null
   170   if (!checkpoint) {
   171     return {
   172       kind: 'stop',
   173       stopReason: 'STATE_CONFLICT',
   174       response: 'Resume input arrived without a durable checkpoint.'
   175     }
   176   }
   177   if (resume.kind === 'approval') {
   178     if (run.state.pendingQuestion) {
   179       // Wrong-kind resume: the execution is paused for an answer, not for an
   180       // approval. Never let this open the loop with a question unresolved.
   181       return {
   182         kind: 'stop',
   183         stopReason: 'STATE_CONFLICT',
   184         response: 'Approval resume arrived while a user question is pending.',
   185         preserveCheckpoint: true
   186       }
   187     }
   188     if (!run.state.pendingApprovalId) {
   189       if (run.state.stopReason === 'APPROVAL_REQUIRED') {
   190         return {
   191           kind: 'stop',
   192           stopReason: 'STATE_CONFLICT',
   193           response: 'The paused approval binding is missing.',
   194           preserveCheckpoint: true
   195         }
   196       }
   197       // The approval was already consumed and the checkpoint cleared the
   198       // binding; a retry of the same attempt is a safe no-op.
   199       return null
   200     }
   201     if (run.state.pendingApprovalId !== resume.approvalId) {
   202       return {
   203         kind: 'stop',
   204         stopReason: 'STATE_CONFLICT',
   205         response: 'Approval does not match the paused execution binding.',
   206         preserveCheckpoint: true
   207       }
   208     }
   209     return null
   210   }
   211   if (run.state.pendingApprovalId) {
   212     return {
   213       kind: 'stop',
   214       stopReason: 'STATE_CONFLICT',
   215       response: 'User input arrived while an approval is pending.',
   216       preserveCheckpoint: true
   217     }
   218   }
   219   if (!run.state.pendingQuestion) {
   220     const alreadyResolved = Object.values(run.state.resolvedInputs).includes(
   221       resume.message
   222     )
   223     if (alreadyResolved) {
   224       if (ctx.options.toolResultContext)
   225         run.resumeObservation = userResumeObservation(ctx, run, resume.message)
   226       return null
   227     }
   228     return {
   229       kind: 'stop',
   230       stopReason: 'STATE_CONFLICT',
   231       response: 'User input arrived without a pending question.',
   232       preserveCheckpoint: true
   233     }
   234   }
   235   const stepNumber = Math.max(1, run.state.stepNumber)
   236   const observation = userResumeObservation(ctx, run, resume.message)
   237   if (ctx.options.toolResultContext) run.resumeObservation = observation
   238   run.observations = ctx.options.toolResultContext
   239     ? retainCompleteObservations([...run.observations, observation])
   240     : [...run.observations, observation].slice(-MAX_CHECKPOINT_OBSERVATIONS)
   241   const pendingQuestion = run.state.pendingQuestion
   242   run.state = {
   243     ...run.state,
   244     pendingQuestion: undefined,
   245     openQuestions: run.state.openQuestions.filter(
   246       (question) => question !== pendingQuestion.promptIntent
   247     ),
   248     resolvedInputs: {
   249       ...run.state.resolvedInputs,
   250       [`answer_${stepNumber}`]: resume.message
   251     }
   252   }
   253   return null
   254 }

   256 function userResumeObservation(
   257   ctx: IterativeDispatchContext,
   258   run: LoopRun,
   259   message: string
   260 ): Observation {
   261   const stepNumber = Math.max(1, run.state.stepNumber)
   262   return ctx.makeObservation(run, {
   263     stepId: `step_${run.executionId}_${stepNumber}_user_input`,
   264     stepNumber,
   265     type: 'USER_MESSAGE',
   266     source: 'USER',
   267     trust: 'UNTRUSTED',
   268     payload: { message },
   269     summary: 'User answered the pending question.',
   270     provenance: { sourceId: 'user' }
   271   })

   278 /** The error code a step carries when a control blocked its call. */
   279 function blockedErrorCode(gate: KernelStop): string {
   280   switch (gate.cause) {
   281     case 'policy_denied':
   282     case 'policy_handoff':
   283     case 'approval_denied':
   284       return gate.cause
   285     case 'policy_unsupported':
   286       return 'policy_outcome_invalid'
   287     default:
   288       return gate.stopReason === 'MAX_DURATION'
   289         ? 'deadline'
   290         : gate.stopReason === 'CANCELLED'
   291           ? 'cancelled'
   292           : 'not_started'
   293   }
   294 }

   296 /** The step recorded for a blocked call, by the control that decided. */
   297 function blockedStepRecord(
   298   gate: KernelStop,
   299   stepNumber: number,
   300   reasonCode: ExecutionStep['reasonCode']
   301 ): StepRecord {
   302   const errorCode = blockedErrorCode(gate)
   303   switch (gate.cause) {
   304     case 'policy_denied':
   305     case 'policy_unsupported':
   306       return {
   307         stepNumber,
   308         stepType: 'POLICY',
   309         status: 'FAILED',
   310         sideEffecting: false,
   311         reasonCode: 'POLICY_REQUIRED',
   312         errorCode
   313       }
   314     case 'approval_denied':
   315       return {
   316         stepNumber,
   317         stepType: 'APPROVAL',
   318         status: 'FAILED',
   319         sideEffecting: false,
   320         reasonCode: 'POLICY_REQUIRED',
   321         errorCode
   322       }
   323     default:
   324       return {
   325         stepNumber,
   326         stepType: 'TOOL',
   327         status: 'FAILED',
   328         sideEffecting: true,
   329         reasonCode,
   330         errorCode
   331       }
   332   }

   335 export async function dispatchTool(
   336   ctx: IterativeDispatchContext,
   337   run: LoopRun,
   338   decision: LoopDecision,
   339   stepNumber: number,
   340   forceApproval: boolean
   341 ): Promise<DispatchOutcome> {
   342   // A previous successful dispatch is trajectory evidence, not the result
   343   // of this attempt. Only this dispatch's success may populate toolResult.
   344   run.lastToolObservation = null
   345   // A resumed in-flight step was already counted when it reached dispatch.
   346   const alreadyCounted =
   347     run.inFlightCounted && run.state.pendingStepNumber === stepNumber
   348   if (
   349     !alreadyCounted &&
   350     run.usage.toolCalls + 1 > run.input.budget.maxToolCalls
   351   ) {
   352     return ctx.stop(
   353       'MAX_TOOL_CALLS',
   354       'Tool-call budget exhausted before the tool could run.'
   355     )
   356   }
   357   const tool = ctx.options.tools.resolve(
   358     decision.toolId as string,
   359     decision.toolVersion
   360   )
   361   if (!tool) {
   362     return ctx.stop(
   363       'STATE_CONFLICT',
   364       `Tool "${decision.toolId}" is unavailable.`
   365     )
   366   }
   367   const operationKey = ctx.operationKey(run, stepNumber, tool.id)
   368   if (ctx.options.toolResultContext) {
   369     if (!ctx.preflightCompleteTool)
   370       return ctx.stop(
   371         'STATE_CONFLICT',
   372         'Complete tool dispatch requires its native capacity guard.'
   373       )
   374     ctx.preflightCompleteTool(run, decision, stepNumber)
   375   }
   376   const invocation = {
   377     toolId: tool.id,
   378     ...(tool.version ? { toolVersion: tool.version } : {}),
   379     input: decision.toolInput ?? {},
   380     operationKey
   381   }
   382   if (ctx.remainingDuration(run) <= 0) {
   383     return ctx.stop(
   384       'MAX_DURATION',
   385       'Duration budget exhausted before tool execution.'
   386     )
   387   }
   388 
   389   const stepId = `step_${run.executionId}_${stepNumber}_tool`
   390   // A resumed step already holds this step number as a tool step (waiting for
   391   // approval or in flight), and the dispatch checkpoint opens it as RUNNING;
   392   // a control that then blocks the call closes that same step instead of
   393   // recording a second one under the same number, or leaving it open while
   394   // the checkpoint ends (AUD-0599 F02).
   395   let stepOpen = run.state.pendingStepNumber === stepNumber
   396   // Opened as RUNNING by this invocation's dispatch checkpoint.
   397   let openedHere = false
   398   // An operator pause proves that nothing ran and nothing stays reserved: the
   399   // step waits for the resume instead of looking in flight, so a later
   400   // terminal stop can close it as not executed (AUD-0600 F02).
   401   const parkForResume = async (): Promise<void> => {
   402     if (!openedHere) return
   403     await ctx.recordStep(run, {
   404       stepNumber,
   405       stepType: 'TOOL',
   406       status: 'WAITING',
   407       sideEffecting: true,
   408       reasonCode: decision.reasonCode
   409     })
   410   }
   411   const recordDenial = (record: StepRecord): Promise<void> =>
   412     ctx.recordStep(
   413       run,
   414       stepOpen
   415         ? {
   416             stepNumber,
   417             stepType: 'TOOL',
   418             status: 'FAILED',
   419             sideEffecting: true,
   420             reasonCode: decision.reasonCode,
   421             errorCode: record.errorCode
   422           }
   423         : record
   424     )
   425   const call: ToolCallState = {
   426     turn: ctx.turn(run),
   427     tool,
   428     invocation,
   429     needsApproval: forceApproval,
   430     executionRef: run.executionId
   431   }
   432   let policyRecorded = false
   433   const recordPolicy = async (): Promise<void> => {
   434     if (policyRecorded || !call.policyDecision) return
   435     policyRecorded = true
   436     await ctx.recordObservation(
   437       run,
   438       ctx.makeObservation(run, {
   439         stepId,
   440         stepNumber,
   441         type: 'POLICY_DECISION',
   442         source: 'POLICY',
   443         trust: 'TRUSTED',
   444         payload: {
   445           outcome: call.policyDecision.outcome,
   446           policyVersion: call.policyDecision.policyVersion
   447         },
   448         summary: `Policy outcome ${call.policyDecision.outcome}.`,
   449         provenance: { sourceId: 'policy-engine' }
   450       })
   451     )
   452   }
   453 
   454   // Policy, approval, guards (pause) and cancellation run in the kernel
   455   // pipeline (SPEC 0181); this loop only adds its checkpoint and its steps.
   456   const pipeline = await runToolCall(ctx.kernel(run), call, {
   457     beforeDispatch: async () => {
   458       await recordPolicy()
   459       const approvalId = call.grant?.approvalId
   460       if (
   461         approvalId &&
   462         run.state.pendingApprovalId &&
   463         run.state.pendingApprovalId !== approvalId
   464       ) {
   465         return {
   466           kind: 'stop',
   467           stopReason: 'STATE_CONFLICT',
   468           response: 'Approval identity changed while resuming the execution.'
   469         }
   470       }
   471       // Budget accounting is committed with the in-flight decision so a crash
   472       // cannot replay the same step against an unconsumed counter; the
   473       // approval is only reserved after this checkpoint.
   474       if (!alreadyCounted) run.usage.toolCalls += 1
   475       run.inFlightCounted = false
   476       // The step is in flight from here: a stop reason left by an earlier
   477       // approval wait would hide that this call is already counted when the
   478       // execution resumes (AUD-0598 R02).
   479       run.state = {
   480         ...run.state,
   481         pendingDecision: decision,
   482         pendingStepNumber: stepNumber,
   483         ...(approvalId ? { pendingApprovalId: approvalId } : {}),
   484         stopReason: undefined
   485       }
   486       await ctx.persistCheckpoint(run)
   487       await ctx.recordStep(run, {
   488         stepNumber,
   489         stepType: 'TOOL',
   490         status: 'RUNNING',
   491         sideEffecting: true,
   492         reasonCode: decision.reasonCode
   493       })
   494       stepOpen = true
   495       openedHere = true
   496       return undefined
   497     }
   498   })
   499 
   500   const recordUnsupported = (): Promise<void> =>
   501     ctx.recordObservation(
   502       run,
   503       ctx.makeObservation(run, {
   504         stepId,
   505         stepNumber,
   506         type: 'POLICY_DECISION',
   507         source: 'POLICY',
   508         trust: 'TRUSTED',
   509         payload: { outcome: 'UNSUPPORTED' },
   510         summary: 'Policy returned an unsupported outcome.',
   511         provenance: { sourceId: 'policy-engine' }
   512       })
   513     )
   514 
   515   if (pipeline.kind === 'stopped') {
   516     const gate = pipeline.gate
   517     const blockedStep = (decided: KernelStop): Promise<void> =>
   518       recordDenial(blockedStepRecord(decided, stepNumber, decision.reasonCode))
   519     if (pipeline.blocked) {
   520       // The closing record was lost: this is neither a clean pause, handoff
   521       // nor denial, although nothing ran. The step is still closed for the
   522       // control's own decision, and the turn ends on the missing evidence
   523       // with that decision in its response (AUD-0599 F02).
   524       const blocked = pipeline.blocked
   525       if (blocked.cause === 'policy_unsupported') await recordUnsupported()
   526       else await recordPolicy()
   527       await blockedStep(blocked)
   528       return ctx.stop(gate.stopReason, gate.response)
   529     }
   530     if (gate.cause === 'operator_paused') {
   531       // Nothing ran and nothing is reserved: keep the step resumable (F03).
   532       await parkForResume()
   533       return {
   534         ...ctx.stop(gate.stopReason, gate.response),
   535         preserveCheckpoint: true
   536       }
   537     }
   538     if (gate.cause === 'policy_unsupported') {
   539       await recordUnsupported()
   540       await blockedStep(gate)
   541       return ctx.stop(gate.stopReason, gate.response)
   542     }
   543     await recordPolicy()
   544     switch (gate.cause) {
   545       case 'policy_handoff':
   546         if (stepOpen) {
   547           await blockedStep(gate)
   548           break
   549         }
   550         await ctx.recordStep(run, {
   551           stepNumber,
   552           stepType: 'HANDOFF',
   553           status: 'SUCCEEDED',
   554           sideEffecting: false,
   555           reasonCode: 'HANDOFF_REQUIRED'
   556         })
   557         break
   558       case 'approval_pending':
   559         return {
   560           kind: 'pause',
   561           pausedKind: 'APPROVAL_REQUIRED',
   562           decision,
   563           ...(gate.approvalId ? { approvalId: gate.approvalId } : {}),
   564           pauseStepNumber: stepNumber
   565         }
   566       case 'policy_denied':
   567       case 'approval_denied':
   568         await blockedStep(gate)
   569         break
   570       default:
   571         // Any other block (a guard, cancellation, a conflicting resume) ends
   572         // the run, so a step it left open is closed with it.
   573         if (stepOpen) await blockedStep(gate)
   574         break
   575     }
   576     return ctx.stop(gate.stopReason, gate.response)
   577   }
   578 
   579   const outcome = pipeline.outcome
   580   const reserved = call.approval !== undefined
   581   const failStep = (errorCode: string) =>
   582     ctx.recordStep(run, {
   583       stepNumber,
   584       stepType: 'TOOL',
   585       status: 'FAILED',
   586       sideEffecting: true,
   587       reasonCode: decision.reasonCode,
   588       errorCode
   589     })
   590 
   591   switch (outcome.kind) {
   592     case 'not_started': {
   593       const gate = outcome.gate
   594       if (pipeline.logFailure) {
   595         // The closing record was lost: this is neither a clean pause nor a
   596         // clean denial, although nothing ran (AUD-0598 R04, AUD-0599 F02).
   597         await failStep(blockedErrorCode(gate))
   598         const lost = unrecordedBlock(pipeline.logFailure, gate)
   599         return ctx.stop(lost.stopReason, lost.response)
   600       }
   601       if (gate.cause === 'operator_paused') {
   602         // The reservation was released as certain; the step resumes later.
   603         await parkForResume()
   604         return {
   605           ...ctx.stop(gate.stopReason, gate.response),
   606           preserveCheckpoint: true
   607         }
   608       }
   609       await failStep(blockedErrorCode(gate))
   610       return ctx.stop(gate.stopReason, gate.response)
   611     }
   612     case 'deadline':
   613       await failStep(reserved ? 'unknown_effect' : 'deadline')
   614       return ctx.stop(
   615         reserved ? 'TOOL_FAILURE' : 'MAX_DURATION',
   616         reserved
   617           ? 'unknown_effect: duration budget exhausted during tool execution.'
   618           : 'Duration budget exhausted during tool execution.'
   619       )
   620     case 'cancelled':
   621       await failStep(reserved ? 'unknown_effect' : 'cancelled')
   622       return ctx.stop(
   623         reserved ? 'TOOL_FAILURE' : 'CANCELLED',
   624         reserved ? `unknown_effect: ${CANCELLED_RESPONSE}` : CANCELLED_RESPONSE
   625       )
   626     case 'threw':
   627     case 'failed': {
   628       const toolError =
   629         outcome.kind === 'threw' ? outcome.error : outcome.result.error
   630       const unknownEffect =
   631         outcome.kind === 'threw' ||
   632         toolError?.startsWith('unknown_effect:') === true
   633       // `unknownEffect` is false only for a settled failure, so the fallback
   634       // status is unreachable for a thrown outcome.
   635       const failureStatus =
   636         outcome.kind === 'failed' ? outcome.result.status : undefined
   637       await ctx.recordObservation(
   638         run,
   639         ctx.makeObservation(run, {
   640           stepId,
   641           stepNumber,
   642           type: 'TOOL_RESULT',
   643           source: 'TOOL',
   644           trust: 'UNTRUSTED',
   645           payload: {
   646             status: unknownEffect ? 'REJECTED' : (failureStatus as string),
   647             error: toolError ?? 'tool execution failed'
   648           },
   649           summary: unknownEffect
   650             ? 'Tool outcome is uncertain and requires reconciliation.'
   651             : 'Tool execution failed.',
   652           provenance: {
   653             sourceId: tool.id,
   654             ...(tool.version ? { sourceVersion: tool.version } : {}),
   655             operationKey,
   656             ...(unknownEffect ? { effectRef: operationKey } : {})
   657           }
   658         })
   659       )
   660       // Preserve this kernel settlement cause losslessly in the step's error
   661       // field. Bounded observations cannot carry restart control evidence.
   662       await failStep(
   663         outcome.kind === 'threw' &&
   664           toolError === 'unknown_effect: effect_settlement_failed'
   665           ? toolError
   666           : unknownEffect
   667             ? 'unknown_effect'
   668             : 'tool_failed'
   669       )
   670       return ctx.stop(
   671         'TOOL_FAILURE',
   672         unknownEffect

  1034 export async function evaluate(
  1035   ctx: IterativeDispatchContext,
  1036   run: LoopRun,
  1037   decision: LoopDecision
  1038 ): Promise<Extract<DispatchOutcome, { kind: 'stop' }> | null> {

  1040     goal: run.state.goal,
  1041     state: { ...run.state, observations: run.observations },
  1042     observations: run.observations,
  1043     lastDecision: decision,
  1044     ...(run.responseText ? { result: run.responseText } : {})
  1045   })

  1049     run.usage.outputTokens += evaluation.usage.outputTokens
  1050     run.usage.costUsd += evaluation.usage.costUsd

  1059     if (afterEvaluation) {
  1060       return ctx.stop(afterEvaluation.reason, afterEvaluation.response)
  1061     }
  1062   }
  1063 
  1064   if (evaluation.outcome === 'COMPLETE') {
  1065     if (decision.decisionType === 'RESPOND') {
  1066       return ctx.stop('COMPLETED', run.responseText ?? 'Execution completed.')
  1067     }
  1068     if (
  1069       decision.decisionType === 'STOP' &&
  1070       decision.reasonCode === 'USER_REQUESTED_STOP'
  1071     ) {
  1072       return ctx.stop('COMPLETED', 'Execution stopped at the user request.')
  1073     }
  1074     // Completion requires an explicit RESPOND step.
  1075     return null
  1076   }
  1077   if (evaluation.outcome === 'FAILED') {
  1078     return ctx.stop(
  1079       mapEvaluationToStopReason(evaluation),
  1080       run.responseText ??
  1081         `Completion evaluation failed: ${evaluation.detail ?? evaluation.reasonCode}.`
  1082     )
  1083   }
  1084   if (evaluation.outcome === 'INSUFFICIENT_EVIDENCE') {
  1085     if (
  1086       decision.decisionType === 'RESPOND' ||
  1087       decision.decisionType === 'STOP'
  1088     ) {
  1089       return ctx.stop(
  1090         'INSUFFICIENT_EVIDENCE',
  1091         run.responseText ??
  1092           'The available evidence is insufficient for a complete answer.'
  1093       )
  1094     }
  1095   }
  1096   // A finalising decision with incomplete evidence is terminal: the agent
  1097   // tried to answer and the runtime will not let it keep improvising.
  1098   if (
  1099     evaluation.outcome === 'INCOMPLETE' &&
  1100     evaluation.reasonCode === 'EVIDENCE_INCOMPLETE' &&
  1101     (decision.decisionType === 'RESPOND' || decision.decisionType === 'STOP')
  1102   ) {
  1103     return ctx.stop(
  1104       'INSUFFICIENT_EVIDENCE',
  1105       'The available evidence is incomplete for a final answer.'
  1106     )
  1107   }
  1108   return null
  1109 }

   276             reasonCode: 'EVIDENCE_INCOMPLETE',

   278             detail: 'Evidence sources conflict; the answer must be qualified.'

   280       }

   282

   284       decision.decisionType === 'RESPOND' &&
   285       !decision.responseText?.trim() &&
   286       !decision.responseIntent?.trim()
   287     ) {
   288       return {
   289         outcome: 'INCOMPLETE',
   290         reasonCode: 'EVIDENCE_INCOMPLETE',
   291         deterministic: true,

   293       }
   294     }
   295 
   296     return {
   297       outcome: 'COMPLETE',
   298       reasonCode: 'COMPLETION_CONFIRMED',
   299       deterministic: true,
   300       detail:
   301         this.strategy === 'DETERMINISTIC'
   302           ? 'Structured completion accepted by the deterministic strategy.'
   303           : 'Structured completion accepted.'
   304     }

    92   public register(registration: CapabilityRegistration): CapabilityRegistry {
    93     const normalized = normalizeRegistration(registration)
    94     if (
    95       getRegistrations(this).some(
    96         (candidate) =>
    97           candidate.descriptor.id === normalized.descriptor.id &&
    98           candidate.descriptor.version === normalized.descriptor.version
    99       )
   100     ) {
   101       throw new CapabilityRegistryError(
   102         'duplicate_capability',
   103         `Capability already registered: ${normalized.descriptor.id}@${normalized.descriptor.version}`
   104       )
   105     }
   106 
   107     return CapabilityRegistry.fromStored([
   108       ...getRegistrations(this),
   109       normalized
   110     ])
   111   }
   112 
   113   public listDescriptors(): readonly CapabilityDescriptor[] {
   114     return sorted(getRegistrations(this)).map(({ descriptor }) =>
   115       cloneDescriptor(descriptor)
   116     )
   117   }

   119   public resolveDescriptor(
   120     capabilityId: string,
   121     version: string
   122   ): CapabilityDescriptor | undefined {
   123     if (!isBoundedIdentifier(capabilityId, MAX_ID_LENGTH)) return undefined
   124     if (!isBoundedVersion(version)) return undefined
   125     const registration = getRegistrations(this).find(
   126       (candidate) =>
   127         candidate.descriptor.id === capabilityId &&
   128         candidate.descriptor.version === version
   129     )
   130     return registration ? cloneDescriptor(registration.descriptor) : undefined
   131   }
   132 
   133   public compositionFingerprint(): string {
   134     const descriptors = sorted(getRegistrations(this)).map(
   135       ({ descriptor }) => descriptor
   136     )
   137     return createHash('sha256')
   138       .update(
   139         stableSerialize({
   140           schemaVersion: 'capability-composition-v1',
   141           descriptors
   142         })
   143       )
   144       .digest('hex')
   145   }

   151       CapabilityRegistry.prototype
   152     ) as CapabilityRegistry
   153     registrationsByRegistry.set(registry, Object.freeze([...registrations]))
   154     return registry
   155   }

   184 export function createCapabilityToolRegistry(
   185   registry: CapabilityRegistryPort,
   186   prepareResult?: (result: unknown) => ToolResult
   187 ): ToolRegistry {
   188   const snapshot = sorted(getRegistrations(registry))
   189   return {

   191       snapshot.map((registration) => toTool(registration, prepareResult)),

   198       }
   199       const registration = snapshot.find(
   200         (candidate) =>
   201           candidate.descriptor.id === capabilityId &&
   202           candidate.descriptor.version === version
   203       )
   204       return registration ? toTool(registration, prepareResult) : undefined
   205     }
   206   }

   209 function normalizeDescriptor(value: unknown): CapabilityDescriptor {
   210   if (!isPlainRecord(value)) {
   211     throw new CapabilityRegistryError(
   212       'invalid_descriptor',
   213       'Capability descriptor must be a plain object'
   214     )
   215   }
   216 
   217   const descriptor = value as Record<string, unknown>
   218   if (!isBoundedIdentifier(descriptor.id, MAX_ID_LENGTH)) {
   219     throw invalidDescriptor('Capability id is invalid')
   220   }
   221   if (!isBoundedVersion(descriptor.version)) {
   222     throw invalidDescriptor('Capability version must be exact and bounded')
   223   }
   224   if (
   225     typeof descriptor.description !== 'string' ||
   226     descriptor.description.trim().length === 0 ||
   227     descriptor.description.length > MAX_DESCRIPTION_LENGTH
   228   ) {
   229     throw invalidDescriptor('Capability description is invalid')
   230   }
   231   if (!ORIGINS.includes(descriptor.origin as CapabilityOrigin)) {
   232     throw invalidDescriptor('Capability origin is invalid')
   233   }
   234   if (!isBoundedIdentifier(descriptor.providerId, MAX_ID_LENGTH)) {
   235     throw invalidDescriptor('Capability provider id is invalid')
   236   }
   237   if (!isBoundedVersion(descriptor.providerVersion)) {
   238     throw invalidDescriptor('Capability provider version is invalid')
   239   }
   240   if (
   241     !isRisk(descriptor.risk) ||
   242     !isSideEffect(descriptor.sideEffect) ||
   243     typeof descriptor.idempotent !== 'boolean' ||
   244     typeof descriptor.requiresApproval !== 'boolean'
   245   ) {
   246     throw invalidDescriptor('Capability governance metadata is invalid')
   247   }
   248 
   249   const inputSchema = cloneJsonValue(descriptor.inputSchema)
   250   const outputSchema = cloneJsonValue(descriptor.outputSchema)
   251   return deepFreeze({
   252     id: descriptor.id,
   253     version: descriptor.version,
   254     description: descriptor.description.trim(),
   255     inputSchema,
   256     outputSchema,
   257     risk: descriptor.risk,
   258     sideEffect: descriptor.sideEffect,
   259     idempotent: descriptor.idempotent,
   260     requiresApproval: descriptor.requiresApproval,
   261     origin: descriptor.origin,
   262     providerId: descriptor.providerId,
   263     providerVersion: descriptor.providerVersion
   264   }) as CapabilityDescriptor
   265 }

   267 function normalizeRegistration(value: unknown): StoredRegistration {
   268   try {
   269     if (!isPlainRecord(value)) {
   270       throw new CapabilityRegistryError(
   271         'invalid_descriptor',
   272         'Capability registration must be a plain object'
   273       )
   274     }
   275     const registration = value as Record<string, unknown>
   276     return {
   277       descriptor: normalizeDescriptor(registration.descriptor),
   278       implementation: normalizeImplementation(registration.implementation)
   279     }
   280   } catch (error) {
   281     if (error instanceof CapabilityRegistryError) throw error
   282     throw new CapabilityRegistryError(
   283       'invalid_descriptor',
   284       'Capability registration contains unsupported data'
   285     )
   286   }
   287 }

   290   registry: CapabilityRegistryPort

   292   const registrations = registrationsByRegistry.get(registry as object)
   293   if (!registrations) {
   294     throw new CapabilityRegistryError(
   295       'invalid_descriptor',
   296       'Capability registry was not created by the neutral registry factory'
   297     )
   298   }
   299   return registrations
   300 }

   302 function normalizeImplementation(value: unknown): CapabilityImplementation {
   303   if (typeof value !== 'object' || value === null || Array.isArray(value)) {
   304     throw new CapabilityRegistryError(
   305       'invalid_implementation',
   306       'Capability implementation requires input validation, execution, and output validation'
   307     )
   308   }
   309 
   310   let validateInput: CapabilityImplementation['validateInput']
   311   let execute: CapabilityImplementation['execute']
   312   let validateOutput: CapabilityImplementation['validateOutput']
   313   try {
   314     const implementation = value as Record<string, unknown>
   315     if (
   316       typeof implementation.validateInput !== 'function' ||
   317       typeof implementation.execute !== 'function' ||
   318       typeof implementation.validateOutput !== 'function'
   319     ) {
   320       throw new Error('missing implementation method')
   321     }
   322     validateInput = (
   323       implementation.validateInput as CapabilityImplementation['validateInput']
   324     ).bind(value)
   325     execute = (
   326       implementation.execute as CapabilityImplementation['execute']
   327     ).bind(value)
   328     validateOutput = (
   329       implementation.validateOutput as CapabilityImplementation['validateOutput']
   330     ).bind(value)
   331   } catch {
   332     throw new CapabilityRegistryError(
   333       'invalid_implementation',
   334       'Capability implementation requires input validation, execution, and output validation'
   335     )
   336   }
   337   return Object.freeze({
   338     validateInput,
   339     execute,
   340     validateOutput
   341   }) as CapabilityImplementation
   342 }

   344 function toTool(
   345   registration: StoredRegistration,
   346   prepareResult?: (result: unknown) => ToolResult
   347 ): ToolDefinition {
   348   const descriptor = cloneDescriptor(registration.descriptor)
   349   return {
   350     ...descriptor,
   351     execute: (input: unknown, context: ToolExecutionContext) =>
   352       executeBound(registration.implementation, input, context, prepareResult)
   353   }
   354 }
   355 
   356 async function executeBound(
   357   implementation: CapabilityImplementation,
   358   input: unknown,
   359   context: ToolExecutionContext,
   360   prepareResult?: (result: unknown) => ToolResult
   361 ): Promise<ToolResult> {
   362   let safeInput: unknown
   363   try {
   364     safeInput = cloneJsonValue(input)
   365     if (!inputMatchesExecutionContext(safeInput, context)) {
   366       return { status: 'REJECTED', error: 'capability_input_invalid' }
   367     }
   368     if (implementation.validateInput(safeInput) !== true) {
   369       return { status: 'REJECTED', error: 'capability_input_invalid' }
   370     }
   371   } catch {
   372     return { status: 'REJECTED', error: 'capability_input_invalid' }
   373   }
   374 
   375   let result: ToolResult
   376   let preparing = false
   377   try {
   378     const safeContext = Object.freeze({ ...context }) as ToolExecutionContext
   379     const rawResult = await implementation.execute(safeInput, safeContext)
   380     preparing = Boolean(prepareResult)
   381     const normalizedResult = prepareResult
   382       ? prepareResult(rawResult)
   383       : normalizeToolResult(rawResult)
   384     preparing = false
   385     if (!normalizedResult) {
   386       return { status: 'REJECTED', error: 'capability_result_invalid' }
   387     }
   388     result = normalizedResult
   389   } catch {
   390     if (preparing)
   391       throw new Error('unknown_effect: complete_result_not_representable')
   392     return { status: 'FAILED', error: 'capability_execution_failed' }
   393   }
   394 
   395   if (result.status === 'SUCCEEDED') {
   396     try {
   397       if (implementation.validateOutput(result.output) !== true) {
   398         return { status: 'REJECTED', error: 'capability_output_invalid' }
   399       }
   400     } catch {
   401       return { status: 'REJECTED', error: 'capability_output_invalid' }
   402     }
   403   }
   404 
   405   try {
   406     if (prepareResult) return prepareResult(result)
   407     return {
   408       status: result.status,
   409       ...(result.output !== undefined
   410         ? { output: cloneJsonValue(result.output) }
   411         : {}),
   412       ...(result.error !== undefined ? { error: result.error } : {})
   413     }
   414   } catch {
   415     return { status: 'REJECTED', error: 'capability_result_invalid' }
   416   }
   417 }

   429 function sorted(
   430   registrations: readonly StoredRegistration[]
   431 ): readonly StoredRegistration[] {
   432   return [...registrations].sort((left, right) => {
   433     const idOrder = compareCodeUnits(left.descriptor.id, right.descriptor.id)
   434     if (idOrder !== 0) return idOrder
   435     return compareCodeUnits(left.descriptor.version, right.descriptor.version)
   436   })
   437 }
   438 
   439 function compareCodeUnits(left: string, right: string): number {
   440   return left < right ? -1 : left > right ? 1 : 0
   441 }

   443 function inputMatchesExecutionContext(
   444   value: unknown,
   445   context: ToolExecutionContext
   446 ): boolean {
   447   if (Array.isArray(value)) {
   448     return value.every((child) => inputMatchesExecutionContext(child, context))
   449   }
   450   if (!isPlainRecord(value)) return true
   451 
   452   for (const [key, child] of Object.entries(value)) {
   453     if (
   454       (AUTHORITY_FIELDS as readonly string[]).includes(key) &&
   455       child !== context[key as (typeof AUTHORITY_FIELDS)[number]]
   456     ) {
   457       return false
   458     }
   459     if (!inputMatchesExecutionContext(child, context)) return false
   460   }
   461   return true
   462 }

   506 function normalizeToolResult(value: unknown): ToolResult | undefined {
   507   try {
   508     const safeValue = cloneJsonValue(value)
   509     if (!isPlainRecord(safeValue)) return undefined
   510     const status = safeValue.status
   511     if (!isToolStatus(status)) return undefined
   512     const error = safeValue.error
   513     if (
   514       error !== undefined &&
   515       (typeof error !== 'string' || error.length > 240)
   516     ) {
   517       return undefined
   518     }
   519     const output = safeValue.output
   520     return {
   521       status,
   522       ...(output !== undefined ? { output } : {}),
   523       ...(error !== undefined ? { error } : {})
   524     }
   525   } catch {
   526     return undefined
   527   }
   528 }

   538 interface SerializationState {
   539   readonly seen: WeakSet<object>
   540   nodes: number
   541 }
   542 
   543 function stableSerialize(
   544   value: unknown,
   545   state: SerializationState = { seen: new WeakSet<object>(), nodes: 0 },
   546   depth = 0
   547 ): string {
   548   state.nodes += 1
   549   if (state.nodes > MAX_JSON_NODES) {
   550     throw new Error('JSON node budget exceeded')
   551   }
   552   if (depth > MAX_JSON_DEPTH) {
   553     throw new Error('JSON depth budget exceeded')
   554   }
   555 
   556   if (value === null) return 'null'
   557   if (typeof value === 'string') {
   558     if (value.length > MAX_JSON_STRING_LENGTH) {
   559       throw new Error('JSON string budget exceeded')
   560     }
   561     return JSON.stringify(value)
   562   }
   563   if (typeof value === 'boolean') return value ? 'true' : 'false'
   564   if (typeof value === 'number') {
   565     if (!Number.isFinite(value)) throw new Error('Non-finite value')
   566     return JSON.stringify(value)
   567   }
   568   if (typeof value !== 'object' || value === undefined) {
   569     throw new Error('Value is not JSON serializable')
   570   }
   571   if (state.seen.has(value)) throw new Error('Value contains a cycle')
   572   state.seen.add(value)
   573   try {
   574     if (Array.isArray(value)) {
   575       if (value.length > MAX_JSON_ARRAY_LENGTH) {
   576         throw new Error('JSON array budget exceeded')
   577       }
   578       const items: string[] = []
   579       for (let index = 0; index < value.length; index += 1) {
   580         if (!Object.prototype.hasOwnProperty.call(value, index)) {
   581           throw new Error('Sparse arrays are not supported')
   582         }
   583         items.push(stableSerialize(value[index], state, depth + 1))
   584       }
   585       return boundedSerialized(`[${items.join(',')}]`)
   586     }
   587     if (!isPlainRecord(value)) throw new Error('Value is not a plain object')
   588     const keys = Object.keys(value)
   589     if (
   590       keys.length > MAX_JSON_KEYS ||
   591       Object.getOwnPropertySymbols(value).length > 0 ||
   592       Object.getOwnPropertyNames(value).length !== keys.length
   593     ) {
   594       throw new Error('JSON object budget or shape exceeded')
   595     }
   596     const parts: string[] = []
   597     for (const key of keys.sort()) {
   598       if (key.length > MAX_JSON_KEY_LENGTH || UNSAFE_JSON_KEYS.has(key)) {
   599         throw new Error('Unsafe JSON object key')
   600       }
   601       const property = Object.getOwnPropertyDescriptor(value, key)
   602       if (!property || !('value' in property)) {
   603         throw new Error('Accessor properties are not supported')
   604       }
   605       parts.push(
   606         `${JSON.stringify(key)}:${stableSerialize(property.value, state, depth + 1)}`
   607       )
   608     }
   609     return boundedSerialized(`{${parts.join(',')}}`)
   610   } finally {
   611     state.seen.delete(value)
   612   }
   613 }
   614 
   615 function boundedSerialized(value: string): string {
   616   if (value.length > MAX_JSON_BYTES) {
   617     throw new Error('JSON byte budget exceeded')
   618   }
   619   return value
   620 }

   622 function deepFreeze<T>(value: T): T {
   623   if (typeof value !== 'object' || value === null) return value
   624   Object.freeze(value)
   625   for (const child of Object.values(value)) deepFreeze(child)
   626   return value
   627 }

   629 function isPlainRecord(value: unknown): value is Record<string, unknown> {
   630   try {
   631     if (typeof value !== 'object' || value === null || Array.isArray(value)) {
   632       return false
   633     }
   634     const prototype = Object.getPrototypeOf(value)
   635     return prototype === Object.prototype || prototype === null
   636   } catch {
   637     return false
   638   }
