   345   public readonly requests: ApprovalRequest[] = []

   348   /** Single-use execution lifecycle, mirroring the durable authority. */
   349   public execution: ApprovalExecutionPort = {
   350     begin: async (request) => {
   351       const state = this.executionStates.get(request.approvalId)
   352       if (state !== undefined) {
   353         throw new Error(`Approval execution is ${state.toLowerCase()}`)
   354       }
   355       this.executionStates.set(request.approvalId, 'RESERVED')
   356       return {
   357         approvalId: request.approvalId,
   358         reservationId: `reservation_${request.approvalId}`

   360     },
   361     complete: async ({ request }) => {
   362       this.executionStates.set(request.approvalId, 'CONSUMED')
   363     },
   364     fail: async ({ request }) => {
   365       this.executionStates.delete(request.approvalId)
   366     },
   367     uncertain: async ({ request }) => {
   368       this.executionStates.set(request.approvalId, 'UNCERTAIN')
   369     }
   370   }
   371 
   372   public async request(request: ApprovalRequest): Promise<ApprovalDecision> {
   373     this.requests.push(request)
   374     const existing = this.decisions.get(request.operationKey)
   375     if (existing) return existing
   376     if (this.approveOnRequest) {
   377       const decision: ApprovalDecision = {
   378         status: 'APPROVED',
   379         approvalId: `approval_${request.operationKey}` as never,
   380         reason: 'approved by fixture'
   381       }
   382       this.decisions.set(request.operationKey, decision)
   383       return decision
   384     }
   385     const decision: ApprovalDecision = {
   386       status: 'PENDING',
   387       approvalId: `approval_${request.operationKey}` as never,
   388       reason: 'approval pending'
   389     }
   390     this.decisions.set(request.operationKey, decision)
   391     return decision
   392   }
   393 
   394   public approve(operationKey: string): void {
   395     this.decisions.set(operationKey, {
   396       status: 'APPROVED',

    78 /** Waterfall over a gate point; a throwing listener becomes a stop (I7). */
    79 export function runGate<S>(
    80   host: KernelHost,
    81   point: GatePoint,
    82   state: S
    83 ): Promise<KernelGate> {
    84   const listeners = host.listeners(point) as unknown as readonly {
    85     readonly plugin: string
    86     readonly fn: (
    87       state: S,
    88       next: () => Promise<KernelGate>
    89     ) => Promise<KernelGate>
    90   }[]
    91   const run = (index: number): Promise<KernelGate> => {
    92     const entry = listeners[index]
    93     if (!entry) return Promise.resolve(PROCEED)
    94     return entry

   100         )
   101       )
   102   }
   103   return run(0)

   138 /** Races an operation against the turn budget and the cancel signal (I8). */
   139 export async function race<T>(
   140   turn: TurnState,
   141   operation: () => Promise<T>
   142 ): Promise<T | typeof DEADLINE | typeof CANCELLED> {
   143   const signal = turn.signal
   144   if (signal.aborted) return CANCELLED
   145   let onAbort!: () => void
   146   const cancelled = new Promise<typeof CANCELLED>((resolve) => {
   147     onAbort = () => resolve(CANCELLED)
   148     signal.addEventListener('abort', onAbort, { once: true })
   149   })
   150   try {
   151     return await Promise.race([turn.budget.withDeadline(operation), cancelled])
   152   } finally {
   153     signal.removeEventListener('abort', onAbort)
   154   }
   155 }

   158  * Like `race`, but always starts the operation first: used once the kernel has
   159  * committed to dispatch, so the cancel signal is passed to the body instead of
   160  * skipping it.

   163   turn: TurnState,
   164   operation: () => Promise<T>
   165 ): Promise<T | typeof DEADLINE | typeof CANCELLED> {
   166   const running = operation()

   180       cancelled
   181     ])
   182   } finally {
   183     signal.removeEventListener('abort', onAbort)
   184   }

   254    * effect; the iterative loop persists its checkpoint here.
   255    */
   256   readonly beforeDispatch?: (
   257     call: ToolCallState
   258   ) => Promise<KernelStop | undefined>

   261 /**
   262  * log `tool/call` → `tool/pre-execute` → guards → cancellation → hook →
   263  * `tool/execute` around the body → `tool/post-execute` → log `tool/result`.
   264  */
   265 export async function runToolCall(
   266   host: KernelHost,
   267   call: ToolCallState,
   268   hooks: ToolPipelineHooks = {}
   269 ): Promise<ToolPipelineResult> {
   270   const { turn, tool, invocation } = call
   271   const correlationId = turn.input.correlationId
   272   // I5: the call is on the log before any policy runs or the body starts.
   273   const logged = await appendLog(host, {
   274     type: 'tool/call',
   275     correlationId,
   276     toolId: tool.id,
   277     operationKey: invocation.operationKey,
   278     input: invocation.input
   279   })
   280   if (logged) return { kind: 'stopped', gate: logged }
   281 
   282   // Every call that was logged is closed exactly once (I5, AUD-0597 F06).
   283   const closeNotStarted = async (
   284     blocked: KernelStop
   285   ): Promise<ToolPipelineResult> => {
   286     const unrecorded = await appendLog(host, {
   287       type: 'tool/result',
   288       correlationId,
   289       toolId: tool.id,
   290       outcome: 'not_started'
   291     })
   292     // A lost closing record is not hidden behind the denial (AUD-0598 R04):
   293     // the call stays blocked, and the turn reports the missing evidence.
   294     return unrecorded
   295       ? { kind: 'stopped', gate: unrecordedBlock(unrecorded, blocked), blocked }
   296       : { kind: 'stopped', gate: blocked }
   297   }
   298 
   299   const gate = await runGate(host, 'tool/pre-execute', call)
   300   if (gate.kind === 'stop') return closeNotStarted(gate)
   301 
   302   const denial = await revalidate(host, call)
   303   if (denial) return closeNotStarted(denial)
   304 
   305   const sandbox = await enforceSandbox(host, call)
   306   if (sandbox) return closeNotStarted(sandbox)
   307 
   308   let hooked: KernelStop | undefined
   309   try {
   310     hooked = await hooks.beforeDispatch?.(call)
   311   } catch (error) {
   312     // The loop's own checkpoint failed: the logged call is still closed once
   313     // and the original failure reaches the loop unchanged (AUD-0598 R03).
   314     await appendLog(host, {
   315       type: 'tool/result',
   316       correlationId,
   317       toolId: tool.id,
   318       outcome: 'not_started'
   319     })
   320     throw error
   321   }
   322   if (hooked) return closeNotStarted(hooked)
   323 
   324   // The hook may have taken time (checkpoint): re-check before reserving.
   325   const late = await revalidate(host, call)
   326   if (late) return closeNotStarted(late)
   327 
   328   const settled = await postExecute(host, call, await dispatch(host, call))
   329   const logFailure = await appendLog(host, {
   330     type: 'tool/result',
   331     correlationId,
   332     toolId: tool.id,
   333     outcome: settled.kind
   334   })
   335   return logFailure
   336     ? { kind: 'settled', outcome: settled, logFailure }
   337     : { kind: 'settled', outcome: settled }
   338 }

   340 /**
   341  * Derives the containment a tool needs from its descriptor (SPEC 0195 §2/§5).
   342  * EXTERNAL effects require an EGRESS gate; WRITE effects require WORKSPACE;
   343  * READ effects require at least IN_PROCESS. A mode-`NONE` capability is pure
   344  * in-memory and never reaches the sandbox gate.
   345  */
   346 /**
   347  * The mode is declared on the descriptor (SPEC 0195 §5). Absent or `'NONE'`
   348  * means no gate; the harness never infers containment from `sideEffect`, so
   349  * legacy side-effecting tools keep running until they opt in.
   350  */
   351 export function requiredSandboxMode(descriptor: ToolDescriptor): SandboxMode {
   352   return descriptor.sandbox ?? 'NONE'
   353 }
   354 
   355 /**
   356  * Enforcement point between every control decision and the capability body
   357  * (SPEC 0195 §3/§4). Fails closed: a missing gate, an `unavailable` outcome
   358  * or a thrown gate stops the call as `UNSAFE_REQUEST` and never downgrades
   359  * to in-process execution. Pure (`NONE`) tools skip the gate entirely.
   360  */
   361 async function enforceSandbox(
   362   host: KernelHost,
   363   call: ToolCallState
   364 ): Promise<KernelStop | undefined> {
   365   const requiredMode = requiredSandboxMode(call.tool)
   366   if (requiredMode === 'NONE') return undefined
   367   if (!host.has('sandbox')) {
   368     return stop(
   369       'UNSAFE_REQUEST',
   370       `Tool "${call.tool.id}" requires sandbox mode ${requiredMode} but no sandbox gate is configured.`
   371     )
   372   }
   373   const gate = host.get('sandbox')
   374   if (!gate) {
   375     // host.has was checked; a missing entry here is a host-integrity fault,
   376     // so fail closed the same way as an unconfigured gate.
   377     return stop(
   378       'UNSAFE_REQUEST',
   379       `Tool "${call.tool.id}" requires sandbox mode ${requiredMode} but no sandbox gate is configured.`
   380     )
   381   }
   382   try {
   383     const outcome: SandboxGateOutcome = await gate.evaluate({
   384       descriptor: call.tool,
   385       context: {
   386         tenantId: call.turn.input.tenantId,
   387         agentId: call.turn.input.agent.id,
   388         correlationId: call.turn.input.correlationId,
   389         traceId: call.turn.input.traceId,
   390         operationKey: call.invocation.operationKey
   391       },
   392       policy: {
   393         requiredMode,
   394         workspaceScoped: requiredMode === 'WORKSPACE',
   395         networkAllowed: requiredMode === 'EGRESS',
   396         subprocessAllowed: requiredMode === 'PROCESS'
   397       }
   398     })
   399     if (outcome.kind === 'denied') {
   400       return stop(
   401         'UNSAFE_REQUEST',
   402         `Sandbox denied tool "${call.tool.id}": ${outcome.reason}.`
   403       )
   404     }
   405     if (outcome.kind === 'unavailable') {
   406       return stop(
   407         'UNSAFE_REQUEST',
   408         `Sandbox mode ${requiredMode} unavailable for tool "${call.tool.id}": ${outcome.reason}.`
   409       )
   410     }
   411     const granted = outcome.kind === 'allowed' ? outcome.handle : undefined
   412     const handle = granted && {
   413       mode: granted.mode,
   414       allowNetwork: granted.allowNetwork,
   415       allowSubprocess: granted.allowSubprocess,
   416       workspaceRoot: granted.workspaceRoot,
   417       deadlineMs: granted.deadlineMs
   418     }
   419     if (
   420       outcome.kind !== 'allowed' ||
   421       outcome.mode !== requiredMode ||
   422       !handle ||
   423       handle.mode !== requiredMode ||
   424       typeof handle.allowNetwork !== 'boolean' ||
   425       typeof handle.allowSubprocess !== 'boolean' ||
   426       (handle.deadlineMs !== undefined &&
   427         (!Number.isFinite(handle.deadlineMs) || handle.deadlineMs <= 0)) ||
   428       (requiredMode === 'WORKSPACE' && handle.workspaceRoot === undefined) ||
   429       (handle.workspaceRoot !== undefined &&
   430         (typeof handle.workspaceRoot !== 'string' ||
   431           !isAbsolute(handle.workspaceRoot) ||
   432           handle.workspaceRoot.includes('\0')))
   433     ) {
   434       return stop(
   435         'UNSAFE_REQUEST',
   436         `Sandbox gate returned an invalid handle for tool "${call.tool.id}".`
   437       )
   438     }
   439     // Copy only validated grants. Neither a later gate mutation nor model
   440     // JSON can rewrite the authority passed to the capability body.
   441     call.sandboxHandle = Object.freeze({
   442       mode: handle.mode,
   443       allowNetwork: handle.allowNetwork,
   444       allowSubprocess: handle.allowSubprocess,
   445       ...(handle.workspaceRoot !== undefined
   446         ? { workspaceRoot: handle.workspaceRoot }
   447         : {}),
   448       ...(handle.deadlineMs !== undefined
   449         ? { deadlineMs: handle.deadlineMs }
   450         : {})
   451     })
   452     return undefined
   453   } catch (error) {
   454     return stop(
   455       'UNSAFE_REQUEST',
   456       `Sandbox gate failed for tool "${call.tool.id}": ${errorMessage(error)}.`
   457     )
   458   }
   459 }

   461 async function dispatch(
   462   host: KernelHost,
   463   call: ToolCallState
   464 ): Promise<ToolOutcome> {
   465   const { turn } = call
   466   const ref = {
   467     tenantId: turn.input.tenantId,
   468     correlationId: turn.input.correlationId,
   469     toolId: call.tool.id,
   470     operationKey: call.invocation.operationKey
   471   }
   472   let started = false
   473   const terminal = async (): Promise<ToolOutcome> => {
   474     if (turn.budget.remainingMs() <= 0) {
   475       return {
   476         kind: 'not_started',
   477         gate: stop(
   478           'MAX_DURATION',
   479           'Duration budget exhausted before tool execution.'
   480         )
   481       }
   482     }
   483     // Last decision point: after every wrapper (the approval reservation),
   484     // a pause or a cancellation still prevents the effect (F01/F02).
   485     const blocked = await revalidate(host, call)
   486     if (blocked) return { kind: 'not_started', gate: blocked }
   487     try {
   488       await host.get('effects').started(ref)
   489     } catch (error) {
   490       return {
   491         kind: 'not_started',
   492         gate: stop(
   493           'INSUFFICIENT_EVIDENCE',
   494           `Effect ledger could not record the start: ${errorMessage(error)}.`
   495         )
   496       }
   497     }
   498     started = true
   499     const startedAt = Date.now()
   500     const toolController = new AbortController()
   501     try {
   502       // Past the decision point the body is always invoked, so a recorded
   503       // start never pairs with a body that did not run.
   504       const value = await raceInvoked(turn, () =>
   505         call.tool.execute(call.invocation.input, {
   506           tenantId: turn.input.tenantId,
   507           agentId: turn.input.agent.id,
   508           correlationId: turn.input.correlationId,
   509           traceId: turn.input.traceId,
   510           operationKey: call.invocation.operationKey,
   511           ...(call.sandboxHandle ? { sandboxHandle: call.sandboxHandle } : {}),
   512           signal: AbortSignal.any([toolController.signal, turn.signal])
   513         })
   514       )
   515       const durationMs = Date.now() - startedAt
   516       if (value === DEADLINE || value === CANCELLED) {
   517         toolController.abort()
   518         return {
   519           kind: value === DEADLINE ? 'deadline' : 'cancelled',
   520           durationMs
   521         }
   522       }
   523       return value.status === 'SUCCEEDED'
   524         ? { kind: 'succeeded', result: value, durationMs }
   525         : { kind: 'failed', result: value, durationMs }
   526     } catch (error) {
   527       return {
   528         kind: 'threw',
   529         error: errorMessage(error),
   530         durationMs: Date.now() - startedAt
   531       }
   532     }
   533   }
   534   const listeners = host.listeners('tool/execute')
   535   const run = (index: number): Promise<ToolOutcome> => {
   536     const entry = listeners[index]
   537     if (!entry) return terminal()
   538     return entry
   539       .fn(call, () => run(index + 1))
   540       .catch(
   541         (error: unknown): ToolOutcome =>
   542           started
   543             ? {
   544                 kind: 'threw',
   545                 error: `${entry.plugin} failed: ${errorMessage(error)}`,
   546                 durationMs: 0
   547               }
   548             : {
   549                 kind: 'not_started',
   550                 gate: stop(
   551                   'INSUFFICIENT_EVIDENCE',
   552                   `${entry.plugin} failed before dispatch: ${errorMessage(error)}.`
   553                 )
   554               }
   555       )
   556   }
   557   const outcome = await run(0)
   558   if (started) {
   559     try {
   560       await host.get('effects').finished(ref, outcome.kind)
   561     } catch {
   562       // The body already ran. A missing settlement is uncertain, never a
   563       // successful effect or permission to retry the body/ledger.
   564       return {
   565         kind: 'threw',
   566         error: 'unknown_effect: effect_settlement_failed',
   567         durationMs: 'durationMs' in outcome ? outcome.durationMs : 0
   568       }
   569     }
   570   }
   571   return outcome
   572 }

   574 async function postExecute(
   575   host: KernelHost,
   576   call: ToolCallState,
   577   outcome: ToolOutcome
   578 ): Promise<ToolOutcome> {
   579   let current = outcome
   580   for (const entry of host.listeners('tool/post-execute')) {
   581     try {
   582       current = await entry.fn(call, current)
   583     } catch (error) {
   584       // I7: a failing settlement after an effect never reads as success.
   585       if (current.kind === 'succeeded') {
   586         current = {
   587           kind: 'confirm_failed',
   588           result: current.result,
   589           error: `${entry.plugin} failed: ${errorMessage(error)}`,
   590           durationMs: current.durationMs
   591         }
   592       }
   593     }
   594   }
   595   return current
   596 }

   611 /**
   612  * `model/before-call` → log `model/request` (I6) → call raced against the
   613  * budget and the cancel signal → log `model/result`.
   614  */
   615 export async function runModelCall(
   616   host: KernelHost,
   617   turn: TurnState,
   618   request: Omit<ModelRequest, 'signal'>
   619 ): Promise<ModelPipelineResult> {
   620   const controller = new AbortController()
   621   const full: ModelRequest = {
   622     ...request,
   623     signal: AbortSignal.any([controller.signal, turn.signal])
   624   }
   625   const preGate = turn.budget.beginModelRequest()
   626   if (preGate) return { kind: 'stopped', gate: preGate }
   627   const gate = await runGate(host, 'model/before-call', { turn, request: full })
   628   if (gate.kind === 'stop') return { kind: 'stopped', gate }
   629   if (turn.signal.aborted) return { kind: 'cancelled' }
   630   // I6: log the request as it will be sent, after every control adjusted it.
   631   const { signal: _signal, ...effective } = full
   632   void _signal
   633   const logged = await appendLog(host, {
   634     type: 'model/request',
   635     correlationId: turn.input.correlationId,
   636     request: snapshot(effective)
   637   })
   638   if (logged) return { kind: 'stopped', gate: logged }
   639   try {
   640     const outcome = await race(turn, () => host.get('model').complete(full))
   641     if (outcome === DEADLINE || outcome === CANCELLED) {
   642       controller.abort()
   643       return { kind: outcome === DEADLINE ? 'deadline' : 'cancelled' }
   644     }
   645     const unrecorded = await appendLog(host, {
   646       type: 'model/result',
   647       correlationId: turn.input.correlationId,
   648       provider: outcome.provider,
   649       inputTokens: outcome.inputTokens,
   650       outputTokens: outcome.outputTokens
   651     })
   652     if (unrecorded) {
   653       return { kind: 'unrecorded', result: outcome, gate: unrecorded }
   654     }
   655     const accounted = turn.budget.recordModelResult({
   656       inputTokens: outcome.inputTokens,
   657       outputTokens: outcome.outputTokens,
   658       costUsd: outcome.costUsd
   659     })
   660     if (accounted) return { kind: 'stopped', gate: accounted }
   661     return { kind: 'completed', result: outcome }
   662   } catch (error) {
   663     return { kind: 'failed', error: errorMessage(error) }
   664   }
   665 }
