   171 function lane(ctx = context()) {
   172   const events: string[] = []
   173   const records: DecisionAdmissionRecord[] = []
   174   const snapshots: DecisionModelRequestSnapshot[] = []
   175   const requests: ModelRequest[] = []
   176   const settlements: DecisionAdmissionSettlement[] = []
   177   let liveTokens =
   178     ctx.budget.maxTokens -
   179     ctx.budgetUsage.inputTokens -
   180     ctx.budgetUsage.outputTokens
   181   let liveCost = ctx.budget.maxCostUsd - ctx.budgetUsage.costUsd
   182   let calls = ctx.budgetUsage.modelCalls
   183   const measure = vi.fn((request: DecisionModelRequestSnapshot) => {
   184     expect(request.maxOutputTokens).toBe(512)
   185     return measurement()
   186   })
   187   const reserve = vi.fn(
   188     async (input: Parameters<DecisionAdmissionPort['reserve']>[0]) => {
   189       events.push('reserve')
   190       const { request, measurement: counted } = input
   191       if (
   192         calls >= ctx.budget.maxModelCalls ||
   193         liveTokens < counted.inputUpperBoundTokens + 512 ||
   194         liveCost < counted.costUpperBoundUsd
   195       )
   196         throw new ToolResultContextError(
   197           'ADMISSION_UNAVAILABLE',
   198           'Synthetic native balance exhausted',
   199           'MAX_TOKENS'
   200         )
   201       calls += 1
   202       liveTokens -= counted.inputUpperBoundTokens + 512
   203       liveCost -= counted.costUpperBoundUsd
   204       snapshots.push(request)
   205       const record: DecisionAdmissionRecord = {
   206         admissionId: `admission.${records.length}`,
   207         tenantId: ctx.tenantId,
   208         executionId: ctx.executionId,
   209         stepNumber: ctx.stepNumber,
   210         attempt: records.length,
   211         purpose: request.purpose as 'ORCHESTRATION' | 'REPAIR',
   212         requestSha256: digest(request),
   213         measurement: counted,
   214         reserved: {
   215           modelCalls: 1,
   216           inputTokens: counted.inputUpperBoundTokens,
   217           outputTokens: 512,
   218           costUsd: counted.costUpperBoundUsd
   219         },
   220         status: 'RESERVED'
   221       }
   222       records.push(record)
   223       return record
   224     }
   225   )
   226   const settle = vi.fn(async (input: DecisionAdmissionSettlement) => {
   227     events.push(`settle.${input.kind}`)
   228     settlements.push(input)
   229     const record = records.find(
   230       (record) => record.admissionId === input.admissionId
   231     )!
   232     if (input.kind === 'UNKNOWN')
   233       return {
   234         ...record,
   235         status: 'UNKNOWN' as const,
   236         unknownReason: input.reason
   237       }
   238     const actual = {
   239       modelCalls: 1,
   240       inputTokens: input.result.inputTokens,
   241       outputTokens: input.result.outputTokens,
   242       costUsd: input.result.costUsd
   243     }
   244     liveTokens +=
   245       record.reserved.inputTokens +
   246       record.reserved.outputTokens -
   247       actual.inputTokens -
   248       actual.outputTokens
   249     liveCost += record.reserved.costUsd - actual.costUsd
   250     return {
   251       ...record,
   252       status: 'SETTLED' as const,
   253       actual,
   254       resultSha256: digest(input.result)
   255     }
   256   })
   257   const complete = vi.fn(
   258     async (request: ModelRequest): Promise<ModelResult> => {
   259       events.push('model')
   260       requests.push(request)
   261       const { signal: _signal, ...wire } = request
   262       void _signal // Local cancellation is deliberately absent from the wire hash.

   268         )
   269         .map(
   270           (item: { value: Observation }) =>
   271             (item.value.payload as { output: unknown }).output
   272         )
   273       return result({
   274         text: JSON.stringify({
   275           ...decision,
   276           responseText: JSON.stringify(outputs)
   277         })
   278       })
   279     }
   280   )
   281   requestEvidence.push({
   282     kind: 'native-port-lane',
   283     events,
   284     records,
   285     snapshots,
   286     settlements
   287   })

   291     records,

   293     requests,
   294     settlements,

   296     reserve,
   297     settle,

   301       decisionInputCounter: { measure },

   303     },
