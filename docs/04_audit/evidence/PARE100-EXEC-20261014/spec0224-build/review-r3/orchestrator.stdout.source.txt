    61 function parseDecisionText(text: string): unknown {
    62   const trimmed = text.trim()
    63   const withoutFence = trimmed
    64     .replace(/^```(?:json)?/i, '')
    65     .replace(/```$/, '')
    66     .trim()
    67   const start = withoutFence.indexOf('{')
    68   const end = withoutFence.lastIndexOf('}')
    69   if (start === -1 || end === -1 || end <= start) {
    70     throw new OrchestratorDecisionError(
    71       'decision_unparseable',
    72       'Orchestrator output did not contain a JSON object'
    73     )
    74   }
    75   try {
    76     return JSON.parse(withoutFence.slice(start, end + 1))
    77   } catch {
    78     throw new OrchestratorDecisionError(
    79       'decision_unparseable',
    80       'Orchestrator output was not valid JSON'
    81     )
    82   }
    83 }

   108     `State: ${context.stateSummary}`,
   109     `Knowledge available: ${context.knowledgeAvailable ? 'yes' : 'no'}`,

   222     throw new ToolResultContextError(

   225     )

   227   const seen = new Set<string>()
   228   const ids = new Set<string>()
   229   const selected = context.contextItems.map((item) => {
   230     if (ids.has(item.contextId)) {
   231       throw new ToolResultContextError(
   232         'STATE_CONFLICT',

   238       contextId: item.contextId,

   264       observation.source !== 'TOOL' ||

   266       observation.trust !== 'UNTRUSTED' ||

   304 /** Race only pure counting / the model; durable reserve and settle stay runtime-owned. */
   305 async function withinDecisionDeadline<T>(
   306   operation: () => T | Promise<T>,
   307   remainingMs: number,
   308   signal?: AbortSignal
   309 ): Promise<T> {
   310   if (signal?.aborted || remainingMs <= 0) {
   311     throw new ToolResultContextError(
   312       'ADMISSION_UNAVAILABLE',
   313       'Decision deadline or cancellation',
   314       'MAX_DURATION'
   315     )
   316   }
   317   let timer: ReturnType<typeof setTimeout> | undefined
   318   let cancel: (() => void) | undefined
   319   const deadline = Date.now() + remainingMs
   320   try {
   321     const stopped = new Promise<never>((_resolve, reject) => {
   322       cancel = () =>
   323         reject(
   324           new ToolResultContextError(
   325             'ADMISSION_UNAVAILABLE',
   326             'Decision cancelled or late',
   327             'MAX_DURATION'
   328           )
   329         )
   330       signal?.addEventListener('abort', cancel, { once: true })
   331       timer = setTimeout(cancel, Math.min(remainingMs, 2_147_483_647))
   332     })
   333     return await Promise.race([
   334       Promise.resolve().then(() => {
   335         if (signal?.aborted || Date.now() >= deadline) {
   336           throw new ToolResultContextError(
   337             'ADMISSION_UNAVAILABLE',
   338             'Decision cancelled before starting',
   339             'MAX_DURATION'
   340           )
   341         }
   342         return operation()
   343       }),
   344       stopped
   345     ])
   346   } finally {
   347     if (timer !== undefined) clearTimeout(timer)
   348     if (cancel) signal?.removeEventListener('abort', cancel)
   349   }
   350 }

   352 function checkReceipt(
   353   receipt: DecisionAdmissionRecord,
   354   context: StepContext,
   355   request: DecisionModelRequestSnapshot,
   356   measurement: DecisionInputMeasurement,
   357   requestSha256: string,
   358   status: 'RESERVED' | 'SETTLED',
   359   previous?: DecisionAdmissionRecord
   360 ): void {
   361   if (
   362     !receipt ||
   363     typeof receipt.admissionId !== 'string' ||
   364     !receipt.admissionId ||
   365     receipt.status !== status ||
   366     receipt.tenantId !== context.tenantId ||
   367     receipt.executionId !== context.executionId ||
   368     receipt.stepNumber !== context.stepNumber ||
   369     receipt.purpose !== request.purpose ||
   370     receipt.requestSha256 !== requestSha256 ||
   371     !Number.isSafeInteger(receipt.attempt) ||
   372     receipt.attempt < 0 ||
   373     completeDigest(receipt.measurement) !== completeDigest(measurement) ||
   374     receipt.reserved?.modelCalls !== 1 ||
   375     receipt.reserved.inputTokens !== measurement.inputUpperBoundTokens ||
   376     receipt.reserved.outputTokens !== COMPLETE_DECISION_OUTPUT_TOKENS ||
   377     receipt.reserved.costUsd !== measurement.costUpperBoundUsd ||
   378     (previous &&
   379       (previous.admissionId !== receipt.admissionId ||
   380         previous.attempt !== receipt.attempt))
   381   ) {
   382     throw new ToolResultContextError(
   383       'STATE_CONFLICT',
   384       'Decision admission receipt does not bind the measured request',
   385       'STATE_CONFLICT'
   386     )

   390 /**
   391  * Hybrid orchestrator: deterministic rules first, probabilistic decision
   392  * support only when semantics are actually required. It selects actions but
   393  * never authorizes them; policy, approval and budgets remain outside.
   394  */
   395 export class HybridOrchestrator implements IterativeOrchestrator {
   396   public readonly contextProtocol = 'complete-tool-v1' as const
   397   public readonly version: string
   398   private readonly decisionModel: ModelGateway | undefined
   399   private readonly decisionInputCounter: DecisionInputCounter | undefined
   400   private readonly decisionModelSupportsMaxOutputTokens: boolean
   401   private decisionPairBreached = false
   402   private readonly rules: readonly OrchestrationRule[]
   403   private readonly maxDecisionRepairs: number
   404 
   405   public constructor(options: HybridOrchestratorOptions = {}) {
   406     this.decisionModel = options.decisionModel
   407     this.decisionInputCounter = options.decisionInputCounter
   408     this.decisionModelSupportsMaxOutputTokens =
   409       options.decisionModelSupportsMaxOutputTokens === true
   410     this.rules = options.rules ?? []
   411     const requested = options.maxDecisionRepairs ?? 1
   412     this.maxDecisionRepairs = Number.isFinite(requested)
   413       ? Math.min(5, Math.max(0, Math.trunc(requested)))
   414       : 1
   415     this.version = options.version ?? 'hybrid-1.0.0'
   416   }

   418   public async decide(
   419     input: IterativeOrchestratorInput
   420   ): Promise<IterativeOrchestratorTurn> {
   421     const context = input.context
   422     for (const rule of this.rules) {
   423       const decision = rule(context)
   424       if (!decision) continue
   425       const validation = validateLoopDecision(decision)
   426       if (!validation.valid) {
   427         throw new OrchestratorDecisionError(
   428           'decision_invalid',
   429           `Deterministic rule produced an invalid decision: ${validation.errors.join('; ')}`
   430         )
   431       }
   432       return context.toolResultContext
   433         ? {
   434             decision,
   435             decisionAdmissionIds: [
   436               ...(context.toolResultContext.settledDecisionAdmissionIds ?? [])
   437             ]
   438           }
   439         : { decision, usage: EMPTY_MODEL_USAGE }
   440     }
   441 
   442     if (context.toolResultContext) return this.decideComplete(input)
   443 
   444     if (!this.decisionModel) {
   445       throw new OrchestratorDecisionError(
   446         'orchestrator_unavailable',
   447         'No deterministic rule matched and no decision model is configured'
   448       )
   449     }
   450 
   451     let usage: ModelUsage = EMPTY_MODEL_USAGE
   452     let lastErrors: readonly string[] = []
   453     let repairAttempt = 0
   454     while (true) {
   455       if (input.signal?.aborted) {
   456         throw new OrchestratorDecisionError(
   457           'orchestrator_unavailable',
   458           'Decision was cancelled by the runtime deadline'
   459         )
   460       }
   461       const prompt = buildDecisionPrompt({
   462         ...context,
   463         ...(repairAttempt > 0
   464           ? {
   465               decisionRepair: {
   466                 attempt: repairAttempt,
   467                 errors: lastErrors
   468               }
   469             }
   470           : {})
   471       })
   472       const modelResult = await this.decisionModel.complete({
   473         messages: [
   474           { role: 'system', content: prompt },
   475           { role: 'user', content: context.userMessage }
   476         ],
   477         context: {
   478           values: { goal: context.goal },
   479           sourceIds: ['hybrid-orchestrator'],
   480           capturedAt: new Date().toISOString()
   481         },
   482         budget: context.budget,
   483         correlationId: context.correlationId,
   484         purpose: repairAttempt > 0 ? 'REPAIR' : 'ORCHESTRATION',
   485         ...(input.signal ? { signal: input.signal } : {})
   486       })
   487       usage = addUsage(usage, modelResult)
   488       let parsed: unknown
   489       try {
   490         parsed = parseDecisionText(modelResult.text)
   491       } catch (error) {
   492         if (repairAttempt >= this.maxDecisionRepairs) {
   493           throw error
   494         }
   495         repairAttempt += 1
   496         lastErrors = [
   497           error instanceof Error ? error.message : 'unparseable decision'
   498         ]
   499         continue
   500       }
   501       const validation = validateLoopDecision(parsed)
   502       const errors = validation.valid
   503         ? disallowedTypeErrors(sanitizeLoopDecision(parsed), context)
   504         : validation.errors
   505       if (errors.length === 0) {
   506         return {
   507           decision: sanitizeLoopDecision(parsed),
   508           usage: resultToUsage(usage)
   509         }
   510       }
   511       if (repairAttempt >= this.maxDecisionRepairs) {
   512         throw new OrchestratorDecisionError(
   513           'decision_invalid',
   514           `Orchestrator decision stayed invalid after ${repairAttempt} repairs: ${errors.join('; ')}`
   515         )
   516       }
   517       repairAttempt += 1
   518       lastErrors = errors
   519     }
   520   }

   522   private async decideComplete(
   523     input: IterativeOrchestratorInput
   524   ): Promise<IterativeOrchestratorTurn> {
   525     const { context, decisionAdmission } = input
   526     const model = this.decisionModel
   527     const counter = this.decisionInputCounter
   528     if (
   529       !model ||
   530       !counter ||
   531       !this.decisionModelSupportsMaxOutputTokens ||
   532       !decisionAdmission ||
   533       this.decisionPairBreached
   534     ) {
   535       throw new ToolResultContextError(
   536         'ADMISSION_UNAVAILABLE',
   537         'Complete decisions require a qualified counter, output cap and native admission port'
   538       )
   539     }
   540     const startedAt = Date.now()
   541     const remainingMs = () =>
   542       context.budget.maxDurationMs -
   543       context.budgetUsage.activeDurationMs -
   544       (Date.now() - startedAt)
   545     const ensureTime = () => {
   546       if (
   547         !finiteNonnegative(context.budget.maxDurationMs) ||
   548         !finiteNonnegative(context.budgetUsage.activeDurationMs) ||
   549         input.signal?.aborted ||
   550         remainingMs() <= 0
   551       ) {
   552         throw new ToolResultContextError(
   553           'ADMISSION_UNAVAILABLE',
   554           'Decision deadline or cancellation',
   555           'MAX_DURATION'
   556         )
   557       }
   558     }
   559     let selected = completeItems(context)
   560     // Native validation may re-enter the same step after a settled turn.
   561     // Report those receipts too; their usage has already been charged.
   562     const decisionAdmissionIds: string[] = [
   563       ...(context.toolResultContext?.settledDecisionAdmissionIds ?? [])
   564     ]
   565     let errors: readonly string[] = context.decisionRepair?.errors ?? []
   566     let repairs = 0
   567     while (true) {
   568       ensureTime()
   569       const attempt = (context.decisionRepair?.attempt ?? 0) + repairs
   570       const isRepair = context.decisionRepair !== undefined || repairs > 0
   571       let request: DecisionModelRequestSnapshot
   572       let measurement: DecisionInputMeasurement
   573       // Trimming always reconstructs and recounts the whole request. Estimates
   574       // on ContextItem are deliberately never consulted here.
   575       while (true) {
   576         ensureTime()
   577         let overflow: ToolResultContextError | undefined
   578         try {
   579           const system = completeSystem(
   580             context,
   581             selected.map(({ item }) => item)
   582           )
   583           const user = canonicalizeCompleteJson(
   584             {
   585               schema: 'cvg.decision-context',
   586               version: 1,
   587               goal: context.goal,
   588               userMessage: context.userMessage,
   589               state: context.stateSummary,
   590               items: selected.map(({ frame }) => frame),
   591               repair: isRepair ? { attempt, errors } : null
   592             },
   593             REQUEST_JSON_LIMITS
   594           )
   595           if (
   596             system.length > COMPLETE_DECISION_SYSTEM_MAX_UNITS ||
   597             user.length > COMPLETE_DECISION_USER_MAX_UNITS
   598           ) {
   599             throw new ToolResultContextError(
   600               'CONTEXT_OVERFLOW',
   601               'Complete decision message exceeds its limit'
   602             )
   603           }
   604           // Serialize once to detach caller-owned objects, then recursively
   605           // freeze the exact snapshot shared with counter and native reserve.
   606           request = freezeComplete(
   607             JSON.parse(
   608               canonicalizeCompleteJson(
   609                 {
   610                   messages: [
   611                     { role: 'system', content: system },
   612                     { role: 'user', content: user }
   613                   ],
   614                   context: {
   615                     values: { goal: context.goal },
   616                     sourceIds: ['hybrid-orchestrator'],
   617                     capturedAt: new Date().toISOString()
   618                   },
   619                   budget: context.budget,
   620                   correlationId: context.correlationId,
   621                   purpose: isRepair ? 'REPAIR' : 'ORCHESTRATION',
   622                   maxOutputTokens: COMPLETE_DECISION_OUTPUT_TOKENS
   623                 },
   624                 REQUEST_JSON_LIMITS
   625               )
   626             )
   627           ) as DecisionModelRequestSnapshot
   628         } catch (error) {
   629           if (error instanceof ToolResultContextError) overflow = error
   630           else if (
   631             error instanceof Error &&
   632             'code' in error &&
   633             error.code === 'RESULT_TOO_LARGE'
   634           ) {
   635             overflow = new ToolResultContextError(
   636               'CONTEXT_OVERFLOW',
   637               'Complete decision request exceeds its byte limit'
   638             )
   639           } else throw error
   640         }
   641         if (!overflow) {
   642           let measured: DecisionInputMeasurement
   643           try {
   644             measured = await withinDecisionDeadline(
   645               () => counter.measure(request),
   646               remainingMs(),
   647               input.signal
   648             )
   649             ensureTime()
   650             if (!validMeasurement(measured))
   651               throw new Error('invalid measurement')
   652             measurement = freezeComplete(
   653               JSON.parse(
   654                 canonicalizeCompleteJson(measured, REQUEST_JSON_LIMITS)
   655               )
   656             ) as DecisionInputMeasurement
   657           } catch {
   658             throw new ToolResultContextError(
   659               'ADMISSION_UNAVAILABLE',
   660               'Decision counter failed or returned an invalid measurement'
   661             )
   662           }
   663           const tokenBudget = context.tokenBudget ?? 4_000
   664           if (!Number.isSafeInteger(tokenBudget) || tokenBudget < 0) {
   665             throw new ToolResultContextError(
   666               'ADMISSION_UNAVAILABLE',
   667               'Invalid context token budget'
   668             )
   669           }
   670           if (
   671             measurement.inputUpperBoundTokens > tokenBudget ||
   672             measurement.inputUpperBoundTokens +
   673               COMPLETE_DECISION_OUTPUT_TOKENS >
   674               measurement.contextWindowTokens
   675           ) {
   676             overflow = new ToolResultContextError(
   677               'CONTEXT_OVERFLOW',
   678               'Complete decision exceeds context budget or model window'
   679             )
   680           } else {
   681             // This is only the step's baseline. Repairs change live balances;
   682             // reserve remains the final authority, never a local usage sum.
   683             const usage = context.budgetUsage
   684             const budget = context.budget
   685             if (
   686               ![
   687                 usage.inputTokens,
   688                 usage.outputTokens,
   689                 usage.costUsd,
   690                 usage.modelCalls,
   691                 usage.decisionRepairs,
   692                 budget.maxTokens,
   693                 budget.maxCostUsd,
   694                 budget.maxModelCalls
   695               ].every(finiteNonnegative)
   696             ) {
   697               throw new ToolResultContextError(
   698                 'ADMISSION_UNAVAILABLE',
   699                 'Invalid execution budget'
   700               )
   701             }
   702             if (
   703               measurement.inputUpperBoundTokens +
   704                 COMPLETE_DECISION_OUTPUT_TOKENS >
   705               budget.maxTokens - usage.inputTokens - usage.outputTokens
   706             ) {
   707               overflow = new ToolResultContextError(
   708                 'CONTEXT_OVERFLOW',
   709                 'Insufficient execution tokens',
   710                 'MAX_TOKENS'
   711               )
   712             } else if (
   713               measurement.costUpperBoundUsd >
   714               budget.maxCostUsd - usage.costUsd
   715             ) {
   716               overflow = new ToolResultContextError(
   717                 'CONTEXT_OVERFLOW',
   718                 'Insufficient execution cost',
   719                 'MAX_COST'
   720               )
   721             } else if (usage.modelCalls >= budget.maxModelCalls) {
   722               throw new ToolResultContextError(
   723                 'ADMISSION_UNAVAILABLE',
   724                 'No model call balance',
   725                 'MAX_MODEL_CALLS'
   726               )
   727             } else if (
   728               isRepair &&
   729               budget.maxDecisionRepairs !== undefined &&
   730               (!finiteNonnegative(budget.maxDecisionRepairs) ||
   731                 usage.decisionRepairs >= budget.maxDecisionRepairs)
   732             ) {
   733               throw new ToolResultContextError(
   734                 'ADMISSION_UNAVAILABLE',
   735                 'No decision repair balance',
   736                 'MODEL_FAILURE'
   737               )
   738             } else break
   739           }
   740         }
   741         let remove = -1
   742         for (let index = 0; index < selected.length; index += 1) {
   743           const candidate = selected[index]!.item
   744           if (
   745             optionalItem(candidate) &&
   746             (remove < 0 ||
   747               CONTEXT_PRIORITIES.indexOf(candidate.priority) >=
   748                 CONTEXT_PRIORITIES.indexOf(selected[remove]!.item.priority))
   749           )
   750             remove = index
   751         }
   752         if (remove < 0) throw overflow
   753         selected = selected.filter((_entry, index) => index !== remove)
   754       }
   755       ensureTime()
   756       const snapshot = request!
   757       const measured = measurement!
   758       const requestSha256 = completeDigest(snapshot)
   759       const reserved = await decisionAdmission.reserve({
   760         request: snapshot,
   761         measurement: measured,
   762         attempt
   763       })
   764       checkReceipt(
   765         reserved,
   766         context,
   767         snapshot,
   768         measured,
   769         requestSha256,
   770         'RESERVED'
   771       )
   772       if (decisionAdmissionIds.includes(reserved.admissionId)) {
   773         throw new ToolResultContextError(
   774           'STATE_CONFLICT',
   775           'Admission ID reused across attempts',
   776           'STATE_CONFLICT'
   777         )
   778       }
   779       let result: ModelResult | undefined
   780       try {
   781         ensureTime()
   782         result = await withinDecisionDeadline(
   783           () =>
   784             model.complete(
   785               Object.freeze({
   786                 ...snapshot,
   787                 ...(input.signal ? { signal: input.signal } : {})
   788               })
   789             ),
   790           remainingMs(),
   791           input.signal
   792         )
   793         ensureTime()
   794         if (
   795           !result ||
   796           typeof result.text !== 'string' ||
   797           result.provider !== measured.provider ||
   798           result.model !== measured.model ||
   799           !Number.isSafeInteger(result.inputTokens) ||
   800           result.inputTokens < 0 ||
   801           result.inputTokens > measured.inputUpperBoundTokens ||
   802           !Number.isSafeInteger(result.outputTokens) ||
   803           result.outputTokens < 0 ||
   804           result.outputTokens > COMPLETE_DECISION_OUTPUT_TOKENS ||
   805           !finiteNonnegative(result.costUsd) ||
   806           result.costUsd > measured.costUpperBoundUsd
   807         ) {
   808           this.decisionPairBreached = true
   809           throw new ToolResultContextError(
   810             'ADMISSION_BREACH',
   811             'Native model identity or usage violates qualified measurement'
   812           )
   813         }
   814         result = freezeComplete(
   815           JSON.parse(canonicalizeCompleteJson(result, REQUEST_JSON_LIMITS))
   816         ) as ModelResult
   817       } catch (error) {
   818         if (
   819           result &&
   820           Number.isSafeInteger(result.inputTokens) &&
   821           result.inputTokens >= 0 &&
   822           Number.isSafeInteger(result.outputTokens) &&
   823           result.outputTokens >= 0 &&
   824           finiteNonnegative(result.costUsd) &&
   825           error instanceof ToolResultContextError &&
   826           error.code === 'ADMISSION_BREACH'
   827         ) {
   828           // Let the native runtime retain finite observed overuse. An UNKNOWN
   829           // without this result would charge only the breached upper bound.
   830           await decisionAdmission.settle({
   831             kind: 'RESULT',
   832             admissionId: reserved.admissionId,
   833             requestSha256,
   834             result
   835           })
   836           throw error
   837         }
   838         await decisionAdmission.settle({
   839           kind: 'UNKNOWN',
   840           admissionId: reserved.admissionId,
   841           requestSha256,
   842           reason:
   843             error instanceof ToolResultContextError &&
   844             error.code === 'ADMISSION_BREACH'
   845               ? 'ADMISSION_BREACH'
   846               : 'DECISION_USAGE_UNKNOWN'
   847         })
   848         if (
   849           error instanceof ToolResultContextError &&
   850           error.code === 'ADMISSION_BREACH'
   851         )
   852           throw error
   853         throw new ToolResultContextError(
   854           'DECISION_USAGE_UNKNOWN',
   855           'Model call failed, was cancelled, returned invalid data or missed its deadline'
   856         )
   857       }
   858       const settled = await decisionAdmission.settle({
   859         kind: 'RESULT',
   860         admissionId: reserved.admissionId,
   861         requestSha256,
   862         result
   863       })
   864       checkReceipt(
   865         settled,
   866         context,
   867         snapshot,
   868         measured,
   869         requestSha256,
   870         'SETTLED',
   871         reserved
   872       )
   873       if (
   874         settled.actual?.modelCalls !== 1 ||
   875         settled.actual.inputTokens !== result.inputTokens ||
   876         settled.actual.outputTokens !== result.outputTokens ||
   877         settled.actual.costUsd !== result.costUsd ||
   878         (settled.resultSha256 !== undefined &&
   879           settled.resultSha256 !== completeDigest(result))
   880       ) {
   881         throw new ToolResultContextError(
   882           'STATE_CONFLICT',
   883           'Settlement does not bind the native result',
   884           'STATE_CONFLICT'
   885         )
   886       }
   887       decisionAdmissionIds.push(settled.admissionId)
   888       ensureTime()
   889       let parsed: unknown
   890       try {
   891         parsed = parseDecisionText(result.text)
   892       } catch (error) {
   893         if (repairs >= this.maxDecisionRepairs) throw error
   894         repairs += 1
   895         errors = [
   896           error instanceof Error ? error.message : 'unparseable decision'
   897         ]
   898         continue
   899       }
   900       const validation = validateLoopDecision(parsed)
   901       errors = validation.valid
   902         ? disallowedTypeErrors(sanitizeLoopDecision(parsed), context)
   903         : validation.errors
   904       if (errors.length === 0)
   905         return { decision: sanitizeLoopDecision(parsed), decisionAdmissionIds }
   906       if (repairs >= this.maxDecisionRepairs) {
   907         throw new OrchestratorDecisionError(
   908           'decision_invalid',
   909           `Orchestrator decision stayed invalid after ${repairs} repairs: ${errors.join('; ')}`
   910         )
   911       }
   912       repairs += 1
   913     }
   914   }
   915 }
