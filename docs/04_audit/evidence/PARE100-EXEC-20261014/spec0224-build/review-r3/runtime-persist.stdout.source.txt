  1789   private makeObservation(
  1790     run: LoopRun,
  1791     input: {
  1792       stepId: string
  1793       stepNumber: number
  1794       type: Observation['type']
  1795       source: Observation['source']
  1796       trust: Observation['trust']
  1797       payload: unknown
  1798       summary: string
  1799       provenance: Observation['provenance']
  1800     }
  1801   ): Observation {
  1802     let payload = input.payload
  1803     if (this.options.toolResultContext && input.type === 'TOOL_RESULT') {
  1804       const value = payload as {
  1805         schemaVersion?: number
  1806         status?: string
  1807         output?: unknown
  1808         error?: unknown
  1809       }
  1810       const tool = this.options.tools.resolve(
  1811         input.provenance.sourceId,
  1812         input.provenance.sourceVersion
  1813       )
  1814       if (!tool)
  1815         throw new ToolResultContextError(
  1816           'STATE_CONFLICT',
  1817           'Tool observation has no native registry binding.',
  1818           'STATE_CONFLICT'
  1819         )
  1820       if (value.schemaVersion !== 1) {
  1821         payload =
  1822           value.status === 'SUCCEEDED'
  1823             ? completeToolPayload(value.output, tool.sideEffect)
  1824             : {
  1825                 schemaVersion: 1,
  1826                 status: value.status === 'REJECTED' ? 'REJECTED' : 'FAILED',
  1827                 sideEffect: tool.sideEffect,
  1828                 completion: 'NOT_APPLICABLE',
  1829                 resultAvailability: 'UNAVAILABLE',
  1830                 error:
  1831                   typeof value.error === 'string'
  1832                     ? value.error.slice(0, 512)
  1833                     : 'tool_failed'
  1834               }
  1835       }
  1836     }
  1837     const observation: Observation = {
  1838       observationId: `obs_${randomUUID()}`,
  1839       executionId: run.executionId,
  1840       stepId: input.stepId,
  1841       stepNumber: input.stepNumber,
  1842       type: input.type,
  1843       source: input.source,
  1844       trust: input.trust,
  1845       payload:
  1846         this.options.toolResultContext && input.type === 'TOOL_RESULT'
  1847           ? payload
  1848           : boundedPayload(payload, this.maxObservationPayloadChars),
  1849       summary: input.summary.slice(0, 500),
  1850       provenance: input.provenance,
  1851       timestamp: this.clock().toISOString()
  1852     }
  1853     if (this.options.toolResultContext && input.type === 'TOOL_RESULT')
  1854       assertCompleteToolObservation(observation)
  1855     return observation
  1856   }

  1862     run.observations = this.options.toolResultContext

  1865     run.state = { ...run.state, observations: run.observations }

  1871   private async recordStep(
  1872     run: LoopRun,
  1873     input: {
  1874       stepNumber: number
  1875       stepType: ExecutionStep['stepType']
  1876       status: ExecutionStep['status']
  1877       sideEffecting: boolean
  1878       decisionType?: ExecutionStep['decisionType']
  1879       reasonCode?: ExecutionStep['reasonCode']
  1880       errorCode?: string
  1881       observationRefs?: readonly string[]
  1882     }
  1883   ): Promise<void> {
  1884     const stepId = `step_${run.executionId}_${input.stepNumber}_${input.stepType.toLowerCase()}`
  1885     const terminal = input.status === 'SUCCEEDED' || input.status === 'FAILED'
  1886     const decisionType =
  1887       input.decisionType ?? run.lastDecision?.decisionType ?? undefined
  1888     await this.options.stepStore.recordStep({
  1889       stepId,
  1890       executionId: run.executionId,
  1891       tenantId: run.input.tenantId,
  1892       stepNumber: input.stepNumber,
  1893       stepType: input.stepType,
  1894       status: input.status,
  1895       attempt: 1,
  1896       sideEffecting: input.sideEffecting,
  1897       startedAt: this.clock().toISOString(),
  1898       ...(terminal ? { completedAt: this.clock().toISOString() } : {}),
  1899       ...(decisionType ? { decisionType } : {}),
  1900       ...(input.reasonCode ? { reasonCode: input.reasonCode } : {}),
  1901       observationRefs: input.observationRefs ?? [],
  1902       ...(input.errorCode ? { errorCode: input.errorCode } : {})
  1903     })
  1904     if (
  1905       (terminal || input.status === 'WAITING') &&
  1906       !run.countedSteps.has(input.stepNumber)
  1907     ) {
  1908       run.countedSteps.add(input.stepNumber)
  1909       run.usage.steps += 1
  1910     }
  1911     await this.safeAudit(
  1912       run,
  1913       input.status === 'RUNNING' ? 'step.started' : 'step.completed',
  1914       { result: `${input.stepType}:${input.status}`, tool: null }
  1915     )
  1916   }

  1918   private async persistCheckpoint(run: LoopRun): Promise<void> {
  1919     const checkpoint = sealCheckpoint({
  1920       executionId: run.executionId,
  1921       tenantId: run.input.tenantId,
  1922       checkpointVersion: this.options.toolResultContext ? 2 : 1,
  1923       runtimeProfile: 'iterative',
  1924       runtimeVersion: RUNTIME_V2_VERSION,
  1925       orchestratorVersion: this.orchestratorVersion,
  1926       stepNumber: run.state.stepNumber,
  1927       state: {
  1928         ...run.state,
  1929         observations: this.options.toolResultContext
  1930           ? retainCompleteObservations(run.observations)
  1931           : run.observations.slice(-MAX_CHECKPOINT_OBSERVATIONS)
  1932       },
  1933       budgetUsage: { ...run.usage }
  1934     })
  1935     if (this.options.toolResultContext) {
  1936       const result = await this.options.stepStore.compareAndSwapCheckpoint!({
  1937         tenantId: run.input.tenantId,
  1938         executionId: run.executionId,
  1939         expected: run.checkpointToken ?? null,
  1940         next: checkpoint
  1941       })
  1942       if (
  1943         result.status !== 'APPLIED' ||
  1944         result.token.checkpointId !== checkpoint.checkpointId ||
  1945         result.token.digest !== checkpoint.digest
  1946       ) {
  1947         run.checkpointConflict = true
  1948         throw new ToolResultContextError(
  1949           'STATE_CONFLICT',
  1950           'Checkpoint compare-and-swap did not commit this writer.',
  1951           'STATE_CONFLICT'
  1952         )
  1953       }
  1954       run.checkpointToken = result.token
  1955     } else {
  1956       await this.options.stepStore.saveCheckpoint(checkpoint)
  1957     }
  1958   }

  1960   private preflightCompleteTool(
  1961     run: LoopRun,
  1962     decision: LoopDecision,
  1963     stepNumber: number
  1964   ): void {
  1965     if (!this.options.toolResultContext) return
  1966     const tool = this.options.tools.resolve(
  1967       decision.toolId ?? '',
  1968       decision.toolVersion
  1969     )
  1970     if (!tool)
  1971       throw new ToolResultContextError(
  1972         'STATE_CONFLICT',
  1973         'Missing native tool.',
  1974         'STATE_CONFLICT'
  1975       )
  1976     if (
  1977       run.observations.filter((value) => value.type === 'TOOL_RESULT').length >=
  1978       MAX_CHECKPOINT_OBSERVATIONS
  1979     ) {
  1980       throw new ToolResultContextError(
  1981         'CONTEXT_OVERFLOW',
  1982         'No retention slot remains before tool dispatch.'
  1983       )
  1984     }
  1985     const placeholder = this.makeObservation(run, {
  1986       stepId: `step_${run.executionId}_${stepNumber}_tool`,
  1987       stepNumber,
  1988       type: 'TOOL_RESULT',
  1989       source: 'TOOL',
  1990       trust: 'UNTRUSTED',
  1991       payload: { status: 'SUCCEEDED', output: 'x'.repeat(32_766) },
  1992       summary: 'x'.repeat(500),
  1993       provenance: {
  1994         sourceId: tool.id,
  1995         sourceVersion: tool.version,
  1996         operationKey: this.operationKey(run, stepNumber, tool.id),
  1997         effectRef: this.operationKey(run, stepNumber, tool.id)
  1998       }
  1999     })
  2000     const prospective = {
  2001       executionId: run.executionId,
  2002       tenantId: run.input.tenantId,
  2003       checkpointVersion: 2,
  2004       runtimeProfile: 'iterative',
  2005       runtimeVersion: RUNTIME_V2_VERSION,
  2006       orchestratorVersion: this.orchestratorVersion,
  2007       stepNumber: run.state.stepNumber,
  2008       state: {
  2009         ...run.state,
  2010         pendingDecision: decision,
  2011         pendingStepNumber: stepNumber,
  2012         observations: retainCompleteObservations([
  2013           ...run.observations,
  2014           placeholder
  2015         ])
  2016       },
  2017       budgetUsage: { ...run.usage, toolCalls: run.usage.toolCalls + 1 }
  2018     }
  2019     try {
  2020       // Optional internal fields are absent on the wire. Real saves use the
  2021       // stricter schema-owned checkpoint codec, not this size-only projection.
  2022       canonicalizeCompleteJson(JSON.parse(JSON.stringify(prospective)), {
  2023         maxBytes: COMPLETE_CHECKPOINT_MAX_BYTES - 65_536,
  2024         maxDepth: 64,
  2025         maxNodes: 1_000_000
  2026       })
  2027     } catch {
  2028       throw new ToolResultContextError(
  2029         'CONTEXT_OVERFLOW',
  2030         'Prospective checkpoint cannot reserve full tool result and step overhead.'
  2031       )
  2032     }
  2033   }

  2074   /**
  2075    * A terminal stop never leaves its pending tool step open (AUD-0600 F02).
  2076    * A WAITING step never reached its effect (approval wait, or parked by an
  2077    * operator pause), so it closes with the stop's own code. A step still
  2078    * RUNNING here may have started its effect before a crash or a failure
  2079    * after dispatch, so it is closed as `unknown_effect` for reconciliation
  2080    * instead of being declared not executed.
  2081    */
  2082   private async closePendingStep(
  2083     run: LoopRun,
  2084     stopReason: StopReason
  2085   ): Promise<void> {
  2086     const stepNumber = run.state.pendingStepNumber
  2087     if (!stepNumber) return
  2088     const steps = await this.options.stepStore.listSteps(
  2089       run.input.tenantId,
  2090       run.executionId
  2091     )
  2092     const step = steps.find(
  2093       (candidate) =>
  2094         candidate.stepNumber === stepNumber && candidate.stepType === 'TOOL'
  2095     )
  2096     if (!step || (step.status !== 'WAITING' && step.status !== 'RUNNING')) {
  2097       return
