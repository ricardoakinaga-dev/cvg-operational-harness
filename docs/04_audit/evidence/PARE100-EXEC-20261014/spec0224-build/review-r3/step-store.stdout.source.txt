    38 export function checkpointDigest(checkpoint: UnsignedCheckpoint): string {
    39   if (checkpoint.checkpointVersion === 2)
    40     return completeCheckpointDigest(checkpoint)
    41   return createHash('sha256')
    42     .update(JSON.stringify(normalizeForHash(checkpoint)))
    43     .digest('hex')
    44 }

    46 function normalizeForHash(value: unknown): unknown {
    47   if (Array.isArray(value)) return value.map(normalizeForHash)
    48   if (value && typeof value === 'object') {
    49     const record = value as Record<string, unknown>
    50     return Object.fromEntries(
    51       Object.keys(record)
    52         .filter((key) => record[key] !== undefined)
    53         .sort()
    54         .map((key) => [key, normalizeForHash(record[key])])
    55     )
    56   }
    57   return value
    58 }

    71 export function validateCheckpointIntegrity(
    72   checkpoint: ExecutionCheckpoint,
    73   expected: { readonly tenantId: string; readonly executionId: string }
    74 ): void {
    75   if (
    76     checkpoint.checkpointVersion !== CHECKPOINT_VERSION &&
    77     checkpoint.checkpointVersion !== 2
    78   ) {
    79     throw new CheckpointError(
    80       'checkpoint_version',
    81       `Checkpoint version ${checkpoint.checkpointVersion} is not supported`
    82     )
    83   }
    84   if (
    85     checkpoint.checkpointVersion === 2 &&
    86     (typeof checkpoint.checkpointId !== 'string' ||
    87       !checkpoint.checkpointId ||
    88       checkpoint.checkpointId.includes('\u0000') ||
    89       typeof checkpoint.createdAt !== 'string' ||
    90       !Number.isFinite(Date.parse(checkpoint.createdAt)))
    91   ) {
    92     throw new CheckpointError(
    93       'checkpoint_integrity',
    94       'Invalid V2 checkpoint metadata'
    95     )
    96   }
    97   if (
    98     checkpoint.tenantId !== expected.tenantId ||
    99     checkpoint.executionId !== expected.executionId
   100   ) {
   101     throw new CheckpointError(
   102       'checkpoint_binding',
   103       'Checkpoint does not belong to the requested execution/tenant'
   104     )
   105   }
   106   const {
   107     checkpointId: _id,
   108     createdAt: _createdAt,
   109     digest,
   110     ...unsigned
   111   } = checkpoint
   112   void _id
   113   void _createdAt
   114   if (checkpointDigest(unsigned) !== digest) {
   115     throw new CheckpointError(
   116       'checkpoint_integrity',
   117       'Checkpoint digest does not match its content'
   118     )
   119   }
   120 }

   130 /** Process-local step/checkpoint store used by controlled proofs. */
   131 export class InMemoryExecutionStepStore implements ExecutionStepStore {
   132   private readonly steps = new Map<string, Map<number, ExecutionStep>>()
   133   private readonly checkpoints = new Map<string, ExecutionCheckpoint>()
   134 
   135   public async recordStep(step: ExecutionStep): Promise<void> {
   136     const identity = key(step.tenantId, step.executionId)
   137     const steps = this.steps.get(identity) ?? new Map<number, ExecutionStep>()
   138     const existing = steps.get(step.stepNumber)
   139     if (existing) {
   140       if (existing.stepId !== step.stepId) {
   141         throw new CheckpointError(
   142           'step_conflict',
   143           'A different step already occupies this step number'
   144         )
   145       }
   146       if (
   147         existing.status !== step.status &&
   148         !isStepTransitionAllowed(existing.status, step.status)
   149       ) {
   150         throw new CheckpointError(
   151           'step_transition',
   152           `Step transition ${existing.status} -> ${step.status} is not allowed`
   153         )
   154       }
   155     } else {
   156       if (!Number.isSafeInteger(step.stepNumber) || step.stepNumber < 1) {
   157         throw new CheckpointError(
   158           'step_conflict',
   159           'A step number must be a positive integer'
   160         )
   161       }
   162       const blocking = [...steps.values()].filter(
   163         (candidate) =>
   164           candidate.stepNumber < step.stepNumber &&
   165           !['SUCCEEDED', 'FAILED', 'SKIPPED', 'WAITING'].includes(
   166             candidate.status
   167           )
   168       )
   169       if (blocking.length > 0) {
   170         throw new CheckpointError(
   171           'step_transition',
   172           `Step ${step.stepNumber} cannot start before step ${blocking[0]?.stepNumber} settles`
   173         )
   174       }
   175     }
   176     steps.set(step.stepNumber, copy(step))
   177     this.steps.set(identity, steps)
   178   }
   179 
   180   public async listSteps(
   181     tenantId: string,
   182     executionId: string
   183   ): Promise<readonly ExecutionStep[]> {
   184     const steps = this.steps.get(key(tenantId, executionId))
   185     if (!steps) return []
   186     return [...steps.values()]
   187       .sort((left, right) => left.stepNumber - right.stepNumber)
   188       .map((step) => copy(step))
   189   }
   190 
   191   public async saveCheckpoint(checkpoint: ExecutionCheckpoint): Promise<void> {
   192     if (
   193       checkpoint.checkpointVersion === 2 ||
   194       this.checkpoints.get(key(checkpoint.tenantId, checkpoint.executionId))
   195         ?.checkpointVersion === 2
   196     ) {
   197       throw new CheckpointError(
   198         'checkpoint_version',
   199         'V2 checkpoints require compareAndSwapCheckpoint'
   200       )
   201     }
   202     validateCheckpointIntegrity(checkpoint, {
   203       tenantId: checkpoint.tenantId,
   204       executionId: checkpoint.executionId
   205     })
   206     this.checkpoints.set(
   207       key(checkpoint.tenantId, checkpoint.executionId),
   208       copy(checkpoint)
   209     )
   210   }
   211 
   212   public async compareAndSwapCheckpoint(
   213     input: CheckpointCompareAndSwapInput
   214   ): Promise<CheckpointCompareAndSwapResult> {
   215     // The entire comparison/validation/replacement is synchronous: no await gap.
   216     validateCheckpointIntegrity(input.next, input)
   217     const identity = key(input.tenantId, input.executionId)
   218     const current = this.checkpoints.get(identity) ?? null
   219     if (current) validateCheckpointIntegrity(current, input)
   220     if (
   221       input.expected &&
   222       input.expected.checkpointId === input.next.checkpointId
   223     ) {
   224       throw new CheckpointError(
   225         'checkpoint_integrity',
   226         'CAS requires a fresh checkpoint ID'
   227       )
   228     }
   229     const token = current
   230       ? { checkpointId: current.checkpointId, digest: current.digest }
   231       : undefined
   232     if (
   233       input.expected === null
   234         ? current !== null
   235         : !current ||
   236           input.expected.checkpointId !== current.checkpointId ||
   237           input.expected.digest !== current.digest
   238     ) {
   239       return { status: 'CONFLICT', ...(token ? { currentToken: token } : {}) }
   240     }
   241     validateCompleteCheckpointTransition(current, input.next)
   242     const next = copy(input.next)
   243     if (next.checkpointVersion === 2) {
   244       // Exercise the same physical codec as persistent storage without leaking its wrapper to callers.
   245       const state = decodeCompleteCheckpointState(
   246         encodeCompleteCheckpointState(next.state)
   247       )
   248       this.checkpoints.set(identity, { ...next, state })
   249     } else this.checkpoints.set(identity, next)
   250     return {
   251       status: 'APPLIED',
   252       token: { checkpointId: next.checkpointId, digest: next.digest }
   253     }
   254   }
   255 
   256   public async loadCheckpoint(
   257     tenantId: string,
   258     executionId: string
   259   ): Promise<ExecutionCheckpoint | null> {
   260     const checkpoint = this.checkpoints.get(key(tenantId, executionId))
   261     if (!checkpoint) return null
   262     validateCheckpointIntegrity(checkpoint, { tenantId, executionId })
   263     return copy(checkpoint)
   264   }
   265 }

   267 export function createInMemoryExecutionStepStore(): InMemoryExecutionStepStore {
   268   return new InMemoryExecutionStepStore()
