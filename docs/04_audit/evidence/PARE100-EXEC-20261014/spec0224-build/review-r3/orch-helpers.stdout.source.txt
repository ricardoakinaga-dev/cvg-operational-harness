    61 function parseDecisionText(text: string): unknown {
    62   const trimmed = text.trim()
    63   const withoutFence = trimmed
    64     .replace(/^```(?:json)?/i, '')
    65     .replace(/```$/, '')
    66     .trim()
    67   const start = withoutFence.indexOf('{')
    68   const end = withoutFence.lastIndexOf('}')
    69   if (start === -1 || end === -1 || end <= start) {
    70     throw new OrchestratorDecisionError(
    71       'decision_unparseable',
    72       'Orchestrator output did not contain a JSON object'
    73     )
    74   }
    75   try {
    76     return JSON.parse(withoutFence.slice(start, end + 1))
    77   } catch {
    78     throw new OrchestratorDecisionError(
    79       'decision_unparseable',
    80       'Orchestrator output was not valid JSON'
    81     )
    82   }
    83 }

   153 function freezeComplete<T>(value: T): T {
   154   if (value !== null && typeof value === 'object') {
   155     for (const child of Object.values(value)) freezeComplete(child)
   156     Object.freeze(value)
   157   }
   158   return value

   216 function completeItems(context: StepContext) {
   217   const required = context.toolResultContext?.requiredObservationIds ?? []
   218   if (
   219     context.toolResultContext?.version !== 1 ||
   220     new Set(required).size !== required.length
   221   ) {
   222     throw new ToolResultContextError(
   223       'STATE_CONFLICT',
   224       'Invalid complete context version or required IDs'
   225     )
   226   }
   227   const seen = new Set<string>()
   228   const ids = new Set<string>()
   229   const selected = context.contextItems.map((item) => {
   230     if (ids.has(item.contextId)) {
   231       throw new ToolResultContextError(
   232         'STATE_CONFLICT',
   233         'Duplicate selected context ID'
   234       )
   235     }
   236     ids.add(item.contextId)
   237     const metadata = {
   238       contextId: item.contextId,
   239       priority: item.priority,
   240       trust: item.trust,
   241       source: item.source,
   242       ...(item.evidenceRefs ? { evidenceRefs: item.evidenceRefs } : {})
   243     }
   244     if (item.priority !== 'FRESH_TOOL_RESULTS') {
   245       return { item, frame: { ...metadata, value: item.content } }
   246     }
   247     let observation
   248     try {
   249       observation = JSON.parse(item.content)
   250       if (
   251         canonicalizeCompleteJson(observation, REQUEST_JSON_LIMITS) !==
   252         item.content
   253       )
   254         throw new Error('noncanonical')
   255     } catch {
   256       throw new ToolResultContextError(
   257         'STATE_CONFLICT',
   258         'Selected tool observation is not canonical complete JSON'
   259       )
   260     }
   261     if (
   262       !observation ||
   263       observation.type !== 'TOOL_RESULT' ||
   264       observation.source !== 'TOOL' ||
   265       observation.executionId !== context.executionId ||
   266       observation.trust !== 'UNTRUSTED' ||
   267       item.trust !== 'UNTRUSTED' ||
   268       !required.includes(observation.observationId) ||
   269       seen.has(observation.observationId) ||
   270       !observation.provenance ||
   271       observation.provenance.sourceId !== item.source
   272     ) {
   273       throw new ToolResultContextError(
   274         'STATE_CONFLICT',
   275         'Selected tool observation bindings are inconsistent'
   276       )
   277     }
   278     seen.add(observation.observationId)
   279     return {
   280       item,
   281       frame: {
   282         ...metadata,
   283         observationId: observation.observationId,
   284         provenance: observation.provenance,
   285         value: observation
   286       }
   287     }
   288   })
   289   if (seen.size !== required.length) {
   290     throw new ToolResultContextError(
   291       'STATE_CONFLICT',
   292       'Required complete tool observation is missing'
   293     )
   294   }
   295   return selected
   296 }

   304 /** Race only pure counting / the model; durable reserve and settle stay runtime-owned. */
   305 async function withinDecisionDeadline<T>(
   306   operation: () => T | Promise<T>,
   307   remainingMs: number,
   308   signal?: AbortSignal
   309 ): Promise<T> {
   310   if (signal?.aborted || remainingMs <= 0) {
   311     throw new ToolResultContextError(
   312       'ADMISSION_UNAVAILABLE',
   313       'Decision deadline or cancellation',
   314       'MAX_DURATION'
   315     )
   316   }
   317   let timer: ReturnType<typeof setTimeout> | undefined
   318   let cancel: (() => void) | undefined
   319   const deadline = Date.now() + remainingMs
   320   try {
   321     const stopped = new Promise<never>((_resolve, reject) => {
   322       cancel = () =>
   323         reject(
   324           new ToolResultContextError(
   325             'ADMISSION_UNAVAILABLE',
   326             'Decision cancelled or late',
   327             'MAX_DURATION'
   328           )
   329         )
   330       signal?.addEventListener('abort', cancel, { once: true })
   331       timer = setTimeout(cancel, Math.min(remainingMs, 2_147_483_647))
   332     })
   333     return await Promise.race([
   334       Promise.resolve().then(() => {
   335         if (signal?.aborted || Date.now() >= deadline) {
   336           throw new ToolResultContextError(
   337             'ADMISSION_UNAVAILABLE',
   338             'Decision cancelled before starting',
   339             'MAX_DURATION'
   340           )
   341         }
   342         return operation()
   343       }),
   344       stopped
   345     ])
   346   } finally {
   347     if (timer !== undefined) clearTimeout(timer)
   348     if (cancel) signal?.removeEventListener('abort', cancel)
   349   }
   350 }

   352 function checkReceipt(
   353   receipt: DecisionAdmissionRecord,
   354   context: StepContext,
   355   request: DecisionModelRequestSnapshot,
   356   measurement: DecisionInputMeasurement,
   357   requestSha256: string,
   358   status: 'RESERVED' | 'SETTLED',
   359   previous?: DecisionAdmissionRecord
   360 ): void {
   361   if (
   362     !receipt ||
   363     typeof receipt.admissionId !== 'string' ||
   364     !receipt.admissionId ||
   365     receipt.status !== status ||
   366     receipt.tenantId !== context.tenantId ||
   367     receipt.executionId !== context.executionId ||
   368     receipt.stepNumber !== context.stepNumber ||
   369     receipt.purpose !== request.purpose ||
   370     receipt.requestSha256 !== requestSha256 ||
   371     !Number.isSafeInteger(receipt.attempt) ||
   372     receipt.attempt < 0 ||
   373     completeDigest(receipt.measurement) !== completeDigest(measurement) ||
   374     receipt.reserved?.modelCalls !== 1 ||
   375     receipt.reserved.inputTokens !== measurement.inputUpperBoundTokens ||
   376     receipt.reserved.outputTokens !== COMPLETE_DECISION_OUTPUT_TOKENS ||
   377     receipt.reserved.costUsd !== measurement.costUpperBoundUsd ||
