   100 function text(value: unknown, label: string, max = 200): string {
   101   if (typeof value !== 'string' || value.trim().length === 0) {
   102     throw new CheckpointError('checkpoint_binding', `${label} is required`)
   103   }
   104   if (value.length > max) {
   105     throw new CheckpointError('checkpoint_binding', `${label} is too long`)
   106   }
   107   return value.trim()
   108 }

   110 function asIso(value: Date | string | null): string | null {
   111   if (value === null) return null
   112   return value instanceof Date
   113     ? value.toISOString()
   114     : new Date(value).toISOString()
   115 }

   117 function jsonValue(value: unknown): unknown {
   118   if (typeof value !== 'string') return value
   119   try {
   120     return JSON.parse(value) as unknown
   121   } catch {
   122     throw new CheckpointError(
   123       'checkpoint_integrity',
   124       'Persisted step JSON is invalid'
   125     )
   126   }
   127 }
   128 
   129 function jsonText(value: unknown, label: string): string {
   130   try {
   131     const serialized = JSON.stringify(value)
   132     if (serialized === undefined) throw new Error('undefined')
   133     return serialized
   134   } catch {
   135     throw new CheckpointError(
   136       'checkpoint_integrity',
   137       `${label} cannot be serialized`
   138     )
   139   }

   142 function mapStep(row: StepRow): ExecutionStep {
   143   const refs = jsonValue(row.observation_refs)
   144   return {
   145     stepId: row.step_id,
   146     executionId: row.execution_id,
   147     tenantId: row.tenant_id,
   148     stepNumber: row.step_number,
   149     stepType: row.step_type,
   150     status: row.status,
   151     attempt: row.attempt,
   152     sideEffecting: row.side_effecting,
   153     startedAt: asIso(row.started_at) as string,
   154     ...(row.completed_at
   155       ? { completedAt: asIso(row.completed_at) as string }
   156       : {}),
   157     ...(row.decision_type ? { decisionType: row.decision_type } : {}),
   158     ...(row.reason_code
   159       ? {
   160           reasonCode: row.reason_code as NonNullable<
   161             ExecutionStep['reasonCode']
   162           >
   163         }
   164       : {}),
   165     observationRefs: Array.isArray(refs) ? (refs as string[]) : [],
   166     ...(row.error_code ? { errorCode: row.error_code } : {})
   167   }
   168 }

   170 function mapCheckpoint(row: CheckpointRow): ExecutionCheckpoint {
   171   return {
   172     checkpointId: row.checkpoint_id,
   173     executionId: row.execution_id,
   174     tenantId: row.tenant_id,
   175     checkpointVersion: row.checkpoint_version,
   176     runtimeProfile:

   178     runtimeVersion: row.runtime_version,
   179     orchestratorVersion: row.orchestrator_version,
   180     stepNumber: row.step_number,
   181     state:
   182       row.checkpoint_version === 2
   183         ? decodeCompleteCheckpointState(row.state)

   185     budgetUsage: jsonValue(

   188     createdAt: asIso(row.created_at) as string,
