   190       throw new ToolResultContextError(

   198         run.usage.outputTokens +

   204         'CONTEXT_OVERFLOW',

   206         'MAX_TOKENS'
   207       )
   208     }

   214         'CONTEXT_OVERFLOW',
   215         'Insufficient decision cost reserve.',
   216         'MAX_COST'
   217       )
   218     }
   219     if (
   220       purpose === 'REPAIR' &&
   221       run.usage.decisionRepairs + 1 > (run.input.budget.maxDecisionRepairs ?? 1)
   222     ) {
   223       throw new ToolResultContextError(
   224         'ADMISSION_UNAVAILABLE',
   225         'Decision repair limit exhausted.',

   242         inputTokens,

   245       },
   246       status: 'RESERVED'
   247     }

   249     run.usage.inputTokens += record.reserved.inputTokens
   250     run.usage.outputTokens += record.reserved.outputTokens
   251     run.usage.costUsd += record.reserved.costUsd

   253     run.state.toolResultContext = {

   255       pendingDecisionAdmission: record.admissionId,
   256       decisionAdmissions: [...state.decisionAdmissions, record]
   257     }
   258     await this.options.persist()

   260     // The receipt is returned only after the checkpoint commit and audit.
   261     return structuredClone(record)

   311   private async settleNow(
   312     input: DecisionAdmissionSettlement
   313   ): Promise<DecisionAdmissionRecord> {
   314     const { run } = this.options
   315     const record = this.state().decisionAdmissions.find(
   316       (value) => value.admissionId === input.admissionId
   317     )
   318     if (
   319       !record ||
   320       record.stepNumber !== this.options.stepNumber ||
   321       record.tenantId !== run.input.tenantId ||
   322       record.executionId !== run.executionId ||
   323       record.requestSha256 !== input.requestSha256
   324     )
   325       throw this.conflict()
   326     if (record.status === 'SETTLED') {
   327       if (
   328         input.kind !== 'RESULT' ||
   329         digest(input.result) !== record.resultSha256
   330       )
   331         throw this.conflict()
   332       return structuredClone(record)
   333     }
   334     if (
   335       record.status !== 'RESERVED' ||
   336       this.state().pendingDecisionAdmission !== record.admissionId
   337     )
   338       throw this.conflict()
   339     if (input.kind === 'UNKNOWN') return this.unknown(record, input.reason)
   340     if (
   341       this.closed ||
   342       this.options.signal.aborted ||
   343       this.options.remainingMs() <= 0
   344     ) {
   345       await this.unknown(record, 'decision_closed')
   346       throw new ToolResultContextError(
   347         'DECISION_USAGE_UNKNOWN',
   348         'A late decision result cannot be accepted.'
   349       )
   350     }
   351     const actual = finiteUsage(input.result)
   352     if (
   353       !actual ||
   354       input.result.provider !== record.measurement.provider ||
   355       input.result.model !== record.measurement.model ||
   356       actual.inputTokens > record.reserved.inputTokens ||
   357       actual.outputTokens > record.reserved.outputTokens ||
   358       actual.costUsd > record.reserved.costUsd
   359     ) {
   360       await this.unknown(record, 'ADMISSION_BREACH', actual ?? undefined)
   361       await this.options.audit(
   362         'decision.admission.breach',
   363         record.requestSha256
   364       )
   365       throw new ToolResultContextError(
   366         'ADMISSION_BREACH',
   367         'The model violated its qualified decision bounds.'
   368       )
   369     }
   370     const resultSha256 = digest(input.result)
   371     const usage = run.usage
   372     const previous = {
   373       inputTokens: usage.inputTokens,
   374       outputTokens: usage.outputTokens,
   375       costUsd: usage.costUsd
   376     }
   377     usage.inputTokens += actual.inputTokens - record.reserved.inputTokens
   378     usage.outputTokens += actual.outputTokens - record.reserved.outputTokens
   379     usage.costUsd += actual.costUsd - record.reserved.costUsd
   380     const settled: DecisionAdmissionRecord = {
   381       ...record,
   382       status: 'SETTLED',
   383       actual,
   384       resultSha256
   385     }
   386     try {
   387       await this.replace(settled)
   388     } catch (error) {
   389       // No acknowledged settlement: retain the pre-call conservative charge.
   390       Object.assign(usage, previous)
   391       throw error
   392     }
   393     await this.options.audit('decision.admission.settled', record.requestSha256)
   394     return structuredClone(settled)
   395   }
