const ts=require('../../worktree/node_modules/typescript');
const fs=require('node:fs');
const claim=JSON.parse(fs.readFileSync(__dirname+'/claim.json','utf8'));
const config=ts.readConfigFile('tsconfig.base.json',ts.sys.readFile);
const parsed=ts.parseJsonConfigFileContent(config.config,ts.sys,process.cwd());
const program=ts.createProgram(claim.owned,{...parsed.options,noEmit:true,composite:false,incremental:false,declaration:false,sourceMap:false});
const diagnostics=claim.owned.flatMap(path=>{const file=program.getSourceFile(path);return file?[...program.getSyntacticDiagnostics(file),...program.getSemanticDiagnostics(file)]:[]});
console.log(ts.formatDiagnosticsWithColorAndContext(diagnostics,{getCurrentDirectory:ts.sys.getCurrentDirectory,getCanonicalFileName:x=>x,getNewLine:()=>"\n"}));
console.log('Scoped type diagnostics: '+diagnostics.length+' across '+claim.owned.length+' owned files; no emit, no global diagnostics requested.');
process.exitCode=diagnostics.length?1:0;
