import subprocess,sys,json,time
from pathlib import Path
p=Path(__file__).parent
label=sys.argv[1]; cmd=sys.argv[2:]
r=subprocess.run(cmd,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
meta={"command":cmd,"exitCode":r.returncode,"emptyStreamsOmitted":[]}
for name,data in [("stdout",r.stdout),("stderr",r.stderr)]:
 if data: (p/(label+"."+name+".log")).write_bytes(data)
 else: meta["emptyStreamsOmitted"].append(name)
(p/(label+".json")).write_text(json.dumps(meta,indent=2))
print(json.dumps(meta)); print(r.stdout.decode(errors="replace")); print(r.stderr.decode(errors="replace"))
sys.exit(r.returncode)
