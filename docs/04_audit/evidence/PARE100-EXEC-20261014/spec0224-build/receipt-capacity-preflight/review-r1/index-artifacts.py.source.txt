from pathlib import Path
import json,hashlib
p=Path(__file__).parent
entries=[];empty=[];derived=[];caches=[]
for f in sorted(p.rglob('*')):
 if not f.is_file() or f.name=='artifact-manifest.json':continue
 rel=f.relative_to(p).as_posix();b=f.read_bytes()
 if not b:role='empty';empty.append(rel)
 elif rel.startswith(('cache/','tmp/')):role='cache';caches.append(rel)
 elif rel.startswith('native.') or rel=='finalize.py':role='derived';derived.append(rel)
 else:role='primary'
 entries.append(dict(path=rel,role=role,bytes=len(b),sha256=hashlib.sha256(b).hexdigest()))
assert all(e['bytes']>0 for e in entries if e['role']=='primary')
m=dict(version=1,root=str(p),self='artifact-manifest.json is the index and is excluded from its own content hash.',primary=[e for e in entries if e['role']=='primary'],empty=[e for e in entries if e['role']=='empty'],derived=[e for e in entries if e['role']=='derived'],caches=[e for e in entries if e['role']=='cache'],directories=[f.relative_to(p).as_posix() for f in sorted(p.rglob('*')) if f.is_dir()])
(p/'artifact-manifest.json').write_text(json.dumps(m,indent=2)+'\n')
print(json.dumps(dict(primary=len(m['primary']),empty=len(m['empty']),derived=len(m['derived']),cacheFiles=len(m['caches']),indexedFiles=len(entries))))
