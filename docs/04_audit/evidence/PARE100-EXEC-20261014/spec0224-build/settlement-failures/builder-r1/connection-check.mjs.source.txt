import {createRequire} from 'node:module';
import {readFileSync,writeFileSync} from 'node:fs';
const require=createRequire(process.env.SETTLE_WORKTREE+'/package.json');const {Client}=require('pg');
const resource=JSON.parse(readFileSync(process.env.SETTLE_RESOURCE,'utf8'));
const client=new Client({connectionString:resource.test_database_url});
try {
 await client.connect();
 const result=await client.query('SELECT count(*)::int AS clients FROM pg_stat_activity WHERE datname=$1 AND (usename=$2 OR application_name LIKE $3) AND pid<>pg_backend_pid()',['pare224_synthetic',resource.roleName,'settlement-writer-%']);
 const identity=(await client.query('SELECT current_user,session_user,current_database(),pg_backend_pid() AS backend_pid')).rows[0];
 const row=result.rows[0];
 writeFileSync(process.env.SETTLE_CONNECTION_PROOF,JSON.stringify({identity,serviceRole:resource.roleName,serviceClients:row.clients,closed:row.clients===0,preservedData:true})+'\n',{flag:'wx',mode:0o600});
 if(row.clients!==0)process.exitCode=1;
}finally{await client.end();}
