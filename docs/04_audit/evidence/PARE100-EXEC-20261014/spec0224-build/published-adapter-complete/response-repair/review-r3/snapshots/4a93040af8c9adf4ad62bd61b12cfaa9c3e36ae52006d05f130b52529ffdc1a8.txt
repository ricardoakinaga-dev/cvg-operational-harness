const canon = (v) => canonicalizeCompleteJson(v, {

	maxDepth: 64,

const sha = (v) => createHash("sha256").update(canon(v)).digest("hex");
function find(v) {
	if (!v || typeof v !== "object") return undefined;
	if (v.type === "TOOL_RESULT" && v.payload?.completion === "COMPLETE") return v.payload.output;
	for (const c of Object.values(v)) {
		const f = find(c);
		if (f !== undefined) return f;
	}
	return undefined;
}
function delivered(r) {
	for (const m of r.messages) {
		try {
			const f = find(JSON.parse(m.content));
			if (f !== undefined) return f;
		} catch {}
	}
	return undefined;
}
