import pathlib,json,sys
b=json.loads(pathlib.Path(sys.argv[1]).read_text());a=json.loads(pathlib.Path(sys.argv[2]).read_text());errors=[]
if b['entries']!=a['entries']:errors.append('immutable-entry-change')
if a['mismatches']or b['mismatches']:errors.append('manifest-content-mismatch')
if a['candidateCount']!=81 or any(not x['pass']for x in a['bindings']):errors.append('candidate-packet-binding')
if a['manifestSHA']!='810490908f79f606a4e2c35f8cdd640625dfaf39b2f5e457fbd1b1cc66025b94':errors.append('manifest-identity')
if a['candidateSHA']!='d2edb86c5058708c8b48ebd9f4ecc6f13b526fd881fddebbd52ea2b7c31ac204':errors.append('candidate-identity')
print(json.dumps({'clean':not errors,'errors':errors,'entryCount':len(a['entries']),'candidateBindings':a['candidateCount']},indent=2));sys.exit(bool(errors))
