export function createSkillRegistry(skills) {
	const byId = new Map();
	for (const skill of skills) {
		if (byId.has(skill.id)) {
			throw new Error(`skill id duplicado no registro: ${skill.id} (versões ${byId.get(skill.id)?.version} e ${skill.version})`);
		}
		byId.set(skill.id, skill);
	}
	return {
		get(id) {
			return byId.get(id);
		},
		list() {
			return [...byId.values()];
		}
	};
}
