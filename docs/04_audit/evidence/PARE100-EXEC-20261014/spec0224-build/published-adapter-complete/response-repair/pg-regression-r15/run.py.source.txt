def inspect():
    x = subprocess.run(['docker', 'inspect', '--format', '{{json .Id}} {{json .State.Status}} {{json .Image}} {{json .NetworkSettings.Ports}}', container], capture_output=True, check=True)
    decoder = json.JSONDecoder()
    remaining = x.stdout.decode().strip()
    values = []
    while remaining:
        value, end = decoder.raw_decode(remaining)
        values.append(value)
        remaining = remaining[end:].lstrip()
    return dict(zip(['id', 'status', 'image', 'ports'], values))


def save(name, value):
    (p / name).write_text(json.dumps(value, indent=2) + '\n')
