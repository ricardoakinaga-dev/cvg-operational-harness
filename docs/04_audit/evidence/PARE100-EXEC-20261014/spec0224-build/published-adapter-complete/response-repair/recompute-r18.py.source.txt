"""Independent raw response/decision accounting reconstruction, no CVG imports."""
import hashlib
import json
import math
import sys
from pathlib import Path

path = Path(sys.argv[1])
raw = path.read_bytes()
cases = json.loads(raw)
checks = 0
snapshots = 0
canonical = lambda v: json.dumps(v, ensure_ascii=False, sort_keys=True, separators=(',', ':'), allow_nan=False).encode()
sha = lambda b: hashlib.sha256(b).hexdigest()
toy = lambda v: math.ceil(len(canonical(v)) / 64)


def check(value, message):
    global checks
    checks += 1
    if not value:
        raise AssertionError(message)


def checkpoint(cp):
    global snapshots
    if cp is None or cp['checkpointVersion'] != 2:
        return
    snapshots += 1
    unsigned = {k: v for k, v in cp.items() if k not in ['checkpointId', 'createdAt', 'digest']}
    check(sha(b'cvg.checkpoint.v2\n' + canonical(unsigned)) == cp['digest'], 'CP domain digest')
    check(cp['budgetUsage']['steps'] >= cp['stepNumber'] == cp['state']['stepNumber'], 'CP native step count')


check([c['case'] for c in cases] == ['RR%02d' % i for i in range(1, 31)], 'Thirty required cases')
reports = []
for case in cases:
    cid = case['case']
    requests = case['requests']
    results = case['modelResults']
    entries = case['modelEntryCheckpoints']
    final = case['checkpoint']
    checkpoint(final)
    check((len(case['checkpointWriteErrors']) > 0) if cid in ['RR16','RR17','RR18','RR19','RR20','RR21','RR24','RR25','RR30'] else case['checkpointWriteErrors'] == [], cid + ': declared CAS fault visibility')
    for saved in case.get('checkpointSaves', []):
        checkpoint(saved['next'])
    check(case['bodies'] == len(case['journals']) == 1, cid + ': tool replay/missing journal')
    journal = case['journals'][0]
    check(journal['state'] == 'CONFIRMED', cid + ': tool not confirmed')
    check(journal['result'] == case['output'], cid + ': journal result changed')
    check(case['expected'] == 'sha256:' + sha(canonical(case['output'])), cid + ': fixture expected digest')
    if int(cid[2:]) >= 24:
        actual_calls = 0 if cid == 'RR29' else 1
        expected_results = 0 if cid in ['RR28','RR29'] else 1
        check(len(requests) == len(entries) == actual_calls and len(results) == expected_results, cid + ': actual provider entry/result distinction')
        check(case['result']['modelCalls'] == final['budgetUsage']['modelCalls'] == actual_calls, cid + ': shared native counter')
        expected_stop = 'MODEL_FAILURE' if cid == 'RR28' else 'INTERNAL_FAILURE'
        check(case['result']['stopReason'] == final['state']['stopReason'] == expected_stop, cid + ': native failure category changed')
        check(final['stepNumber'] == final['state']['stepNumber'] == final['budgetUsage']['steps'] == 2, cid + ': stopped native cursor')
        check(case['measurements'] == 0 and final['state']['toolResultContext']['decisionAdmissions'] == [], cid + ': response got new decision receipt/counter')
        native_status = 'SUCCEEDED' if cid == 'RR27' else 'FAILED'
        check(any(v['stepNumber'] == 2 and v['stepType'] == 'RESPOND' and v['status'] == native_status for v in case['nativeSteps']), cid + ': lost native acknowledged terminal fact')
        observations = [o for o in final['state']['observations'] if o['type'] == 'TOOL_RESULT']
        check(len(observations) == 1 and observations[0]['payload']['output'] == case['output'], cid + ': full tool observation lost')
        for request, cp in zip(requests, entries):
            checkpoint(cp)
            check(cp['budgetUsage']['modelCalls'] == 1, cid + ': no committed native slot before entry')
            check(request['purpose'] == 'RESPONSE' and len(request['messages']) == 2 and 'maxOutputTokens' not in request, cid + ': native response framing/contract')
            check(len(request['messages'][0]['content']) <= 16000 and len(request['messages'][1]['content']) <= 32000 and len(canonical(request)) <= 131072, cid + ': wire limits')
            frame = json.loads(request['messages'][1]['content'])
            items = [i for i in frame['items'] if i['priority'] == 'FRESH_TOOL_RESULTS']
            check(len(items) == 1 and items[0]['trust'] == 'UNTRUSTED' and items[0]['value'] == observations[0], cid + ': exact full observation and native provenance')
        for request,result in zip(requests,results):
            check(result['inputTokens'] == toy(request) and result['outputTokens'] == toy(result['text']) and result['costUsd'] == 0.0625, cid + ': synthetic observed usage')
            check(result['text'] == ('' if cid == 'RR25' else case['expected']), cid + ': actual model result')
        observed = {k:sum(v[k] for v in results) for k in ['inputTokens','outputTokens','costUsd']}
        for key,value in observed.items():
            check(case['result']['usage'][key] == final['budgetUsage'][key] == value, cid + ': exact known usage retained ' + key)
        replay = case['replay']
        check(replay['result']['stopReason'] == 'STATE_CONFLICT', cid + ': stopped complete attempt reopened')
        check(replay['checkpoint'] == final and replay['nativeSteps'] == case['nativeSteps'] and replay['journals'] == case['journals'] and replay['bodies'] == 1, cid + ': stopped recovery mutated durable state/effects')
        check(replay['requests'] == requests and replay['modelResults'] == results, cid + ': stopped recovery invoked model again')
        check('already stopped' in replay['result']['response'] and 'reconciliation' in replay['result']['response'], cid + ': stopped attempt not identified')
        reports.append({'case':cid,'actualCalls':actual_calls,'knownResults':expected_results,'usageKnown':False if cid == 'RR28' else True,'recordedUsage':observed,'durableStop':expected_stop,'nativeRespond':native_status,'replay':'REFUSED_UNCHANGED_CP_JOURNAL_NO_MODEL','limits':'Model throw has no observed token/cost result; zero recorded usage is not an invented historical bill.'})
        continue
    if int(cid[2:]) >= 16:
        scenario = case['scenario']
        actual_calls = 1 if cid in ['RR16','RR17','RR18','RR21'] else 0
        held_slots = 0 if cid in ['RR22','RR23'] else 1
        check(len(requests) == len(results) == len(entries) == actual_calls, cid + ': actual provider entries')
        check(case['result']['modelCalls'] == final['budgetUsage']['modelCalls'] == held_slots, cid + ': native held slot, not billing')
        check(case['result']['stopReason'] != 'COMPLETED', cid + ': writer uncertainty delivered success')
        check(case['measurements'] == 0 and final['state']['toolResultContext']['decisionAdmissions'] == [], cid + ': new response gate/receipt')
        observations = [o for o in final['state']['observations'] if o['type'] == 'TOOL_RESULT']
        check(len(observations) == 1 and observations[0]['payload']['output'] == case['output'], cid + ': lost native full result')
        check(any(v['stepNumber'] == 2 and v['stepType'] == 'RESPOND' and v['status'] == 'FAILED' for v in case['nativeSteps']), cid + ': no acknowledged native failed response')
        for request, result, cp in zip(requests, results, entries):
            checkpoint(cp)
            check(cp['budgetUsage']['modelCalls'] == 1, cid + ': native slot not committed before actual model entry')
            check(result['inputTokens'] == toy(request) and result['outputTokens'] == toy(result['text']) and result['costUsd'] == 0.03125, cid + ': declared synthetic actual usage')
            check(request['purpose'] == 'RESPONSE' and len(request['messages']) == 2 and 'maxOutputTokens' not in request, cid + ': native wire/owner changed')
            check(len(request['messages'][0]['content']) <= 16000 and len(request['messages'][1]['content']) <= 32000 and len(canonical(request)) <= 131072, cid + ': frame limits')
            frame = json.loads(request['messages'][1]['content'])
            items = [v for v in frame['items'] if v['priority'] == 'FRESH_TOOL_RESULTS']
            check(len(items) == 1 and items[0]['trust'] == 'UNTRUSTED' and items[0]['value'] == observations[0], cid + ': full selected observation/provenance')
            payload = items[0]['value']['payload']
            check(payload['outputBytes'] == len(canonical(case['output'])) and payload['outputSha256'] == sha(canonical(case['output'])), cid + ': actual full output binding')
            check(result['text'] == ('' if cid == 'RR17' else case['expected']), cid + ': model text not based on actual declared input')
        actual = {k: sum(r[k] for r in results) for k in ['inputTokens','outputTokens','costUsd']}
        for key,value in actual.items():
            check(case['result']['usage'][key] == value, cid + ': actual observed usage returned once ' + key)
        if cid in ['RR16','RR17','RR18']:
            check(final == entries[0] and final['budgetUsage']['costUsd'] == 0 and final['state'].get('stopReason') is None, cid + ': unacknowledged result falsely settled')
            check(any(v['stepNumber'] > final['stepNumber'] and v['stepType'] == 'RESPOND' for v in case['nativeSteps']), cid + ': no durable attempt ahead of old checkpoint')
        elif cid == 'RR21':
            check(final['budgetUsage']['costUsd'] == 0.03125, cid + ': observed usage committed before lost ACK erased')
        else:
            check(actual_calls == 0 and final['budgetUsage']['costUsd'] == 0, cid + ': invocation before required ACK')
        replay = case['replay']
        check(replay['result']['stopReason'] == 'STATE_CONFLICT', cid + ': uncertain response resumed')
        check(replay['requests'] == requests and replay['modelResults'] == results, cid + ': second provider invocation')
        check(replay['checkpoint'] == final and replay['journals'] == case['journals'] and replay['nativeSteps'] == case['nativeSteps'] and replay['bodies'] == 1, cid + ': recovery rewrote unknown accounting/effects')
        if cid in ['RR16','RR17','RR18']:
            check('uncertain' in replay['result']['response'] and 'reconciliation' in replay['result']['response'], cid + ': missing explicit uncertainty/handoff')
        reports.append({'case':cid,'actualCalls':actual_calls,'nativeHeldSlots':held_slots,'observedUsage':actual,'durableUsage':final['budgetUsage'],'durableStop':final['state'].get('stopReason'),'recovery':'REFUSED_NO_NEW_MODEL_NO_STATE_REWRITE','qualification':'UnACKed actual cost remains UNKNOWN; durable pre-call zero is not claimed settled cost. Native hold may precede zero actual entries.'})
        continue
    check(len(requests) == len(results) == len(entries) == case['result']['modelCalls'], cid + ': actual model count')
    actual = {k: sum(r[k] for r in results) for k in ['inputTokens', 'outputTokens', 'costUsd']}
    for key, value in actual.items():
        check(case['result']['usage'][key] == value == final['budgetUsage'][key], cid + ': native shared usage ' + key)
    check(final['budgetUsage']['modelCalls'] == len(requests), cid + ': durable call count')
    receipts = final['state'].get('toolResultContext', {}).get('decisionAdmissions', [])
    orchestration = [(q, r) for q, r in zip(requests, results) if q['purpose'] == 'ORCHESTRATION']
    check(len(receipts) == len(orchestration) == case['measurements'], cid + ': decision accounting ownership')
    for receipt, (request, result) in zip(receipts, orchestration):
        check(receipt['status'] == 'SETTLED', cid + ': receipt not settled')
        check(receipt['requestSha256'] == sha(canonical(request)), cid + ': decision request binding')
        check(receipt['resultSha256'] == sha(canonical(result)), cid + ': decision result binding')
        check(receipt['measurement']['inputUpperBoundTokens'] == toy(request), cid + ': declared Toy64 input')
        check(receipt['reserved']['outputTokens'] == request['maxOutputTokens'] == 512, cid + ': unchanged decision cap')
        check(receipt['actual'] == dict(modelCalls=1, inputTokens=result['inputTokens'], outputTokens=result['outputTokens'], costUsd=result['costUsd']), cid + ': actual receipt usage')
    for request, result, cp in zip(requests, results, entries):
        checkpoint(cp)
        check(result['inputTokens'] == toy(request) and result['outputTokens'] == toy(result['text']), cid + ': actual synthetic usage')
        check(result['provider'] == 'synthetic.response' and result['model'] == 'toy64-v1' and result['costUsd'] == 0, cid + ': synthetic pair')
        if cid in ['RR04', 'RR13']:
            check(request['messages'][1]['content'] == case['input']['userMessage'], 'Legacy wire changed')
            continue
        check(len(request['messages']) == 2, cid + ': message count')
        check(len(request['messages'][0]['content']) <= 16000 and len(request['messages'][1]['content']) <= 32000, cid + ': message limits')
        check(len(canonical(request)) <= 131072, cid + ': request byte limit')
        frame = json.loads(request['messages'][1]['content'])
        check(frame['schema'] == 'cvg.decision-context' and frame['version'] == 1, cid + ': frame schema')
        tool_items = [i for i in frame['items'] if i['priority'] == 'FRESH_TOOL_RESULTS']
        observations = [o for o in cp['state']['observations'] if o['type'] == 'TOOL_RESULT']
        check(len(tool_items) == len(observations), cid + ': required actual model-entry observations')
        for item, observation in zip(tool_items, observations):
            check(item['trust'] == 'UNTRUSTED' and item['value'] == observation, cid + ': selected full observation')
            payload = observation['payload']
            check(payload['output'] == case['output'] and payload['completion'] == 'COMPLETE', cid + ': actual full output')
            check(payload['outputBytes'] == len(canonical(case['output'])) and payload['outputSha256'] == sha(canonical(case['output'])), cid + ': full value byte/hash')
        if request['purpose'] == 'RESPONSE':
            check(len(observations) == 1, cid + ': response has no full durable result')
            check(result['text'] == case['expected'] if cid not in ['RR14', 'RR15'] else result['text'].strip() == '', cid + ': response not frame-grounded or declared empty')
            check('maxOutputTokens' not in request, cid + ': new response output contract invented')
        if cid == 'RR08':
            for marker in ['UNTRUSTED_TOOL_224', 'UNTRUSTED_USER_224', 'UNTRUSTED_INTENT_224']:
                check(marker not in request['messages'][0]['content'] and marker in request['messages'][1]['content'], 'Data became system instruction or was lost')
    if cid in ['RR03', 'RR07']:
        check(requests == [] and case['contextBuilds'] == 3, cid + ': incomplete selection/framing called model')
        check('CONTEXT_OVERFLOW' in case['result']['response'] and case['result']['stopReason'] == final['state']['stopReason'] == 'INSUFFICIENT_EVIDENCE', cid + ': no durable context refusal')
    elif cid in ['RR14', 'RR15']:
        check(case['result']['stopReason'] == final['state']['stopReason'] == 'INSUFFICIENT_EVIDENCE', cid + ': empty response refusal missing')
        replay = case['replay']
        check(replay['result']['stopReason'] == 'STATE_CONFLICT', cid + ': empty response resumed')
        check(replay['requests'] == requests and replay['modelResults'] == results, cid + ': replay made a new model call')
        check(replay['checkpoint'] == final and replay['journals'] == case['journals'] and replay['nativeSteps'] == case['nativeSteps'] and replay['bodies'] == 1, cid + ': terminal replay changed durable effects/accounting')
    elif cid in ['RR10', 'RR11', 'RR12', 'RR13']:
        check(case['result']['stopReason'] == final['state']['stopReason'] == 'INSUFFICIENT_EVIDENCE', cid + ': fault not durably conservative')
        check(len(requests) == 1 == final['budgetUsage']['modelCalls'], cid + ': duplicated fault-path call')
        if cid in ['RR11', 'RR12']:
            check(len(case['auditVetoes']) == 1, cid + ': missing actual audit veto')
    elif cid == 'RR06':
        check([q['purpose'] for q in requests] == ['ORCHESTRATION', 'ORCHESTRATION'], 'Budget refusal called RESPONSE')
        check(case['result']['stopReason'] == final['state']['stopReason'] == 'MAX_MODEL_CALLS', 'No durable budget refusal')
        check(final['stepNumber'] == final['budgetUsage']['steps'] == 2, 'Failed response step not recorded')
        check(any(s['stepNumber'] == 2 and s['stepType'] == 'RESPOND' and s['status'] == 'FAILED' for s in case['nativeSteps']), 'No actual native failed RESPOND')
    else:
        check(case['result']['stopReason'] == final['state']['stopReason'] == 'COMPLETED', cid + ': positive incomplete')
        if cid != 'RR04':
            check(case['result']['response'] == case['expected'], cid + ': wrong final digest')
    if cid != 'RR13':
        check(final['stepNumber'] == final['state']['stepNumber'] == final['budgetUsage']['steps'] == 2, cid + ': acknowledged native terminal cursor')
        check('acceptedDecision' not in final['state'].get('toolResultContext', {}), cid + ': accepted decision not cleared')
        status = 'FAILED' if cid in ['RR03', 'RR06', 'RR07', 'RR10', 'RR14', 'RR15'] else 'SUCCEEDED'
        check(any(s['stepNumber'] == 2 and s['stepType'] == 'RESPOND' and s['status'] == status for s in case['nativeSteps']), cid + ': terminal step contradicts native acknowledged fact')
    reports.append({'case': cid, 'actualPurposes': [q['purpose'] for q in requests], 'actualCalls': len(requests),
                    'nativeDecisionReceipts': len(receipts), 'nativeUsage': actual, 'durableStop': final['state']['stopReason']})
print(json.dumps({'status': 'PASS_SCOPED_RESPONSE_RAW_RECONSTRUCTION', 'checks': checks, 'checkpointOccurrences': snapshots,
                  'cases': reports, 'source': str(path), 'sourceSHA256': sha(raw),
                  'limits': 'Synthetic Toy64, native memory/public factory. No real tokenizer/billing/PG/publishedconsumer/global proof.'}, indent=2))
