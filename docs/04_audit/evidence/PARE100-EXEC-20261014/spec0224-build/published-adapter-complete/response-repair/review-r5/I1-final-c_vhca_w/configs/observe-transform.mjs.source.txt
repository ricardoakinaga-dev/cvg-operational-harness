import {writeFileSync,appendFileSync} from 'node:fs';
import {createHash} from 'node:crypto';
export default {name:'i1-observe-post-transform',enforce:'post',transform(code,id){
 const dir=process.env.I1_REVIEW+'/observed/transformed/';
 const digest=createHash('sha256').update(code).digest('hex');
 writeFileSync(dir+digest+'.js',code);
 appendFileSync(process.env.I1_REVIEW+'/observed/transform-v3.jsonl',JSON.stringify({pid:process.pid,id,bytes:Buffer.byteLength(code),sha256:digest,phase:'Vite post transform hook, not V8 line coverage'})+'\n');
 }};
