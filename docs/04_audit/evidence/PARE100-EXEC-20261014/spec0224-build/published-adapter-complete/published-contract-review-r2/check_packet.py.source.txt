#!/usr/bin/env python3
"""Readonly input verification; writes results only in the assigned critic tree.

This does not import, execute or test any copied application code or SQL.
"""
import hashlib
import json
from pathlib import Path

ROOT = Path('/tmp/cvg-pare100-exec-20261014-6az3hlay/spec0224-build/published-adapter-complete/published-contract-review-packet-r2')
OUT = Path('/tmp/cvg-pare100-exec-20261014-6az3hlay/spec0224-build/published-adapter-complete/published-contract-review-r2')
MANIFEST_SHA = '13e29710e0055ffed26955b85ae8d03e6aed49b672a3159ac15de0492e31c99b'


def sha(data):
    return hashlib.sha256(data).hexdigest()


def write(name, obj):
    (OUT / name).write_text(json.dumps(obj, ensure_ascii=False, indent=2) + '\n')


def main():
    manifest_path = ROOT / 'manifest.json'
    raw = manifest_path.read_bytes()
    manifest = json.loads(raw)
    checks = []

    def check(name, passed, detail):
        checks.append({'id': name, 'status': 'PASS' if passed else 'FAIL', 'detail': detail,
                       'kind': 'STATIC_DOCUMENTARY_CHECK'})

    check('manifest_identity', sha(raw) == MANIFEST_SHA, sha(raw))
    check('manifest_scope', manifest['readRoot'] == str(ROOT) and manifest['ownedOutput'] == str(OUT),
          'Literal assigned readRoot and ownedOutput')
    entries = manifest['inputs']
    check('exact_input_count', len(entries) == 53, len(entries))
    check('unique_ids_paths', len({i['id'] for i in entries}) == 53 and len({i['copy'] for i in entries}) == 53,
          '53 distinct IDs and copy paths')
    texts = {}
    rows = []
    for item in entries:
        p = Path(item['copy'])
        scope = p.is_relative_to(ROOT) and p.resolve() == p and not p.is_symlink()
        if not scope:
            raise ValueError('Refusing input outside the exact copied scope: ' + item['id'])
        b = p.read_bytes()
        text = b.decode('utf-8')
        texts[item['id']] = text
        rows.append({'id': item['id'], 'path': str(p), 'bytes': len(b), 'sha256': sha(b),
                     'match': len(b) == item['bytes'] and sha(b) == item['sha256'], 'scope': scope})
    check('all_copy_fingerprints', all(i['match'] and i['scope'] for i in rows), '53/53 size, SHA-256 and resolved paths')
    before = json.loads((OUT / 'fingerprints-before.json').read_text())
    after = {'manifest': {'path': str(manifest_path), 'bytes': len(raw), 'sha256': sha(raw), 'match': sha(raw) == MANIFEST_SHA},
             'count': len(rows), 'inputs': rows}
    check('before_after_immutable', before == after, 'Exact equality of all 54 fingerprint records')
    write('fingerprints-after.json', after)
    source_manifest = json.loads(texts['source-manifest.json'])
    bindings = source_manifest['sources']
    direct = [i for i in entries if i['id'] not in ['PROPOSED-SPEC.md', 'ACCEPTANCE-PLAN.md', 'DECISIONS-AND-RISKS.md', 'source-manifest.json']]
    check('direct_source_bindings', bindings == direct, {'inner_count': len(bindings), 'expected': len(direct)})
    check('pending_225_identity', next(i for i in rows if i['id'] == 'P225')['sha256'] == '04b92ba4d13b1ed58157a24258cade87d8d6d31df833ef6c1969059772cc6da5', 'Proposal only; no authority inferred')
    anchors = {
        'native_historical_decision_branch': ('N02', "if (accepted.stepNumber !== run.stepNumber + 1)"),
        'native_pending_rejected_with_accepted_decision': ('N04', "if (!validateLoopDecision(accepted.decision).valid || pending !== null)"),
        'native_response_reserve_excludes_response_today': ('S11', "(purpose !== 'ORCHESTRATION' && purpose !== 'REPAIR')"),
        'native_direct_unrecorded_arithmetic': ('S12', 'run.usage.inputTokens += outcome.result.inputTokens'),
        'native_breach_records_actual': ('S11', "await this.unknown(record, 'ADMISSION_BREACH', actual ?? undefined)"),
        'proposal_closes_dispatch_on_breach': ('PROPOSED-SPEC.md', 'fecha dispatch da execução'),
        'proposal_diagnostic_forbids_new_actual': ('PROPOSED-SPEC.md', 'somente RESERVED→UNKNOWN sem actual novo'),
        'proposal_retains_accepted_decision': ('PROPOSED-SPEC.md', 'conserva acceptedDecision até a'),
        'native_old_ack_executes_callback': ('N05', 'result = await rawInput.effect(mapDurableOutboxRow(prepared.event))'),
        'proposal_ack_prepare_no_write': ('PROPOSED-SPEC.md', 'ack_prepare não escreve'),
        'oracle_old_ack_zero_effects': ('ACCEPTANCE-PLAN.md', 'Old ACK0 mutations/effects'),
    }
    for name, (input_id, needle) in anchors.items():
        locations = [n for n, line in enumerate(texts[input_id].splitlines(), 1) if needle in line]
        check(name, bool(locations), {'input': input_id, 'lines': locations, 'literal': needle})
    old_ack = texts['N05'].split('const prepared = await withOutboxTransaction(', 1)[1].split("if (prepared.kind === 'processed'", 1)[0]
    check('old_ack_prepare_no_dml_literal', not any(word in old_ack.upper() for word in ['INSERT INTO ', 'UPDATE OUTBOX_', 'DELETE FROM ']),
          'Text slice only; SELECT FOR UPDATE is a lock and is not DML. No claim of a runtime probe.')
    check('all_future_acceptance_not_run', 'PA01–PA18' in texts['ACCEPTANCE-PLAN.md'] and '**NOT_RUN**' in texts['ACCEPTANCE-PLAN.md'],
          'Document declares NOT_RUN; no acceptance tests run by this checker')
    result = {'schema': 'cvg.critic-i1.documentary-checks/1', 'scope': 'Only the enumerated 53 copies and exact manifest; own fingerprint baseline also read',
              'checks': checks, 'overall': 'PASS' if all(c['status'] == 'PASS' for c in checks) else 'FAIL',
              'check_errors': [{'command': 'python - <<... (initial fingerprint attempt)', 'exit_code': 127,
                                'error': '/bin/bash: line 1: python: command not found',
                                'correction': 'Used python3; first attempt executed no checker body and wrote no file.'}],
              'limits': 'Static anchors and fingerprints only. They do not execute application code, SQL, tests or acceptance scenarios; PASS authenticates the review inputs/anchors, not the proposal.'}
    write('check-results.json', result)
    print(json.dumps({'checks': len(checks), 'status': result['overall'], 'inputs': len(rows), 'immutable': before == after}))


if __name__ == '__main__':
    main()
