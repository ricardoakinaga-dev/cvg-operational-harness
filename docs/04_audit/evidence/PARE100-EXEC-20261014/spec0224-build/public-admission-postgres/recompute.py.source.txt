"""Independent raw-evidence verifier; imports no CVG codec/test/runtime."""
import sys
import base64
import copy
import hashlib
import json
from pathlib import Path

base = Path(__file__).parent
run = Path(sys.argv[1])
case_dirs = sorted((run/'cases').iterdir())
rows = [json.loads((p/'observed.json').read_text()) for p in case_dirs]
checks = 0
snapshots = 0
def check(value, message):
    global checks
    checks += 1
    if not value:
        raise AssertionError(message)
def sha(raw):
    return hashlib.sha256(raw).hexdigest()
def canonical(value):
    # Fixture numbers are integers and exact binary fractions; strings ASCII.
    return json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(',', ':')).encode()
def reconstruct(value):
    global snapshots
    snapshots += 1
    cp = copy.deepcopy(value['checkpointWithoutFixturePadding'])
    recipe = value['paddingRecipe']
    check(recipe['character'] in ['p', '\u0000'], 'Padding alphabet')
    padding = recipe['character'] * recipe['repeat'] + recipe.get('suffix', '')
    check(sha(padding.encode()) == recipe['sha256'], 'Padding SHA')
    if recipe['present']:
        cp['state']['resolvedInputs']['fixturePadding'] = padding
    unsigned = {k:v for k,v in cp.items() if k not in ['checkpointId', 'createdAt', 'digest']}
    raw = canonical(unsigned)
    check(len(raw) == value['logicalUnsignedBytes'], 'Unsigned byte count')
    check(8388608 - len(raw) == value['remainingBytes'], 'Remaining bytes')
    check(sha(b'cvg.checkpoint.v2\n' + raw) == cp['digest'], 'Domain digest')
    check(len(raw) <= 8388608, 'Overflow stored CP')
    return cp


pre = json.loads((run/'preexistence-check.json').read_text())
check(pre == {'schema_exists':False,'role_exists':False}, 'Resource existed before provisioning')
identity = json.loads((run/'serving-identity-grants-rls-migrations.json').read_text())
role = identity['identity']['current_user']
check(identity['identity']['session_user']==role, 'Unexpected assumed role')
for key in ['rolsuper','rolbypassrls','rolinherit','rolcreatedb','rolcreaterole']:
    check(identity['identity'][key] is False, 'Unexpected privilege '+key)
for key in ['memberships','owned_schemas','owned_relations']:
    check(identity['identity'][key]==0, 'Service ownership '+key)
check(identity['identity']['rolcanlogin'] is True, 'Service cannot login')
expected_tables={'operational_executions','operational_execution_checkpoints','operational_execution_steps','operational_effect_journal'}
check({e['relname'] for e in identity['rls']}==expected_tables, 'RLS table inventory')
for table in identity['rls']:
    check(table['relrowsecurity'] and table['relforcerowsecurity'] and table['owner']!=role, 'Forced RLS/nonowner')
check({(e['table_name'],e['privilege_type']) for e in identity['grants']}=={(t,p) for t in expected_tables for p in ['INSERT','SELECT','UPDATE']}, 'Unexpected grants')
for policy in identity['policies']:
    check('cvg.tenant_id' in policy['qual'] and 'tenant_isolation_quarantined' in policy['qual'], 'Tenant policy')
    check(policy['with_check']==policy['qual'], 'RLS write policy')
check(len(identity['policies'])==4, 'Policy inventory')
migrations=list((base/'review-packet-r1/artifact/packages/persistence/migrations').glob('*.sql'))
versions={p.stem:sha(p.read_bytes()) for p in migrations}
for record in identity['migrationRows']:
    check(record['version'] in versions and record['checksum']==versions[record['version']], 'Applied migration checksum')
check(len({e['version'] for e in identity['migrationRows']})==len(identity['migrationRows']), 'Repeated migration identity')
cases=[]
for row in rows:
    before = reconstruct(row['before'])
    after = reconstruct(row['after'])
    again = reconstruct(row['afterRepeat']) if row['afterRepeat'] else None
    old = before['state']['toolResultContext']['decisionAdmissions']
    new = after['state']['toolResultContext']['decisionAdmissions']
    refusal = row['remainingSeedBytes'] == 2000
    prior = '-prior-true-' in row['name']
    check(row['before']['remainingBytes'] == (2000 if refusal else 20000), 'Wrong seed bytes')
    check(row['modelsBefore'] == (2 if prior else 0), 'Wrong real prior models')
    check(row['actualCompleteEntries'] == row['modelsBefore'] + int(not refusal), 'Actual complete entries')
    check(row['actualModelsAfter'] == row['actualCompleteEntries'], 'Unobserved failed complete entry')
    check(row['bodiesAfter'] == row['bodiesBefore'] == int(prior), 'New or missing tool body')
    check(new[:len(old)] == old, 'Prior receipts changed')
    check(row['oldJournal'] == row['afterJournal'], 'Prior journal changed')
    old_tools = [o for o in before['state']['observations'] if o['type']=='TOOL_RESULT']
    new_tools = [o for o in after['state']['observations'] if o['type']=='TOOL_RESULT']
    check(old_tools == new_tools, 'Prior TOOL_RESULT changed')
    check(before['state']['resolvedInputs']['fixturePadding'] == after['state']['resolvedInputs']['fixturePadding'], 'Seed data changed')
    for item in row['writes']:
        cp = reconstruct(item['checkpoint'])
        check(item['result']['status'] == 'APPLIED', 'CAS not applied')
        check(item['result']['token']['checkpointId'] == cp['checkpointId'], 'CAS ID')
        check(item['result']['token']['digest'] == cp['digest'], 'CAS digest')
    for measured in row['measurements']:
        reconstruct(measured['checkpointBeforeReserve'])
        raw = canonical(measured['request'])
        check(measured['measurement']['inputUpperBoundTokens'] == (len(raw)+63)//64, 'Qualified toy input count')
        check(measured['request']['maxOutputTokens'] == 512, 'Output cap')
    for called in row['calls']:
        cp = reconstruct(called['checkpointAtComplete'])
        receipt = called['receipt']
        ctx = cp['state']['toolResultContext']
        check(receipt['status'] == 'RESERVED', 'Model had no reserved receipt')
        check(ctx['pendingDecisionAdmission'] == receipt['admissionId'], 'Model pending receipt identity')
        check(receipt in ctx['decisionAdmissions'], 'Model receipt not actually stored')
        check(receipt['requestSha256'] == sha(canonical(called['request'])), 'Wrong actual request hash')
        check(receipt['reserved']['outputTokens'] == 512, 'Reserved output cap')
    fields=['modelCalls','inputTokens','outputTokens','costUsd','toolCalls','steps']
    if refusal:
        check(row['result']['stopReason'] == 'INSUFFICIENT_EVIDENCE', 'Wrong first handoff')
        check(row['result']['response'] == 'CONTEXT_OVERFLOW: No checkpoint space for the decision receipt.', 'Wrong local receipt diagnostic')
        check(row['repeated']['stopReason'] == 'STATE_CONFLICT', 'Terminal resume not refused')
        check(new == old == again['state']['toolResultContext']['decisionAdmissions'], 'Refusal stored new receipt')
        check(after['state']['stopReason'] == 'INSUFFICIENT_EVIDENCE', 'Handoff not acknowledged')
        check('pendingQuestion' not in after['state'], 'Answered question still pending')
        check(again == after, 'Terminal resume mutated checkpoint')
        check(row['oldJournal'] == row['afterRepeatJournal'], 'Repeated refusal changed journal')
        for field in fields:
            check(before['budgetUsage'][field] == after['budgetUsage'][field] == again['budgetUsage'][field], 'Acknowledged budget changed: '+field)
        check(after['budgetUsage']['activeDurationMs'] > before['budgetUsage']['activeDurationMs'], 'Refusal duration not acknowledged')
        resumed_measurements=[m for m in row['measurements'] if m['resumed']]
        check(len(resumed_measurements)==1, 'Terminal resume re-entered counter')
        check(row['result']['modelCalls'] == before['budgetUsage']['modelCalls'], 'Unpersisted live model debit')
        for field in ['inputTokens','outputTokens','costUsd']:
            check(row['result']['usage'][field] == before['budgetUsage'][field], 'Unpersisted live debit '+field)

    else:
        check(row['result']['stopReason'] == 'COMPLETED', 'Positive incomplete')
        check(len(new)==len(old)+1 and new[-1]['status']=='SETTLED', 'New receipt not settled')
        for field in ['modelCalls','inputTokens','outputTokens','costUsd']:
            check(after['budgetUsage'][field] == sum(r['actual'][field] for r in new), 'Settled accounting: '+field)
        check(after['budgetUsage']['costUsd'] == before['budgetUsage']['costUsd']+0.125, 'Exact synthetic cost')
    cases.append({'case':row['name'],'actualCompleteEntries':row['actualCompleteEntries'],'priorActualModels':row['modelsBefore'],'returnedModelDebits':row['result']['modelCalls'],'acknowledgedModelDebits':after['budgetUsage']['modelCalls'],'acknowledgedCost':after['budgetUsage']['costUsd'],'stopReason':row['result']['stopReason']})

    physical = json.loads((case_dirs[len(cases)-1]/'physical-checkpoints.json').read_text())
    check(bool(physical), 'No direct SQL physical observations')
    known_ids = {before['checkpointId'], after['checkpointId']}
    if again: known_ids.add(again['checkpointId'])
    for called in row['calls']: known_ids.add(called['checkpointAtComplete']['checkpointWithoutFixturePadding']['checkpointId'])
    for write in row['writes']: known_ids.add(write['checkpoint']['checkpointWithoutFixturePadding']['checkpointId'])
    for measured in physical:
        cp = reconstruct(measured['nativeLogicalCheckpoint'])
        check(cp['checkpointId'] in known_ids or measured['label']=='actual-native-pause-before-padding', 'Unbound direct SQL checkpoint')
        check(measured['freshPhysicalClient'] is True, 'Reader was not fresh')
        identity = measured['readerIdentity']
        check(identity['backend_pid'] != measured['writerPid'], 'Writer reused for committed SQL evidence')
        check(identity['current_user'] == identity['session_user'], 'Unexpected service role switch')
        check(identity['tenant_context'] == cp['tenantId'], 'Wrong SQL tenant')
        state = canonical(cp['state'])
        data = base64.b64encode(state).decode('ascii')
        header = measured['wrapperHeader']
        wire = measured['wire']
        check(header['storageCodec']=='cvg.checkpoint-state/2' and header['encoding']=='base64', 'Physical codec')
        check(header['decodedBytes']==len(state), 'Physical decoded bytes')
        check(header['decodedSha256']==sha(state), 'Physical decoded SHA')
        check(wire['expectedStateCanonicalBytes']==len(state) and wire['expectedStateCanonicalSha256']==sha(state), 'State reconstruction')
        check(wire['base64Chars']==len(data) and wire['base64Sha256']==sha(data.encode()), 'Physical base64 binding')
        wrapper = dict(header, data=data)
        check(wire['canonicalWrapperSha256']==sha(canonical(wrapper)), 'Physical wrapper canonical hash')
        text = (wire['prefixBeforeQuotedBase64Data'] + json.dumps(data) + wire['suffixAfterQuotedBase64Data']).encode()
        check(wire['sqlTextBytes']==len(text) and wire['sqlTextSha256']==sha(text), 'Actual state::text reconstruction')
        check(json.loads(text)==wrapper, 'SQL wrapper differs from reconstructed state')
        for sql, key in [('checkpoint_id','checkpointId'),('execution_id','executionId'),('tenant_id','tenantId'),('checkpoint_version','checkpointVersion'),('runtime_profile','runtimeProfile'),('runtime_version','runtimeVersion'),('orchestrator_version','orchestratorVersion'),('step_number','stepNumber'),('digest','digest'),('budget_usage','budgetUsage'),('created_at','createdAt')]:
            check(measured['rowHeader'][sql]==cp[key], 'SQL row logical binding '+sql)
    events = {e['sequence']:e for e in row['events']}
    check(sorted(events)==list(range(1,len(events)+1)), 'Event sequence holes/duplicates')
    for call in row['calls']:
        entry=events[call['completeEntry']]; cas=events[call['priorCASEvent']]; read=events[call['readEvent']]
        cp = call['checkpointAtComplete']['checkpointWithoutFixturePadding']
        check(cas['type']=='native-CAS-return-after-transaction' and cas['result']['status']=='APPLIED', 'No completed native transaction')
        check(entry['type']=='model.complete-entry' and read['type']=='fresh-service-SQL-before-response', 'Wrong event identity')
        check(cas['sequence']<entry['sequence']<read['sequence'], 'Reserve commit/model entry/SQL read order')
        check(read['priorCASEvent']==cas['sequence'] and read['completeEntry']==entry['sequence'], 'Read event linkage')
        for event in [cas,read]:
            check(event['checkpointId']==cp['checkpointId'] and event['digest']==cp['digest'], 'Event checkpoint binding')
            check(event['pendingReceipt']==call['receipt']['admissionId'], 'Event receipt binding')
        physical_match=[p for p in physical if p.get('readEvent')==read['sequence']]
        check(len(physical_match)==1, 'Missing fresh SQL read record')
        check(physical_match[0]['readerIdentity']['backend_pid']==read['backendPid'], 'SQL event backend binding')
        response_events=[e for e in row['events'] if e['type']=='synthetic-response-computation' and e.get('completeEntry')==entry['sequence']]
        check(len(response_events)==1 and response_events[0]['sequence']>read['sequence'], 'Response computed before committed read')
    if refusal:
        check(row['repeatCounts']['before']==row['repeatCounts']['after'], 'Repeat performed guarded work')

check(len(cases)==6,'Missing cases')
result={'status':'PASS','checks':checks,'snapshots':snapshots,'cases':cases,'scope':'Lead independent Python reconstruction of synthetic logical checkpoints, SQL physical state wrapper and ordered observations; not runtime rerun, process, fence or fullTR/global acceptance.'}
print(json.dumps(result,indent=2))
