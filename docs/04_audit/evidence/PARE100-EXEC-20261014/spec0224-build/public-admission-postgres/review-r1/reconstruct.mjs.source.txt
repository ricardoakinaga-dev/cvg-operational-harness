function canonical(v){
 if(v===null||typeof v!=='object')return JSON.stringify(v);
 const pieces=[];
 if(Array.isArray(v)){for(const x of v)pieces.push(canonical(x));return '['+pieces.join(',')+']';}
 for(const k of Object.keys(v).sort())pieces.push(JSON.stringify(k)+':'+canonical(v[k]));
 return '{'+pieces.join(',')+'}';
}

function restore(s){

 assert.equal(p.field,'state.resolvedInputs.fixturePadding');
 const pad=p.character.repeat(p.repeat)+p.suffix;
 assert.equal(H(pad),p.sha256);
 if(p.present)cp.state.resolvedInputs.fixturePadding=pad;
 else assert.equal(pad,'');
 const {checkpointId,createdAt,digest,...unsigned}=cp;
 const raw=canonical(unsigned);
 assert.equal(bytes(raw),s.logicalUnsignedBytes);

 assert.equal(H('cvg.checkpoint.v2\n'+raw),digest);
 assert.equal(MAX-bytes(raw),s.remainingBytes);
 assert.ok(bytes(raw)<=MAX);
 snapshotCount++;
 return cp;
}

 assert.equal(data.length,4*Math.ceil(bytes(state)/3));

 assert.equal(bytes(wire),w.sqlTextBytes);assert.equal(bytes(wire),p.rowHeader.physical_state_bytes);
 assert.equal(H(wire),w.sqlTextSha256);assert.equal(H(canonical(wrapper)),w.canonicalWrapperSha256);
 assert.equal(data.length,w.base64Chars);assert.equal(H(data),w.base64Sha256);
 assert.equal(bytes(state),w.expectedStateCanonicalBytes);assert.equal(H(state),w.expectedStateCanonicalSha256);
 const map={tenant_id:'tenantId',execution_id:'executionId',checkpoint_id:'checkpointId',checkpoint_version:'checkpointVersion',runtime_profile:'runtimeProfile',runtime_version:'runtimeVersion',orchestrator_version:'orchestratorVersion',step_number:'stepNumber',budget_usage:'budgetUsage',digest:'digest',created_at:'createdAt'};

}
