# Independent encoding for this ASCII-key, binary-exact-cost synthetic packet.
# Not a general replacement for the native strict JSON serializer.
def canon(v):return json.dumps(v,sort_keys=True,separators=(',',':'),ensure_ascii=False,allow_nan=False)
def digest(v):return hashlib.sha256(canon(v).encode()).hexdigest()
def sha(b):return hashlib.sha256(b).hexdigest()
def reqtokens(v):return math.ceil(len(canon(v).encode())/64)
def unsigned(cp):return {k:v for k,v in cp.items() if k not in ('checkpointId','createdAt','digest')}
def cpseal(cp):return sha(('cvg.checkpoint.v2\n'+canon(unsigned(cp))).encode())

 raw=canon(unsigned(cp)).encode();check(len(raw)==s['logicalUnsignedBytes'],'independent checkpoint byte count');check(MAX-len(raw)==s['remainingBytes'],'remaining byte count');check(len(raw)<=MAX,'native byte cap');check(cpseal(cp)==cp['digest'],'independent checkpoint digest');return cp
def receipts(cp):return cp['state']['toolResultContext']['decisionAdmissions']
def results(cp):return [q for q in cp['state']['observations'] if q['type']=='TOOL_RESULT']
def projection(pre,m,record=None):
 cp=copy.deepcopy(pre);ctx=cp['state']['toolResultContext'];step=cp['state']['stepNumber']+1
 record=record or {'admissionId':'admission_'+'0'*36,'tenantId':cp['tenantId'],'executionId':cp['executionId'],'stepNumber':step,'attempt':sum(r['stepNumber']==step for r in receipts(cp)),'purpose':m['request']['purpose'],'requestSha256':digest(m['request']),'measurement':m['measurement'],'reserved':{'modelCalls':1,'inputTokens':m['measurement']['inputUpperBoundTokens'],'outputTokens':512,'costUsd':m['measurement']['costUpperBoundUsd']},'status':'RESERVED'}
 # Native post-tool evaluation runs after the last persisted post-tool snapshot.
 # Independently reconstruct its sole deterministic value before the second seed reserve.
 if not m['resumed'] and results(cp) and 'lastEvaluation' not in cp['state']:
  cp['state']['lastEvaluation']={'outcome':'INCOMPLETE','reasonCode':'EVIDENCE_INCOMPLETE','deterministic':True,'detail':'Decision CALL_TOOL requires another loop step.'}
 ctx['decisionAdmissions'].append(copy.deepcopy(record));ctx['pendingDecisionAdmission']=record['admissionId']
 for k,v in record['reserved'].items():cp['budgetUsage'][k]+=v
 if record['purpose']=='REPAIR':cp['budgetUsage']['decisionRepairs']+=1
