const fs=require('node:fs'),crypto=require('node:crypto'),Module=require('node:module');
const seen=new Set(),original=Module._extensions['.js'];
Module._extensions['.js']=function(m,f){if(!seen.has(f)){seen.add(f);const b=fs.readFileSync(f);fs.appendFileSync(process.env.I1_LOADED_FILES,JSON.stringify({path:f,bytes:b.length,sha256:crypto.createHash('sha256').update(b).digest('hex')})+'\n');}return original(m,f);};
