const fs=require('node:fs'),crypto=require('node:crypto'),Module=require('node:module'),{fileURLToPath}=require('node:url');
const seen=new Set();
Module.registerHooks({load(url,ctx,next){if(url.startsWith('file:')){const f=fileURLToPath(url);if(!seen.has(f)){seen.add(f);const b=fs.readFileSync(f);fs.appendFileSync(process.env.I1_LOADED_FILES,JSON.stringify({path:f,bytes:b.length,sha256:crypto.createHash('sha256').update(b).digest('hex'),phase:'before-module-load'})+'\n');}}return next(url,ctx);}});
