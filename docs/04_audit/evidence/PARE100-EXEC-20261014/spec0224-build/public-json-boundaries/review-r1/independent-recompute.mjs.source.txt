function checkpoint(cp,label){
 let {checkpointId,createdAt,digest,...unsigned}=cp;
 check(digest===sha('cvg.checkpoint.v2\n'+canon(unsigned)),label+': logical checkpoint digest');

}
function physical(snapshot,label){
 const cp=snapshot.checkpoint;checkpoint(cp,label);
 check(snapshot.rows.checkpoints.length===1,label+': one physical checkpoint');
 const p=snapshot.rows.checkpoints[0],w=p.state;
 check(w.storageCodec==='cvg.checkpoint-state/2'&&w.encoding==='base64',label+': physical storage codec');
 const b=Buffer.from(w.data,'base64');const raw=b.toString('utf8');
 check(b.toString('base64')===w.data,label+': base64 exact');
 check(b.length===w.decodedBytes&&sha(b)===w.decodedSha256,label+': physical bytes and hash');
 check(canon(JSON.parse(raw))===raw,label+': physical state canonical bytes');
 check(raw===canon(cp.state),label+': physical and logical state agree');
 check(canon(p.budget_usage)===canon(cp.budgetUsage),label+': physical and logical budget agree');
 check(p.digest===cp.digest&&p.checkpoint_id===cp.checkpointId,label+': physical identity and digest');
 return {codec:w.storageCodec,decodedBytes:b.length,decodedSha256:sha(b),checkpointDigest:cp.digest};
}
function accounting(cp,raw,casTrace,label){
 const ctx=cp.state.toolResultContext,receipts=ctx.decisionAdmissions;
 check(ctx.pendingDecisionAdmission===null,label+': no pending reserve');
 check(receipts.length===raw.requests.length&&receipts.length===raw.responses.length,label+': calls equal receipts');
 const totals={modelCalls:0,inputTokens:0,outputTokens:0,costUsd:0};let admissions=[];
 for(let i=0;i<receipts.length;i++){
  const r=receipts[i],request=raw.requests[i],response=raw.responses[i];
  const requestHash=sha(canon(request));
  check(r.status==='SETTLED',label+': settled '+i);
  check(r.executionId===cp.executionId&&r.tenantId===cp.tenantId,label+': binding '+i);
  check(r.requestSha256===requestHash,label+': request hash '+i);
  check(r.resultSha256===sha(canon(response)),label+': native result hash '+i);
  check(request.maxOutputTokens===512&&r.reserved.outputTokens===512,label+': output cap '+i);
  const actual={modelCalls:1,inputTokens:response.inputTokens,outputTokens:response.outputTokens,costUsd:response.costUsd};
  check(canon(r.actual)===canon(actual),label+': receipt actual '+i);
  check(r.measurement.inputUpperBoundTokens===Math.ceil(Buffer.byteLength(canon(request))/64),label+': toy64 exact '+i);
  check(response.inputTokens===r.measurement.inputUpperBoundTokens&&response.outputTokens<=512,label+': measured native usage '+i);
  check(raw.counters.some(c=>canon(c.request)===canon(request)&&canon(c.measurement)===canon(r.measurement)),label+': actual request equals counted request '+i);
  const applied=casTrace.filter(t=>t.result?.status==='APPLIED');
  const reserved=applied.findIndex(t=>t.input.next.state.toolResultContext.decisionAdmissions.some(x=>x.admissionId===r.admissionId&&x.status==='RESERVED'));
  const settled=applied.findIndex(t=>t.input.next.state.toolResultContext.decisionAdmissions.some(x=>x.admissionId===r.admissionId&&x.status==='SETTLED'));
  check(reserved>=0&&settled>reserved,label+': durable CAS reserve then settle '+i);
  admissions.push({admissionId:r.admissionId,requestHash,resultHash:r.resultSha256,reservedCASIndex:reserved,settledCASIndex:settled,actual});
  for(const k of Object.keys(totals))totals[k]+=actual[k];
 }
 for(const k of Object.keys(totals))check(cp.budgetUsage[k]===totals[k],label+': exact usage total '+k);
 return {totals,admissions};
}
