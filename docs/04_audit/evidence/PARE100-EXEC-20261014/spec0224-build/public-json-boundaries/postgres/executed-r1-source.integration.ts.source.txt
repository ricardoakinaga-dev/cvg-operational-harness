// Independent oracle, called ONLY for known-valid fixtures / parsed model frames.
// Unsafe native body outputs are never passed to this or JSON.stringify.
function golden(value: unknown): string {
  if (value === null || typeof value !== 'object') return JSON.stringify(value)
  if (Array.isArray(value)) return '[' + value.map(golden).join(',') + ']'
  return (
    '{' +
    Object.keys(value)
      .sort()
      .map(
        (key) =>
          JSON.stringify(key) +
          ':' +
          golden((value as Record<string, unknown>)[key])
      )
      .join(',') +
    '}'
  )
}

async function snapshot(
  connection: Pool,
  store: PostgresExecutionStepStore,
  executionId: string
) {
  const rows = await withTenantContext(connection, tenantId, async (c) => ({
    execution: (
      await c.query(
        'SELECT * FROM operational_executions WHERE tenant_id=$1 AND id=$2',
        [tenantId, executionId]
      )
    ).rows,
    checkpoints: (
      await c.query(
        'SELECT * FROM operational_execution_checkpoints WHERE tenant_id=$1 AND execution_id=$2',
        [tenantId, executionId]
      )
    ).rows,
    steps: (
      await c.query(
        'SELECT * FROM operational_execution_steps WHERE tenant_id=$1 AND execution_id=$2 ORDER BY step_number',
        [tenantId, executionId]
      )
    ).rows,
    journal: (
      await c.query(
        'SELECT * FROM operational_effect_journal WHERE tenant_id=$1 AND operation_key LIKE $2 ORDER BY operation_key',
        [tenantId, executionId + ':%']
      )
    ).rows
  }))
  return {
    rows,
    checkpoint: await store.loadCheckpoint(tenantId, executionId),
    steps: await store.listSteps(tenantId, executionId)
  }
}

          })
          expect(golden(payload!.output)).toBe(c.canonical)
          expect(payload!.output).toEqual(JSON.parse(c.canonical!))
          expect(obs).toMatchObject({ source: 'TOOL', trust: 'UNTRUSTED' })
          expect(obs!.provenance.operationKey).toBe(journal.operation_key)
          if (c.kind === 'context-overflow') {
            expect(counts.model).toBe(1)
            expect((first as { stopReason: string }).stopReason).toBe(
              'CONTEXT_OVERFLOW'
            )
            checks.admissionClassification = 'CAPTURE_AVAILABLE_CONTEXT_REFUSED'
          } else {
            expect(counts.model).toBe(2)
            expect(frameValues).toHaveLength(1)
            const observed = frameValues[0] as {
              payload: Record<string, unknown>
            }
            expect(golden(observed.payload.output)).toBe(c.canonical)
            expect(observed.payload.output).toEqual(JSON.parse(c.canonical!))
            expect(observed.payload).toEqual(payload)
            expect((first as { stopReason: string }).stopReason).toBe(
              'COMPLETED'
            )
            expect((first as { response: string }).response).toBe(receipts[0])
            checks.admissionClassification = 'FULL_VALUE_ACTUALLY_DELIVERED'
          }
        }
      }
      // Cross-tenant SELECT and store read controls exercise native tenant context.
      const isolation = await withTenantContext(
        pool!,
        otherTenant,
        async (client) =>
          (
            await client.query(
              'SELECT * FROM operational_effect_journal WHERE operation_key LIKE $1',
              [executionId + ':%']
            )
          ).rows
      )
      expect(isolation).toEqual([])
      expect(
        await initial.store.loadCheckpoint(otherTenant, executionId)
      ).toBeNull()
      checks.crossTenantDenied = true
      expect(after?.checkpoint).toEqual(before.checkpoint)
      checks.checkpointUnchangedOnRerun = true
    } catch (error) {
      checkError = error
      throw error
    } finally {
      const observation = {
        caseId: c.id,
        kind: c.kind,
        executionId,
        counts,
        first,
        rerun,
        checks,
        verdict: checkError
          ? 'RED'
          : c.kind === 'nul'
            ? 'EXPECTED_SAFE_REQUIRED_PATH_FAIL'
            : 'BUILD_TEST_EVIDENCE',
        error: checkError ? errorInfo(checkError) : undefined
      }
      summaries.push(observation)
      await save(join(dir, 'observed.json'), observation)
      await save(join(dir, 'raw-requests-responses.json'), {
        requests,
        responses,
        counters,
        frameValues,
        receipts
      })
      await save(join(dir, 'native-journal-cas-trace.json'), {
        journalTrace,
        casTrace
      })
      await save(join(dir, 'audit-telemetry.json'), {
        audit: audit.events,
        telemetry: telemetry.events
      })
    }
  })
})
