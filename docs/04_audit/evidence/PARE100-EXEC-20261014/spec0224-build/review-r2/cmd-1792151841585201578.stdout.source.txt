   286	  private stateSummary(
   287	    input: ContextBuildInput,
   288	    observations: readonly Observation[]
   289	  ): string {
   290	    const resolved = Object.entries(input.state.resolvedInputs)
   291	      .map(([key, value]) => `${key}=${safeString(value)}`)
   292	      .join(', ')
   293	    const lastObservation = observations[observations.length - 1]
   294	    const parts = [
   295	      `goal=${input.goal}`,
   296	      `step=${input.state.stepNumber}`,
   297	      `observations=${observations.length}`,
   298	      `openQuestions=${input.state.openQuestions.length}`
   299	    ]
   300	    if (resolved) parts.push(`resolved={${resolved}}`)
   301	    if (input.state.pendingApprovalId) {
   302	      parts.push(`pendingApproval=${input.state.pendingApprovalId}`)
   303	    }
   304	    if (input.state.pendingQuestion) {
   305	      parts.push(`pendingQuestion=${input.state.pendingQuestion.promptIntent}`)
   306	    }
   307	    if (input.state.lastEvaluation) {
   308	      parts.push(
   309	        `lastEvaluation=${input.state.lastEvaluation.outcome}/${input.state.lastEvaluation.reasonCode}`
   310	      )
   311	    }
   312	    if (lastObservation) {
   313	      parts.push(
   314	        `lastObservation=${lastObservation.source}/${lastObservation.type}:${lastObservation.summary}`
   315	      )
   316	    }
   317	    return parts.join(' | ')
   318	  }

   320	  private trim(items: readonly ContextItem[]): ContextItem[] {
   321	    const selected: ContextItem[] = []
   322	    let used = 0
   323	    for (const priority of PRIORITY_ORDER) {
   324	      const group = items.filter((entry) => entry.priority === priority)
   325	      for (const entry of group) {
   326	        const mandatory = MANDATORY_PRIORITIES.has(entry.priority)
   327	        if (!mandatory && used + entry.tokenEstimate > this.tokenBudget) {
   328	          continue
   329	        }
   330	        selected.push(entry)
   331	        used += entry.tokenEstimate
   332	      }
   333	    }
   334	    return selected
   335	  }

   353	            ...(currentToken ? { currentToken } : {})
   354	          }
   355	        }
   356	        validateCompleteCheckpointTransition(current, next)
   357	        await guard?.recheck()
   358	        const result =
   359	          expected === null
   360	            ? await client.query(
   361	                `INSERT INTO operational_execution_checkpoints
   362	          (tenant_id,execution_id,checkpoint_id,checkpoint_version,runtime_profile,runtime_version,
   363	           orchestrator_version,step_number,state,budget_usage,digest,created_at)
   364	          VALUES ($1,$2,$3,$4,$5,$6,$7,$8,$9::jsonb,$10::jsonb,$11,$12)
   365	          ON CONFLICT (tenant_id,execution_id) DO NOTHING RETURNING checkpoint_id`,
   366	                values
   367	              )
   368	            : await client.query(
   369	                `UPDATE operational_execution_checkpoints SET
   370	          checkpoint_id=$3,checkpoint_version=$4,runtime_profile=$5,runtime_version=$6,
   371	          orchestrator_version=$7,step_number=$8,state=$9::jsonb,budget_usage=$10::jsonb,digest=$11,created_at=$12
   372	          WHERE tenant_id=$1 AND execution_id=$2 AND checkpoint_id=$13 AND digest=$14
   373	          RETURNING checkpoint_id`,
   374	                [...values, expected.checkpointId, expected.digest]
   375	              )
   376	        await guard?.recheck()
   377	        return result.rows.length
   378	          ? {
   379	              status: 'APPLIED',
   380	              token: { checkpointId: next.checkpointId, digest: next.digest }
   381	            }
   382	          : { status: 'CONFLICT' }
   383	      })
   384	    )
   385	  }

   387	  public loadCheckpoint(
   388	    tenantId: string,
   389	    executionId: string
   390	  ): Promise<ExecutionCheckpoint | null> {
   391	    const tenant = TenantIdSchema.parse(tenantId)
   392	    const id = text(executionId, 'executionId')
   393	    return this.run(tenant, async (client) => {
   394	      const result = await client.query<CheckpointRow>(
   395	        `SELECT ${boundedCheckpointColumns}
   396	           FROM operational_execution_checkpoints
   397	          WHERE tenant_id = $1 AND execution_id = $2`,
   398	        [tenant, id]
   399	      )
   400	      const row = result.rows[0]
   401	      if (!row) return null
   402	      const checkpoint = mapCheckpoint(row)
   403	      validateCheckpointIntegrity(checkpoint, { tenantId, executionId: id })
   404	      return checkpoint
   405	    })
   406	  }

   408	  private async run<T>(
   409	    tenant: TenantId,
   410	    operation: (client: PostgresQueryable) => Promise<T>
   411	  ): Promise<T> {
   412	    if (isPool(this.connection)) {
   413	      return withTenantContext(
   414	        this.connection,
   415	        tenant,
   416	        (client: PostgresPoolClient) => operation(client)
   417	      )
   418	    }
   419	    return operation(this.connection)
   420	  }
