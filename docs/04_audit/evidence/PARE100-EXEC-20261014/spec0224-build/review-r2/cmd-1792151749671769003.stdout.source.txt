   163	export function applyResume(
   164	  ctx: IterativeDispatchContext,
   165	  run: LoopRun,
   166	  checkpoint: ExecutionCheckpoint | null
   167	): Extract<DispatchOutcome, { kind: 'stop' }> | null {
   168	  const resume = run.input.resume
   169	  if (!resume) return null
   170	  if (!checkpoint) {
   171	    return {
   172	      kind: 'stop',
   173	      stopReason: 'STATE_CONFLICT',
   174	      response: 'Resume input arrived without a durable checkpoint.'
   175	    }
   176	  }
   177	  if (resume.kind === 'approval') {
   178	    if (run.state.pendingQuestion) {
   179	      // Wrong-kind resume: the execution is paused for an answer, not for an
   180	      // approval. Never let this open the loop with a question unresolved.
   181	      return {
   182	        kind: 'stop',
   183	        stopReason: 'STATE_CONFLICT',
   184	        response: 'Approval resume arrived while a user question is pending.',
   185	        preserveCheckpoint: true
   186	      }
   187	    }
   188	    if (!run.state.pendingApprovalId) {
   189	      if (run.state.stopReason === 'APPROVAL_REQUIRED') {
   190	        return {
   191	          kind: 'stop',
   192	          stopReason: 'STATE_CONFLICT',
   193	          response: 'The paused approval binding is missing.',
   194	          preserveCheckpoint: true
   195	        }
   196	      }
   197	      // The approval was already consumed and the checkpoint cleared the
   198	      // binding; a retry of the same attempt is a safe no-op.
   199	      return null
   200	    }
   201	    if (run.state.pendingApprovalId !== resume.approvalId) {
   202	      return {
   203	        kind: 'stop',
   204	        stopReason: 'STATE_CONFLICT',
   205	        response: 'Approval does not match the paused execution binding.',
   206	        preserveCheckpoint: true
   207	      }
   208	    }
   209	    return null
   210	  }
   211	  if (run.state.pendingApprovalId) {
   212	    return {
   213	      kind: 'stop',
   214	      stopReason: 'STATE_CONFLICT',
   215	      response: 'User input arrived while an approval is pending.',
   216	      preserveCheckpoint: true
   217	    }
   218	  }
   219	  if (!run.state.pendingQuestion) {
   220	    const alreadyResolved = Object.values(run.state.resolvedInputs).includes(
   221	      resume.message
   222	    )
   223	    if (alreadyResolved) {
   224	      if (ctx.options.toolResultContext)
   225	        run.resumeObservation = userResumeObservation(ctx, run, resume.message)
   226	      return null
   227	    }
   228	    return {
   229	      kind: 'stop',
   230	      stopReason: 'STATE_CONFLICT',
   231	      response: 'User input arrived without a pending question.',
   232	      preserveCheckpoint: true
   233	    }
   234	  }
   235	  const stepNumber = Math.max(1, run.state.stepNumber)
   236	  const observation = userResumeObservation(ctx, run, resume.message)
   237	  if (ctx.options.toolResultContext) run.resumeObservation = observation
   238	  run.observations = ctx.options.toolResultContext
   239	    ? retainCompleteObservations([...run.observations, observation])
   240	    : [...run.observations, observation].slice(-MAX_CHECKPOINT_OBSERVATIONS)
   241	  const pendingQuestion = run.state.pendingQuestion
   242	  run.state = {
   243	    ...run.state,
   244	    pendingQuestion: undefined,
   245	    openQuestions: run.state.openQuestions.filter(
   246	      (question) => question !== pendingQuestion.promptIntent
   247	    ),
   248	    resolvedInputs: {
   249	      ...run.state.resolvedInputs,
   250	      [`answer_${stepNumber}`]: resume.message
   251	    }
   252	  }
   253	  return null
   254	}

   256	function userResumeObservation(
   257	  ctx: IterativeDispatchContext,
   258	  run: LoopRun,
   259	  message: string
   260	): Observation {
   261	  const stepNumber = Math.max(1, run.state.stepNumber)
   262	  return ctx.makeObservation(run, {
   263	    stepId: `step_${run.executionId}_${stepNumber}_user_input`,
   264	    stepNumber,
   265	    type: 'USER_MESSAGE',
   266	    source: 'USER',
   267	    trust: 'UNTRUSTED',
   268	    payload: { message },
   269	    summary: 'User answered the pending question.',
   270	    provenance: { sourceId: 'user' }
   920	            },
   921	            startedAtMs
   922	          )
   923	        }
   924	        await this.persistCheckpoint(run)
   925	      }
   926	      await this.safeAudit(run, 'runtime_v2.started', {
   927	        result: checkpoint ? 'resumed' : 'started'
   928	      })
   929	      const outcome = await this.runLoop(run)
   930	      return await this.finalize(run, outcome, startedAtMs)
   931	    } catch (error) {
   932	      return await this.finalize(
   933	        run,
   934	        {
   935	          kind: 'stop',
   936	          stopReason:
   937	            error instanceof ToolResultContextError
   938	              ? error.stopReason
   939	              : 'INTERNAL_FAILURE',
   940	          response:
   941	            error instanceof ToolResultContextError
   942	              ? `${error.code}: ${error.message}`
   943	              : `Execution failed internally: ${errorMessage(error)}.`,
   944	          ...(run.checkpointConflict ? { preserveCheckpoint: true } : {})
   945	        },
   946	        startedAtMs
   947	      )
   948	    }

   951	  private rejectCheckpoint(
   952	    checkpoint: ExecutionCheckpoint,
   953	    input: RuntimeInput
   954	  ): string | null {
   955	    if (checkpoint.checkpointVersion === 2 && !this.options.toolResultContext) {
   956	      return 'Complete checkpoint requires the explicitly configured complete-tool-v1 mode.'
   957	    }
   958	    if (checkpoint.runtimeProfile !== 'iterative') {
   959	      return 'Execution checkpoint belongs to a different runtime profile.'
   960	    }
   961	    if (checkpoint.runtimeVersion !== RUNTIME_V2_VERSION) {
   962	      return `Execution checkpoint requires runtime ${checkpoint.runtimeVersion}, this runtime is ${RUNTIME_V2_VERSION}.`
   963	    }
   964	    if (input.runtimeProfile && input.runtimeProfile !== 'iterative') {
   965	      return 'Execution checkpoint conflicts with the requested runtime profile.'
   966	    }
   967	    if (
   968	      this.options.capabilityFingerprint &&
   969	      checkpoint.state.capabilityFingerprint !==
   970	        this.options.capabilityFingerprint
   971	    ) {
   972	      return 'Execution checkpoint belongs to a different capability composition.'
   973	    }
   974	    if (
   975	      checkpoint.state.stopReason &&
   976	      TERMINAL_STOPS.has(checkpoint.state.stopReason) &&
   977	      checkpoint.state.stopReason !== 'NEEDS_USER_INPUT' &&
   978	      checkpoint.state.stopReason !== 'APPROVAL_REQUIRED'
   979	    ) {
   980	      return 'Terminal execution cannot be resumed.'
   981	    }
   982	    return null
   983	  }

   985	  private applyResume(
   986	    run: LoopRun,
   987	    checkpoint: ExecutionCheckpoint | null
   988	  ): Extract<DispatchOutcome, { kind: 'stop' }> | null {
   989	    return applyResume(this.dispatchContext, run, checkpoint)
   990	  }

   992	  private async runLoop(
   993	    run: LoopRun
   994	  ): Promise<Extract<DispatchOutcome, { kind: 'stop' }>> {
   995	    if (run.state.pendingQuestion) {
   996	      // Deterministic rule: a pending question must not be answered by a new
   997	      // model decision. Re-pause without spending a model call.
   998	      return this.pause(run, Math.max(1, run.state.stepNumber), {
   999	        kind: 'pause',
  1000	        pausedKind: 'NEEDS_USER_INPUT',
  1001	        decision: null
  1002	      })
  1003	    }
  1004	    while (true) {
  1005	      // I8: cancellation is honored before every step.
  1006	      if (this.#turns.get(run)?.signal.aborted) {
  1007	        return this.stop('CANCELLED', CANCELLED_RESPONSE)
  1008	      }
  1009	      // I12: pause (and any other step control) is checked before each step.
  1010	      const beforeStep = await runGate(
  1011	        this.kernelFor(run),
  1012	        'turn/before-step',
  1013	        this.turnFor(run)
  1014	      )
  1015	      if (beforeStep.kind === 'stop') {
  1016	        // An operator pause is not terminal: the durable checkpoint, with any
  1017	        // pending approval or question, stays resumable (AUD-0597 F03).
  1018	        return {
  1019	          ...this.stop(beforeStep.stopReason, beforeStep.response),
  1020	          /* v8 ignore next -- the pause control is the only before-step control */
  1021	          ...(beforeStep.cause === 'operator_paused'
  1022	            ? { preserveCheckpoint: true }
  1023	            : {})
  1024	        }
  1025	      }
  1026	      const budgetStop = this.checkBudget(run)
  1027	      if (budgetStop) return this.stop(budgetStop.reason, budgetStop.response)
  1028	
  1029	      let decision: LoopDecision
  1030	      let stepNumber: number
  1031	      let decisionAdmissionIds: readonly string[] | undefined
  1032	      let modelDecision = false
  1033	
  1034	      if (run.state.pendingDecision && run.state.pendingStepNumber) {
  1035	        decision = run.state.pendingDecision
  1036	        stepNumber = run.state.pendingStepNumber
  1037	        run.lastDecision = decision
  1038	      } else if (
  1039	        this.options.toolResultContext &&
  1040	        run.state.toolResultContext?.acceptedDecision
  1041	      ) {
  1042	        const accepted = run.state.toolResultContext.acceptedDecision
  1043	        if (accepted.stepNumber !== run.stepNumber + 1)
  1044	          throw new ToolResultContextError(
  1045	            'STATE_CONFLICT',
  1046	            'Accepted decision is bound to another step.',
  1047	            'STATE_CONFLICT'
  1048	          )
  1049	        decision = accepted.decision
  1050	        stepNumber = accepted.stepNumber
  1051	        decisionAdmissionIds = accepted.decisionAdmissionIds
  1052	        run.lastDecision = decision
  1053	      } else {
  1054	        stepNumber = run.stepNumber + 1
  1055	        if (stepNumber > run.input.budget.maxSteps) {
  1056	          return this.stop('MAX_STEPS', 'Step budget exhausted.')
  1057	        }
  1058	        const decisionResult = await this.decide(run, stepNumber)
  1059	        if (decisionResult.outcome) return decisionResult.outcome
  1060	        decision = sanitizeLoopDecision(decisionResult.decision)
  1061	        decisionAdmissionIds = decisionResult.decisionAdmissionIds
  1062	        modelDecision = decisionResult.modelDecision ?? false
  1063	        run.lastDecision = decision
  1064	      }
  1065	
  1066	      const validation = this.validateDecision(run, decision)
  1067	      if (!validation.valid) {
  1068	        if (!this.options.toolResultContext) run.usage.decisionRepairs += 1
  1069	        run.lastDecisionErrors = validation.errors
  1070	        const repairBudget = run.input.budget.maxDecisionRepairs ?? 1
