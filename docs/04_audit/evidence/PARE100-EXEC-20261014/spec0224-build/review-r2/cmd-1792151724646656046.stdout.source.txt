   308	function validateState(
   309	  state: AgentLoopState,
   310	  binding?: { tenantId: string; executionId: string }
   311	): void {
   312	  const ctx = state.toolResultContext!
   313	  if (
   314	    !state.resolvedInputs ||
   315	    Array.isArray(state.resolvedInputs) ||
   316	    typeof state.resolvedInputs !== 'object'
   317	  )
   318	    fail()
   319	  if (
   320	    state.pendingStepNumber !== undefined &&
   321	    state.pendingStepNumber !== state.stepNumber + 1 &&
   322	    state.pendingStepNumber !== state.stepNumber
   323	  )
   324	    fail('Invalid pending step')
   325	  if (state.observations.length > MAX_CHECKPOINT_OBSERVATIONS)
   326	    fail('Complete checkpoint observation capacity exceeded')
   327	  const observations = new Set<string>(),
   328	    toolSteps = new Set<number>()
   329	  const executionId =
   330	    binding?.executionId ??
   331	    state.observations[0]?.executionId ??
   332	    ctx.decisionAdmissions[0]?.executionId
   333	  const tenantId = binding?.tenantId ?? ctx.decisionAdmissions[0]?.tenantId
   334	  for (const item of state.observations) {
   335	    if (
   336	      observations.has(item.observationId) ||
   337	      item.stepNumber >
   338	        Math.max(state.stepNumber, state.pendingStepNumber ?? 0) ||
   339	      item.executionId !== executionId
   340	    )
   341	      fail('Observation binding mismatch')
   342	    observations.add(item.observationId)
   343	    if (item.type !== 'TOOL_RESULT') continue
   344	    if (
   345	      item.source !== 'TOOL' ||
   346	      item.trust !== 'UNTRUSTED' ||
   347	      item.stepNumber < 1
   348	    )
   349	      fail('Invalid tool observation')
   350	    toolSteps.add(item.stepNumber)
   351	    const payload = item.payload as Record<string, unknown>
   352	    if (payload?.completion === 'COMPLETE') {
   353	      succeeded(payload)
   354	      const raw = canonicalizeCompleteJson(payload.output)
   355	      if (
   356	        payload.outputBytes !== Buffer.byteLength(raw) ||
   357	        payload.outputSha256 !== sha(raw)
   358	      )
   359	        fail('Complete output integrity mismatch')
   360	    } else if (payload?.completion === 'UNAVAILABLE') unavailable(payload)
   361	    else failed(payload)
   362	  }
   363	  if (
   364	    ctx.deliveredThroughStep > state.stepNumber ||
   365	    (ctx.deliveredThroughStep !== 0 && !toolSteps.has(ctx.deliveredThroughStep))
   366	  )
   367	    fail('Invalid delivery cursor')
   368	  const admissions = new Map<string, (typeof ctx.decisionAdmissions)[number]>(),
   369	    attempts = new Set<string>()
   370	  let pending: string | null = null
   371	  for (const entry of ctx.decisionAdmissions) {
   372	    if (
   373	      admissions.has(entry.admissionId) ||
   374	      entry.stepNumber > state.stepNumber + 1 ||
   375	      entry.tenantId !== tenantId ||
   376	      entry.executionId !== executionId
   377	    )
   378	      fail('Admission binding mismatch')
   379	    const attempt = `${entry.stepNumber}:${entry.attempt}`
   380	    if (attempts.has(attempt)) fail('Duplicate admission attempt')
   381	    attempts.add(attempt)
   382	    admissions.set(entry.admissionId, entry)
   383	    const m = entry.measurement,
   384	      r = entry.reserved
   385	    if (
   386	      r.modelCalls !== 1 ||
   387	      r.inputTokens !== m.inputUpperBoundTokens ||
   388	      r.outputTokens !== 512 ||
   389	      r.costUsd !== m.costUpperBoundUsd ||
   390	      m.inputUpperBoundTokens + 512 > m.contextWindowTokens
   391	    )
   392	      fail('Invalid admission reservation')
   393	    if (entry.status === 'RESERVED') {
   394	      if (
   395	        pending !== null ||
   396	        entry.actual !== undefined ||
   397	        entry.unknownReason !== undefined ||
   398	        entry.resultSha256 !== undefined
   399	      )
   400	        fail('Invalid pending admission')
   401	      pending = entry.admissionId
   402	    } else if (entry.status === 'SETTLED') {
   403	      if (
   404	        !entry.actual ||
   405	        !entry.resultSha256 ||
   406	        entry.unknownReason !== undefined ||
   407	        entry.actual.modelCalls !== 1 ||
   408	        entry.actual.inputTokens > r.inputTokens ||
   409	        entry.actual.outputTokens > r.outputTokens ||
   410	        entry.actual.costUsd > r.costUsd
   411	      )
   412	        fail('Invalid settled admission')
   413	    } else if (
   414	      !entry.unknownReason ||
   415	      (entry.actual !== undefined &&
   416	        entry.unknownReason !== 'ADMISSION_BREACH') ||
   417	      entry.resultSha256 !== undefined
   418	    )
   419	      fail('Invalid unknown admission')
   420	  }
   421	  if (pending !== ctx.pendingDecisionAdmission)
   422	    fail('Pending admission does not match ledger')
   423	  const accepted = ctx.acceptedDecision
   424	  if (accepted) {
   425	    if (
   426	      accepted.stepNumber !== state.stepNumber + 1 &&
   427	      accepted.stepNumber !== state.stepNumber
   428	    )
   429	      fail('Accepted decision step mismatch')
   430	    if (!validateLoopDecision(accepted.decision).valid || pending !== null)
   431	      fail('Invalid accepted decision')
   432	    const ids = new Set(accepted.decisionAdmissionIds)
   433	    if (ids.size !== accepted.decisionAdmissionIds.length)
   434	      fail('Duplicate accepted receipt')
   435	    for (const admissionId of ids) {
   436	      const entry = admissions.get(admissionId)
   437	      if (
   438	        !entry ||
   439	        entry.status !== 'SETTLED' ||
   440	        entry.stepNumber !== accepted.stepNumber
   441	      )
   442	        fail('Accepted receipt binding mismatch')
   443	    }
   444	    const nativeIds = ctx.decisionAdmissions
   445	      .filter((r) => r.stepNumber === accepted.stepNumber)
   446	      .map((r) => r.admissionId)
   447	    if (
   448	      nativeIds.length !== ids.size ||
   449	      nativeIds.some((admissionId) => !ids.has(admissionId))
   450	    )
   451	      fail('Accepted decision omits native receipt')
   452	    if (
   453	      state.pendingDecision &&
   454	      (state.pendingStepNumber !== accepted.stepNumber ||
   455	        canonicalizeCompleteJson(state.pendingDecision, LIMITS) !==
   456	          canonicalizeCompleteJson(accepted.decision, LIMITS))
   457	    )
   458	      fail('Pending and accepted decisions disagree')
   459	  }
   460	}

   463	  preflight(value)
   464	  const state = stateSchema(value) as AgentLoopState
   465	  validateState(state)
   466	  canonicalizeCompleteJson(state, LIMITS)
   467	  return state

   470	function normalizedUnsigned(value: unknown): UnsignedCheckpoint {
   471	  preflight(value)
   472	  const unsigned = unsignedSchema(value) as UnsignedCheckpoint
   473	  validateState(unsigned.state, unsigned)
   474	  canonicalizeCompleteJson(unsigned, LIMITS)
   475	  if (
   476	    unsigned.state.stepNumber !== unsigned.stepNumber ||
   477	    unsigned.budgetUsage.steps < unsigned.stepNumber
   478	  )
   479	    fail('Checkpoint step mismatch')
   480	  const sum = { modelCalls: 0, inputTokens: 0, outputTokens: 0, costUsd: 0 }
   481	  let repairs = 0
   482	  for (const r of unsigned.state.toolResultContext!.decisionAdmissions) {
   483	    for (const key of Object.keys(sum) as (keyof typeof sum)[]) {
   484	      sum[key] +=
   485	        r.status === 'SETTLED'
   486	          ? r.actual![key]
   487	          : Math.max(r.reserved[key], r.actual?.[key] ?? 0)
   488	    }
   489	    if (r.purpose === 'REPAIR') repairs++
   490	  }
   491	  for (const key of Object.keys(sum) as (keyof typeof sum)[]) {
   492	    if (unsigned.budgetUsage[key] + (key === 'costUsd' ? 1e-12 : 0) < sum[key])
   493	      fail('Budget does not cover admission ledger')
   494	  }
   495	  if (unsigned.budgetUsage.decisionRepairs < repairs)
   496	    fail('Repair budget does not cover ledger')
   497	  return unsigned
   498	}

   521	/** Reject excessive nesting/node counts before JSON.parse allocates a tree. */
   522	function boundedJsonText(raw: string): void {
   523	  let inString = false,
   524	    escaped = false,
   525	    nodes = 0,
   526	    inAtom = false,
   527	    stringStart = 0
   528	  const stack: (Set<string> | null)[] = []
   529	  const value = (): void => {
   530	    if (stack.length > LIMITS.maxDepth || ++nodes > LIMITS.maxNodes)
   531	      fail('JSON complexity limit')
   532	  }
   533	  for (let i = 0; i < raw.length; i++) {
   534	    const char = raw[i]!
   535	    if (inString) {
   536	      if (escaped) escaped = false
   537	      else if (char === '\\') escaped = true
   538	      else if (char === '"') {
   539	        inString = false
   540	        let next = i + 1
   541	        while (next < raw.length && /\s/.test(raw[next]!)) next++
   542	        if (raw[next] === ':') {
   543	          const keys = stack[stack.length - 1]
   544	          if (!keys) fail('Invalid JSON object key')
   545	          const key = JSON.parse(raw.slice(stringStart, i + 1)) as string
   546	          if (keys!.has(key)) fail('Duplicate JSON key')
   547	          keys!.add(key)
   548	          if (keys!.size > LIMITS.maxNodes) fail('JSON node limit')
   549	        } else value()
   550	      }
   551	      continue
   552	    }
   553	    if (char === '"') {
   554	      inString = true
   555	      stringStart = i
   556	      inAtom = false
   557	    } else if (char === '{' || char === '[') {
   558	      value()
   559	      stack.push(char === '{' ? new Set() : null)
   560	      inAtom = false
   561	    } else if (char === '}' || char === ']') {
   562	      stack.pop()
   563	      inAtom = false
   564	    } else if (char === ',' || char === ':' || /\s/.test(char)) inAtom = false
   565	    else if (!inAtom) {
   566	      value()
   567	      inAtom = true
   568	    }
   569	  }
   570	}

   572	/** Bound work before schema normalization. This is a lower-bound measurement,
   573	 * not validation: only the schema may omit an undefined optional field. */
   574	function preflight(value: unknown): void {
   575	  let nodes = 0,
   576	    bytes = 0
   577	  const ancestors = new Set<object>()
   578	  const visit = (input: unknown, depth: number): void => {
   579	    if (input === undefined) return
   580	    if (++nodes > LIMITS.maxNodes || depth > LIMITS.maxDepth)
   581	      fail('Checkpoint complexity limit')
   582	    bytes += typeof input === 'string' ? input.length + 2 : 1
   583	    if (bytes > LIMITS.maxBytes) fail('Checkpoint byte limit')
   584	    if (!input || typeof input !== 'object') return
   585	    if (ancestors.has(input)) fail('Checkpoint cycle')
   586	    ancestors.add(input)
   587	    const array = Array.isArray(input)
   588	    if (array && input.length > LIMITS.maxNodes - nodes)
   589	      fail('Checkpoint node limit')
   590	    const keys = Reflect.ownKeys(input)
   591	    if (keys.length > LIMITS.maxNodes + 1) fail('Checkpoint node limit')
   592	    for (const key of keys) {
   593	      if (array && key === 'length') continue
   594	      if (typeof key !== 'string') fail('Checkpoint symbol')
   595	      const descriptor = Object.getOwnPropertyDescriptor(input, key)
   596	      if (!descriptor || !('value' in descriptor)) fail('Checkpoint accessor')
   597	      if (descriptor.value === undefined) continue
   598	      if (!array) bytes += (key as string).length + 3
   599	      if (bytes > LIMITS.maxBytes) fail('Checkpoint byte limit')
   600	      visit(descriptor.value, depth + 1)
   601	    }
   602	    ancestors.delete(input)
   603	  }
   604	  visit(value, 0)
   605	}

   607	export function decodeCompleteCheckpointState(raw: unknown): AgentLoopState {
   608	  try {
   609	    if (typeof raw === 'string') {
   610	      if (Buffer.byteLength(raw) > COMPLETE_CHECKPOINT_STORAGE_MAX_BYTES)
   611	        fail('Physical checkpoint limit')
   612	      boundedJsonText(raw)
   613	      raw = JSON.parse(raw) as unknown
   614	    }
   615	    const wrapper = record({
   616	      storageCodec: one('cvg.checkpoint-state/2'),
   617	      encoding: one('base64'),
   618	      decodedBytes: integer,
   619	      decodedSha256: digest,
   620	      data: str
   621	    })(raw) as CompleteCheckpointStorageState
   622	    if (
   623	      wrapper.decodedBytes > COMPLETE_CHECKPOINT_MAX_BYTES ||
   624	      wrapper.data.length > COMPLETE_CHECKPOINT_DATA_MAX_CHARS
   625	    )
   626	      fail('Physical checkpoint limit')
   627	    if (
   628	      Buffer.byteLength(
   629	        canonicalizeCompleteJson(wrapper, {
   630	          ...LIMITS,
   631	          maxBytes: COMPLETE_CHECKPOINT_STORAGE_MAX_BYTES
   632	        })
   633	      ) > COMPLETE_CHECKPOINT_STORAGE_MAX_BYTES
   634	    )
   635	      fail()
   636	    if (
   637	      wrapper.data.length !== 4 * Math.ceil(wrapper.decodedBytes / 3) ||
   638	      !/^[A-Za-z0-9+/]*={0,2}$/.test(wrapper.data)
   639	    )
   640	      fail('Invalid base64')
   641	    const bytes = Buffer.from(wrapper.data, 'base64')
   642	    if (
   643	      bytes.length !== wrapper.decodedBytes ||
   644	      bytes.toString('base64') !== wrapper.data ||
   645	      sha(bytes) !== wrapper.decodedSha256
   646	    )
   647	      fail('Physical checkpoint integrity mismatch')
   648	    const text = new TextDecoder('utf-8', {
   649	      fatal: true,
   650	      ignoreBOM: true
   651	    }).decode(bytes)
   652	    boundedJsonText(text)
   653	    const state = normalizedState(JSON.parse(text) as unknown)
   654	    if (canonicalizeCompleteJson(state, LIMITS) !== text)
   655	      fail('Noncanonical checkpoint state')
   656	    return state
   657	  } catch (error) {
   658	    if (error instanceof CheckpointError) throw error
   659	    fail('Malformed complete checkpoint encoding')
   660	  }
   661	}

   663	/** CAS transition validation; cannot erase retained observations or settled receipts. */
   664	export function validateCompleteCheckpointTransition(
   665	  current: ExecutionCheckpoint | null,
   666	  next: ExecutionCheckpoint
   667	): void {
   668	  if (!current) return
   669	  if (current.checkpointId === next.checkpointId)
   670	    fail('Checkpoint ID must change on every CAS')
   671	  const fromLegacy =
   672	    current.checkpointVersion === 1 &&
   673	    next.checkpointVersion === 2 &&
   674	    !current.state.observations.some((o) => o.type === 'TOOL_RESULT')
   675	  if (current.checkpointVersion !== next.checkpointVersion && !fromLegacy)
   676	    fail('Checkpoint mode cannot change through CAS')
   677	  if (next.checkpointVersion !== 2) return
   678	  if (
   679	    current.runtimeProfile !== next.runtimeProfile ||
   680	    current.runtimeVersion !== next.runtimeVersion ||
   681	    current.orchestratorVersion !== next.orchestratorVersion ||
   682	    current.state.capabilityFingerprint !== next.state.capabilityFingerprint ||
   683	    current.state.goal !== next.state.goal ||
   684	    next.stepNumber < current.stepNumber ||
   685	    next.state.toolResultContext!.deliveredThroughStep <
   686	      (current.state.toolResultContext?.deliveredThroughStep ?? 0)
   687	  )
   688	    fail('Checkpoint binding or cursor changed')
   689	  if (fromLegacy) return
   690	  const accepted = current.state.toolResultContext!.acceptedDecision
   691	  if (
   692	    accepted &&
   693	    next.stepNumber < accepted.stepNumber &&
   694	    canonicalizeCompleteJson(accepted, LIMITS) !==
   695	      canonicalizeCompleteJson(
   696	        next.state.toolResultContext!.acceptedDecision,
   697	        LIMITS
   698	      )
   699	  )
   700	    fail('Accepted decision cannot change before its step completes')
   701	  const nextObservations = new Map(
   702	    next.state.observations.map((o) => [o.observationId, o])
   703	  )
   704	  for (const old of current.state.observations) {
   705	    if (
   706	      old.type === 'TOOL_RESULT' &&
   707	      canonicalizeCompleteJson(
   708	        nextObservations.get(old.observationId),
   709	        LIMITS
   710	      ) !== canonicalizeCompleteJson(old, LIMITS)
   711	    )
   712	      fail('Tool observation retention violated')
   713	  }
   714	  const receipts = new Map(
   715	    next.state.toolResultContext!.decisionAdmissions.map((r) => [
   716	      r.admissionId,
   717	      r
   718	    ])
   719	  )
   720	  for (const old of current.state.toolResultContext!.decisionAdmissions) {
   721	    const updated = receipts.get(old.admissionId)
   722	    if (!updated) fail('Admission retention violated')
   723	    if (old.status === 'RESERVED') {
   724	      const {
   725	        status: _s,
   726	        actual: _a,
   727	        unknownReason: _u,
   728	        resultSha256: _h,
   729	        ...immutable
   730	      } = updated!
   731	      const { status: _old, ...previous } = old
   732	      void _s
   733	      void _a
   734	      void _u
   735	      void _h
   736	      void _old
   737	      if (
   738	        canonicalizeCompleteJson(immutable, LIMITS) !==
   739	        canonicalizeCompleteJson(previous, LIMITS)
   740	      )
   741	        fail('Admission binding changed')
   742	    } else if (
   743	      canonicalizeCompleteJson(old, LIMITS) !==
   744	      canonicalizeCompleteJson(updated, LIMITS)
   745	    )
   746	      fail('Settled admission changed')
   747	  }
   748	}
    75	 * Canonical public composition root for the neutral harness packages. Product
    76	 * adapters provide the ports; this function only wires them together and
    77	 * resolves the runtime profile through a registry-like selection.
    78	 */
    79	export function createOperationalHarness(
    80	  options: OperationalHarnessOptions
    81	): OperationalHarness {
    82	  if (
    83	    options.toolResultContext &&
    84	    (!options.iterativeOrchestrator || !options.stepStore)
    85	  ) {
    86	    throw new Error(
    87	      'complete-tool-v1 requires iterative orchestration and a checkpoint store.'
    88	    )
    89	  }
    90	  const capabilityFingerprint = options.capabilities?.compositionFingerprint()
    91	  if (options.capabilities && !options.effectJournal) {
    92	    throw new Error(
    93	      'Capability composition requires an explicit effect journal'
    94	    )
    95	  }
    96	  const tools = resolveTools(options, capabilityFingerprint)
    97	  const singlePass = new SinglePassGovernedRuntime({
    98	    orchestrator: options.orchestrator ?? new SinglePassOrchestrator(),
    99	    modelGateway: options.modelGateway,
   100	    policy: options.policy,
   101	    approvals: options.approvals,
   102	    tools,
   103	    ...(capabilityFingerprint ? { capabilityFingerprint } : {}),
   104	    audit: options.audit,
   105	    telemetry: options.telemetry,
   106	    ...(options.pause ? { pause: options.pause } : {}),
   107	    ...(options.effects ? { effects: options.effects } : {}),
   108	    ...(options.log ? { log: options.log } : {}),
   109	    ...(options.sandbox ? { sandbox: options.sandbox } : {})
   110	  })
   111	
   112	  const iterative =
   113	    options.iterativeOrchestrator && options.stepStore
   114	      ? new IterativeGovernedRuntime({
   115	          orchestrator: options.iterativeOrchestrator,
   116	          modelGateway: options.modelGateway,
   117	          policy: options.policy,
   118	          approvals: options.approvals,
   119	          tools,
   120	          ...(capabilityFingerprint ? { capabilityFingerprint } : {}),
   121	          audit: options.audit,
   122	          telemetry: options.telemetry,
   123	          ...(options.pause ? { pause: options.pause } : {}),
   124	          ...(options.effects ? { effects: options.effects } : {}),
   125	          ...(options.sandbox ? { sandbox: options.sandbox } : {}),
   126	          ...(options.log ? { log: options.log } : {}),
   127	          stepStore: options.stepStore,
   128	          ...(options.toolResultContext
   129	            ? { toolResultContext: options.toolResultContext }
   130	            : {}),
   131	          ...(options.contextEngine
   132	            ? { contextEngine: options.contextEngine }
   133	            : {}),
   134	          ...(options.knowledge ? { knowledge: options.knowledge } : {}),
   135	          ...(options.sufficiencyEvaluator
   136	            ? { sufficiencyEvaluator: options.sufficiencyEvaluator }
   137	            : {}),
   138	          ...(options.completionEvaluator
   139	            ? { completionEvaluator: options.completionEvaluator }
   140	            : {}),
   141	          ...(options.claimExtractor
   142	            ? { claimExtractor: options.claimExtractor }
   143	            : {}),
   144	          ...(options.clock ? { clock: options.clock } : {})
   145	        })
   146	      : undefined
   147	
   148	  const resolveProfile = (input: RuntimeInput): RuntimeProfile => {
   149	    const profile =
   150	      input.runtimeProfile ?? options.defaultRuntimeProfile ?? 'single_pass'
   151	    if (options.toolResultContext && profile !== 'iterative') {
   152	      throw new Error(
   153	        'complete-tool-v1 is only supported by the iterative profile.'
   154	      )
   155	    }
   156	    return profile
   157	  }
   158	
   159	  const execute = async (input: RuntimeInput): Promise<RuntimeResult> => {
   160	    const governedInput = capabilityFingerprint
   161	      ? { ...input, capabilityFingerprint }
   162	      : input
   163	    if (
   164	      capabilityFingerprint &&
   165	      input.executionId &&
   166	      input.capabilityFingerprint === undefined
   167	    ) {
   168	      await safeAudit(options, input, 'runtime.stopped', 'STATE_CONFLICT')
   169	      return {
   170	        response:
   171	          'Capability composition fingerprint is required for durable execution.',
   172	        stopReason: 'STATE_CONFLICT',
   173	        steps: 0,
   174	        modelCalls: 0,
   175	        toolCalls: 0,
   176	        usage: { inputTokens: 0, outputTokens: 0, costUsd: 0 }
   177	      }
   178	    }
   179	    if (
   180	      capabilityFingerprint &&
   181	      input.capabilityFingerprint !== undefined &&
   182	      input.capabilityFingerprint !== capabilityFingerprint
   183	    ) {
   184	      await safeAudit(options, input, 'runtime.stopped', 'STATE_CONFLICT')
   185	      return {
   186	        response:
   187	          'Capability composition fingerprint does not match the runtime.',
   188	        stopReason: 'STATE_CONFLICT',
   189	        steps: 0,
   190	        modelCalls: 0,
   191	        toolCalls: 0,
   192	        usage: { inputTokens: 0, outputTokens: 0, costUsd: 0 }
   193	      }
   194	    }
   195	    const profile = resolveProfile(governedInput)
   196	    if (profile === 'iterative') {
   197	      if (!iterative) {
   198	        await safeAudit(
   199	          options,
   200	          governedInput,
   201	          'runtime.stopped',
   202	          'STATE_CONFLICT'
   203	        )
   204	        return {
   205	          response:
   206	            'The iterative runtime profile is not configured in this composition.',
   207	          stopReason: 'STATE_CONFLICT',
   208	          steps: 0,
   209	          modelCalls: 0,
   210	          toolCalls: 0,
   211	          usage: { inputTokens: 0, outputTokens: 0, costUsd: 0 }
   212	        }
   213	      }
   214	      return iterative.execute(governedInput)
   215	    }
   216	    return singlePass.execute(governedInput)
   217	  }
   218	
   219	  return {
   220	    run: (input) => execute(input),
   221	    execute,
   222	    resolveProfile,
   223	    ...(options.capabilities
   224	      ? { capabilityFingerprint: options.capabilities.compositionFingerprint() }
   225	      : {})
   226	  }
   227	}

   229	function resolveTools(
   230	  options: OperationalHarnessOptions,
   231	  capabilityFingerprint: string | undefined
   232	): ToolRegistry {
   233	  if (options.capabilities && options.tools) {
   234	    throw new Error(
   235	      'Operational harness composition accepts capabilities or tools, not both'
   236	    )
   237	  }
   238	  const baseRegistry = options.capabilities
   239	    ? createCapabilityToolRegistry(
   240	        options.capabilities,
   241	        options.toolResultContext ? prepareCompleteToolResult : undefined
   242	      )
   243	    : options.tools
   244	  if (!baseRegistry) {
   245	    throw new Error(
   246	      'Operational harness composition requires an explicit capability or tool registry'
   247	    )
   248	  }
   249	  const registry = options.toolResultContext
   250	    ? prepareCompleteToolRegistry(baseRegistry)
   251	    : baseRegistry
   252	  if (!options.effectJournal) return registry
   253	  return createJournaledToolRegistry(registry, options.effectJournal, {
   254	    ...options.effectJournalOptions,
   255	    ...(capabilityFingerprint
   256	      ? { compositionFingerprint: capabilityFingerprint }
   257	      : {})
   258	  })
   259	}
