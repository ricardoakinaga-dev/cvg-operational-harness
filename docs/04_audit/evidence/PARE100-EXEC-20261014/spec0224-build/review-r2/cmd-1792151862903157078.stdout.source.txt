    27	  for (let index = observations.length - 1; index >= 0; index -= 1) {

    40	function toolObservationIsEffect(observation: Observation): boolean {
    41	  const payload = observation.payload as {
    42	    status?: unknown
    43	    sideEffect?: unknown
    44	  } | null
    45	  if (!payload || typeof payload !== 'object') return false
    46	  if (payload.status !== 'SUCCEEDED') return false
    47	  return payload.sideEffect === 'WRITE' || payload.sideEffect === 'EXTERNAL'
    48	}

    50	/**
    51	 * A confirmed action must be grounded in the claimed effect: either the exact
    52	 * tool named by the decision succeeded with a side effect, or (when no tool is
    53	 * named) at least one side-effecting tool succeeded. A previous successful
    54	 * read never satisfies an action-confirmation claim.
    55	 */
    56	function hasConfirmedEffect(
    57	  decision: { readonly toolId?: string },
    58	  observations: readonly Observation[]
    59	): boolean {
    60	  const effects = observations.filter(
    61	    (observation) =>
    62	      observation.type === 'TOOL_RESULT' && toolObservationIsEffect(observation)
    63	  )
    64	  if (decision.toolId) {
    65	    return effects.some(
    66	      (observation) => observation.provenance.sourceId === decision.toolId
    67	    )
    68	  }
    69	  return effects.length > 0
    70	}

    74	  if (!payload || typeof payload !== 'object') return false
    75	  return payload.status === 'FAILED' || payload.status === 'REJECTED'
    76	}

   101	  private readonly strategy: CompletionStrategy
   102	  private readonly semanticJudge: SemanticCompletionJudge | undefined
   103	
   104	  public constructor(options: CompletionEvaluatorOptions = {}) {
   105	    this.strategy = options.strategy ?? 'DETERMINISTIC'
   106	    this.semanticJudge = options.semanticJudge

   109	  public async evaluate(
   110	    input: CompletionEvaluationInput
   111	  ): Promise<CompletionEvaluation> {
   112	    const deterministic = this.evaluateDeterministic(input)
   113	    if (deterministic.outcome !== 'COMPLETE') {
   114	      return deterministic
   115	    }
   116	    if (this.strategy !== 'HYBRID' || !this.semanticJudge) {
   117	      return deterministic
   118	    }
   119	    try {
   120	      const judged = await this.semanticJudge.judge(input)
   121	      return judged
   122	    } catch {
   123	      return {
   124	        ...deterministic,
   125	        // Unreachable: every deterministic COMPLETE carries a detail.
   126	        /* v8 ignore next 2 */
   127	        detail:
   128	          `${deterministic.detail ?? ''} semantic_judge_unavailable`.trim(),
   129	        reasonCode: 'EVALUATOR_UNAVAILABLE'
   130	      }
   131	    }
