    69	/** Only declared optional internal fields may omit undefined. Arbitrary JSON is never repaired. */
    70	function record(
    71	  required: Record<string, Rule>,
    72	  optional: Record<string, Rule> = {}
    73	): Rule {
    74	  return (value) => {
    75	    if (!value || typeof value !== 'object' || Array.isArray(value)) fail()
    76	    const prototype = Object.getPrototypeOf(value)
    77	    if (prototype !== Object.prototype && prototype !== null) fail()
    78	    const keys = Reflect.ownKeys(value)
    79	    if (
    80	      keys.length >
    81	      Object.keys(required).length + Object.keys(optional).length
    82	    )
    83	      fail('Unknown checkpoint field')
    84	    const normalized = Object.create(null) as Record<string, unknown>
    85	    for (const key of keys) {
    86	      if (
    87	        typeof key !== 'string' ||
    88	        (!Object.hasOwn(required, key) && !Object.hasOwn(optional, key))
    89	      )
    90	        fail('Unknown checkpoint field')
    91	      const descriptor = Object.getOwnPropertyDescriptor(value, key)
    92	      if (!descriptor || !('value' in descriptor))
    93	        fail('Checkpoint accessor is forbidden')
    94	      if (descriptor.value === undefined && Object.hasOwn(optional, key))
    95	        continue
    96	      normalized[key as string] = (required[key as string] ??
    97	        optional[key as string])!(descriptor.value)
    98	    }
    99	    for (const key of Object.keys(required))
   100	      if (!Object.hasOwn(normalized, key)) fail('Missing checkpoint field')
   101	    return normalized
   102	  }
   103	}
   104	function array(rule: Rule): Rule {
   105	  return (value) => {
   106	    if (
   107	      !Array.isArray(value) ||
   108	      Object.getPrototypeOf(value) !== Array.prototype
   109	    )
   110	      fail()
   111	    const length = Object.getOwnPropertyDescriptor(value, 'length')!
   112	      .value as number
   113	    if (
   114	      length > LIMITS.maxNodes ||
   115	      Reflect.ownKeys(value).length !== length + 1
   116	    )
   117	      fail('Invalid checkpoint array')
   118	    const normalized: unknown[] = []
   119	    for (let i = 0; i < length; i++) {
   120	      const descriptor = Object.getOwnPropertyDescriptor(value, String(i))
   121	      if (!descriptor || !('value' in descriptor))
   122	        fail('Invalid checkpoint array')
   123	      normalized.push(rule(descriptor.value))
   124	    }
   125	    return normalized
   126	  }
