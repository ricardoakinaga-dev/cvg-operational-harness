import { execFileSync } from 'node:child_process'
import {
  mkdtempSync,
  readFileSync,
  writeFileSync,
  rmSync,
  cpSync,
  mkdirSync
} from 'node:fs'
import { tmpdir } from 'node:os'
import { join, resolve } from 'node:path'
import { afterAll, describe, expect, it } from 'vitest'

const ROOT = resolve(process.cwd())
const SCRIPT = join(ROOT, 'scripts', 'gen-tool-catalog.mjs')
const CATALOG = join(ROOT, 'docs', 'architecture', 'TOOL_CATALOG.md')

const tempRoot = mkdtempSync(join(tmpdir(), 'pdsh-014-'))
afterAll(() => {
  rmSync(tempRoot, { recursive: true, force: true })
})

function runCheck(): { status: number; stdout: string; stderr: string } {
  try {
    const stdout = execFileSync('node', [SCRIPT, '--check'], {
      encoding: 'utf8',
      cwd: ROOT
    })
    return { status: 0, stdout, stderr: '' }
  } catch (error) {
    const e = error as {
      status?: number
      stdout?: string
      stderr?: string
    }
    return {
      status: e.status ?? 1,
      stdout: e.stdout ?? '',
      stderr: e.stderr ?? ''
    }
  }
}

describe('PDSH-014 tool catalog drift-check', () => {
  it('committed TOOL_CATALOG.md is up to date (exit 0)', () => {
    const result = runCheck()
    expect(result.status).toBe(0)
    expect(result.stdout).toContain('up to date')
  })

  it('catalog has the GENERATED header and expected tools/skills', () => {
    const content = readFileSync(CATALOG, 'utf8')
    expect(content).toContain('GENERATED by scripts/gen-tool-catalog.mjs')
    expect(content).toContain('DO NOT EDIT BY HAND')
    for (const id of [
      'fs.read',
      'fs.list',
      'fs.write',
      'fs.edit',
      'shell.run',
      'exam.upload',
      'exam.download'
    ]) {
      expect(content).toContain(`\`${id}\``)
    }
    expect(content).toContain('`exam-intake`')
  })

  it('generation is deterministic (two runs produce identical output)', () => {
    const first = readFileSync(CATALOG, 'utf8')
    execFileSync('node', [SCRIPT], { cwd: ROOT, encoding: 'utf8' })
    const second = readFileSync(CATALOG, 'utf8')
    expect(second).toBe(first)
  })

  it('stale catalog (drift) → check fails with exit 1', () => {
    // Work in a temp copy of the repo surface the script touches.
    const fakeRoot = join(tempRoot, 'drift')
    mkdirSync(join(fakeRoot, 'scripts'), { recursive: true })
    mkdirSync(join(fakeRoot, 'docs', 'architecture'), { recursive: true })
    mkdirSync(join(fakeRoot, 'skills'), { recursive: true })
    mkdirSync(join(fakeRoot, 'packages', 'sandbox', 'src'), { recursive: true })
    mkdirSync(join(fakeRoot, 'packages', 'channel-gateway', 'src'), {
      recursive: true
    })
    cpSync(
      join(ROOT, 'packages', 'channel-gateway', 'src', 'exam-egress.ts'),
      join(fakeRoot, 'packages', 'channel-gateway', 'src', 'exam-egress.ts')
    )
    cpSync(SCRIPT, join(fakeRoot, 'scripts', 'gen-tool-catalog.mjs'))
    cpSync(CATALOG, join(fakeRoot, 'docs', 'architecture', 'TOOL_CATALOG.md'))
    cpSync(
      join(ROOT, 'packages', 'sandbox', 'src', 'fs-tools.ts'),
      join(fakeRoot, 'packages', 'sandbox', 'src', 'fs-tools.ts')
    )
    cpSync(
      join(ROOT, 'packages', 'sandbox', 'src', 'shell-tool.ts'),
      join(fakeRoot, 'packages', 'sandbox', 'src', 'shell-tool.ts')
    )
    cpSync(
      join(ROOT, 'skills', 'exam-intake'),
      join(fakeRoot, 'skills', 'exam-intake'),
      { recursive: true }
    )

    // Add a new skill without regenerating → drift.
    const newSkill = join(fakeRoot, 'skills', 'ghost-skill')
    mkdirSync(newSkill, { recursive: true })
    writeFileSync(
      join(newSkill, 'skill.json'),
      JSON.stringify({
        id: 'ghost-skill',
        version: '1.0.0',
        description: 'Skill added without regenerating the catalog.',
        requiredTools: [],
        requiredKnowledge: [],
        riskLevel: 'LOW'
      })
    )

    let status: number
    let stderr: string
    try {
      execFileSync(
        'node',
        [join(fakeRoot, 'scripts', 'gen-tool-catalog.mjs'), '--check'],
        { encoding: 'utf8', cwd: fakeRoot }
      )
      status = 0
      stderr = ''
    } catch (error) {
      const e = error as { status?: number; stderr?: string }
      status = e.status ?? 1
      stderr = e.stderr ?? ''
    }
    expect(status).toBe(1)
    expect(stderr).toContain('stale')
  })
})
