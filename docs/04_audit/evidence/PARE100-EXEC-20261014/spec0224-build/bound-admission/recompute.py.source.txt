"""Independent Python evidence reconstruction. No CVG imports, no database mutations."""
import sys,json,hashlib,base64,math,datetime
from pathlib import Path
run=Path(sys.argv[1]);checks=0;snapshots=0;calls=0
canonical=lambda v:json.dumps(v,ensure_ascii=False,sort_keys=True,separators=(',',':')).encode()
sha=lambda b:hashlib.sha256(b).hexdigest()
read=lambda p:json.loads((run/p).read_text())
def check(v,m):
 global checks
 checks+=1
 if not v:raise AssertionError(m)
def iso(v):return datetime.datetime.fromisoformat(v.replace('Z','+00:00'))
def toy(v):return math.ceil(len(canonical(v))/64)
def checkpoint(cp):
 global snapshots
 snapshots+=1
 unsigned={k:v for k,v in cp.items() if k not in ['checkpointId','createdAt','digest']}
 raw=canonical(unsigned);check(cp['checkpointVersion']==2,'CP version');check(len(raw)<=8388608,'CP size');check(sha(b'cvg.checkpoint.v2\n'+raw)==cp['digest'],'CP domain digest')
 return cp

 check(set(wrapper)=={'storageCodec','encoding','decodedBytes','decodedSha256','data'},'Wrapper fields');check(wrapper['storageCodec']=='cvg.checkpoint-state/2' and wrapper['encoding']=='base64','Wrapper codec')
 raw=base64.b64decode(wrapper['data'],validate=True);check(base64.b64encode(raw).decode()==wrapper['data'],'Canonical base64');check(len(raw)==wrapper['decodedBytes'] and sha(raw)==wrapper['decodedSha256'],'Physical state hash/bytes');check(raw==canonical(cp['state']),'Physical state canonical binding')
 for sql,key in [('checkpoint_id','checkpointId'),('execution_id','executionId'),('tenant_id','tenantId'),('checkpoint_version','checkpointVersion'),('runtime_profile','runtimeProfile'),('runtime_version','runtimeVersion'),('orchestrator_version','orchestratorVersion'),('step_number','stepNumber'),('digest','digest'),('created_at','createdAt'),('budget_usage','budgetUsage')]:check(row[sql]==cp[key],'Physical row header '+sql)
 identity=value['identity'];authority=value['authority'];check(identity['current_user']==identity['session_user'],'Fresh authenticated service role');check(identity['tenant_context']==cp['tenantId'],'Tenant context');check(authority['tenant_id']==cp['tenantId'] and authority['execution_id']==cp['executionId'],'Authority binding');check(authority['phase_state']=='OPEN' and authority['open_phase_acked'] is False and authority['execution_state']=='RUNNING','Actual bound OPEN phase')
 runtime=authority['admission_runtime'];check(authority['prepared_runtime']==runtime,'Native prepared runtime unchanged');check(authority['request']['runtime']=={k:v for k,v in runtime.items() if k!='executionId'} and runtime['executionId']==authority['execution_id'],'Native parent normalization removes only envelope executionId');check(runtime['context']['values']['adapterVersion']=='controlled-published-v1','Actual bound request marker');check(runtime['context']['values']['requestDigest']==authority['request_digest'],'Native request digest');check(runtime['capabilityFingerprint']==authority['capability_fingerprint']==cp['state']['capabilityFingerprint'],'Capability binding')
 return cp

def attempt(value,authority):
 global calls
 events=value['events'];check([e['sequence'] for e in events]==list(range(1,len(events)+1)),'Local event sequence');check(all(e['pid']==value['pid'] for e in events),'Event process identity');check(value['input']==authority['admission']['runtime'],'Input is native admission runtime')

 for event in events:
  if event['type']=='CAS-entry':checkpoint(event['checkpoint'])
  elif event['type']=='CAS-return-after-transaction':
   prior=entries[event['entry']];cp=prior['checkpoint'];check(prior['type']=='CAS-entry' and prior['sequence']<event['sequence'],'CAS order');check(cp['checkpointId']==event['checkpointId'] and cp['digest']==event['digest'],'CAS proposed token')
   if event['result']['status']=='APPLIED':check(event['result']['token']=={'checkpointId':cp['checkpointId'],'digest':cp['digest']},'Applied token')
   else:check(event['result']['status']=='CONFLICT','Unknown CAS disposition')
  elif event['type']=='fresh-service-SQL':
   calls+=1;cp=physical(event['physical']);start=entries[event['entry']];commit=entries[event['commit']];ctx=cp['state']['toolResultContext'];receipt=next(r for r in ctx['decisionAdmissions'] if r['admissionId']==ctx['pendingDecisionAdmission'])

   request=start['request'];check(receipt['requestSha256']==sha(canonical(request)),'Actual request digest');check(receipt['measurement']['inputUpperBoundTokens']==toy(request),'Toy input counter');check(request['maxOutputTokens']==receipt['reserved']['outputTokens']==512,'Output cap');check(receipt['reserved']['modelCalls']==1 and receipt['reserved']['costUsd']==0,'Native zero-cost reserve');check(iso(event['physical']['authority']['expires_at'])>iso(event['physical']['authority']['observed_at']),'Model began with live native prepared deadline')
   returns=[e for e in events if e['type']=='native-fake-provider-return' and e['entry']==start['sequence']];check(len(returns)==1 and returns[0]['sequence']>event['sequence'],'Native fake provider ordering');returned=returns[0]['nativeResult'];check(returned['externalCall'] is False and returned['provider']=='fake' and returned['model']=='deterministic-v1','Native fake provider identity');frame=json.loads(request['messages'][1]['content']);check(json.loads(returned['text'])['responseText']=='Bound response: '+frame['goal'],'Response from actual frame only')
 if value['final']:checkpoint(value['final'])
 return events

  cp=physical(read('negative-expired/terminal-physical.json'));check(cp==value['final'],'Terminal physical state');check(cp['state']['stopReason']=='INTERNAL_FAILURE' and cp['state']['toolResultContext']['decisionAdmissions']==[],'Expired deadline dispatched or invented receipt')

  prior={r['row']['execution_id']:r for r in snap['before']['operational_execution_checkpoints']};after={r['row']['execution_id']:r for r in snap['after']['operational_execution_checkpoints']};check(all(after[k]==v for k,v in prior.items()),'Expiry altered old checkpoints')
  observation=read('negative-expired/terminal-physical.json')['authority'];check(iso(observation['observed_at'])>iso(observation['expires_at']),'Deadline did not expire');check(iso(observation['lease_until'])>iso(observation['observed_at']),'Terminal used expired parent lease')
races=[]
for kind in ['initial','reserve']:
