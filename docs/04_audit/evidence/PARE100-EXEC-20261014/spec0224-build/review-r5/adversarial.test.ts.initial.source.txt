const toolId = 'synthetic.recovery.write'
type JournalMode = 'none' | 'factory' | 'prewrapped'

function fixture(mode: JournalMode, fixtureOptions: any = {}) {
  const journal = new InMemoryEffectJournal()
  const initialStore = new InMemoryExecutionStepStore()
  const base = phase3RuntimeInput()
  const input = {
    ...base,
    agent: phase3AgentProfile({ tools: [toolId] }),
    userMessage: 'Perform one synthetic in-memory operation.',
    budget: { ...base.budget, maxDurationMs: 5000 }
  }
  const requests: any[] = []
  const operationKeys: string[] = []
  let bodies = 0
  let captured: ExecutionCheckpoint
  let capturedSteps: readonly ExecutionStep[]
  let entered!: () => void
  const bodyEntered = new Promise<void>((resolve) => {
    entered = resolve
  })
  const factory = (
    store: InMemoryExecutionStepStore,
    suspend: boolean,
    recovery?: { mode?: JournalMode; journal?: EffectJournal; version?: string; sideEffect?: any; policy?: any; approvals?: any; flags?: any; wrapperOptions?: any; model?: boolean }
  ) => {
    const selectedMode = recovery?.mode ?? mode
    const authority = recovery?.journal ?? journal
    const tool: ToolDefinition = {
      id: toolId,
      version: recovery?.version ?? '1',
      ...(recovery?.flags ?? {}),
      description: 'Synthetic in-memory counter only',
      inputSchema: { type: 'object' },
      outputSchema: {},
      risk: 'LOW',
      sideEffect: recovery?.sideEffect ?? 'WRITE',
      idempotent: false,
      requiresApproval: false,
      execute: async (_input, context) => {
        bodies++
        operationKeys.push(context.operationKey!)
        if (suspend) {
          captured = (await initialStore.loadCheckpoint(
            input.tenantId,
            input.executionId!
          ))!
          capturedSteps = await initialStore.listSteps(
            input.tenantId,
            input.executionId!
          )
          entered()
          await new Promise<void>(() => {})
        }
        return { status: 'SUCCEEDED', output: fixtureOptions.output ?? 'synthetic operation done' }
      }
    }
    const raw: ToolRegistry = {
      list: () => [tool],
      resolve: (id) => (id === toolId ? tool : undefined)
    }
    return createOperationalHarness({
      toolResultContext: { version: 1 },
      ...(selectedMode === 'factory' ? { effectJournal: authority, effectJournalOptions: recovery?.wrapperOptions } : {}),
      tools:
        selectedMode === 'prewrapped'
          ? createJournaledToolRegistry(raw, authority, recovery?.wrapperOptions)
          : raw,
      stepStore: store,
      iterativeOrchestrator: new HybridOrchestrator({
        ...(recovery?.model ? { decisionModel: model, decisionInputCounter: counter, decisionModelSupportsMaxOutputTokens: true } : {}),
        rules: [
          (context) =>
            context.stepNumber === 1
              ? {
                  decisionType: 'CALL_TOOL',
                  reasonCode: 'TOOL_REQUIRED',
                  toolId,
                  toolVersion: '1',
                  toolInput: {}
                }
              : recovery?.model ? undefined : {
                  decisionType: 'RESPOND',
                  reasonCode: 'GOAL_SATISFIED',
                  responseText: 'Synthetic operation done.'
                }
        ]
      }),
      modelGateway: {
        complete: async () => {
          throw new Error('Must never call a model')
        }
      },
      policy: recovery?.policy ?? new ScriptedPolicyEngine(),
      approvals: recovery?.approvals ?? new InMemoryApprovalEngine(),
      audit: new RecordingAuditSink(),
      telemetry: new RecordingTelemetrySink()
    })
  }
  const model = { complete: async ({ signal: _signal, ...request }: any) => {
    requests.push(structuredClone(request))
    return { text: JSON.stringify({ decisionType: 'RESPOND', reasonCode: 'GOAL_SATISFIED', responseText: 'Synthetic result recovered.' }), provider: 'synthetic.i1', model: 'toy-i1', inputTokens: 20, outputTokens: 10, costUsd: 0.00001 }
  } }
  const counter = { measure: async () => ({ counterId: 'toy-i1', counterVersion: '1', provider: 'synthetic.i1', model: 'toy-i1', formattingVersion: '1', inputUpperBoundTokens: 500, contextWindowTokens: 100000, costUpperBoundUsd: 0.001 }) }
  const interrupt = async (failed = false) => {
    const controller = new AbortController()
    const running = factory(initialStore, true).run({
      ...input,
      signal: controller.signal
    })
    await bodyEntered
    controller.abort()
    await running
    // Recreate only state durable before the body completed. This is a memory
    // crash reconstruction; the interrupted process's later writes are absent.
    const restored = new InMemoryExecutionStepStore()
    await restored.compareAndSwapCheckpoint({
      tenantId: input.tenantId,
      executionId: input.executionId!,
      expected: null,
      next: captured
    })
    for (const step of capturedSteps)
      await restored.recordStep(
        failed && step.stepType === 'TOOL'
          ? { ...step, status: 'FAILED', errorCode: 'unknown_effect' }
          : step
      )
    return restored
  }
  return {
    input,
    factory,
    initialStore,
    interrupt,
    journal,
    operationKeys,
    requests,
    bodyCount: () => bodies,
    checkpoint: () => captured
  }
}

async function snapshot(lane: any, store: any) {
 return { checkpoint: await store.loadCheckpoint(lane.input.tenantId, lane.input.executionId), steps: await store.listSteps(lane.input.tenantId, lane.input.executionId) }
}
function noExtraBill(result: any, before: any) {
 expect(result.toolCalls).toBe(before.checkpoint.budgetUsage.toolCalls)
 expect(result.modelCalls).toBe(before.checkpoint.budgetUsage.modelCalls)
 for (const key of ['inputTokens','outputTokens','costUsd']) expect(result.usage[key]).toBe(before.checkpoint.budgetUsage[key])
 expect(result.toolResult).toBeUndefined()
}
async function confirm(lane: any, output: any) {
 const operationKey=lane.operationKeys[0]!
 const rec=await lane.journal.get(lane.input.tenantId,operationKey)
 await lane.journal.confirm({tenantId: lane.input.tenantId, operationKey, attemptId: rec!.attemptId, result: output})
 return await lane.journal.get(lane.input.tenantId,operationKey)
}
