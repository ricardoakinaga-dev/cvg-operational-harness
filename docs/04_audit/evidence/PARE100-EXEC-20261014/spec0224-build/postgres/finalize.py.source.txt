import pathlib, json, hashlib, datetime
B = pathlib.Path(__file__).resolve().parent
W = B.parent.parent / 'worktree'
def digest(p):
    data = p.read_bytes()
    return {'sha256': hashlib.sha256(data).hexdigest(), 'bytes': len(data)}
def save(name, data):
    (B / name).write_text(json.dumps(data, indent=2, ensure_ascii=True)+'\n')

baseline = json.loads((B / 'sources-r1.json').read_text())['files']
sources = {p: digest(W / p) for p in baseline}
own = 'packages/persistence/src/__tests__/complete-tool-process-restart.test.ts'
frozen = json.loads((B.parent / 'codec/candidate-manifest.json').read_text())
matches = {x['path']: digest(W / x['path'])['sha256'] == x['sha256'] for x in frozen}
assert all(matches.values()), matches
assert sources['docs/02_spec/0224_pare100_complete_tool_result_context.md']['sha256'] == 'f3bc9a6115d17abb4185644cfc998f7aef39164774afff00042a64202b2c49ea'
drift = {p: {'earlier': value, 'current': sources[p]} for p, value in baseline.items() if p != own and sources[p] != value}
assert set(drift) <= {'packages/contracts/src/complete-tool-context.ts'}, drift
save('sources-final.json', {'observedAt': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'files': sources, 'codecBuilderManifestMatches': matches, 'concurrentDrift': drift, 'note': 'sources-r1 was captured during first run, not a claimed pre-execution baseline; builder frozen manifest independently matched for all seven files. Current hashes are observational, not a claim that all sources were frozen during execution. Shared contract type drift is Lead-owned.'})

resource = json.loads((B / 'resource.json').read_text())
final = json.loads((B / 'container-final.stdout.log').read_text())[0]

assert final['Mounts'][0]['Name'] == resource['anonymousVolume']

restart = [json.loads(p.read_text()) | {'artifact': str(p.relative_to(B))} for p in B.glob('restart-*/comparison.json')]
assert len(restart) == 1

for p in B.glob('race-*/*.result.json'):
