"""Fixed read-only focused quality commands with private Node environment."""
from pathlib import Path
import datetime
import hashlib
import json
import subprocess
import sys
import time

P=Path('/tmp/cvg-pare100-exec-20261014-6az3hlay/sandbox-metadata-observation-r3')
W=P.parent/'worktree'
D=Path(__file__).resolve().parent
sys.path.insert(0,str(P))
from common import env_for, NODE, put
folder=D/'quality'
folder.mkdir()
for name in ['home','tmp','cache','config','data']:
    (folder/name).mkdir()
env=env_for(folder)
# Inherit the repository's focused recommended rules through a private config.
config=folder/'eslint.config.cjs'
config.write_text("module.exports = require("+json.dumps(str(W/'eslint.config.js'))+")\n")
commands=[
    ('format',[str(NODE),str(W/'node_modules/prettier/bin/prettier.cjs'),'--check',str(P/'observe.mts')],None),
    ('typecheck',[str(NODE),str(W/'node_modules/typescript/bin/tsc'),'-p',str(P/'tsconfig.json'),'--noEmit','--incremental','false'],None),
    ('lint',[str(NODE),str(W/'node_modules/eslint/bin/eslint.js'),'--config',str(config),'--no-cache','--stdin','--stdin-filename',str(W/'tests/differential/sandbox-metadata-observation.mts')],(P/'observe.mts').read_bytes()),
]
results=[]
for label,argv,data in commands:
    start=datetime.datetime.now(datetime.timezone.utc).isoformat()
    t=time.monotonic()
    result=subprocess.run(argv,cwd=W,env=env,input=data,stdout=subprocess.PIPE,stderr=subprocess.PIPE,timeout=60)
    (folder/(label+'.stdout')).write_bytes(result.stdout)
    (folder/(label+'.stderr')).write_bytes(result.stderr)
    record={'label':label,'argv':argv,'cwd':str(W),'env':env,'startUTC':start,'elapsedSeconds':time.monotonic()-t,'exit':result.returncode,'stdoutBytes':len(result.stdout),'stderrBytes':len(result.stderr),'truncated':False,'inputSHA256':hashlib.sha256(data).hexdigest() if data is not None else None,'emission':'noEmit; incremental false; format check; lint no-cache; private HOME/TMP/XDG; no installs'}
    put(folder/(label+'.command.json'),record)
    results.append(record)
    print(label,result.returncode,result.stdout.decode(),result.stderr.decode(),flush=True)
put(folder/'results.json',results)
sys.exit(0 if all(x['exit']==0 for x in results) else 1)
