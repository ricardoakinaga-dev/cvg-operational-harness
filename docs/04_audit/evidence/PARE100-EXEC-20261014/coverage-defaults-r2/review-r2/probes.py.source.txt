def summary(root):
    dims=['statements','branches','functions','lines']
    metric=lambda t,c:dict(total=t,covered=c,skipped=0,pct=100*c/t)
    return {'total':{k:metric(100,85 if k=='branches' else 90) for k in dims},
            str(root/'packages/contracts/src/synthetic.ts'):{k:metric(20,20) for k in dims},
            str(root/'packages/other/src/synthetic.ts'):{k:metric(80,65 if k=='branches' else 70) for k in dims}}

def fixture(name,mode='missing',flag='garbage',gate='coverage',prior='file',snapshot=False):
    root=O/'fixtures'/name;root.mkdir(parents=True);graph('scripts/ci-bar.mjs',root)
    (root/'package.json').write_text('{"type":"module"}\n');(root/'bin').mkdir()
    (root/'bin/npm').write_text(producer);(root/'bin/npm').chmod(0o700)
    a=root/'artifacts';(a/'gates').mkdir(parents=True)
    dump(a/'ci-bar-state.json',{'scope':scope,'runId':'synthetic-i1-r2','candidateId':'a'*64,'nodeVersion':'22.23.2','npmVersion':'synthetic','startedAt':'2026-10-16T00:00:00Z','candidateFiles':[],'gates':[]})
    (root/'coverage').mkdir();(root/'coverage/untouched.txt').write_text('coverage sibling must survive')
    report=root/'coverage/coverage-summary.json';target=root/'previous-target.json'
    dump(target,summary(root));target_before=sha(target)
    if prior=='file':shutil.copyfile(target,report)
    elif prior=='symlink':report.symlink_to(target)
    elif prior=='dangling':report.symlink_to(root/'nonexistent.json')
    elif prior=='directory':report.mkdir();(report/'child').write_text('directory child must survive')
    elif prior!='none':raise ValueError(prior)
    if snapshot:dump(a/'gate-outputs/coverage/coverage/coverage-summary.json',summary(root))
    e=env(name);e.update(PATH=str(root/'bin')+':'+e['PATH'],CI_ARTIFACT_DIR=str(a),CVG_TEST_SCOPE='all',SYNTHETIC_REPORT=mode,I1_PRESERVE_ENV='unmodified')
    if flag is not None:e['CVG_COVERAGE_PER_FILE']=flag
    code,logs=run(name,[NODE,'scripts/ci-bar.mjs','gate',gate],root,e,15)
    observed=json.loads((root/'observed.json').read_text()) if (root/'observed.json').exists() else None
    entry=json.loads(logs['stdout']) if logs['stdout'].strip() else None
    facts={'name':name,'exit':code,'entry':entry,'observed':observed,'report_exists':report.exists(),'report_is_symlink':report.is_symlink(),
           'target_unchanged':target_before==sha(target),'sibling_unchanged':(root/'coverage/untouched.txt').read_text()=='coverage sibling must survive',
           'directory_child_survives':(report/'child').exists() if report.is_dir() else None}
    dump(O/'observations'/f'{name}.json',facts)
    assert facts['target_unchanged'] and facts['sibling_unchanged']
    if observed:
        assert observed['env']['I1_PRESERVE_ENV']=='unmodified' and observed['scope']=='core'
        if gate in ['coverage','certify']:assert observed['perFile']=='1'
        else:assert observed['perFile']==flag
    return facts
