def run(name,argv,cwd,e,timeout=80):
    assert 0<timeout<=90
    assert 'NODE_OPTIONS' not in e
    assert str(cwd).startswith(str(O))
    out=O/'commands'/name;out.mkdir(parents=True,exist_ok=True)
    started=time.time();p=subprocess.Popen([str(a) for a in argv],cwd=cwd,env=e,stdout=subprocess.PIPE,stderr=subprocess.PIPE,start_new_session=True)
    sel=selectors.DefaultSelector();sel.register(p.stdout,selectors.EVENT_READ,'stdout');sel.register(p.stderr,selectors.EVENT_READ,'stderr')
    logs={'stdout':bytearray(),'stderr':bytearray()};reason=None
    while sel.get_map():
        if time.time()-started>timeout: reason='timeout'
        if reason and p.poll() is None: os.killpg(p.pid,signal.SIGKILL)
        for key,_ in sel.select(.1):
            b=os.read(key.fileobj.fileno(),65536)
            if not b:sel.unregister(key.fileobj);continue
            remaining=2*1024*1024-sum(map(len,logs.values()))
            logs[key.data].extend(b[:max(0,remaining)])
            if len(b)>remaining:reason='output_bound'
    p.wait();duration=time.time()-started
    alive=[]
    for item in Path('/proc').iterdir():
        if item.name.isdigit():
            try:
                data=(item/'stat').read_text().rsplit(')',1)[1].split()
                if int(data[2])==p.pid:alive.append(int(item.name))
            except (OSError,ValueError,IndexError):pass
    for pid in alive:
        try:os.kill(pid,signal.SIGKILL)
        except ProcessLookupError:pass
    for k,b in logs.items():(out/(k+'.log')).write_bytes(b)
    record={'argv':[str(a) for a in argv],'cwd':str(cwd),'environment':e,'pid':p.pid,'process_group':p.pid,
            'timeout_seconds':timeout,'output_bound_bytes':2*1024*1024,'termination_reason':reason,
            'exit':p.returncode,'elapsed_seconds':round(duration,4),'started_unix':started,'remaining_process_group':alive,
            'logs':{k:{'path':str(out/(k+'.log')),'sha256':sha(out/(k+'.log')),'bytes':len(b)} for k,b in logs.items()}}
    dump(out/'command.json',record)
    assert reason is None and not alive,record
    print(name,'exit',p.returncode,'seconds',round(duration,3),flush=True)
    return p.returncode,{k:bytes(b).decode(errors='replace') for k,b in logs.items()}
