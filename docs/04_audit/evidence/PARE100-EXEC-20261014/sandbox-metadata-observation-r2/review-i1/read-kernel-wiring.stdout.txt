 153 /**
 154  * Single-turn governed loop on the plugin kernel (SPEC 0181). Every phase runs
 155  * through a named interception point; governance lives in control plugins.
 156  * Behavior matches `SinglePassGovernedRuntime` and closes the conformance gaps
 157  * I5 (call logged before dispatch), I6 (model request logged), I7 (listener
 158  * failures normalized) and I8 (cancellation by `RuntimeInput.signal`).
 159  */
 160 export class KernelRuntime implements HarnessRuntime {
 161   #host: Promise<KernelHost> | undefined
 162 
 163   private constructor(
 164     private readonly plugins: readonly KernelPlugin[],
 165     host: KernelHost | undefined,
 166     private readonly options: KernelRuntimeOptions = {}
 167   ) {
 168     if (host) this.#host = Promise.resolve(host)
 169   }
 170 
 171   /** Boots eagerly and fails loudly when the profile is incomplete. */
 172   public static async boot(
 173     plugins: readonly KernelPlugin[],
 174     options: KernelRuntimeOptions = {}
 175   ): Promise<KernelRuntime> {
 176     const host = await KernelHost.boot(plugins, REQUIRED_SERVICES)
 177     return new KernelRuntime(plugins, host, options)
 178   }
 179 
 180   /** Boots on first use; a boot failure becomes a fail-closed result. */
 181   public static lazy(
 182     plugins: readonly KernelPlugin[],
 183     options: KernelRuntimeOptions = {}
 184   ): KernelRuntime {
 185     return new KernelRuntime(plugins, undefined, options)
 186   }
 187 
 188   public async dispose(): Promise<void> {
 189     if (!this.#host) return
 190     const host = await this.#host.catch(() => undefined)
 191     await host?.dispose()
 192   }
 193 
 194   public async execute(input: RuntimeInput): Promise<RuntimeResult> {
 195     let host: KernelHost
 196     try {
 197       this.#host ??= KernelHost.boot(this.plugins, REQUIRED_SERVICES)
 198       host = await this.#host
 199     } catch (error) {
 200       return result({
 201         response:
 202           this.options.bootFailureResponse ??
 203           `Kernel could not start: ${errorMessage(error)}.`,
 204         stopReason: 'INTERNAL_FAILURE',
 205         modelCalls: 0,
 206         toolCalls: 0
 207       })
 208     }
 209     return new Turn(host, input).run()
 210   }

 213 /** One execution of the loop over a booted, sealed host. */
 214 class Turn {
 215   readonly #host: KernelHost
 216   readonly #controller = new AbortController()
 217   readonly #state: TurnState
 218 
 219   public constructor(host: KernelHost, input: RuntimeInput) {
 220     this.#host = host
 221     const startedAt = Date.now()
 222     if (input.signal?.aborted) this.#controller.abort()
 223     input.signal?.addEventListener('abort', () => this.#controller.abort(), {
 224       once: true
 225     })
 226     this.#state = createTurnState(
 227       input,
 228       startedAt,
 229       host.get('budget').begin(input, startedAt),
 230       this.#controller.signal
 231     )
 232   }
 233 
 234   public async run(): Promise<RuntimeResult> {
 235     const input = this.#state.input
 236     if (!hasValidBoundaryInput(input)) {
 237       return this.#finish(
 238         result({
 239           response: 'Runtime identity or execution budget is invalid.',
 240           stopReason: 'UNSAFE_REQUEST',
 241           modelCalls: 0,
 242           toolCalls: 0
 243         })
 244       )
 245     }
 246     if (input.budget.maxSteps < 1) {
 247       return this.#finish(
 248         fromStop(
 249           stop('MAX_STEPS', 'Execution budget exhausted before the first step.')
 250         )
 251       )
 252     }
 253     if (input.budget.maxDurationMs < 1) {
 254       return this.#finish(
 255         fromStop(
 256           stop('MAX_DURATION', 'Execution duration budget is not available.')
 257         )
 258       )
 259     }
 260 
 261     const logged = await appendLog(this.#host, {
 262       type: 'turn/start',
 263       correlationId: input.correlationId
 264     })
 265     if (logged) return this.#finish(fromStop(logged))
 266 
 267     const gate = await runGate(this.#host, 'turn/before-step', this.#state)
 268     if (gate.kind === 'stop') return this.#finish(fromStop(gate))
 269 
 270     try {
 271       if (this.#state.signal.aborted) {
 272         return this.#finish(fromStop(stop('CANCELLED', CANCELLED_RESPONSE)))
 273       }
 274       if (this.#state.budget.remainingMs() <= 0) {
 275         return this.#finish(
 276           fromStop(
 277             stop('MAX_DURATION', 'Duration budget exhausted before planning.')
 278           )
 279         )
 280       }
 281       const decision = await race(this.#state, () =>
 282         this.#host.get('planner').decideNextStep({
 283           runtime: input,
 284           availableTools: this.#availableTools(),
 285           step: 1
 286         })
 287       )
 288       if (decision === CANCELLED) {
 289         return this.#finish(fromStop(stop('CANCELLED', CANCELLED_RESPONSE)))
 290       }
 291       if (decision === DEADLINE) {
 292         return this.#finish(
 293           fromStop(
 294             stop('MAX_DURATION', 'Duration budget exhausted while planning.')
 295           )
 296         )
 297       }
 298       switch (decision.action) {
 299         case 'RESPOND':
 300           return await this.#respond(decision.response)
 301         case 'CALL_TOOL':
 302           return await this.#callTool(decision.toolInvocation)
 303         case 'ASK_USER':
 304           return this.#finish(
 305             fromStop(
 306               stop(
 307                 'NEEDS_USER_INPUT',
 308                 decision.response ??
 309                   decision.reason ??
 310                   'More information is required.'
 311               )
 312             )
 313           )
 314         case 'REQUEST_APPROVAL':
 315           return this.#finish(
 316             fromStop(
 317               stop(
 318                 'APPROVAL_REQUIRED',
 319                 decision.response ??
 320                   decision.reason ??
 321                   'Approval is required before continuing.'
 322               )
 323             )
 324           )
 325         case 'HANDOFF':
 326           return this.#finish(
 327             fromStop(
 328               stop(
 329                 'HUMAN_TAKEOVER',
 330                 decision.response ??
 331                   decision.reason ??
 332                   'Human takeover is required.'
 333               )
 334             )
 335           )
 336         case 'STOP':
 337           return this.#finish(
 338             fromStop(
 339               stop(
 340                 'INSUFFICIENT_EVIDENCE',
 341                 decision.response ?? decision.reason ?? 'Execution stopped.'
 342               )
 343             )
 344           )
 345         case 'RETRIEVE':
 346         case 'VERIFY':
 347           return this.#finish(
 348             fromStop(
 349               stop(
 350                 'INSUFFICIENT_EVIDENCE',
 351                 'This single-pass compatibility runtime does not execute retrieval or verification steps.'
 352               )
 353             )
 354           )
 355       }
 356     } catch (error) {
 357       return this.#finish(
 358         fromStop(
 359           stop(
 360             'INSUFFICIENT_EVIDENCE',
 361             `Execution could not be planned: ${errorMessage(error)}.`
 362           )
 363         )
 364       )
 365     }
 366   }
 367 
 368   async #respond(response: string | undefined): Promise<RuntimeResult> {
 369     if (response?.trim()) {
 370       return this.#finish(
 371         result({
 372           response,
 373           stopReason: 'COMPLETED',
 374           modelCalls: 0,
 375           toolCalls: 0
 376         })
 377       )
 378     }
 379     const input = this.#state.input
 380     // Single pre-model gate: runModelCall (kernel/pipeline.ts) consumes the
 381     // model-call slot via turn.budget.beginModelRequest() and records tokens/
 382     // cost via recordModelResult() (PDSH-004 / I11). Gating here too would
 383     // double-consume the slot for one turn.
 384     const outcome = await runModelCall(this.#host, this.#state, {
 385       messages: [
 386         {
 387           role: 'system',
 388           content: `${input.agent.objective}\n${input.agent.instructions.join('\n')}`
 389         },
 390         { role: 'user', content: input.userMessage }
 391       ],
 392       context: input.context,
 393       budget: input.budget,
 394       correlationId: input.correlationId
 395     })
 396     switch (outcome.kind) {
 397       case 'stopped':
 398         return this.#finish(fromStop(outcome.gate))
 399       case 'cancelled':
 400         return this.#finish(
 401           fromStop(stop('CANCELLED', CANCELLED_RESPONSE, { modelCalls: 1 }))
 402         )
 403       case 'deadline':
 404         return this.#finish(
 405           fromStop(
 406             stop(
 407               'MAX_DURATION',
 408               'Duration budget exhausted during model execution.',
 409               { modelCalls: 1 }
 410             )
 411           )
 412         )
 413       case 'failed':
 414         return this.#finish(
 415           fromStop(
 416             stop('MODEL_FAILURE', `Model execution failed: ${outcome.error}.`, {
 417               modelCalls: 1
 418             })
 419           )
 420         )
 421       case 'unrecorded':
 422         return this.#finish(
 423           fromStop({
 424             ...outcome.gate,
 425             modelCalls: 1,
 426             usage: {
 427               inputTokens: outcome.result.inputTokens,
 428               outputTokens: outcome.result.outputTokens,
 429               costUsd: outcome.result.costUsd
 430             }
 431           })
 432         )
 433       case 'completed': {
 434         const model = outcome.result
 435         const metadata = this.#state.metadata
 436         metadata.provider = model.provider
 437         metadata.inputTokens = model.inputTokens
 438         metadata.outputTokens = model.outputTokens
 439         metadata.costUsd = model.costUsd
 440         const exhausted = this.#state.budget.recordModelResult(model)
 441         if (exhausted) return this.#finish(fromStop(exhausted))
 442         return this.#finish(
 443           result({
 444             response: model.text,
 445             stopReason: 'COMPLETED',
 446             modelCalls: 1,
 447             toolCalls: 0,
 448             inputTokens: model.inputTokens,
 449             outputTokens: model.outputTokens,
 450             costUsd: model.costUsd
 451           })
 452         )
 453       }
 454     }
 455   }

 457   async #callTool(
 458     requested: ToolInvocation | undefined
 459   ): Promise<RuntimeResult> {
 460     const input = this.#state.input
 461     const invocation = requested ?? input.requestedTool
 462     if (!invocation) {
 463       return this.#finish(
 464         fromStop(stop('TOOL_FAILURE', 'No tool invocation was supplied.'))
 465       )
 466     }
 467     if (input.budget.maxToolCalls < 1) {
 468       return this.#finish(
 469         fromStop(
 470           stop(
 471             'MAX_TOOL_CALLS',
 472             'Tool-call budget exhausted before the tool could run.'
 473           )
 474         )
 475       )
 476     }
 477     const tool = this.#host
 478       .get('tools')
 479       .resolve(invocation.toolId, invocation.toolVersion)
 480     this.#state.metadata.tool = invocation.toolId
 481     if (!tool) {
 482       return this.#finish(
 483         fromStop(
 484           stop('TOOL_FAILURE', `Tool "${invocation.toolId}" is unavailable.`, {
 485             toolCalls: 1
 486           })
 487         )
 488       )
 489     }
 490     if (!exposes(input.agent.tools, tool.id)) {
 491       return this.#finish(
 492         fromStop(
 493           stop(
 494             'POLICY_DENIED',
 495             `Tool "${tool.id}" is not exposed to this agent profile.`
 496           )
 497         )
 498       )
 499     }
 500 
 501     const call: ToolCallState = {
 502       turn: this.#state,
 503       tool,
 504       invocation,
 505       needsApproval: false
 506     }
 507 
 508     const pipeline = await runToolCall(this.#host, call)
 509     if (pipeline.kind === 'stopped') {
 510       const approvalId = pipeline.gate.approvalId ?? call.grant?.approvalId
 511       return this.#finish(
 512         fromStop({
 513           ...pipeline.gate,
 514           ...(approvalId !== undefined ? { approvalId } : {})
 515         })
 516       )
 517     }
 518     const settled = pipeline.outcome
 519     this.#state.metadata.toolDurationMs =
 520       'durationMs' in settled ? settled.durationMs : 0
 521     const toolResult = this.#toolResult(call, settled)
 522     if (pipeline.logFailure) {
 523       // F04: the effect fact stays in the result; the turn is not a success.
 524       // A call blocked after its reservation keeps the reason it was blocked
 525       // (AUD-0599 F01).
 526       return this.#finish({
 527         ...toolResult,
 528         response:
 529           settled.kind === 'not_started'
 530             ? unrecordedBlock(pipeline.logFailure, settled.gate).response
 531             : pipeline.logFailure.response,
 532         stopReason: 'INSUFFICIENT_EVIDENCE'
 533       })
 534     }
 535     return this.#finish(toolResult)
 536   }
 537 
 538   #toolResult(call: ToolCallState, outcome: ToolOutcome): RuntimeResult {
 539     const approvalId = call.grant?.approvalId
 540     const withApproval = approvalId !== undefined ? { approvalId } : {}
 541     switch (outcome.kind) {
 542       case 'not_started':
 543         return fromStop({ ...outcome.gate, ...withApproval })
 544       case 'deadline':
 545         return result({
 546           response: call.approval
 547             ? 'unknown_effect: duration budget exhausted during tool execution.'
 548             : 'Duration budget exhausted during tool execution.',
 549           stopReason: call.approval ? 'TOOL_FAILURE' : 'MAX_DURATION',
 550           ...withApproval,
 551           modelCalls: 0,
 552           toolCalls: 1
 553         })
 554       case 'cancelled':
 555         return result({
 556           response: call.approval
 557             ? `unknown_effect: ${CANCELLED_RESPONSE}`
 558             : CANCELLED_RESPONSE,
 559           stopReason: call.approval ? 'TOOL_FAILURE' : 'CANCELLED',
 560           ...withApproval,
 561           modelCalls: 0,
 562           toolCalls: 1
 563         })
 564       case 'threw': {
 565         const toolResult: ToolResult = {
 566           status: 'FAILED',
 567           error: outcome.error
 568         }
 569         return result({
 570           response: call.approval
 571             ? `unknown_effect: ${outcome.error}`
 572             : outcome.error,
 573           stopReason: 'TOOL_FAILURE',
 574           ...withApproval,
 575           modelCalls: 0,
 576           toolCalls: 1,
 577           toolResult
 578         })
 579       }
 580       case 'failed':
 581         return result({
 582           response: outcome.result.error ?? 'Tool execution failed.',
 583           stopReason: 'TOOL_FAILURE',
 584           ...withApproval,
 585           modelCalls: 0,
 586           toolCalls: 1,
 587           toolResult: outcome.result
 588         })
 589       case 'confirm_failed':
 590         return result({
 591           response: `unknown_effect: approval confirmation failed: ${outcome.error}.`,
 592           stopReason: 'TOOL_FAILURE',
 593           ...withApproval,
 594           modelCalls: 0,
 595           toolCalls: 1,
 596           toolResult: outcome.result
 597         })
 598       case 'succeeded':
 599         if (
 600           Date.now() - this.#state.startedAt >
 601           this.#state.input.budget.maxDurationMs
 602         ) {
 603           return result({
 604             response: 'Duration budget exhausted after tool execution.',
 605             stopReason: 'MAX_DURATION',
 606             modelCalls: 0,
