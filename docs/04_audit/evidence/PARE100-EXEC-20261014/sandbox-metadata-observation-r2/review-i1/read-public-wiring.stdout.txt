  71   public constructor(registrations: readonly CapabilityRegistration[] = []) {
  72     const stored: StoredRegistration[] = []
  73     for (const registration of registrations) {
  74       const normalized = normalizeRegistration(registration)
  75       if (
  76         stored.some(
  77           (candidate) =>
  78             candidate.descriptor.id === normalized.descriptor.id &&
  79             candidate.descriptor.version === normalized.descriptor.version
  80         )
  81       ) {
  82         throw new CapabilityRegistryError(
  83           'duplicate_capability',
  84           `Capability already registered: ${normalized.descriptor.id}@${normalized.descriptor.version}`
  85         )

  88     }

  98           candidate.descriptor.version === normalized.descriptor.version

 103         `Capability already registered: ${normalized.descriptor.id}@${normalized.descriptor.version}`

 105     }

 207 function normalizeDescriptor(value: unknown): CapabilityDescriptor {
 208   if (!isPlainRecord(value)) {
 209     throw new CapabilityRegistryError(
 210       'invalid_descriptor',
 211       'Capability descriptor must be a plain object'
 212     )
 213   }
 214 
 215   const descriptor = value as Record<string, unknown>
 216   if (!isBoundedIdentifier(descriptor.id, MAX_ID_LENGTH)) {
 217     throw invalidDescriptor('Capability id is invalid')
 218   }
 219   if (!isBoundedVersion(descriptor.version)) {
 220     throw invalidDescriptor('Capability version must be exact and bounded')
 221   }
 222   if (
 223     typeof descriptor.description !== 'string' ||
 224     descriptor.description.trim().length === 0 ||
 225     descriptor.description.length > MAX_DESCRIPTION_LENGTH
 226   ) {
 227     throw invalidDescriptor('Capability description is invalid')
 228   }
 229   if (!ORIGINS.includes(descriptor.origin as CapabilityOrigin)) {
 230     throw invalidDescriptor('Capability origin is invalid')
 231   }
 232   if (!isBoundedIdentifier(descriptor.providerId, MAX_ID_LENGTH)) {
 233     throw invalidDescriptor('Capability provider id is invalid')
 234   }
 235   if (!isBoundedVersion(descriptor.providerVersion)) {
 236     throw invalidDescriptor('Capability provider version is invalid')
 237   }
 238   if (
 239     !isRisk(descriptor.risk) ||
 240     !isSideEffect(descriptor.sideEffect) ||
 241     typeof descriptor.idempotent !== 'boolean' ||
 242     typeof descriptor.requiresApproval !== 'boolean'
 243   ) {
 244     throw invalidDescriptor('Capability governance metadata is invalid')
 245   }
 246 
 247   const inputSchema = cloneJsonValue(descriptor.inputSchema)
 248   const outputSchema = cloneJsonValue(descriptor.outputSchema)
 249   return deepFreeze({
 250     id: descriptor.id,
 251     version: descriptor.version,
 252     description: descriptor.description.trim(),
 253     inputSchema,
 254     outputSchema,
 255     risk: descriptor.risk,
 256     sideEffect: descriptor.sideEffect,

 259     origin: descriptor.origin,

 351 async function executeBound(
 352   implementation: CapabilityImplementation,
 353   input: unknown,
 354   context: ToolExecutionContext
 355 ): Promise<ToolResult> {
 356   let safeInput: unknown
 357   try {
 358     safeInput = cloneJsonValue(input)
 359     if (!inputMatchesExecutionContext(safeInput, context)) {
 360       return { status: 'REJECTED', error: 'capability_input_invalid' }
 361     }
 362     if (implementation.validateInput(safeInput) !== true) {
 363       return { status: 'REJECTED', error: 'capability_input_invalid' }
 364     }
 365   } catch {
 366     return { status: 'REJECTED', error: 'capability_input_invalid' }
 367   }
 368 
 369   let result: ToolResult
 370   try {
 371     const safeContext = Object.freeze({ ...context }) as ToolExecutionContext
 372     const rawResult = await implementation.execute(safeInput, safeContext)
 373     const normalizedResult = normalizeToolResult(rawResult)
 374     if (!normalizedResult) {
 375       return { status: 'REJECTED', error: 'capability_result_invalid' }
 376     }
 377     result = normalizedResult
 378   } catch {
 379     return { status: 'FAILED', error: 'capability_execution_failed' }
 380   }
 381 
 382   if (result.status === 'SUCCEEDED') {
 383     try {
 384       if (implementation.validateOutput(result.output) !== true) {
 385         return { status: 'REJECTED', error: 'capability_output_invalid' }
 386       }
 387     } catch {
 388       return { status: 'REJECTED', error: 'capability_output_invalid' }
 389     }
 390   }
 391 
 392   try {
 393     return {
 394       status: result.status,
 395       ...(result.output !== undefined
 396         ? { output: cloneJsonValue(result.output) }
 397         : {}),
 398       ...(result.error !== undefined ? { error: result.error } : {})
 399     }
 400   } catch {
 401     return { status: 'REJECTED', error: 'capability_result_invalid' }
 402   }

 429 function inputMatchesExecutionContext(
 430   value: unknown,
 431   context: ToolExecutionContext
 432 ): boolean {
 433   if (Array.isArray(value)) {
 434     return value.every((child) => inputMatchesExecutionContext(child, context))
 435   }
 436   if (!isPlainRecord(value)) return true
 437 
 438   for (const [key, child] of Object.entries(value)) {
 439     if (
 440       (AUTHORITY_FIELDS as readonly string[]).includes(key) &&
 441       child !== context[key as (typeof AUTHORITY_FIELDS)[number]]
 442     ) {
 443       return false
 444     }
 445     if (!inputMatchesExecutionContext(child, context)) return false
 446   }
 447   return true

 499     if (
 500       error !== undefined &&
 501       (typeof error !== 'string' || error.length > 240)
 502     ) {
 503       return undefined
 504     }
 505     const output = safeValue.output
 506     return {
 507       status,
 508       ...(output !== undefined ? { output } : {}),
 509       ...(error !== undefined ? { error } : {})
 510     }
 511   } catch {
 512     return undefined
 513   }
 514 }

 524 interface SerializationState {
 525   readonly seen: WeakSet<object>
 526   nodes: number
 527 }
 528 
 529 function stableSerialize(
 530   value: unknown,
 531   state: SerializationState = { seen: new WeakSet<object>(), nodes: 0 },
 532   depth = 0
 533 ): string {
 534   state.nodes += 1
 535   if (state.nodes > MAX_JSON_NODES) {
 536     throw new Error('JSON node budget exceeded')
 537   }
 538   if (depth > MAX_JSON_DEPTH) {
 539     throw new Error('JSON depth budget exceeded')
 540   }
 541 
 542   if (value === null) return 'null'
 543   if (typeof value === 'string') {
 544     if (value.length > MAX_JSON_STRING_LENGTH) {
 545       throw new Error('JSON string budget exceeded')
 546     }
 547     return JSON.stringify(value)
 548   }
 549   if (typeof value === 'boolean') return value ? 'true' : 'false'
 550   if (typeof value === 'number') {
 551     if (!Number.isFinite(value)) throw new Error('Non-finite value')
 552     return JSON.stringify(value)
 553   }
 554   if (typeof value !== 'object' || value === undefined) {
 555     throw new Error('Value is not JSON serializable')
 556   }
 557   if (state.seen.has(value)) throw new Error('Value contains a cycle')
 558   state.seen.add(value)
 559   try {
 560     if (Array.isArray(value)) {
 561       if (value.length > MAX_JSON_ARRAY_LENGTH) {
 562         throw new Error('JSON array budget exceeded')
 563       }
 564       const items: string[] = []
 565       for (let index = 0; index < value.length; index += 1) {
 566         if (!Object.prototype.hasOwnProperty.call(value, index)) {
 567           throw new Error('Sparse arrays are not supported')
 568         }
 569         items.push(stableSerialize(value[index], state, depth + 1))
 570       }
 571       return boundedSerialized(`[${items.join(',')}]`)
 572     }
 573     if (!isPlainRecord(value)) throw new Error('Value is not a plain object')
 574     const keys = Object.keys(value)
 575     if (
 576       keys.length > MAX_JSON_KEYS ||
 577       Object.getOwnPropertySymbols(value).length > 0 ||
 578       Object.getOwnPropertyNames(value).length !== keys.length
 579     ) {
 580       throw new Error('JSON object budget or shape exceeded')
 581     }
 582     const parts: string[] = []
 583     for (const key of keys.sort()) {
 584       if (key.length > MAX_JSON_KEY_LENGTH || UNSAFE_JSON_KEYS.has(key)) {
 585         throw new Error('Unsafe JSON object key')
 586       }
 587       const property = Object.getOwnPropertyDescriptor(value, key)
 588       if (!property || !('value' in property)) {
 589         throw new Error('Accessor properties are not supported')
 590       }
 591       parts.push(
 592         `${JSON.stringify(key)}:${stableSerialize(property.value, state, depth + 1)}`
 593       )
 594     }
 595     return boundedSerialized(`{${parts.join(',')}}`)
 596   } finally {
 597     state.seen.delete(value)
 598   }
 599 }

 601 function boundedSerialized(value: string): string {
 602   if (value.length > MAX_JSON_BYTES) {
 603     throw new Error('JSON byte budget exceeded')
 604   }
 605   return value
 606 }

 608 function deepFreeze<T>(value: T): T {
 609   if (typeof value !== 'object' || value === null) return value
 610   Object.freeze(value)
 611   for (const child of Object.values(value)) deepFreeze(child)
 612   return value
 613 }

 615 function isPlainRecord(value: unknown): value is Record<string, unknown> {
 616   try {
 617     if (typeof value !== 'object' || value === null || Array.isArray(value)) {
 618       return false
 619     }
 620     const prototype = Object.getPrototypeOf(value)
 621     return prototype === Object.prototype || prototype === null
 622   } catch {
 623     return false
 624   }
