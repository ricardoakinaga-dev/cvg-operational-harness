  32 function capture(value: unknown, depth = 0): unknown {
  33   if (depth > 64) throw new Error("observation depth exceeded");
  34   if (value === undefined) return { $type: "undefined" };
  35   if (typeof value === "function") return { $type: "function" };
  36   if (
  37     value === null ||
  38     typeof value === "string" ||
  39     typeof value === "boolean"
  40   ) {
  41     return value;
  42   }
  43   if (typeof value === "number" && Number.isFinite(value)) return value;
  44   if (value instanceof AbortSignal) {
  45     return {
  46       $type: "AbortSignal",
  47       aborted: value.aborted,
  48       reason: capture(value.reason, depth + 1),
  49     };
  50   }
  51   if (Array.isArray(value))
  52     return value.map((item) => capture(item, depth + 1));
  53   if (typeof value !== "object")
  54     throw new Error("unsupported observation value");
  55   const proto: unknown = Object.getPrototypeOf(value);
  56   if (proto !== Object.prototype && proto !== null) {
  57     throw new Error("unsupported observation object");
  58   }
  59   const result: Record<string, unknown> = {};
  60   for (const key of Reflect.ownKeys(value)) {
  61     if (typeof key !== "string") throw new Error("unsupported observation key");
  62     const property = Object.getOwnPropertyDescriptor(value, key);
  63     if (!property || !("value" in property)) {
  64       throw new Error("observation getter refused");
  65     }
  66     result[key] = capture(property.value, depth + 1);
  67   }
  68   return result;
  69 }
