1: /**
2:  * Local sandbox backend. It selects the platform runner chain (Linux bwrap then
3:  * Landlock; macOS Seatbelt; Windows the ACL restricted-token runner), functionally probes
4:  * competing candidates once, and reports each wrap's enforcement and stderr
5:  * classification facts. Missing or unusable confinement fails closed rather
6:  * than returning the original argv.
7:  *
8:  * The windows-acl rung additionally owns the write grants: the write SID is
9:  * the per-WORKSPACE identity derived from the canonical workspace path
10:  * (`workspaceWriteSid`), while every live session receives a RANDOM private
11:  * temp directory and its own derived capability (`tempWriteSid`). The
12:  * workspace-root ACE materializes once per workspace per server lifetime
13:  * and STANDS (the cross-session reuse cache — the exact-ACE skip makes
14:  * every later provision O(1) instead of re-propagating the tree per
15:  * session); the private-temp ACEs are revoked on dispose. The runner
16:  * receives both SIDs (their presence marks the seam-managed contract) and
17:  * stops managing DACLs itself. The rung reports partial enforcement because
18:  * NTFS hard links alias one file object across paths, reads stay unconfined,
19:  * and a tree another AppContainer tool has ACL'd with a package SID is not
20:  * readable by the Low-integrity child.
21:  * @module @deepseek-ai/dsh-sandbox-local
22:  */

81: /** Probe whether `bwrap` can create the profile; the provider caches the bounded result. */
82: function defaultProbeBwrap(timeoutMs: number): boolean {
83:   const workspaceRoot = mkdtempSync(join(tmpdir(), 'dsh-bwrap-probe-'))
84:   try {
85:     const probe = spawnSync('bwrap', [...bwrapProfileArgs({ mode: 'workspace-write', workspaceRoot }), '--', 'true'], {
86:       timeout: timeoutMs,
87:       stdio: 'ignore',
88:     })
89:     return probe.status === 0
90:   } finally {
91:     rmSync(workspaceRoot, { recursive: true, force: true })
92:   }
93: }

95: /**
96:  * Functional Seatbelt probe: apply the real `read-only` profile through
97:  * `sandbox-exec -p` and run `true` under it — exit 0 means the kernel
98:  * accepted and enforced the profile (`sandbox-exec` exits non-zero when
99:  * `sandbox_init` refuses it). A missing `sandbox-exec` (every non-macOS
100:  * host) fails the spawn and probes `unusable`, exactly like the other
101:  * rungs' absent binaries. Apple marks the CLI deprecated but ships it on
102:  * every macOS; if it ever disappears, this probe is what fails closed.
103:  */
104: function defaultProbeSeatbelt(seatbeltExec: string, timeoutMs: number): boolean {
105:   const probe = spawnSync(seatbeltExec, [...seatbeltProfileArgs({ mode: 'read-only', workspaceRoot: '/' }), '--', 'true'], {
106:     timeout: timeoutMs,
107:     stdio: 'ignore',
108:   })
109:   return probe.status === 0
110: }

112: /**
113:  * Functional windows-acl probe: run the runner in read-only mode (zero grants,
114:  * no ACL mutation) around `cmd /c exit 0` — exit 0 means the runner created
115:  * the restricted token and spawned the child under it. The win32 chain is a
116:  * sole candidate, so the product never probes; the probe exists for override
117:  * chains and mirrors the other rungs' shape.
118:  */
119: function defaultProbeWindowsAcl(runnerInvocation: string[], timeoutMs: number): boolean {
120:   const program = runnerInvocation[0]
121:   if (program === undefined) return false
122:   const probe = spawnSync(program, [
123:     ...runnerInvocation.slice(1),
124:     '--workspace', tmpdir(), '--temp', tmpdir(), '--mode', 'read-only',
125:     '--', 'cmd', '/c', 'exit', '0',
126:   ], {
127:     timeout: timeoutMs,
128:     stdio: 'ignore',
129:   })
130:   return probe.status === 0

262: /**
263:  * Local process-sandbox provider. Registers as `ctx.sandbox`. Caches the
264:  * chain verdict and, on the windows-acl rung, the write grants
265:  * ({@link AclWriteGrant}: the standing workspace-root grant per workspace
266:  * and the revocable private-temp grant per live session/workspace pair, the
267:  * latter revoked on provider dispose); the one-time probes spawn nothing
268:  * else.
269:  */
270: export class LocalSandboxProvider extends SandboxProvider {
271:   // Inline schema call: the config catalog walks `static Config` statically.
272:   static Config: z<Config> = z.object({
273:     runnerCommand: z.array(z.string()).default([]),
274:     runnerFailureSignatures: z.array(z.string()).default([]),
275:     probeTimeoutMs: z.natural().default(5_000),
276:   })
277: 
278:   /** Test hook (mirrors the bash executors' `internals`). */
279:   internals: SandboxInternals = {}
280: 
281:   private readonly runnerCommand: string[] | undefined
282:   private readonly configuredRunnerFailureSignatures: string[]
283:   private readonly probeTimeoutMs: number
284:   /** Cached chain verdict; undefined until the first confined wrap needs it. */
285:   private selectedRunner: SelectedRunner | 'unavailable' | undefined
286:   /**
287:    * Server-lifetime write grants (windows-acl rung): the STANDING
288:    * workspace-root grant per workspace (its ACE is the cross-session reuse
289:    * cache and outlives the provider — never revoked) and the REVOCABLE
290:    * private-temp grant per live session/workspace pair (revoked on provider
291:    * dispose).
292:    */
293:   private readonly workspaceGrants = new Map<string, AclWriteGrant>()
294:   private readonly tempCapabilities = new Map<string, AclTempCapability>()
295: 
296:   constructor(ctx: Context, config: Config) {
297:     super(ctx)
298:     // The schema (static Config) defaults every field — the casts record
299:     // those runtime facts. An empty runnerCommand means "not configured":
300:     // use the platform chain.
301:     const runner = config.runnerCommand as string[]
302:     const runnerFailureSignatures = config.runnerFailureSignatures as string[]
303:     if (runner.length === 0 && runnerFailureSignatures.length > 0) {
304:       throw new Error('sandbox-local: runnerFailureSignatures requires runnerCommand')
305:     }
306:     if (runner.length > 0 && runnerFailureSignatures.length === 0) {
307:       throw new Error('sandbox-local: runnerCommand requires at least one runnerFailureSignatures entry')
308:     }
309:     if (runnerFailureSignatures.some(signature => signature.trim().length === 0 || /[\r\n]/u.test(signature))) {
310:       throw new Error('sandbox-local: runnerFailureSignatures entries must be non-empty single-line strings')
311:     }
312:     this.runnerCommand = runner.length > 0 ? runner : undefined
313:     this.configuredRunnerFailureSignatures = runnerFailureSignatures
314:     this.probeTimeoutMs = config.probeTimeoutMs as number
315:     assertPositiveFinite('probeTimeoutMs', this.probeTimeoutMs)
316:     // An operator-supplied runner does not use the ACL backend. The registry
317:     // remains optional and may be mounted after this provider.
318:     /* v8 ignore next 3 -- Windows-only registration; the Linux coverage lane cannot take this branch */
319:     if (process.platform === 'win32' && this.runnerCommand === undefined) {
320:       ctx.inject(['skills'], (skillsCtx) => { registerAclDiagnosisSkill(skillsCtx) })
321:     }
322:     // The temp grants are revoked with the provider: a clean server
323:     // shutdown leaves no temp ACEs behind (workspace ACEs stand by design —
324:     // the reuse cache; an unclean shutdown leaves them for the next
325:     // provision's exact-ACE skip).
326:     ctx.effect(() => () => {
327:       this.revokeAclGrants()
328:     })
329:   }
330: 
331:   /**
332:    * Wrap `argv` in the selected runner's invocation for `policy` — the configured
333:    * `runnerCommand` when present (the operator's assertion, no probe), else the platform
334:    * chain's runner speaking its own profile dialect.
335:    *
336:    * @param argv - the exact argv the caller is about to spawn.
337:    * @param policy - the file-effect policy this execution runs under.
338:    * @param signal - cancellation before policy resolution or grant creation.
339:    * @returns the wrapped argv plus the selected backend's enforcement completeness, denial
340:    *   signatures, and structured runner-failure rules; throws the fail-closed
341:    *   `SANDBOX_UNAVAILABLE` error when the platform has no usable runner.
342:    */
343:   async confine(argv: readonly string[], policy: SandboxPolicy, signal?: AbortSignal): Promise<ConfinedArgv> {
344:     signal?.throwIfAborted()
345:     policy = { ...policy, workspaceRoot: canonicalPath(policy.workspaceRoot) }
346:     if (this.runnerCommand !== undefined) {
347:       return Promise.resolve<ConfinedArgv>({
348:         argv: [...this.runnerCommand, ...bwrapProfileArgs(policy), '--', ...argv],
349:         enforcement: 'full',
350:         denialSignatures: DENIAL_SIGNATURES.runnerCommand,
351:         runnerFailureRules: [{ fatalSignatures: this.configuredRunnerFailureSignatures }],
352:       })
353:     }
354:     const selected = this.selectRunner(policy.mode)
355:     const runnerArgv = this.runnerArgv(selected.runner, policy)
356:     return Promise.resolve<ConfinedArgv>({
357:       argv: [...runnerArgv, '--', ...argv],
358:       enforcement: selected.enforcement,
359:       denialSignatures: DENIAL_SIGNATURES[selected.runner],
360:       runnerFailureRules: RUNNER_FAILURE_RULES[selected.runner],
361:     })
362:   }
363: 
364:   /**
365:    * Describe the selected runner's temporary-files and identity semantics.
366:    * @param policy - complete policy whose mode determines temporary mounts.
367:    * @returns facts from the same cached selection used by confine.
368:    */
369:   override describe(policy: SandboxPolicy): SandboxExecutionEnvironment {
370:     if (this.runnerCommand !== undefined) {
371:       return { runner: 'configured', enforcement: 'full', temporaryFiles: 'unknown', sharesHostTemporaryFiles: 'unknown', identity: 'unknown' }
372:     }
373:     this.selectedRunner ??= this.chainVerdict()
374:     if (this.selectedRunner === 'unavailable') {
375:       return { runner: 'unavailable', enforcement: 'unavailable', temporaryFiles: 'unknown', sharesHostTemporaryFiles: 'unknown', identity: 'unknown' }
376:     }
377:     const { runner, enforcement } = this.selectedRunner
378:     switch (runner) {
379:       case 'bwrap': {
380:         const privateTemporaryFiles = policy.mode === 'workspace-write'
381:           && policy.workspaceRoot !== '/' && policy.workspaceRoot !== '/tmp'
382:         return { runner, enforcement, temporaryFiles: privateTemporaryFiles ? 'per-call' : 'host', sharesHostTemporaryFiles: !privateTemporaryFiles, identity: 'namespace-dependent' }
383:       }
384:       case 'landlock':
385:       case 'seatbelt':
386:         return { runner, enforcement, temporaryFiles: 'host', sharesHostTemporaryFiles: true, identity: 'host' }
387:       case 'windows-acl':
388:         return { runner, enforcement, temporaryFiles: policy.mode === 'workspace-write' ? policy.sessionId === undefined ? 'per-call' : 'per-session' : 'host', sharesHostTemporaryFiles: policy.mode !== 'workspace-write', identity: 'restricted-token' }
389:       default: return assertNever(runner)
390:     }
391:   }
392: 
393:   /** The selected rung's runner invocation (program + profile arguments) for one policy. */
394:   private runnerArgv(runner: SelectedRunner['runner'], policy: SandboxPolicy): string[] {
395:     switch (runner) {
396:       case 'bwrap': return ['bwrap', ...bwrapProfileArgs(policy)]
397:       case 'landlock': return [this.landlockLauncher(), ...landlockProfileArgs(policy)]
398:       case 'seatbelt': return [this.seatbeltExec(), ...seatbeltProfileArgs(policy)]
399:       case 'windows-acl': return this.windowsAclRunnerArgv(policy)
400:       default: return assertNever(runner)
401:     }
