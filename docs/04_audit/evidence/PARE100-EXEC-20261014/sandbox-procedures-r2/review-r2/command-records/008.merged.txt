20:  * Attribute only Node ENOENT/EACCES failures whose error path equals argv[0]
21:  * after independently ruling out the caller-owned cwd. A supplied error path
22:  * must exactly identify the runner; without one, the syscall must. With a
23:  * usable cwd, these codes describe resolution or execute permission for that
24:  * argv[0] or its shebang interpreter.
25:  * The workdir is checked at classification time, not atomically with spawn;
26:  * concurrent path replacement may change attribution but cannot permit an
27:  * unconfined execution.
28:  * @param error - the original spawn rejection.
29:  * @param runnerProgram - provider argv[0], the executable that establishes confinement.
30:  * @param workdir - the caller-owned spawn cwd, checked independently for usability.
31:  * @returns whether the rejection has executable-specific runner evidence.
32:  */

34:   error: unknown,
35:   runnerProgram: string | undefined,
36:   workdir: string,
37: ): boolean {
38:   if (runnerProgram === undefined || !isUsableWorkdir(workdir)) return false
39:   if (typeof error !== 'object' || error === null) return false
40:   const { code, path, syscall } = error as { code?: unknown; path?: unknown; syscall?: unknown }
41:   if (typeof code !== 'string' || !EXECUTABLE_SPAWN_CODES.has(code)) return false
42:   if (typeof syscall !== 'string') return false
43:   const exactSyscall = `spawn ${runnerProgram}`
44:   if (path === undefined) return syscall === exactSyscall
45:   if (typeof path !== 'string' || path.length === 0 || path !== runnerProgram) return false
46:   return syscall === 'spawn' || syscall === exactSyscall

55: /**
56:  * Classify one settled process against the selected backend's structured
57:  * runner-failure rules. Each rule requires a nonzero exit, its optional
58:  * exit-code gate, and a fatal signature on one stderr line after exact
59:  * informational lines are excluded.
60:  * @param exitCode - process exit code; null means signal termination.
61:  * @param stderr - collected stderr text, left unchanged.
62:  * @param rules - structured runner-failure rules from the active wrap.
63:  * @returns the first matching fatal line, or undefined when evidence is insufficient.
64:  */
65: export function classifyRunnerFailure(
66:   exitCode: number | null,
67:   stderr: string,
68:   rules: readonly RunnerFailureRule[],
69: ): RunnerFailureMatch | undefined {
70:   if (exitCode === null || exitCode === 0) return undefined
71:   const lines = stderr.split(/\r?\n/)
72:   for (const rule of rules) {
73:     if (rule.allowedExitCodes !== undefined && !rule.allowedExitCodes.includes(exitCode)) continue
74:     const informationalLines = new Set((rule.informationalLines ?? []).map(line => line.toLowerCase()))
75:     // An empty or whitespace-only substring is not meaningful runner evidence.
76:     // Ignore it while keeping any valid signatures beside it active.
77:     const fatalSignatures = rule.fatalSignatures
78:       .filter(signature => signature.trim().length > 0)
79:       .map(signature => signature.toLowerCase())
80:     for (const line of lines) {
81:       const lowered = line.toLowerCase()
82:       if (informationalLines.has(lowered)) continue
83:       if (fatalSignatures.some(signature => lowered.includes(signature))) return { detail: line }
84:     }
85:   }
86:   return undefined
87: }

59:    * table (lazily, once per server). Fail-closed: any failure throws — nothing
60:    * is granted yet.

95:    * recorded BEFORE the grant: a post-apply throw (a LocalFree failure after
96:    * SetNamedSecurityInfoW succeeded) must still revoke it, and revoking an
97:    * ungranted path is a no-op merge. Callers treat a throw as a failed
98:    * materialization and dispose the instance to revoke the paths granted so
