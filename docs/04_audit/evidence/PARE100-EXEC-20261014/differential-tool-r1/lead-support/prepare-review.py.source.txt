import hashlib
import json
import shutil
import subprocess
from pathlib import Path

PARENT = Path('/tmp/cvg-pare100-exec-20261014-6az3hlay')
WORK = PARENT / 'worktree'
PRIVATE = PARENT / 'differential-tool-r1'
PACKET = PRIVATE / 'review-packet'
REF = Path('/home/ricardo/deepseek-harness-session-fixes')


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def write(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, indent=2, ensure_ascii=False) + '\n')


assert not PACKET.exists()
PACKET.mkdir()
sources = []


def copy(src, rel):
    dst = PACKET / rel
    dst.parent.mkdir(parents=True, exist_ok=True)
    shutil.copy2(src, dst)
    sources.append({'live': str(src), 'copy': rel, 'sha256': sha(src)})


for src in (WORK / 'tests/differential').iterdir():
    if src.is_file():
        copy(src, str(Path('cvg/tests/differential') / src.name))
for pkg in ['contracts', 'harness', 'orchestrator', 'policy', 'approval-engine', 'shared']:
    directory = WORK / 'packages' / pkg
    if not directory.exists():
        # Resolve the actual public package folder; no silent omission.
        for candidate in (WORK / 'packages').iterdir():
            manifest = candidate / 'package.json'
            if manifest.is_file() and json.loads(manifest.read_text()).get('name') == '@cvg/' + pkg:
                directory = candidate
                break
    assert directory.exists(), pkg
    for src in (directory / 'src').rglob('*'):
        if src.is_file() and '__tests__' not in src.parts:
            copy(src, 'cvg/' + str(src.relative_to(WORK)))
    for name in ['package.json', 'tsconfig.json']:
        copy(directory / name, 'cvg/' + str((directory / name).relative_to(WORK)))
for rel in ['packages/sdk/src/server.ts', 'packages/sdk/src/client.ts', 'package.json', 'package-lock.json', 'vitest.config.mts', 'tsconfig.base.json', 'tsconfig.typecheck.json']:

]
for rel in ref_sources:
    copy(REF / rel, 'reference/' + rel)
