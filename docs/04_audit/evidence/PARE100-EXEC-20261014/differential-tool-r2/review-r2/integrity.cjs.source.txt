const fs=require('fs'),p=require('path'),c=require('crypto');const root=p.resolve('../review-packet'); const hash=x=>c.createHash('sha256').update(fs.readFileSync(x)).digest('hex');const walk=d=>fs.readdirSync(d,{withFileTypes:true}).flatMap(e=>e.isDirectory()?walk(p.join(d,e.name)):[p.join(d,e.name)]);const sm=JSON.parse(fs.readFileSync(root+'/source-manifest.json')),pm=JSON.parse(fs.readFileSync(root+'/packet-manifest.json'));const out={phase:process.argv[2],time:new Date().toISOString(),sources:sm.map(x=>({...x,copyHash:hash(root+'/'+x.copy),liveHash:hash(x.live)})),packet:pm.map(x=>({...x,actual:hash(root+'/'+x.path)})),membership:walk(root).map(x=>({path:p.relative(root,x),sha256:hash(x)})).sort((a,b)=>a.path.localeCompare(b.path))};out.summary={sourcePairs:sm.length,packetEntries:pm.length,members:out.membership.length,sourceMismatch:out.sources.filter(x=>x.sha256!==x.copyHash||x.sha256!==x.liveHash),packetMismatch:out.packet.filter(x=>x.sha256!==x.actual),unlisted:out.membership.filter(x=>!pm.some(y=>y.path===x.path)).map(x=>x.path)};fs.writeFileSync('integrity-'+process.argv[2]+'.json',JSON.stringify(out,null,2));console.log(out.summary);
