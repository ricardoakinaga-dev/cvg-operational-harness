from pathlib import Path
import datetime, hashlib, json, os, signal, subprocess, sys, time
base = Path(__file__).parent
w = base.parents[1] / 'worktree'
label = sys.argv[1]
node = '/home/ricardo/.nvm/versions/node/v22.23.2/bin/node'
commands = {'typecheck': ['npm', 'run', 'typecheck'], 'lint': ['npm', 'run', 'lint'],
            'typecheck-tests': [node, str(base / 'typecheck.cjs')],
            'contract-red': [node, 'node_modules/vitest/vitest.mjs', 'run', '--config', str(base / 'contract.config.mjs'), '--configLoader', 'native', '--reporter=verbose'],
            'contract-after': [node, 'node_modules/vitest/vitest.mjs', 'run', '--config', str(base / 'contract.config.mjs'), '--configLoader', 'native', '--reporter=verbose'],
            'synthetic': [node, 'node_modules/vitest/vitest.mjs', 'run', '--config', str(base / 'synthetic.config.mjs'), '--configLoader', 'native'],
            'format-write': [node, 'node_modules/prettier/bin/prettier.cjs', '--write', *[r['path'] for r in json.loads((base / 'task-short-spec.json').read_text())['scope']]],
            'format-check': [node, 'node_modules/prettier/bin/prettier.cjs', '--check', *[r['path'] for r in json.loads((base / 'task-short-spec.json').read_text())['scope']]]}

assert label in commands
assert not (base / (label + '.json')).exists()
home = base / (label + '-home'); home.mkdir(mode=0o700)
env = {'PATH': str(Path(node).parent) + ':/usr/bin:/bin', 'HOME': str(home), 'TMPDIR': str(home),
       'LANG': 'C.UTF-8', 'TZ': 'UTC', 'NODE_ENV': 'test', 'CI': '1'}
paths = [r['path'] for r in json.loads((base.parent / 'task-short-spec.json').read_text())['maxWriteScope']]
snapshot = lambda: {p: hashlib.sha256((w / p).read_bytes()).hexdigest() for p in paths}
before = snapshot()
t = time.monotonic(); start = datetime.datetime.now(datetime.timezone.utc).isoformat()
proc = subprocess.Popen(commands[label], cwd=w, env=env, start_new_session=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
expired = False
try:
    stdout, stderr = proc.communicate(timeout=180)
except subprocess.TimeoutExpired:
    expired = True; os.killpg(proc.pid, signal.SIGKILL)
    stdout, stderr = proc.communicate(timeout=5)
for stream, data in [('stdout', stdout), ('stderr', stderr)]:
    with (base / (label + '.' + stream)).open('xb') as f: f.write(data)
after = snapshot()
record = {'label': label, 'argv': commands[label], 'cwd': str(w), 'environment': env,
          'startedAt': start, 'seconds': time.monotonic() - t, 'exit': proc.returncode,
          'deadlineExpired': expired, 'before': before, 'after': after, 'sourceUnchanged': before == after,
          'stdoutSHA256': hashlib.sha256(stdout).hexdigest(), 'stderrSHA256': hashlib.sha256(stderr).hexdigest()}
with (base / (label + '.json')).open('x') as f: json.dump(record, f, indent=2); f.write('\n')
print(json.dumps({k: record[k] for k in ['label', 'seconds', 'exit', 'deadlineExpired', 'sourceUnchanged']}))
