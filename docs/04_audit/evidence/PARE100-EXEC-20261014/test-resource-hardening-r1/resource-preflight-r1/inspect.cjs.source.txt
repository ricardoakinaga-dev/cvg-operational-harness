const fs = require('node:fs');
const path = require('node:path');
const crypto = require('node:crypto');
const assert = require('node:assert/strict');
const { execFileSync } = require('node:child_process');
const task = JSON.parse(fs.readFileSync(path.join(__dirname, 'task-short-spec.json')));
const sha = b => crypto.createHash('sha256').update(b).digest('hex');
const binary = '/usr/bin/docker', socket = '/run/docker.sock';
assert.equal(fs.realpathSync(binary), binary); assert(fs.lstatSync(binary).isFile());
assert.equal(fs.realpathSync(socket), socket); assert(fs.lstatSync(socket).isSocket());
const old = JSON.parse(fs.readFileSync(task.readScope[0]));
assert.equal(old.id, task.containerId); assert.equal(old.imageId, task.expectedImageId); assert.equal(old.status, 'exited');
const descriptorBytes = fs.readFileSync(task.readScope[1]);
const descriptor = JSON.parse(descriptorBytes);
assert.equal(descriptor.synthetic, true);
const url = new URL(descriptor.test_database_url);
assert.equal(url.hostname, '127.0.0.1'); assert.equal(url.protocol, 'postgresql:');
assert.equal(url.pathname, '/pare224_synthetic'); assert.equal(url.username, 'pare224');
assert(url.password); assert.equal(descriptor.containerId, task.containerId);
const home = path.join(__dirname, 'home'); fs.mkdirSync(home, { mode: 0o700 });
const env = { PATH: '/usr/bin:/bin', HOME: home, TMPDIR: home, LANG: 'C.UTF-8', DOCKER_HOST: 'unix://' + socket };
const argv = ['inspect', '--format', '{{json .}}', task.containerId];
const start = Date.now();
const bytes = execFileSync(binary, argv, { cwd: __dirname, env, encoding: 'buffer', timeout: 5000, maxBuffer: 1048576, killSignal: 'SIGKILL' });
fs.writeFileSync(path.join(__dirname, 'inspect-private.json'), bytes, { mode: 0o600, flag: 'wx' });
const object = JSON.parse(bytes);
assert.equal(object.Id, task.containerId); assert.equal(object.Name, task.expectedName); assert.equal(object.Image, task.expectedImageId);
assert.equal(object.State.Running, false); assert.equal(object.Config.Image, 'postgres:16.4-alpine');
const ports = object.HostConfig.PortBindings['5432/tcp'];
assert.equal(ports.length, 1); assert.equal(ports[0].HostIp, '127.0.0.1'); assert.equal(ports[0].HostPort, url.port);
const result = { at: new Date().toISOString(), task: task.task, argv: [binary, ...argv], environment: env,
  milliseconds: Date.now() - start, exit: 0,
  dockerSHA256: sha(fs.readFileSync(binary)), socket,
  previousDescriptorSHA256: sha(descriptorBytes), rawInspectSHA256: sha(bytes),
  containerId: object.Id, containerName: object.Name, imageId: object.Image, state: object.State.Status,
  port: { host: ports[0].HostIp, port: ports[0].HostPort },
  mounts: object.Mounts.map(m => ({ type: m.Type, destination: m.Destination, writable: m.RW })),
  resourceScope: 'OWN224_IDENTITY_ONLY_NO_START_OR_DB_CONNECTION',
  nativeP01toP04: 'NOT_RUN', noOldNamespaceAdopted: true };
fs.writeFileSync(path.join(__dirname, 'identity-public.json'), JSON.stringify(result, null, 2) + '\n', { mode: 0o600, flag: 'wx' });
process.stdout.write(JSON.stringify({ state: result.state, name: result.containerName, port: result.port, identityMatch: true, nativeP01toP04: result.nativeP01toP04 }) + '\n');
