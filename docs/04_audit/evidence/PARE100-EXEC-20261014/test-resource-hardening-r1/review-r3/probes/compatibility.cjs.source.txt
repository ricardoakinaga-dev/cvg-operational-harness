function tokens(s){const scan=ts.createScanner(ts.ScriptTarget.Latest,true,ts.LanguageVariant.Standard,s);const list=[];let kind;while((kind=scan.scan())!==ts.SyntaxKind.EndOfFileToken)list.push([kind,scan.getTokenText()]);return JSON.stringify(list)}
function analyze(text){const source=ts.createSourceFile('input.ts',text,ts.ScriptTarget.Latest,true);const expects=[],selection=[],literals=[],functions={};function visit(n){if(ts.isCallExpression(n)&&ts.isIdentifier(n.expression)&&n.expression.text==='expect'){let root=n;while(root.parent&&(ts.isPropertyAccessExpression(root.parent)||ts.isCallExpression(root.parent)||ts.isElementAccessExpression(root.parent))&&root.parent.expression===root)root=root.parent;expects.push(tokens(root.getText(source)))}
