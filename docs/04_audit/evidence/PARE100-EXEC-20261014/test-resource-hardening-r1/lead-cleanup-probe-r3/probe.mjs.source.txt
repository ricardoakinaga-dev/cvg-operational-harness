import { registerHooks } from 'node:module'
import { fileURLToPath } from 'node:url'
import { createHash } from 'node:crypto'
const mock = new URL('./mock-fs.mjs', import.meta.url).href
globalThis.primary = new Error('primary mkdir failure')
globalThis.secondary = new Error('secondary HOME removal failure')
registerHooks({resolve(s,c,next) {
  if (s === 'node:fs/promises' && c.parentURL !== mock) return {url:mock,shortCircuit:true}
  return next(s,c)
}})
const file = '/tmp/cvg-pare100-exec-20261014-6az3hlay/worktree/packages/persistence/src/__tests__/fixtures/complete-process-support.ts'
const {privateEnvironment} = await import(file)
const root = fileURLToPath(new URL('./root', import.meta.url))
let result
try { await privateEnvironment(root); result={accepted:true} }
catch(error) {result={accepted:false, error:error.message, retainsPrimary:error===globalThis.primary || error.cause===globalThis.primary || error.errors?.includes(globalThis.primary), returnsSecondaryOnly:error===globalThis.secondary}}
finally { await fs.rm(root,{recursive:true}) }
