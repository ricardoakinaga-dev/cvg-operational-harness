const ts = require('/tmp/cvg-pare100-exec-20261014-6az3hlay/worktree/node_modules/typescript')
const fs = require('node:fs')
const root = '/tmp/cvg-pare100-exec-20261014-6az3hlay/worktree'
const manifest = JSON.parse(fs.readFileSync(__dirname + '/../builder-inputs/manifest.json'))
const config = ts.readConfigFile(root + '/tsconfig.typecheck.json', ts.sys.readFile)
const parsed = ts.parseJsonConfigFileContent(config.config, ts.sys, root)
const files = manifest.writeScope.map(x => root + '/' + x.path)
const program = ts.createProgram(files, {...parsed.options, incremental:false, composite:false, noEmit:true})
const diagnostics = [...parsed.errors, ...ts.getPreEmitDiagnostics(program)]
console.log(ts.formatDiagnosticsWithColorAndContext(diagnostics, {getCurrentDirectory:()=>root, getCanonicalFileName:x=>x, getNewLine:()=> '\n'}))
console.log('Owned roots:', files.length, 'diagnostics:', diagnostics.length)
process.exitCode = diagnostics.length ? 1 : 0
