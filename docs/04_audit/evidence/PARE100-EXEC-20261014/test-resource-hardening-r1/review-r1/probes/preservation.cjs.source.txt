function visit(n){
 if(ts.isCallExpression(n)&&ts.isIdentifier(n.expression)&&n.expression.text==='expect') {let top=n;while(top.parent&&(ts.isPropertyAccessExpression(top.parent)||ts.isElementAccessExpression(top.parent)||ts.isCallExpression(top.parent))&&top.parent.expression===top)top=top.parent;expects.push(printer.printNode(ts.EmitHint.Unspecified,top,source));}

 if(ts.isStringLiteralLike(n)&&(/\0|NUL|Olá|漢字|😀|🙂/.test(n.text)))literals.push(n.text);
 ts.forEachChild(n,visit);
}visit(source);return {expects,cases,literals};}
