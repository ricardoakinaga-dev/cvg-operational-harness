function server(name) {
  counts[name] = 0;
  const s = net.createServer(socket => { counts[name]++; socket.end(name); });
  servers.push(s);
  return s;
}
function listen(s, ...args) {
  return new Promise((resolve,reject) => {
    s.once('error', reject);
    s.listen(...args, () => { s.removeListener('error', reject); resolve(s.address()); });
  });
}
function attempt(name, expected, invoke) {
  return new Promise(resolve => {
    let sync = true, socket, finished = false;
    const finish = value => {
      if (finished) return;
      finished = true;
      clearTimeout(timer);
      if (socket) socket.destroy();
      const row = { name, expected, ...value, pass: value.result === expected && (expected !== 'CVG_GLOBAL_TEST_NETWORK_BLOCKED' || value.async === true) };
      results.push(row); console.log(JSON.stringify(row)); resolve(row);
    };
    const timer = setTimeout(() => finish({result:'TIMEOUT'}), 1000);
    try {
      socket = invoke();
      socket.once('error', error => finish({result:error.code || error.message, async:!sync}));
      socket.once('connect', () => finish({result:'CONNECTED', async:!sync, remoteAddress:socket.remoteAddress, remotePort:socket.remotePort}));
    } catch (error) { finish({result:error.code || error.message, async:false}); }
    sync = false;
  });
}
const blocked = 'CVG_GLOBAL_TEST_NETWORK_BLOCKED';
function pin(ip, calls) {
  return (host,opts,cb) => {
    if (calls) calls.push({host,opts});
    return opts.all ? cb(null,[{address:ip,family:net.isIP(ip)}]) : cb(null,ip,net.isIP(ip));
  };
}
