def sha(data):
    return hashlib.sha256(data).hexdigest()

def write(name, value):
    (OUT / name).write_text(json.dumps(value, indent=2, ensure_ascii=False) + '\n')

def git(*args):
    return subprocess.check_output(['git', '-C', str(CANDIDATE), *args])

def snapshot(name, data):
    dest = OUT / 'snapshots' / name
    dest.parent.mkdir(parents=True, exist_ok=True)
    dest.write_bytes(data)
    return str(dest.relative_to(OUT))

def unique_object(pairs):
    result = {}
    for key, value in pairs:
        assert key not in result, 'Duplicate JSON key: ' + key
        result[key] = value
    return result

def load(data):
    return json.loads(data, object_pairs_hook=unique_object)

def delta(old, new, prefix=()):
    if type(old) is not type(new):
        return [{'op': 'replace', 'path': list(prefix), 'old': old, 'new': new}]
    if isinstance(old, dict):
        changes = []
        for key in sorted(old.keys() | new.keys()):
            if key not in old:
                changes.append({'op': 'add', 'path': list(prefix + (key,)), 'new': new[key]})
            elif key not in new:
                changes.append({'op': 'remove', 'path': list(prefix + (key,)), 'old': old[key]})
            else:
                changes.extend(delta(old[key], new[key], prefix + (key,)))
        return changes
    if old != new:
        return [{'op': 'replace', 'path': list(prefix), 'old': old, 'new': new}]
    return []
