def main():
    base_path, out_path, edits_path = sys.argv[1:4]
    raw = open(base_path, 'rb').read()
    if hashlib.sha256(raw).hexdigest() != BASE_SHA:
        raise SystemExit('base SHA mismatch')
    base = raw.decode('utf-8')
    text = base
    records = []
    for eid, anchor, old, new in EDITS:
        if base.count(old) != 1:
            raise SystemExit(f'{eid}: old occurs {base.count(old)}x in base')
        if text.count(old) != 1:
            raise SystemExit(f'{eid}: old occurs {text.count(old)}x at apply time')
        text = text.replace(old, new, 1)
        records.append({'id': eid, 'anchor': anchor, 'old': old, 'new': new,
                        'old_sha256': hashlib.sha256(old.encode()).hexdigest()})
    if text.count('\n') != base.count('\n'):
        raise SystemExit('line count changed')
    open(out_path, 'w', encoding='utf-8', newline='').write(text)
    json.dump(records, open(edits_path, 'w', encoding='utf-8'), ensure_ascii=False,
              indent=2)
    print('edits', len(records), 'unformatted_sha256',
          hashlib.sha256(text.encode()).hexdigest())


main()
