vi.mock('/tmp/cvg-aud0601-20261006/candidate/apps/api/src/production-bootstrap.ts',()=>({buildApiWithProductionSessions:seam.build}))

 exit=vi.spyOn(process,'exit').mockImplementation(()=>undefined as never);vi.spyOn(console,'error').mockImplementation(()=>{})
})
