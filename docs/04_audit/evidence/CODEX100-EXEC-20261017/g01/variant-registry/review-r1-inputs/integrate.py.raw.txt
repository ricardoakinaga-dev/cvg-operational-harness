def require(ok,message):
    if not ok:raise AssertionError(message)


def read(p):return json.loads(p.read_text())
def sha(b):return hashlib.sha256(b).hexdigest()
def canon(v):return json.dumps(v,ensure_ascii=False,sort_keys=True,separators=(',',':'),allow_nan=False).encode()
def encoded(v):return (json.dumps(v,ensure_ascii=False,indent=2)+'\n').encode()


def dereference(value,pointer):
    require(pointer=='' or pointer.startswith('/'),'Invalid pointer')
    for token in pointer.split('/')[1:]:
        token=token.replace('~1','/').replace('~0','~')
        value=value[int(token)] if isinstance(value,list) else value[token]
    return value


def strict_lane(lane):
    directory=OUT/lane;manifest=read(directory/'manifest.json');rows=manifest['files']
    actual={str(p.relative_to(directory)) for p in directory.rglob('*') if p.is_file() and p!=directory/'manifest.json'}
    require(len(rows)==len({r['path'] for r in rows}) and actual=={r['path'] for r in rows},'Strict lane file set '+lane)
    require(not any(p.is_symlink() for p in directory.rglob('*')),'Lane symlink')
    require(all(sha((directory/r['path']).read_bytes())==r['sha256'] and (directory/r['path']).stat().st_size==r['bytes'] for r in rows),'Strict lane bytes '+lane)
    return {'lane':lane,'manifest_sha256':sha((directory/'manifest.json').read_bytes()),'files':len(rows)}

def main():
    fresh=build();require(fresh==build(),'Nondeterministic registry join')
    negatives=[]
    for name,mutation in [
      ('drop_functional_requirement',lambda x:x['requirements'].pop()),
      ('drop_family',lambda x:x['families'].pop()),
      ('wrong_owner',lambda x:x['requirements'][0].update(owner='DX100-Z02')),

      ('erase_event_owner_limit',lambda x:x.update(remaining_gaps=[r for r in x['remaining_gaps'] if r['id']!='TUI-INTERNAL-EFFECT-CORRELATION'])),
      ('drop_packaging_pointer',lambda x:x['additional_source_fragments'].pop()),
      ('promote_support_count_to_functions',lambda x:x['counts'].update(functional_requirements=19947)),
      ('close_generic_source_acceptance',lambda x:next(r for r in x['requirements'] if r['historical_capability_supplement']).update(source_variant_acceptance='CLOSED')),
      ('approve_firstjourney',lambda x:x['first_native_journey'].update(build_status='APPROVED')),
      ('fake_parity_receipt',lambda x:x.update(functional_receipts=1,full_g01_closure=True)),
    ]:
        bad=copy.deepcopy(fresh);mutation(bad);require(bad!=fresh,'Ineffective negative '+name)
        negatives.append({'id':name,'result':'REJECTED_BY_EXACT_REGENERATION'})
    report={'result':'PASS','kind':'T1_SOURCE_REGISTRY_JOIN_ONLY','counts':fresh['counts'],'negative_controls':negatives,'functional_receipts':0,'full_g01_closure':False}
    for name,value in [('integration.json',fresh),('verification.json',report)]:
        if '--check' in sys.argv:require((OUT/name).read_bytes()==encoded(value),'Join regeneration differs '+name)
        else:(OUT/name).write_bytes(encoded(value))
    print(json.dumps(report))


if __name__=='__main__':main()
