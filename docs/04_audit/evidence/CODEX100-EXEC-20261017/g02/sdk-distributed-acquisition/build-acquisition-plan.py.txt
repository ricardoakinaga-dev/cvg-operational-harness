def rel(p):
    return os.path.normpath(os.path.join(UNIT, p))


def sha(p):
    with open(rel(p), "rb") as f:
        return hashlib.sha256(f.read()).hexdigest()


def ptr(p, pointer=None):
    d = {"path": p, "sha256": sha(p)}
    if pointer:
        d["pointer"] = pointer
    return d
