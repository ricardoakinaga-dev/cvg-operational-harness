#!/usr/bin/env python3
# DX100-G02:SPECIFY-SDK-SOURCE-CONTINUATION-3 - escreve next-task.json e state.json com os sha atuais dos documentos da especificação.
# Uso: python3 -I -B close-unit-c3.py.txt <spec_dir> <phase>
#   phase: draft (antes da revisão) | final (depois da R2; lê review-r1/review.json, repair-r1.json, review-r2/review.json e,

import hashlib
import json
import os
import sys
from datetime import datetime, timezone

SPEC, PHASE = sys.argv[1], sys.argv[2]
TASK = "DX100-G02:SPECIFY-SDK-SOURCE-CONTINUATION-3"
UNIT = "docs/04_audit/evidence/CODEX100-EXEC-20261017/g02/sdk-source-continuation-3-spec"


def sha(name):
    with open(os.path.join(SPEC, name), "rb") as f:
        return hashlib.sha256(f.read()).hexdigest()


def load(name):
    with open(os.path.join(SPEC, name), encoding="utf-8") as f:
        return json.load(f)


def exists(name):
    return os.path.exists(os.path.join(SPEC, name))


def ref(name):
    return {"path": UNIT + "/" + name, "sha256": sha(name)}


c1, c2 = load("cases-c3-1.json"), load("cases-c3-2.json")

cov, fr, ev, t3 = load("coverage-plan-c3.json"), load("freeze-r4-c3-delta.json"), load("evaluator-r4-c3-delta.json"), load("t3-assessment.json")
val, iv = load("validation.json"), load("input-verification.json")
adm = load("../sdk-batch-2b2-admission/state.json")
review, carry, status = None, [], "IN_REVIEW"
if PHASE == "final":
    r1, rep, r2 = load("review-r1/review.json"), load("repair-r1.json"), load("review-r2/review.json")
    disp = load("review-r2/executor-disposition.json") if exists("review-r2/executor-disposition.json") else None
    review = {"R1": "%s (%s) review-r1/review.json; texto sem edição em review-r1/verdict.raw.txt" % (r1["verdict"], r1.get("summary_counts", "")),
              "repair": "repair-r1.json (1/1), preimagens em archives/r1-preimages/",
              "R2": "%s (%s) review-r2/review.json; texto sem edição em review-r2/verdict.raw.txt" % (r2["verdict"], r2.get("summary_counts", "")),
              "R2_disposition": "review-r2/executor-disposition.json" if disp else None,
              "critic": "subagente novo por rodada, somente leitura; o executor grava brief e veredito sem edição"}
    if disp:
        carry = [{"finding": "C3-SPEC/" + i["id"], "text": i["note"]} for i in disp["items"] if i["disposition"].startswith("CARRIED")]
    open_mandatory = r2.get("open_mandatory", 0)
    status = "COMPLETED_LOCAL" if r2["verdict"] == "APPROVE" and not carry else (
        "COMPLETED_LOCAL_WITH_CARRIED_P2" if open_mandatory == 0 or disp else "BLOCKED_REVIEW")


def slots(doc):
    return [{"slot": c["slot"], "case": c["id"], "kind": c["kind"], "contracts": c["contracts"], "role": c["role"], "gate": bool(c.get("gate")),
             "arms": [b.split(" ")[0] for b in c["branches_targeted"]]} for c in doc["cases"]]


spec = {k: ref(v) for k, v in {
    "cases_c3_1": "cases-c3-1.json", "cases_c3_2": "cases-c3-2.json", "coverage_plan": "coverage-plan-c3.json",

    "t3": "t3-assessment.json", "validation": "validation.json", "input_verification": "input-verification.json"}.items()}
for k, b in list(fr["base"].items()) + list(ev["base"].items()):

        ", ".join(b1["planned"]["qualification_services"]), b1["planned"]["sdk_cases"]),
    "first_step": "Lead registra claim novo. Conferir state.json desta pasta e os sha da especificação acima antes de qualquer escrita.",
    "budget": {"case_slots": b1["planned"]["case_slots"], "sdk_cases": b1["planned"]["sdk_cases"], "control_slots": 1,
               "conditional_slots": b1["planned"]["conditional_slots"], "qualification_services_planned": b1["planned"]["qualification_services_planned"],
