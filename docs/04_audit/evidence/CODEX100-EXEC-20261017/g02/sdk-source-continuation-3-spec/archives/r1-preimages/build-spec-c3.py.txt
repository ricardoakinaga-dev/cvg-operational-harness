def lines_of(key):

def A(key, text, nth=None, span=0):
    """Anchor: unique line (or the nth match) containing `text`; optional span of following lines."""
    ls = lines_of(key)
    hits = [i for i, l in enumerate(ls, 1) if text in l]
    if nth is None:
        if len(hits) != 1:
            ANCHOR_ERRORS.append((key, text, hits))
            return {"input": key, "error": "ambiguous_or_missing"}
        n = hits[0]
    else:
        if len(hits) < nth:
            ANCHOR_ERRORS.append((key, text, hits, nth))
            return {"input": key, "error": "missing"}
        n = hits[nth - 1]
    return {"input": key, "sha256": INPUTS[key]["sha256"], "lines": [n] if not span else [n, n + span], "text": ls[n - 1].strip()}

def tsteps(*items):
    out = []
    for i, s in enumerate(items, 1):
        d = {"step": i}
        d.update(s)
        out.append(d)
    return out

def START(codex, as_, options=None, options_as=None, resume=None):

def THROWS(name, message=None, prefix=None):
    t = {"name": name}

    if prefix is not None:

    return {"throws": t, "spawns": 0}


def ts_err(step, name, message=None, prefix=None, cause_name=None, code=None):
    e = {"name": name}
    if message is not None:
        e["message"] = message
    if prefix is not None:
        e["message_prefix"] = prefix
    if cause_name is not None:
        e["cause_name"] = cause_name
    if code is not None:
        e["code"] = code
    return {"step": step, "error": e}


def argv_meta(argv):
    return {"argv": argv, "argv_json_bytes": len(compact(argv).encode("utf-8")), "argv_count": len(argv)}


def toml_str(s):
    """exec.ts toTomlValue para string: JSON.stringify(value). Para strings, json.dumps(ensure_ascii=False) dá os mesmos bytes
    (escapes curtos \\t \\n \\r \\b \\f, \\u00XX minúsculo para os demais controles abaixo de 0x20, 0x7f literal); o
    validador Node reconfere com JSON.stringify."""
    return json.dumps(s, ensure_ascii=False)

def R(method, reply=None, error=None, subset=None, params=None, absent=False, before=None, after=None, hold=None):
    r = {"match": {"method": method}}
    if params is not None:
        r["match"]["params"] = params
    if subset is not None:
        r["match"]["params_subset"] = subset
    if absent:
        r["match"]["params_absent"] = True
    if before:
        r["before_reply"] = before
    if hold:
        r["hold"] = hold
    if error is not None:
        r["reply"] = {"error": error}
    else:
        r["reply"] = {"result": reply if reply is not None else {}}
    if after:
        r["after_reply"] = after
    return r

def steps(*items):
    out = []
    for i, s in enumerate(items, 1):

        d.update(s)
        out.append(d)
    return out


def S(api, **kw):
    d = {"api": api}
    d.update(kw)
    return d


def CALL(on, method, *args, as_=None, aw=False, **kwargs):
    d = {"api": "call", "on": on, "method": method}
    if args:
        d["args"] = list(args)
    if kwargs:
        d["kwargs"] = kwargs
    if aw:
        d["await"] = True
    if as_:
        d["as"] = as_
    return d


def REQ(on, method, params=None, aw=False):
    return CALL(on, "request", method, params, aw=aw, response_model="InitializeResponse")

    return d

    out, src = {}, {}
    for n in sorted(set(names)):

            out[n] = {"class": NEW_PRED[n]["class"], "oracle": NEW_PRED[n]["oracle"], "source": "NEW (%s)" % NEW_PRED[n]["clause"]}
            continue
        for label, t in prefer:
            if n in t:
                out[n] = {"class": t[n]["class"], "oracle": t[n]["oracle"], "source": label}
                break
        else:
            raise SystemExit("predicate without definition: " + n)
    return out

def case_input_of(c):
    if "control_case_input" in c:
        return c["control_case_input"]
    if c["language"] == "typescript":
        return {"sdk_calls": c["sdk_calls"], "fixture": c["fixture"]}

def finalize(cases, control, sub_lot, language):
    out, slot = [], 0
    for cid, c in list(cases.items()) + [(control["copy_of"]["case"].replace("2B1", "C31").replace("2B2", "C32"), control)]:
        slot += 1
        c = dict(c)
        is_ctrl = "control_case_input" in c
        c["id"] = cid
        c["kind"] = "CONTROL" if is_ctrl else "SDK"
        c["language"] = "python" if is_ctrl else language
        c["sub_lot"] = sub_lot
        c["slot"] = slot
        ci = case_input_of(c)
        for k in ("sdk_calls", "fixture", "peer_script", "driver_mode", "control_case_input"):
            c.pop(k, None)
        if is_ctrl:
            c["execution_order"] = 1
        else:
            c.setdefault("version_dependent", [] if language == "typescript" else PYV)
            if language == "python" and c["version_dependent"] == []:
                c["version_dependent"] = PYV
            c["inconclusive_on_deadline"] = False
            c["credit"] = CREDIT
            c.pop("sub_lot_base", None)
        keys = sorted({a["input"] for a in c["anchors"]})
        for a in c["anchors"]:
            assert "error" not in a, a
            assert a["sha256"] == INPUTS[a["input"]]["sha256"], ("anchor sha", cid, a)
        c["inputs"] = keys
        c["runtime_status"] = "NOT_RUN"
        c["receipt"] = None
        c["case_input_note"] = ("sdk_calls e fixture do slot existem só dentro de case_input (fonte única do supervisor, F4-01)." if language == "typescript" and not is_ctrl
                                else "sdk_calls, peer_script, control_client e driver_mode do slot existem só dentro de case_input (fonte única do supervisor, F4-01).")
        c["case_input"] = ci
        c["case_input_sha256"] = sha_bytes(compact(ci).encode("utf-8"))
        out.append({k: c[k] for k in ORDER if k in c})
        extra = set(c) - set(ORDER)
        assert not extra, (cid, extra)
    return out
