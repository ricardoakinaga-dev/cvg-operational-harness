// Every top-level field that differs from the base must be declared in changed_fields (removed ones under 'removed'); the rest is unchanged_fields.
function fieldAccounting(c, base, holder) {
  const p = []
  const keys = [...new Set([...Object.keys(c), ...Object.keys(base)])].filter((k) => !STRUCTURAL.has(k))
  const same = (k) => JSON.stringify(c[k]) === JSON.stringify(base[k])
  const differ = keys.filter((k) => !same(k)); const removed = differ.filter((k) => !(k in c))
  const want = new Set([...differ.filter((k) => k in c), ...CI_KEYS.filter((k) => JSON.stringify(base.case_input[k]) !== JSON.stringify(c.case_input[k]))])
  const declared = new Set(Object.keys(holder.changed_fields).filter((k) => k !== 'removed' && k !== 'id'))
  for (const k of want) if (!declared.has(k)) p.push(c.id + ': undeclared change ' + k)
  for (const k of declared) if (!want.has(k)) p.push(c.id + ': declared but unchanged ' + k)
  for (const k of removed) if (!(holder.changed_fields.removed ?? '').includes(k)) p.push(c.id + ': removed not declared ' + k)
  if (JSON.stringify([...(holder.unchanged_fields ?? [])].sort()) !== JSON.stringify(keys.filter(same).sort())) p.push(c.id + ': unchanged_fields')
  return p
}

function* objects(x) {
  if (Array.isArray(x)) for (const v of x) yield* objects(v)
  else if (x && typeof x === 'object') { yield x; for (const v of Object.values(x)) yield* objects(v) }
}
function* strings(x) {
  if (typeof x === 'string') yield x
  else if (Array.isArray(x)) for (const v of x) yield* strings(v)
  else if (x && typeof x === 'object') for (const v of Object.values(x)) yield* strings(v)
}

      || ['first_then_witness_then_throw', 'next_abort_at_barrier'].includes(st.consume)) caps.add('CAP-TS-SDKCALLS-C3')

// ------------------------------------------------------------------ per-file checks (pure; reused by the negative controls)
function checkCases(doc, sub, lang, cov) {
  const p = []
  const n = doc.cases.length
  if (doc.task !== 'DX100-G02:SPECIFY-SDK-SOURCE-CONTINUATION-3' || doc.sub_lot !== sub || doc.status !== 'FROZEN_SPEC_NOT_RUN') p.push('header')
  if (doc.functional_receipts !== 0 || doc.parity_credit !== 0) p.push('credit')
  if (n > 24 || doc.counts.slots !== n) p.push('slots ' + n)
  const slots = doc.cases.map((c) => c.slot)
  if (JSON.stringify(slots) !== JSON.stringify([...Array(n).keys()].map((i) => i + 1))) p.push('slot numbering')
  const ctrls = doc.cases.filter((c) => c.kind === 'CONTROL')
  if (ctrls.length !== 1 || ctrls[0].slot !== n || ctrls[0].execution_order !== 1) p.push('control slot')
  for (const [k, spec] of Object.entries(doc.inputs)) {
    let got = null
    try { got = sha(readFileSync(inputPath(k, spec))) } catch { got = null }
    if (got !== spec.sha256) p.push('input sha ' + k)
  }
  const newArms = cov.new_arm_ids.arms
  for (const c of doc.cases) {
    const id = c.id
    const ci = c.case_input
    if (sha(JSON.stringify(ci)) !== c.case_input_sha256) p.push(id + ': case_input sha')
    if (c.runtime_status !== 'NOT_RUN' || c.receipt !== null) p.push(id + ': runtime')
    // anchors
    for (const a of c.anchors) {
      const spec = doc.inputs[a.input]
      if (!spec || !c.inputs.includes(a.input)) { p.push(id + ': anchor input ' + a.input); continue }
      if (a.sha256 !== spec.sha256) p.push(id + ': anchor sha ' + a.input)
      const ls = linesOf(inputPath(a.input, spec))
      if ((ls[a.lines[0] - 1] ?? '').trim() !== a.text) p.push(id + ': anchor text ' + a.input + ':' + a.lines[0])
      if (a.lines.length === 2 && !(a.lines[1] > a.lines[0] && a.lines[1] <= ls.length)) p.push(id + ': anchor span')
    }
    // predicates
    for (const pr of c.predicates) if (!doc.predicate_table[pr]) p.push(id + ': undefined predicate ' + pr)
    if (c.kind === 'CONTROL') {
      const base = C2B[c.copy_of.case]
      const want = clone(base.case_input); want.control_client.argv[want.control_client.argv.length - 1] = id
      if (JSON.stringify(want) !== JSON.stringify(ci)) p.push(id + ': control copy')
      if (sha(JSON.stringify(base)) !== c.copy_of.case_object_sha256) p.push(id + ': control base sha')
      p.push(...fieldAccounting(c, base, c.copy_of))
      if (JSON.stringify(c.predicates) !== JSON.stringify(base.predicates)) p.push(id + ': control predicates')
      continue
    }
    if (c.language !== lang) p.push(id + ': language')
    const wantKeys = lang === 'typescript' ? ['sdk_calls', 'fixture'] : ['sdk_calls', 'peer_script'].concat(ci.driver_mode ? ['driver_mode'] : [])
    if (JSON.stringify(Object.keys(ci)) !== JSON.stringify(wantKeys)) p.push(id + ': case_input keys')
    const stepNums = new Set(ci.sdk_calls.map((s) => s.step))
    if (JSON.stringify([...stepNums]) !== JSON.stringify(ci.sdk_calls.map((_, i) => i + 1))) p.push(id + ': step numbering')
    // arms and discrimination
    const arms = c.branches_targeted.map((b) => b.split(' ')[0])
    for (const a of arms) {
      if (!ARM_RE[lang].test(a)) p.push(id + ': arm format ' + a)
      if (lang === 'typescript' && !TS_CATALOG.has(a)) p.push(id + ': arm not in G01 ' + a)
      if (lang === 'python' && (PY_ARMS_2B.has(a) === (a in newArms))) p.push(id + ': arm id unknown or colliding ' + a)
    }
    if (JSON.stringify([...new Set(arms.map((a) => a.slice(0, 4)))].sort()) !== JSON.stringify([...c.contracts].sort())) p.push(id + ': contracts differ from targeted arms')
    const disc = new Set(c.arm_discrimination.map((d) => d.arm))
    if (JSON.stringify([...new Set(arms)].sort()) !== JSON.stringify([...disc].sort())) p.push(id + ': discrimination arms')
    for (const d of c.arm_discrimination) {
      if (!d.contrary || !d.observable_if_contrary || !d.judged_by?.length) p.push(id + ': discrimination fields ' + d.arm)
      if (!d.judged_by.every((j) => c.predicates.includes(j))) p.push(id + ': judged_by outside predicates ' + d.arm)
      if (!d.judged_by.some((j) => doc.predicate_table[j]?.class === 'client')) p.push(id + ': no client predicate ' + d.arm)
      if (!d.steps.every((s) => stepNums.has(s))) p.push(id + ': discrimination step ' + d.arm)
    }
    if (!c.known_bad.judged_by.every((j) => c.predicates.includes(j) && doc.predicate_table[j]?.class === 'client')) p.push(id + ': known_bad judged_by')
    for (const e of c.expected.api ?? []) if (!stepNums.has(e.step)) p.push(id + ': expected step ' + e.step)
    // vocabulary and capabilities
    const declared = c.infra
    const derived = lang === 'typescript' ? deriveTsInfra(c) : derivePyInfra(c)
    for (const cap of derived) if (!declared.some((d) => d === cap || d.startsWith(cap + ' '))) p.push(id + ': infra missing ' + cap)
    if (lang === 'typescript') {
      for (const st of ci.sdk_calls) {
        if (st.barrier && !st.api) { if (!TS_BARRIERS.has(st.barrier)) p.push(id + ': barrier ' + st.barrier); continue }
        if (st.barrier && !((st.consume === 'first_then_witness_then_throw' && st.barrier === 'SUSPENDED' && st.throw?.name)
          || (st.consume === 'next_abort_at_barrier' && st.barrier === 'CHILD_HELD' && st.abort))) p.push(id + ': consume barrier ' + st.step)
        if (!TS_APIS.has(st.api) && !/^\w+\.(startThread|resumeThread)$/.test(st.api)) p.push(id + ': ts api ' + st.api)
        if (st.consume && !TS_CONSUME.has(st.consume)) p.push(id + ': consume ' + st.consume)
        if (st.expect?.throws) {
          const e = (c.expected.api ?? []).find((x) => x.step === st.step && x.error)
          const t = st.expect.throws
          if (!e || e.error.name !== t.name || e.error.message !== t.message || e.error.message_prefix !== t.message_prefix) p.push(id + ': expect/expected ' + st.step)
        }
      }
      for (const o of objects(ci.sdk_calls)) for (const k of Object.keys(o)) if (k.startsWith('$') && !TS_MARK_OBJ.has(k)) p.push(id + ': marker ' + k)
      for (const s of strings(ci.sdk_calls)) if (/^\$[A-Za-z]/.test(s) && !TS_MARK_STR.has(s)) p.push(id + ': string marker ' + s)
      const witnessSteps = []
      for (const st of ci.sdk_calls) for (const o of objects(st)) if (o.$GETTER?.witness) witnessSteps.push([o.$GETTER.witness, st.step])
      const gw = c.expected.getter_witnesses ?? []
      if (JSON.stringify(witnessSteps.map(([w, n]) => w + '@' + n).sort()) !== JSON.stringify(gw.map((g) => g.id + '@' + g.declared_in_step).sort())) p.push(id + ': getter witnesses')
      for (const g of gw) {
        const inv = ci.sdk_calls.find((x) => x.step === g.step)
        if (!inv || !['run', 'run_streamed', 'events.consume'].includes(inv.api) || g.step < g.declared_in_step) p.push(id + ': getter witness step ' + g.id)
      }
      if (witnessSteps.length && !['tmp.getter_witness_exact', 'tmp.getter_witness_recorded'].every((x) => c.predicates.includes(x))) p.push(id + ': witness predicates')
      if (c.expected.argv) {
        if (c.expected.argv_json_bytes !== Buffer.byteLength(JSON.stringify(c.expected.argv)) || c.expected.argv_count !== c.expected.argv.length) p.push(id + ': argv bytes')
      }
      const errs = (c.expected.api ?? []).filter((x) => x.error)
      const v8msg = errs.some((x) => /is not iterable|Cannot read properties|serialize a BigInt|Maximum call stack|operation was aborted/.test(x.error.message ?? ''))
      if (v8msg && !c.version_dependent.length) p.push(id + ': version_dependent missing')
    } else {
      for (const st of objects(ci.sdk_calls)) {
        if (typeof st.api === 'string' && !PY_APIS.has(st.api)) p.push(id + ': py api ' + st.api)
        if (st.wait_class !== undefined && (!WAIT_CLASSES.has(st.wait_class) || !WAIT_APIS.has(st.api)
          || (st.wait_class === 'CLIENT_EFFECT_UNLESS_ALIVE') !== (typeof st.thread_name === 'string'))) p.push(id + ': wait_class ' + st.step)
        for (const k of Object.keys(st)) if (k.startsWith('$') && !PY_MARKERS.has(k)) p.push(id + ': py marker ' + k)
      }
      if (ci.sdk_calls.some((s) => s.wait_class) && !c.predicates.includes('wait.client_effect_recorded')) p.push(id + ': wait predicate')
      if (ci.sdk_calls.at(-1).api !== 'epilogue') p.push(id + ': epilogue last')
    }
    if (c.gate && id !== 'C3-PY02') p.push(id + ': unexpected gate')
    const zc = (c.expected.fault?.fired ?? []).filter((f) => 'zero_calls_client_effect_unless_started' in f)
    if (zc.length && (id !== 'C3-PY13ER' || !zc.every((f) => ci.sdk_calls.some((s) => s.thread_name === f.zero_calls_client_effect_unless_started)))) p.push(id + ': zero-calls fault rule')
    if (c.replay_of) {
      const base = C2B[c.replay_of.case]
      if (sha(JSON.stringify(base)) !== c.replay_of.case_object_sha256) p.push(id + ': replay base sha')
      if (!Object.keys(c.replay_of.changed_fields).length) p.push(id + ': replay changed_fields')
      p.push(...fieldAccounting(c, base, c.replay_of))
      if (JSON.stringify(base.case_input.peer_script) !== JSON.stringify(ci.peer_script)) p.push(id + ': replay peer_script changed')
      const bs = base.case_input.sdk_calls
      if (id === 'C3-PY13ER') {
        for (const [i, st] of ci.sdk_calls.entries()) {
          const b = clone(bs[i]); const s = clone(st)
          if ([11, 12, 13].includes(st.step)) { delete s.wait_class; if (!('thread_name' in b)) delete s.thread_name; delete s.note; delete b.note }
          if (JSON.stringify(b) !== JSON.stringify(s)) p.push(id + ': replay diff outside steps 11-13 at ' + st.step)
        }
        if (ci.sdk_calls.length !== bs.length) p.push(id + ': replay length')
      }
      if (id === 'C3-PY13F') {
        const b0 = clone(bs[0]); const s0 = clone(ci.sdk_calls[0]); const added = s0.hooks.pop()
        if (JSON.stringify(b0) !== JSON.stringify(s0) || added.code !== 'openai_codex._goal._GoalOperationState.finish') p.push(id + ': replay hooks')
        for (let i = 1; i < bs.length - 1; i++) if (JSON.stringify(bs[i]) !== JSON.stringify(ci.sdk_calls[i])) p.push(id + ': replay step ' + (i + 1))
        const last = ci.sdk_calls.at(-1); const prev = ci.sdk_calls.at(-2)
        if (prev.api !== 'hook.count' || prev.id !== 'h_finish' || last.api !== 'epilogue' || ci.sdk_calls.length !== bs.length + 1) p.push(id + ': replay tail')
      }
    }
  }
  return p
}

function checkTable(doc) {
  const p = []
  for (const [k, v] of Object.entries(doc.predicate_table)) {
    if (!CLASSES.includes(v.class)) p.push('class ' + k)
    if (v.source.startsWith('NEW')) {
      const clause = NEW_PRED_CLAUSES[k]
      if (!clause || !v.source.includes(clause) || !EV.clauses.some((c) => c.id === clause)) p.push('new predicate clause ' + k)
    } else {
      const base = TABLES[v.source]?.[k]
      if (!base || base.class !== v.class || base.oracle !== v.oracle) p.push('base mismatch ' + k)
    }
  }
  const used = new Set(doc.cases.flatMap((c) => c.predicates))
  for (const k of Object.keys(doc.predicate_table)) if (!used.has(k)) p.push('unused ' + k)
  return p
}

function checkControlChars(doc) {
  const p = []
  const c = doc.cases.find((x) => x.id === 'C3-TS07')
  const good = c.case_input.sdk_calls.find((s) => s.api === 'new Codex' && s.as === 'good')
  const s = good.options.config.s
  if (!/[\t\u0001\u001f\u007f]/.test(s)) p.push('control string lost')
  if (c.expected.argv[3] !== 's=' + JSON.stringify(s)) p.push('argv element 3 differs from JSON.stringify')
  if (!c.expected.argv[3].includes('\\t') || !c.expected.argv[3].includes('\\u0001') || !c.expected.argv[3].includes('\u007f')) p.push('escape form')
  return p
}

function checkCoverage(cov, cases) {
  const p = []
  const byId = Object.fromEntries(cases.map((c) => [c.id, c]))
  const ids = cov.residuals.map((r) => r.id)
  if (new Set(ids).size !== ids.length) p.push('residual ids not unique')
  const scope = ['TS01', 'TS02', 'TS03', 'TS04', 'TS05', 'TS06', 'TS07', 'TS08', 'TS09', 'TS10', 'TS11', 'TS12', 'TS13', 'TS14', 'TS15',
    'PY02', 'PY03', 'PY04', 'PY05', 'PY08', 'PY10', 'PY12', 'PY13', 'PY14', 'PY15', 'PY16', 'PY17', 'PY18']
  for (const k of scope) if (!cov.residuals.some((r) => r.contract === k)) p.push('scope contract without residual ' + k)
  for (const k of Object.keys(V5.remaining_open_by_contract)) if (scope.includes(k) && !cov.residuals.some((r) => r.contract === k && r.source.includes('remaining_open_by_contract'))) p.push('v5 remaining not cited ' + k)
  const disp = sha(readFileSync(DISP_PATH))
  for (const r of cov.residuals) {
    if (!(r.class in cov.classes)) { p.push(r.id + ': class'); continue }
    if (r.class === 'IN_SPEC') {
      if (!r.cases?.length || !r.arms?.length) p.push(r.id + ': IN without cases/arms')
      for (const cid of r.cases ?? []) if (!byId[cid]) p.push(r.id + ': unknown case ' + cid)
      for (const a of r.arms ?? []) if (!(r.cases ?? []).some((cid) => byId[cid]?.branches_targeted.some((b) => b.startsWith(a + ' ')))) p.push(r.id + ': arm not targeted ' + a)
    } else if (r.cases) p.push(r.id + ': OUT with cases')
    if (r.class === 'OUT_USER_DECISION' && (!r.questions?.length || !r.questions.every((q) => QIDS.includes(q)) || r.questions_source?.sha256 !== disp)) p.push(r.id + ': questions')
    if (r.class === 'OUT_NEW_CAPABILITY' && !r.capability_needed) p.push(r.id + ': capability')
    if (['OUT_NONDETERMINISTIC', 'OUT_UNREACHABLE', 'OUT_NOT_APPLICABLE', 'ALREADY_OBSERVED'].includes(r.class) && !r.reason) p.push(r.id + ': reason')
  }
  for (const c of cases) {
    if (c.kind !== 'SDK') continue
    for (const b of c.branches_targeted) {
      const a = b.split(' ')[0]
      if (!cov.residuals.some((r) => r.class === 'IN_SPEC' && r.cases.includes(c.id) && r.arms.includes(a))) p.push(c.id + ': arm without residual ' + a)
    }
  }
  if (JSON.stringify(cov.projection_if_all_pass.counts) !== JSON.stringify(V5.counts) || cov.projection_if_all_pass.rows_changing_status.length) p.push('projection')
  for (const a of Object.keys(cov.new_arm_ids.arms)) if (PY_ARMS_2B.has(a)) p.push('new arm collides ' + a)
  for (const row of cov.rows) {
    const want = cases.filter((c) => c.kind === 'SDK' && c.contracts.includes(row.source_contract)).map((c) => c.id)
    if (JSON.stringify(want) !== JSON.stringify(row.cases)) p.push('row cases ' + row.source_contract)
    if (row.status_if_all_cases_pass !== 'OBSERVED_CLIENT_PARTIAL' || row.whole_requirement_covered_after !== false) p.push('row status ' + row.source_contract)
  }
  if (cov.functional_receipts !== 0 || cov.parity_credit !== 0) p.push('credit')
  return p
}

function checkBudget(b, doc, sub) {
  const p = []
  const gated = doc.cases.filter((c) => c.kind === 'SDK' && c.gate).map((c) => c.id)
  if (b.sub_lot !== sub || b.max.case_slots !== 24 || b.max.services !== 12) p.push('max')
  if (b.planned.case_slots !== doc.cases.length || b.planned.case_slots > 24) p.push('slots')
  if (b.planned.qualification_services.length !== b.planned.qualification_services_planned || b.planned.qualification_services_planned > 12) p.push('services')
  if (JSON.stringify(b.planned.conditional_slots) !== JSON.stringify(gated)) p.push('conditional')
  if (b.planned.automatic_retries !== 0 || b.max.automatic_retries !== 0) p.push('retries')
  return p
}
