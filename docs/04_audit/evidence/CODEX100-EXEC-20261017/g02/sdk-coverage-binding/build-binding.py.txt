def sha(p):
    with open(p, "rb") as f:
        return hashlib.sha256(f.read()).hexdigest()


def load(p):
    with open(p) as f:
        return json.load(f)

    return os.path.relpath(p, OUT)

def gaps_of(cid):
    if cid in pyc:
        return pyc[cid]["gaps"]
    return [tsc[cid]["gap_id"]]


GOAL_FACTS_P = os.path.join(G01, "payload-and-goal-facts.json")


def residual_refs(cid):
    # Resíduo autoritativo: o contrato G01 inteiro menos o que o caso mostrou.
    if cid in pyc:
        refs = [{"path": rel_from_out(pyc_p), "sha256": sha(pyc_p),
                 "pointer": f"rows[id={cid}].source_contract|future_positive|future_known_bad"}]
        if cid == "PY14":
            refs.append({"path": rel_from_out(GOAL_FACTS_P), "sha256": sha(GOAL_FACTS_P), "pointer": "goal"})
        return refs
    return [{"path": rel_from_out(tsc_p), "sha256": sha(tsc_p),
             "pointer": f"contracts[id={cid}].expectation|branches"}]


def acc_entry(case_id):
    r = cr[case_id]
    d = os.path.join(ADM, r["attempt_dir"])
    a1 = os.path.join(d, "acceptance.json")
    a2 = os.path.join(d, "acceptance-r2.json")
    j1, j2 = load(a1), load(a2)
    assert j1["case"] == case_id and j2["case"] == case_id
    assert cr2[case_id]["sha256"] == sha(a2), case_id
    return {
        "case": case_id,
        "outcome_service_r1": j1["outcome"],
        "outcome_offline_r2": j2["outcome"],
        "capture_verdict_service_r1": j1["capture_verdict"],
        "capture_verdict_offline_r2": j2["capture_verdict"],
        "predicates_service_r1": len(j1["predicates"]),
        "predicates_offline_r2": len(j2["predicates"]),
        "failed_predicates_service_r1": j1["failed_predicates"],
        "failed_predicates_offline_r2": j2["failed_predicates"],
        "by_judge_offline_r2": cr2[case_id]["by_judge"],
        "client_fact_only": j1["facts"]["client_fact_only"],
        "parity_receipt": j1["facts"]["parity_receipt"],
        "functional_receipts": j1["functional_receipts"],
        "acceptance": {"path": rel_from_out(a1), "sha256": sha(a1)},
        "acceptance_r2": {"path": rel_from_out(a2), "sha256": sha(a2)},
    }

        row["status"] = "OBSERVED_CLIENT_PARTIAL" if all_pass else "OBSERVED_CLIENT_PARTIAL_WITH_FAILURES"

        row["remaining_open_scope_exhaustive"] = False
        row["residual_planned_profile_proposal"] = RESIDUAL_PLAN[cid]

            cited[o["acceptance"]["path"]] = o["acceptance"]["sha256"]
