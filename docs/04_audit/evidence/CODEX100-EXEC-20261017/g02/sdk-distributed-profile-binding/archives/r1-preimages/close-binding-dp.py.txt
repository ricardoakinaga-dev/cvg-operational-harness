import hashlib
import json
import os

def load(rel):
    with open(os.path.join(UNIT, rel), encoding="utf-8") as f:
        return json.load(f)


def sha(rel):
    with open(os.path.join(UNIT, rel), "rb") as f:
        return hashlib.sha256(f.read()).hexdigest()


def save(rel, doc):
    with open(os.path.join(UNIT, rel), "w", encoding="utf-8") as f:
        f.write(json.dumps(doc, ensure_ascii=False, indent=2) + "\n")


v7, g7, val, iv = load("coverage-observed-v7.json"), load("gaps-review-v7.json"), load("validation.json"), load("input-verification.json")

if PHASE == "final":
    rounds = sorted(d for d in os.listdir(UNIT) if d.startswith("review-r") and os.path.isdir(os.path.join(UNIT, d)))
    carried, problems = [], []
    for rd in rounds:
        rv = load(rd + "/review.json")
        disp = load(rd + "/executor-disposition.json") if os.path.exists(os.path.join(UNIT, rd, "executor-disposition.json")) else None
        dmap = {x["id"]: x for x in (disp or {}).get("dispositions", [])}
        review[rd.upper().replace("REVIEW-", "")] = {
            "path": rd + "/review.json", "sha256": sha(rd + "/review.json"), "verdict": rv["verdict"], "subject": rv["subject"],
            "subject_sha256": rv["subject_sha256"],
            "counts": {s: sum(1 for f in rv["findings"] if f["severity"] == s) for s in ("P1", "P2", "P3")},
            "findings": [{"id": f["id"], "severity": f["severity"], "title": f["title"], "status": dmap.get(f["id"], {}).get("status")}
                         for f in rv["findings"]],
            "disposition": {"path": rd + "/executor-disposition.json", "sha256": sha(rd + "/executor-disposition.json")} if disp else None}
    last = load(rounds[-1] + "/review.json")
    last_disp = load(rounds[-1] + "/executor-disposition.json")
    lmap = {x["id"]: x["status"] for x in last_disp["dispositions"]}
    for f in last["findings"]:
        st = lmap.get(f["id"])
        if st is None:
            problems.append("%s sem disposição" % f["id"])
        if f["severity"] in ("P1", "P2") and st != "FIXED":
            problems.append("%s %s com status %s" % (f["id"], f["severity"], st))
        if st == "CARRIED_OVER":
            carried.append(f["id"])
    if last["verdict"] != "APPROVE" or problems:
        status = "REVIEW_REVISE"
    else:
        status = "COMPLETED_LOCAL_WITH_CARRIED_FINDINGS" if carried else "COMPLETED_LOCAL"
    review["status_rule"] = {"problems": problems, "carried_over": carried}
