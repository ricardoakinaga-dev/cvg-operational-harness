def P(rel):
    return os.path.normpath(os.path.join(UNIT, rel))


def load(rel):
    with open(P(rel), encoding="utf-8") as f:
        return json.load(f)


def sha(rel):
    with open(P(rel), "rb") as f:
        return hashlib.sha256(f.read()).hexdigest()


def src(rel):
    return {"path": rel, "sha256": sha(rel)}

def case_qualifications(case):
    s = sub_of[case]
    r = rows_cr[case]
    _, acc = acceptance(case)
    q = [PROFILE_Q]
    if any(p["id"] == "shim.platform_scoped" for p in acc["predicates"]):
        q.append(SHIM_Q)
    if r["capture_form"] == "REAL_CLI_KEYLESS":
        q.append(REALCLI_Q)
    if r["capture_form"] == "CLIENT_TERMINATED_SIGTERM":
        q.append(SIGTERM_Q)
    if r["capture_form"] == "PASS_DECLARED_TERMINATION":
        q.append(DECL_TERM_Q)
    if s == "DP-1":
        q += [DP1_F07, DP1_DV3]
    if case in ("DP-PK06A", "DP-PK08A"):
        q.append(DP1_R2F03)
    if case == "DP-PK09A":
        q.append(DP1_PK09B02)
    if case == "DP-PK07B":
        q += [DP1R_REPL, DP1R_PK07, DP1R_PLATFORM, DP1R_CONTRARY]
    if case == "DP-PY19A":
        q.append(DP2_F04)
    if case == "DP-PY02P":
        q.append(DP2_F05)
    if case == "DP-PY20A":
        q.append(DP2_F09)
    return q
