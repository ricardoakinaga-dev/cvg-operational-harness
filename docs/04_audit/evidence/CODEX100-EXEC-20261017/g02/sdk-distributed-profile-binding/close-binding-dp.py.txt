import json

def load(rel):
    with open(os.path.join(UNIT, rel), encoding="utf-8") as f:
        return json.load(f)


def sha(rel):
    with open(os.path.join(UNIT, rel), "rb") as f:
        return hashlib.sha256(f.read()).hexdigest()


def save(rel, doc):
    with open(os.path.join(UNIT, rel), "w", encoding="utf-8") as f:
        f.write(json.dumps(doc, ensure_ascii=False, indent=2) + "\n")

    if os.path.exists(os.path.join(UNIT, rd, "executor-disposition.json")):
        return rd + "/executor-disposition.json", load(rd + "/executor-disposition.json")
    if rd == "review-r1" and os.path.exists(os.path.join(UNIT, "repair-r1.json")):
        return "repair-r1.json", load("repair-r1.json")
    return None, None

    carried, problems = [], []
    for rd in rounds:
        rv = load(rd + "/review.json")
        dpath, disp = disposition_of(rd)
        dmap = {x["id"]: x for x in (disp or {}).get("dispositions", [])}
        review[rd.upper().replace("REVIEW-", "")] = {

            "subject_sha256": rv["subject_sha256"],

            "findings": [{"id": f["id"], "severity": f["severity"], "title": f["title"], "status": dmap.get(f["id"], {}).get("status")}
                         for f in rv["findings"]],
            "disposition": {"path": dpath, "sha256": sha(dpath)} if disp else None}
    last = load(rounds[-1] + "/review.json")
    _, last_disp = disposition_of(rounds[-1])
    lmap = {x["id"]: x["status"] for x in last_disp["dispositions"]}
    for f in last["findings"]:
        st = lmap.get(f["id"])
        if st is None:
            problems.append("%s sem disposição" % f["id"])
        if f["severity"] in ("P1", "P2") and st != "FIXED":
            problems.append("%s %s com status %s" % (f["id"], f["severity"], st))
        if st == "CARRIED_OVER":
            carried.append(f["id"])
    if PHASE == "repaired":
        status = "REPAIRED_PENDING_R2"
    elif last["verdict"] != "APPROVE" or problems:
        status = "REVIEW_REVISE"
    else:
        status = "COMPLETED_LOCAL_WITH_CARRIED_FINDINGS" if carried else "COMPLETED_LOCAL"
    review["status_rule"] = {"problems": problems, "carried_over": carried}
