def sha(p):
    with open(p, "rb") as f:
        return hashlib.sha256(f.read()).hexdigest()


def load(p):
    with open(p, encoding="utf-8") as f:
        return json.load(f)


def ref(rel, absp):
    return {"path": rel, "sha256": sha(absp)}

def arm_id(a):
    m = ARM_ID.match(a)
    return m.group(1) if m else None


def arms_of(cid, contract):
    out = []
    for b in cases[cid]["branches_targeted"]:
        aid = arm_id(b)
        if aid is None:
            raise SystemExit("branch without id: %s %s" % (cid, b))
        if aid.startswith(contract + "."):
            out.append(b)
    return out


def observation(cid, contract):
    a, r = acc[cid], adj_rows[cid]
    return {
        "case": cid,
        "final_reading": r["final_reading"], "attribution": r["attribution"], "client_observation": r["client_observation"],
        "service_outcome_r4": r["service_outcome_r4"], "offline_outcome_r4": r["offline_outcome_r4"], "offline_equal": r["offline_equal"],
        "composition_rule": a["composition_rule"], "capture_form": a["capture_form"], "capture_reading": r["capture_reading"],
        "predicates": len(a["predicates"]), "failed_predicates": a["failed_predicates"],
        "replaces_2a_case": r.get("replaces_2a_case"),
        "analytic_branch_arms_observed": arms_of(cid, contract),
        "qualifications": [x for x in (r.get("qualification"), EXTRA_QUAL.get(cid), COMMON_QUAL) if x],
        "acceptance_service": ref(ADMR + r["dir"] + "/acceptance.json", os.path.join(ADM, r["dir"], "acceptance.json")),
        "adjudication_pointer": ADMR + "adjudication-r1.json#/rows[case=%s]" % cid,
        "client_fact_only": True, "parity_receipt": None, "functional_receipts": 0,
    }

        if status3[r["source_contract"]] != "OPEN_REQUIRED_FUTURE_PROFILE":

    return out
