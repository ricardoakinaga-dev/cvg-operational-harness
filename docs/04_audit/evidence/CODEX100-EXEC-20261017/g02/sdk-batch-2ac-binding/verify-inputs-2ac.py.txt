#!/usr/bin/env python3
# DX100-G02:BIND-SDK-SOURCE-BATCH-2A-CONTINUATION - passo 0 (somente leitura).
# Uso: python3 -I -B verify-inputs-2ac.py.txt <repo_root> <out_json>
# As pastas g02 estão não rastreadas no checkout da main; a conferência é por sha (unit-manifest, freezes, next-task) e por
# blob git contra os commits de origem (git hash-object local x git ls-tree do commit), sem escrever nada fora do <out_json>.
import hashlib
import json
import os
import re
import subprocess
import sys
from datetime import datetime, timezone

ROOT, OUT = sys.argv[1], sys.argv[2]
E = "docs/04_audit/evidence/CODEX100-EXEC-20261017"
G02 = os.path.join(ROOT, E, "g02")
checks = []


def sha(p):
    with open(p, "rb") as f:
        return hashlib.sha256(f.read()).hexdigest()


def load(p):
    with open(p, encoding="utf-8") as f:
        return json.load(f)


def check(name, ok, detail=None):
    checks.append({"check": name, "ok": bool(ok), "detail": detail})


def git(*a):
    return subprocess.run(["git", "-C", ROOT] + list(a), capture_output=True, text=True)


def same_as_commit(rel, commit, exclude=()):
    """Every file of the commit tree under rel equals the working file (git blob id), and no extra working file exists."""
    lt = git("ls-tree", "-r", commit, "--", rel)
    if lt.returncode != 0:
        return False, ["ls-tree failed"]
    tree = {}
    for line in lt.stdout.splitlines():
        meta, path = line.split("\t", 1)
        tree[path] = meta.split()[2]
    bad = []
    for path, blob in tree.items():
        if os.path.basename(path) in exclude:
            continue
        local = os.path.join(ROOT, path)
        if not os.path.isfile(local):
            bad.append("missing " + path)
            continue
        h = git("hash-object", local).stdout.strip()
        if h != blob:
            bad.append("differs " + path)
    for dp, _, fs in os.walk(os.path.join(ROOT, rel)):
        for f in fs:
            p = os.path.relpath(os.path.join(dp, f), ROOT)
            if p not in tree and f not in exclude:
                bad.append("extra " + p)
    return not bad, {"files_in_commit": len(tree), "problems": bad[:10]}


# 1-4: admission of the continuation (the only source of observations)
adm_rel = E + "/g02/sdk-batch-2ac-admission"
adm = os.path.join(ROOT, adm_rel)
st = load(os.path.join(adm, "state.json"))
check("2ac admission COMPLETED_LOCAL with R2 APPROVE and 8 PASS + CONTROL MISMATCH",
      st["status"] == "COMPLETED_LOCAL" and st["review"]["r2"]["verdict"] == "APPROVE" and st["cases"]["counts"]["PASS_CLIENT_OBSERVATION"] == 8
      and st["control"]["outcome"] == "MISMATCH", {"status": st["status"], "r2": st["review"]["r2"]["verdict"], "counts": st["cases"]["counts"]})
man = load(os.path.join(adm, "unit-manifest.json"))
bad = [p for p, m in man["files"].items() if sha(os.path.join(adm, p)) != m["sha256"]]
cr = load(os.path.join(adm, "commit-receipt.json"))
check("2ac unit-manifest sha equals commit-receipt and all 520 files match", sha(os.path.join(adm, "unit-manifest.json")) == cr["unit_manifest_sha256"]
      and len(man["files"]) == cr["unit_manifest_file_count"] and not bad, {"files": len(man["files"]), "bad": bad[:5]})
ok, d = same_as_commit(adm_rel, cr["content_commit"], exclude=("commit-receipt.json",))
check("2ac folder equals content commit %s (blob ids)" % cr["content_commit"][:8], ok, d)
check("2ac source-freeze-r4 and identity-freeze-r4 sha as recorded in state.json",
      sha(os.path.join(adm, "source-freeze-r4.json")) == st["freezes"]["source-freeze-r4.json"]
      and sha(os.path.join(adm, "identity-freeze-r4.json")) == st["freezes"]["identity-freeze-r4.json"])
adj = load(os.path.join(adm, "adjudication-r1.json"))
want = ["2A-TS03R", "2A-TS06R", "2A-TS07R", "2A-TS12R", "2A-TS10E", "2A-TS06X", "2A-TS07X", "2A-TS08X"]
rows = {r["case"]: r for r in adj["rows"]}
ok = sorted(adj["bind_consumes_only"]) == sorted(want) and all(rows[c]["final_reading"] == "PASS_CLIENT_OBSERVATION" and rows[c]["attribution"] == "CLIENT" for c in want)
check("adjudication bind_consumes_only = 8 SDK cases, all PASS_CLIENT_OBSERVATION/CLIENT", ok, adj["bind_consumes_only"])
acc_ok = {}
for c in want:
    a = load(os.path.join(adm, rows[c]["dir"], "acceptance.json"))
    acc_ok[c] = a["case"] == c and a["outcome"] == "PASS_CLIENT_OBSERVATION" and a["failed_predicates"] == []
check("acceptance.json of the 8 cases agree with the adjudication", all(acc_ok.values()), acc_ok)
nt = load(os.path.join(adm, "next-task.json"))
check("2ac next-task proposes this unit", nt["task"].startswith("DX100-G02:BIND-SDK-SOURCE-BATCH-2A-CONTINUATION") and nt["t3_required"] is False, nt["task"])

# 5-7: continuation spec (plan, cases) and its copy inside the admission
# Reparo R1-F07: commits de origem lidos dos registros das unidades, não fixos no código.
cont_rel = E + "/g02/sdk-source-batch-2a-continuation-spec"
cont_commit = re.search(r"raiz ([0-9a-f]{8})", st["spec_inputs"]["source"]).group(1)
ok, d = same_as_commit(cont_rel, cont_commit)
check("continuation spec folder equals commit %s read from 2ac state.json#/spec_inputs/source (blob ids)" % cont_commit, ok, d)
cnt = load(os.path.join(ROOT, cont_rel, "next-task.json"))
spec_ok = {k: sha(os.path.join(ROOT, v["path"])) == v["sha256"] for k, v in cnt["spec"].items()}
copies = {f: sha(os.path.join(adm, "inputs/spec-2ac", f)) == sha(os.path.join(ROOT, cont_rel, f))
          for f in ("cases-2a-cont.json", "coverage-plan-2a-cont.json", "freeze-r4-contract.json", "evaluator-r4-contract.json",
                    "control-slot-contract.json", "execution-budget.json")}
check("continuation spec sha (next-task) and the admission copies inputs/spec-2ac are identical", all(spec_ok.values()) and all(copies.values()),
      {"spec": spec_ok, "copies": copies})

# 8-9: binding v2 (base of v3, read only)

# 10-11: batch 2B spec (for the projection) and the lead decision on R2-F01
b2_rel = E + "/g02/sdk-source-batch-2b-spec"
b2nt = load(os.path.join(ROOT, b2_rel, "next-task.json"))
b2_ok = {k: sha(os.path.join(ROOT, v["path"])) == v["sha256"] for k, v in b2nt["spec"].items()}
b2st = load(os.path.join(ROOT, b2_rel, "state.json"))

blob_commit = None

    lt = git("ls-tree", commits[0], "--", ld_rel).stdout.split()
    blob_commit = lt[2] if len(lt) >= 3 else None
ld_sha = sha(os.path.join(ROOT, ld_rel))
check("lead decision on 2B-SPEC/R2-F01: ACCEPT_BINDING_RULE, blob equal to the latest commit that contains it",
      ld["decision"] == "ACCEPT_BINDING_RULE" and blob_commit == blob_local,
      {"decision": ld["decision"], "commit": commits[0] if commits else None, "blob": blob_local, "sha256": ld_sha})

# 12: G01 contracts used as residual references
g01 = {"python-contracts.json": "25c48bdd75710c2eb5bde33c6b18379892078ca3e5b78f1448dfe42c94e19ad1",
       "typescript/contracts.json": "b90b17ce1bbcd824ff92c68180b79d4ef428c715ced48a520db379bbfb7275d2"}

doc = {"kind": "INPUT_VERIFICATION", "task": "DX100-G02:BIND-SDK-SOURCE-BATCH-2A-CONTINUATION",
       "result": "PASS" if all(c["ok"] for c in checks) else "FAIL", "checks": checks,
       "repo": {"branch": git("branch", "--show-current").stdout.strip(), "head": git("rev-parse", "--short=8", "HEAD").stdout.strip(),

    f.write(json.dumps(doc, ensure_ascii=False, indent=2) + "\n")
print(doc["result"], sum(c["ok"] for c in checks), "/", len(checks))
