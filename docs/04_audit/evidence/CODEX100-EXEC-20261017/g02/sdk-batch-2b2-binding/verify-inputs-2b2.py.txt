#!/usr/bin/env python3
# DX100-G02:BIND-SDK-SOURCE-BATCH-2B-2 - passo 0 (somente leitura).
# Uso: python3 -I -B verify-inputs-2b2.py.txt <repo_root> <out_json>
# As pastas g02 estão não rastreadas no checkout da main. A conferência é por sha (manifestos, freezes, next-task) e por blob git
# (git hash-object local x git ls-tree do commit). Âncora durável: o commit do PR documental #7 do Lead, que está na ref
# origin/candidate/d9ef-docs-20261019 e leva a especificação 2B, o BIND do 2B-1 e a admissão 2B-2. Nada é escrito fora do <out_json>.
import hashlib
import json
import os
import subprocess
import sys
from datetime import datetime, timezone

ROOT, OUT = sys.argv[1], sys.argv[2]
E = "docs/04_audit/evidence/CODEX100-EXEC-20261017"
G02R = E + "/g02"
LEAD_PR_COMMIT = "7e1da11a4"  # PR documental #7 do Lead; resolvido para o sha completo abaixo
checks = []


def sha(p):
    with open(p, "rb") as f:
        return hashlib.sha256(f.read()).hexdigest()


def load(p):
    with open(p, encoding="utf-8") as f:
        return json.load(f)


def A(rel):
    return os.path.join(ROOT, rel)


def check(name, ok, detail=None):
    checks.append({"check": name, "ok": bool(ok), "detail": detail})


def git(*a):
    return subprocess.run(["git", "-C", ROOT] + list(a), capture_output=True, text=True)


def same_as_commit(rel, commit, exclude=()):
    """Todo arquivo da árvore do commit sob rel é igual ao arquivo local (blob git) e não há arquivo local a mais."""
    lt = git("ls-tree", "-r", commit, "--", rel)
    if lt.returncode != 0:
        return False, {"problems": ["ls-tree failed"]}
    tree = {}
    for line in lt.stdout.splitlines():
        meta, path = line.split("\t", 1)
        tree[path] = meta.split()[2]
    bad = []
    for path, blob in tree.items():
        if os.path.basename(path) in exclude:
            continue
        if not os.path.isfile(A(path)):
            bad.append("missing " + path)
        elif git("hash-object", A(path)).stdout.strip() != blob:
            bad.append("differs " + path)
    for dp, _, fs in os.walk(A(rel)):
        for f in fs:
            p = os.path.relpath(os.path.join(dp, f), ROOT)
            if p not in tree and f not in exclude:
                bad.append("extra " + p)
    return (not bad and len(tree) > 0), {"files_in_commit": len(tree), "problems": bad[:10]}


PR = git("rev-parse", LEAD_PR_COMMIT).stdout.strip()
refs = [x for x in git("for-each-ref", "--contains", PR, "--format=%(refname)").stdout.split() if x]
check("Lead documentary PR commit resolves and is contained in a ref (durable anchor, 2AC-BIND/R2-F04)", len(PR) == 40 and bool(refs), {"commit": PR, "refs": refs})

      st["status"] == "COMPLETED_LOCAL" and st["review"]["r1"]["verdict"] == "APPROVE" and st["cases"]["counts"]["PASS_CLIENT_OBSERVATION"] == 17
      and st["control"]["outcome"] == "MISMATCH" and st["control"]["gate_passed"] is True,
      {"status": st["status"], "r1": st["review"]["r1"]["verdict"], "counts": st["cases"]["counts"]})
man = load(os.path.join(adm, "unit-manifest.json"))
bad = [p for p, m in man["files"].items() if sha(os.path.join(adm, p)) != m["sha256"]]
cr = load(os.path.join(adm, "commit-receipt.json"))
check("2B-2 unit-manifest sha equals commit-receipt and all %d files match (discrimination-check-2b2.json included)" % len(man["files"]),
      sha(os.path.join(adm, "unit-manifest.json")) == cr["unit_manifest_sha256"] and len(man["files"]) == cr["unit_manifest_file_count"] and not bad
      and "discrimination-check-2b2.json" in man["files"], {"files": len(man["files"]), "bad": bad[:5]})
ok, d = same_as_commit(adm_rel, cr["content_commit"], exclude=("commit-receipt.json", "lead-decision-f01.json"))
check("2B-2 folder equals content commit %s (blob ids; commit-receipt and the later lead decision excluded)" % cr["content_commit"][:8], ok, d)
ok, d = same_as_commit(adm_rel, PR)
check("2B-2 folder equals the Lead PR commit %s, lead-decision-f01.json included (blob ids)" % PR[:8], ok, d)
fr = {f: sha(os.path.join(adm, f)) == st["freezes"][f] for f in ("source-freeze-r4-2b2.json", "identity-freeze-r4-2b2.json", "diag-admission-2b2.json")}
check("2B-2 source-freeze, identity-freeze and diag-admission sha as recorded in state.json", all(fr.values()), fr)
adj = load(os.path.join(adm, "adjudication-r1.json"))
rows = {r["case"]: r for r in adj["rows"]}
want = [r["case"] for r in adj["rows"] if r["kind"] == "SDK"]
dc = load(os.path.join(adm, "discrimination-check-2b2.json"))
by_status = {}
for a in dc["arms"]:
    by_status.setdefault(a["status"], []).append(a["arm"])
ok = (sorted(adj["bind_consumes_only"]) == sorted(want) and len(want) == 17 and "2B2-CTRL01" not in adj["bind_consumes_only"]
      and all(rows[c]["final_reading"] == "PASS_CLIENT_OBSERVATION" and rows[c]["attribution"] == "CLIENT" for c in want))
check("adjudication bind_consumes_only = the 17 SDK cases, all PASS_CLIENT_OBSERVATION/CLIENT; CONTROL not consumed", ok, adj["bind_consumes_only"])
row_arms = {s: sorted(a for c in want for a in rows[c]["arms_" + k]) for s, k in (("DISCRIMINATED", "discriminated"), ("PARTIAL_SCOPE", "partial_scope"), ("NOT_DISCRIMINATED", "not_discriminated"))}
ok = (dc["counts"] == adj["arm_discrimination"]["counts"] == {"DISCRIMINATED": 68, "PARTIAL_SCOPE": 8, "NOT_DISCRIMINATED": 1}
      and all(sorted(by_status.get(s, [])) == row_arms[s] for s in row_arms)
      and sorted(by_status["PARTIAL_SCOPE"]) == sorted(adj["arm_discrimination"]["partial_scope"]) and by_status["NOT_DISCRIMINATED"] == adj["arm_discrimination"]["not_discriminated"])
check("discrimination-check-2b2.json agrees with adjudication-r1.json (68/8/1, per-case arms, partial and not discriminated lists)", ok, dc["counts"])
acc_ok = {}
for c in want:
    a = load(os.path.join(adm, rows[c]["dir"], "acceptance.json"))
    acc_ok[c] = (a["case"] == c and a["outcome"] == "PASS_CLIENT_OBSERVATION" and a["failed_predicates"] == [] and a["capture_form"] == rows[c]["capture_form"]
                 and a["predicate_ids_match_plan"] is True and all(p["ok"] for p in a["predicates"]))
check("acceptance.json of the 17 cases agree with the adjudication (outcome, capture_form, all predicates ok)", all(acc_ok.values()), acc_ok)
rv = load(os.path.join(adm, "reevaluation-2b2.json"))

nt = load(os.path.join(adm, "next-task.json"))

ld = load(os.path.join(adm, "lead-decision-f01.json"))
check("lead decision F01: ACCEPT_FAIL_CLOSED_OUTCOME, the 2B-PY13E gets no credit for discriminating 'pular o cleanup'",
      ld["decision"] == "ACCEPT_FAIL_CLOSED_OUTCOME" and "2B-PY13E" in ld["consequence_for_bind"], {"sha256": sha(os.path.join(adm, "lead-decision-f01.json"))})

# 10-12: especificação 2B e BIND do 2B-1 (v4, base do v5)
b2_rel = G02R + "/sdk-source-batch-2b-spec"
ok, d = same_as_commit(b2_rel, PR)
check("2B spec folder equals the Lead PR commit %s (blob ids)" % PR[:8], ok, d)
copies = {f: sha(os.path.join(adm, "inputs/spec-2b", f)) == sha(A(b2_rel + "/" + f)) for f in ("cases-2b-2.json", "coverage-plan-2b.json", "next-task.json")}
check("the admission copies inputs/spec-2b (cases-2b-2, coverage-plan-2b, next-task) equal the 2B spec", all(copies.values()), copies)

check("2B-1 binding folder equals the Lead PR commit %s (blob ids)" % PR[:8], ok, d)
b1st = load(A(b1_rel + "/state.json"))

check("2B-1 binding COMPLETED_LOCAL; v4, R2 and disposition sha as recorded in its next-task; v4 29/15; R2-F01..F03 carried to this BIND",

      {"status": b1st["status"], "records": rec, "counts": cov4["counts"], "disposition": [(i["id"], i["disposition"]) for i in disp["items"]]})

# 13-14: bases de fonte do v4 contra a identity freeze (2B1-BIND/R2-F03) e contratos G01
tree = load(A(G02R + "/sdk-batch-2b1-admission/identity-freeze-r4-2b1.json"))["trees"]["sdk-python"]

      bool(bases) and all(bases.values()) and tree == tree2, bases)
g01 = {"python-contracts.json": "25c48bdd75710c2eb5bde33c6b18379892078ca3e5b78f1448dfe42c94e19ad1",
       "typescript/contracts.json": "b90b17ce1bbcd824ff92c68180b79d4ef428c715ced48a520db379bbfb7275d2"}

       "anchor_commit": {"commit": PR, "what": "commit do PR documental #7 do Lead (CONT-20261018-D9EF), que leva a especificação 2B, o BIND do 2B-1 e a admissão 2B-2",

    f.write(json.dumps(doc, ensure_ascii=False, indent=2) + "\n")
