def sha(p):
    with open(p, "rb") as f:
        return hashlib.sha256(f.read()).hexdigest()

def interventions(cid):
    out = []
    for s in cases[cid]["case_input"]["sdk_calls"]:
        if s["api"] == "shim.install":
            out.append("shim em %s (%s)" % (s["target"], ", ".join(k for k in s if k not in ("step", "api", "target"))))
        elif s["api"] == "fault.install":
            out.append("fault %s em %s.%s" % (s["mode"], s["on"], s["attr"]))
        elif s["api"] == "loop.block_release_wait_cancel":
            out.append("loop bloqueado até o gancho %s (loop.block_release_wait_cancel)" % s["wait_hook"]["id"])
        elif isinstance(s.get("fault"), dict):
            out.append("fault %s no %s" % (json.dumps(s["fault"], ensure_ascii=False, sort_keys=True), s.get("as")))
    return out


def acc_pred(cid, pid):
    for p in acc[cid]["predicates"]:
        if p["id"] == pid:
            return p
    raise SystemExit("predicate %s not in acceptance of %s" % (pid, cid))


def evidence(cid, aid):
    out = []
    for it in ARM_EVIDENCE[cid][aid]:
        p = acc_pred(cid, it["predicate"])
        if not p["ok"] or it["predicate"] not in cases[cid]["predicates"]:
            raise SystemExit("evidence predicate not ok or not frozen: %s %s" % (cid, it))
        out.append(dict(it, **{"class": p["class"], "ok": True}))
    return out
