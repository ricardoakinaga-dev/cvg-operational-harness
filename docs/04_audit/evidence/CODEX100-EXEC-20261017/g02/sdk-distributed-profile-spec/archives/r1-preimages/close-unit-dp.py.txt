#!/usr/bin/env python3
# DX100-G02:SPECIFY-SDK-DISTRIBUTED-PROFILE-KEYLESS - escreve next-task.json e state.json com os sha atuais dos documentos da especificação.
# Uso: python3 -I -B close-unit-dp.py.txt <spec_dir> <phase>
#   phase: draft (antes da revisão) | final (depois da última rodada; lê review-r1/review.json e, se existirem, repair-r1.json,

import json
import os
import sys

def sha(name):
    with open(os.path.join(SPEC, name), "rb") as f:
        return hashlib.sha256(f.read()).hexdigest()


def load(name):
    with open(os.path.join(SPEC, name), encoding="utf-8") as f:
        return json.load(f)


def exists(name):
    return os.path.exists(os.path.join(SPEC, name))


def ref(name):
    return {"path": UNIT + "/" + name, "sha256": sha(name)}

review, carry, status, rounds, repairs = None, [], "IN_REVIEW", 0, 0

    r1 = load("review-r1/review.json")
    rounds = 1
    review = {"R1": "%s (%s) review-r1/review.json; texto sem edição em review-r1/verdict.raw.txt" % (r1["verdict"], r1.get("summary_counts", "")),
              "critic": "subagente novo por rodada, somente leitura; o executor grava brief e veredito sem edição"}
    last = r1
    if exists("repair-r1.json"):
        repairs = 1
        review["repair"] = "repair-r1.json (1/1), preimagens em archives/r1-preimages/"
    if exists("review-r2/review.json"):
        r2 = load("review-r2/review.json")
        rounds, last = 2, r2
        review["R2"] = "%s (%s) review-r2/review.json; texto sem edição em review-r2/verdict.raw.txt" % (r2["verdict"], r2.get("summary_counts", ""))
    disp_name = "review-r2/executor-disposition.json" if rounds == 2 else "review-r1/executor-disposition.json"
    disp = load(disp_name) if exists(disp_name) else None
    if disp:
        review["disposition"] = disp_name
        carry = [{"finding": "DP-SPEC/" + i["id"], "text": i["note"]} for i in disp["items"] if i["disposition"].startswith("CARRIED")]
        p1_open = [i["id"] for i in disp["items"] if i.get("severity") == "P1" and not i["disposition"].startswith("FIXED")]
        p2_open = [i["id"] for i in disp["items"] if i.get("severity") == "P2" and not (i["disposition"].startswith("FIXED") or i["disposition"].startswith("CARRIED"))]
    else:
        p1_open = [f["id"] for f in last.get("findings", []) if f["severity"] == "P1"]
        p2_open = [f["id"] for f in last.get("findings", []) if f["severity"] == "P2"]
    carried_p2 = any("P2" == i.get("severity") and i["disposition"].startswith("CARRIED") for i in (disp or {}).get("items", []))
    if p1_open or p2_open:
        status = "BLOCKED_REVIEW"
    elif last["verdict"] == "APPROVE" and not carry:
        status = "COMPLETED_LOCAL"
    else:
        status = "COMPLETED_LOCAL_WITH_CARRIED_P2" if carried_p2 else "COMPLETED_LOCAL_WITH_CARRIED_P3"
