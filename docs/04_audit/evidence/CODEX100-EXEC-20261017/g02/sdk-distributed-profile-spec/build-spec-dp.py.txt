def sha_bytes(b):
    return hashlib.sha256(b).hexdigest()


def sha_file(p):
    return sha_bytes(open(p, "rb").read())

def compact(o):

def ptr(p, pointer=None):
    d = {"path": p, "sha256": sha_file(rel(p))}
    if pointer:
        d["pointer"] = pointer
    return d

def member_text(aid, name):
    key = "%s:%s" % (aid, name)
    if key in TEXT:
        return TEXT[key]
    path = STORED[aid]
    assert sha_file(path) == ARCH_SHA[aid], aid
    if path.endswith(".whl"):
        with zipfile.ZipFile(path) as z:
            data = z.read(name)
    else:
        with tarfile.open(path, "r:gz") as t:
            data = t.extractfile(t.getmember(name)).read()
    assert sha_bytes(data) == MEMBERS[aid][name]["sha256"], key
    TEXT[key] = {"lines": data.decode("utf-8").split("\n"), "sha256": sha_bytes(data), "archive_sha256": ARCH_SHA[aid],
                 "archive": os.path.basename(path), "mode": MEMBERS[aid][name].get("mode")}
    return TEXT[key]

def lines_of(key):
    if ":" in key and key.split(":", 1)[0] in STORED:
        aid, name = key.split(":", 1)
        return member_text(aid, name)["lines"]
    return open(rel(PIN_FILES[key]), encoding="utf-8").read().split("\n")


def A(key, text, nth=None, span=0):
    ls = lines_of(key)
    hits = [i for i, l in enumerate(ls, 1) if text in l]
    if (nth is None and len(hits) != 1) or (nth is not None and len(hits) < nth):
        ANCHOR_ERRORS.append((key, text, hits))
        return {"input": key, "error": "ambiguous_or_missing"}
    n = hits[0] if nth is None else hits[nth - 1]
    if key in PIN_FILES:
        sha = sha_file(rel(PIN_FILES[key]))
    else:
        sha = member_text(*key.split(":", 1))["sha256"]
    return {"input": key, "sha256": sha, "lines": [n] if not span else [n, n + span], "text": ls[n - 1].strip()}

def finalize(cases, ctrl_id, ctrl, sub_lot, language):
    out = []
    items = list(cases.items()) + [(ctrl_id, ctrl)]
    for slot, (cid, c) in enumerate(items, 1):
        c = dict(c)
        is_ctrl = c.get("kind") == "CONTROL" or "control_case_input" in c
        if "control_case_input" in c:
            ci = c.pop("control_case_input")
        else:
            ci = {"sdk_calls": c.pop("sdk_calls")}
            for k in ("trees", "child_script", "peer_script", "driver_runtime"):
                if c.get(k) not in (None, []):
                    ci[k] = c[k]
                c.pop(k, None)
        for k in ("trees", "child_script", "peer_script", "driver_runtime"):
            c.pop(k, None)
        c.update(id=cid, kind="CONTROL" if is_ctrl else ("ARTIFACT_CHECK" if c.get("role") == "ARTIFACT_CHECK" else "SDK"),
                 language=c.get("language", "python" if language == "python" else "node"), sub_lot=sub_lot, slot=slot)
        if is_ctrl:
            c["execution_order"] = 1
        else:
            c["inconclusive_on_deadline"] = False
            c["credit"] = CREDIT
            c["build_link"] = NBL
        for a in c.get("anchors", []):
            assert "error" not in a, (cid, a)
        # R1-F11: inputs do caso = âncoras + membros copiados nas fixtures + membros lidos por artifact.check + metadados do cache
        used = {a["input"] for a in c.get("anchors", [])} | set(c.pop("extra_inputs", []))
        if not is_ctrl:
            used |= {e["member"] for t in ci.get("trees", []) for e in t["entries"] if e["kind"] == "copy_of_member"}
            used |= {s["member"] for s in ci["sdk_calls"] if s.get("api") == "artifact.check" and "member" in s}
        c["inputs"] = sorted(used)
        c["runtime_status"] = "NOT_RUN"
        c["receipt"] = None
        c["case_input_note"] = "Passos, árvores de fixture, roteiro do filho sintético, roteiro do peer e runtime existem só dentro de case_input (fonte única do supervisor, F4-01)."
        c["case_input"] = ci
        c["case_input_sha256"] = sha_bytes(compact(ci).encode("utf-8"))
        extra = set(c) - set(ORDER)
        assert not extra, (cid, extra)
        out.append({k: c[k] for k in ORDER if k in c})
    return out
