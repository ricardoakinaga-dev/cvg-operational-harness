def sha(p):
    with open(p, "rb") as f:
        return hashlib.sha256(f.read()).hexdigest()


def load(p):
    with open(p, encoding="utf-8") as f:

def ref(rel, absp):
    return {"path": rel, "sha256": sha(absp)}


def arm_id(a):
    m = ARM_ID.match(a)

def acc_pred(cid, pid):
    for p in acc[cid]["predicates"]:
        if p["id"] == pid:
            return p
    raise SystemExit("predicate %s not in acceptance of %s" % (pid, cid))


def evidence(cid, aid):
    items = ARM_EVIDENCE[cid][aid]
    out = []
    for it in items:
        p = acc_pred(cid, it["predicate"])
        if not p["ok"] or it["predicate"] not in cases[cid]["predicates"]:
            raise SystemExit("evidence predicate not ok or not frozen: %s %s" % (cid, it))
        out.append(dict(it, **{"class": p["class"], "ok": True}))
    return out
