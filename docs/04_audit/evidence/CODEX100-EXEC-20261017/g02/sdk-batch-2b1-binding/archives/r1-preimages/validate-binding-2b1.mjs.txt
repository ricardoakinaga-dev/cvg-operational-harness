// Recalcula a partir das entradas: casos consumidos, status por linha, braços observados e abertos, contagens, sha das fontes e dos
// acceptance.json citados, lacunas e projeção do 2B-2. Confere o conteúdo (2AC-BIND/R2-F02): cada evidência de braço contra o
// acceptance (predicado aprovado e congelado, chaves de probe, passos, tokens, ganchos, consumidores, regras), os marcadores das
// qualificações, os escopos, as erratas do v3 e a cópia carried_from_v3 linha a linha; roda controles negativos em cópias em memória.
import { createHash } from 'node:crypto'
import { readFileSync, writeFileSync, readdirSync, statSync } from 'node:fs'

const G02 = 'docs/04_audit/evidence/CODEX100-EXEC-20261017/g02'
const UNIT = G02 + '/sdk-batch-2b1-binding'

const sha = (b) => createHash('sha256').update(b).digest('hex')
const shaFile = (p) => sha(readFileSync(p))
const load = (p) => JSON.parse(readFileSync(p, 'utf8'))
const clone = (x) => JSON.parse(JSON.stringify(x))
const J = (x) => JSON.stringify(x)
const armId = (a) => (/^((?:TS|PY|PK)\d\d\.B\d\d)/.exec(a) || [])[1] ?? null
const strings = (x) => (typeof x === 'string' ? [x] : Array.isArray(x) ? x.flatMap(strings) : x && typeof x === 'object' ? Object.values(x).flatMap(strings) : [])

const cNext = load(join(C, 'next-task.json'))

  disp: load(join(C, 'review-r2/executor-disposition.json')), adj,

  cases2: Object.fromEntries(load(join(ADM, 'inputs/spec-2b/cases-2b-2.json')).cases.map((c) => [c.id, c])),
  plan: load(join(ADM, 'inputs/spec-2b/coverage-plan-2b.json')),

// Regras de conteúdo fixas desta unidade (o build escreve os textos; aqui só os marcadores mínimos e as decisões).
const NOT_DISCRIMINATED = new Set(['PY05|PY05.B10'])
const QUAL_MARKERS = {
  'PY02|PY02.B02': ['ordem das chaves não é comparada'], 'PY02|PY02.B06': ['lançador', 'Popen.args'], 'PY02|PY02.B07': ['ordem das chaves não é comparada'],
  'PY02|PY02.B08': ['falha declarada', 'não foi observado'], 'PY08|PY08.B07': ['não é afirmado'], 'PY10|PY10.B04': ['forçada pelo harness', 'não é afirmada'],
}
const CARRIED_QUAL_MARKERS = { 'TS05.B04': ['inferidos', 'não foram testemunhados'], 'TS10.B04': ['inferida'], 'TS12.B01': ['NOT_WITNESSED'] }
const SCOPE_RULES = { 'PY05|PY05.B06': { must: 'no mínimo', mustNot: 'acima do mínimo', notObserved: 'acima do mínimo' } }
const CASE_QUAL_MARKERS = { '2B-PY01A': 'CLIENT_TERMINATED_SIGTERM', '2B-PY01D': 'CLIENT_TERMINATED_SIGTERM', '2B-PY02A': 'CLIENT_TERMINATED_SIGTERM', '2B-PY02K': 'falha declarada' }
const MUST_REMAIN_ADD = new Set(['PY02', 'PY05', 'PY10'])
const MUST_REMAIN_ADD_V3 = ['TS03', 'TS05', 'TS07', 'TS10', 'TS12']
const LIFT = ['source_contract', 'gaps']

function verifyEvidence(cid, it) {
  const a = inputs.acc[cid]
  const p = a.predicates.find((x) => x.id === it.predicate)
  if (!p) return 'predicate absent'
  if (!p.ok) return 'predicate not ok'
  if (!(inputs.cases[cid]?.predicates ?? []).includes(it.predicate)) return 'predicate not frozen for the case'
  if (it.class !== p.class) return 'class differs'
  const d = p.detail
  if (it.keys) for (const k of it.keys) if (!(d && typeof d === 'object' && !Array.isArray(d) && k in d)) return `probe key ${k} absent`
  if (it.steps) for (const s of it.steps) if (!(Array.isArray(d) && d.some((x) => x.step === s && x.ok === true))) return `step ${s} absent`
  if (it.tokens) {
    const pools = it.predicate === 'hook.events_exact' ? d.chains : d
    for (const t of it.tokens) if (!(Array.isArray(pools) && pools.some((o) => o && t in o))) return `token ${t} absent`
  }
  if (it.entries) for (const e of it.entries) if (!(d.entries ?? []).some((x) => x.matched?.length && Object.entries(e).every(([k, v]) => x.entry?.[k] === v))) return `hook entry ${J(e)} absent`
  if (it.workers) for (const w of it.workers) if (d?.[w]?.ok !== true) return `worker ${w} absent`
  if (it.rules) for (const r of it.rules) if (!(d.checked ?? []).some((x) => x.rule === r && x.ok === true)) return `peer rule ${r} absent`
  return null
}

function expected(inp) {
  const consumed = new Set(inp.adj.bind_consumes_only)
  const adjRows = Object.fromEntries(inp.adj.rows.map((r) => [r.case, r]))
  const plan = Object.fromEntries(inp.plan.rows.map((r) => [r.source_contract, r]))
  const out = {}
  for (const r3 of inp.v3.rows) {
    const c = r3.source_contract
    const p = plan[c]
    const in1 = Boolean(p) && J(p.sub_lots) === J(['2B-1'])
    let status = r3.status
    const observed = {}, open = []
    if (in1) {
      for (const x of p.cases.filter((y) => consumed.has(y))) {
        for (const b of inp.cases[x].branches_targeted.filter((y) => y.startsWith(c + '.'))) {
          const id = armId(b)
          if (NOT_DISCRIMINATED.has(`${c}|${id}`)) open.push(`${id}|${x}|${b}`)
          else (observed[id] ??= []).push(`${x}|${b}`)
        }
      }
      const used = p.primary_cases.filter((x) => adjRows[x] && adjRows[x].outcome !== 'NOT_RUN')
      if (r3.status !== 'OBSERVED_CLIENT_PARTIAL') status = used.length && used.every((x) => consumed.has(x)) ? 'OBSERVED_CLIENT_PARTIAL' : 'OPEN_REQUIRED_FUTURE_PROFILE'
    }
    out[c] = { status, in1, p, observed, open, obs: in1 ? p.cases.filter((x) => consumed.has(x)) : [] }
  }
  return out
}

function expectedCarried(inp, c, v4row) {
  const r3 = clone(inp.v3.rows.find((x) => x.source_contract === c))
  for (const k of LIFT) delete r3[k]
  if (c === 'TS07') {
    const a = r3.arms_closed_v3.find((x) => armId(x.arm) === 'TS07.B03')
    Object.assign(a, inp.disp.items.find((i) => i.id === 'R2-F01').erratum_text)
    a.evidence_v4 = v4row?.carried_from_v3?.arms_closed_v3?.find((x) => armId(x.arm) === 'TS07.B03')?.evidence_v4
  }
  if (c === 'TS12') r3.remaining_after_continuation_additions = v4row?.carried_from_v3?.remaining_after_continuation_additions
  return r3
}

function validate(docs, inp) {
  const errors = []
  const err = (m) => errors.push(m)
  const v4 = docs['coverage-observed-v4.json']
  const g4 = docs['gaps-review-v4.json']
  const pj = docs['projection-2b2-on-v4.json']
  const v3 = inp.v3
  if (shaFile(join(C, 'coverage-observed-v3.json')) !== cNext.review_records.coverage_observed_v3.sha256) err('v3 changed on disk')
  const exp = expected(inp)
  const consumed = new Set(inp.adj.bind_consumes_only)
  const adjRows = Object.fromEntries(inp.adj.rows.map((r) => [r.case, r]))
  if (v4.rows.length !== 44 || v3.rows.length !== 44) err('row count')
  if (J([...v4.bound_cases_2b1].sort()) !== J([...consumed].sort()) || consumed.has('2B1-CTRL01')) err('bound cases differ from bind_consumes_only')
  if (v4.control_slot?.counts_as_sdk_observation !== false) err('bound cases: CONTROL counted')
  if (J(v4.conditional_slots) !== J(inp.adj.conditional_slots)) err('conditional slots differ from the adjudication')
  const F01 = inp.disp.items.find((i) => i.id === 'R2-F01')
  const caseArmEvidence = {}
  for (const r of v4.rows) {
    const c = r.source_contract
    const e = exp[c]
    if (!e) { err(`unknown row ${c}`); continue }
    const r3 = v3.rows.find((x) => x.source_contract === c)
    if (r.status !== e.status) err(`${c}: status ${r.status} != ${e.status}`)
    if (r.status_v3 !== r3.status || J(r.gaps) !== J(r3.gaps)) err(`${c}: status_v3 or gaps`)
    if (r.status_changed !== (r.status !== r3.status)) err(`${c}: status_changed flag`)
    // cópia do v3 com só as erratas declaradas
    if (J(r.carried_from_v3) !== J(expectedCarried(inp, c, r))) err(`${c}: carried_from_v3 differs from the v3 row beyond the declared errata`)
    if (c === 'TS07') {
      const a = r.carried_from_v3.arms_closed_v3.find((x) => armId(x.arm) === 'TS07.B03')
      if (a.observed_scope !== F01.erratum_text.observed_scope || a.not_observed !== F01.erratum_text.not_observed || a.observed_scope.includes('chave vazia')) err('TS07.B03 erratum not applied in the row')
      for (const ev of a.evidence_v4 ?? []) {
        const acc = load(normalize(join(C, ev.acceptance)))
        const p = acc.predicates.find((x) => x.id === ev.predicate)
        if (!(p && p.ok && ev.contains.every((s) => strings(p.detail).some((t) => t.includes(s))))) err('TS07.B03 erratum evidence not in the 2A-TS07R acceptance')
      }
      if (!(a.evidence_v4 ?? []).length) err('TS07.B03 erratum evidence missing')
    }
    if (c === 'TS12') {
      const t = r.carried_from_v3.remaining_after_continuation_additions ?? ''
      if (!t.includes('pergunta T3 nova') || t.includes('condicionada à pergunta T3 contingente') || !t.includes('NOT_WITNESSED')) err('TS12 destination (2AC-BIND/R2-F05) not applied')
    }
    for (const a of r.carried_from_v3.arms_closed_v3 ?? []) {
      for (const m of CARRIED_QUAL_MARKERS[armId(a.arm)] ?? []) if (!(a.qualification ?? '').includes(m)) err(`${c}: carried qualification marker '${m}' missing in ${armId(a.arm)}`)
    }
    if (MUST_REMAIN_ADD_V3.includes(c) && !(r.carried_from_v3.remaining_after_continuation_additions ?? '').length) err(`${c}: carried remaining additions missing`)
    if (e.in1) {
      // observações
      if (J((r.observations_2b1 ?? []).map((o) => o.case)) !== J(e.obs)) err(`${c}: observations differ`)
      for (const o of r.observations_2b1 ?? []) {
        const ar = adjRows[o.case]
        if (!consumed.has(o.case)) err(`${c}: ${o.case} not consumable`)
        if (o.final_reading !== ar.final_reading || o.attribution !== ar.attribution) err(`${o.case}: reading not from the adjudication`)
        if (o.acceptance_service.path !== '../sdk-batch-2b1-admission/' + ar.dir + '/acceptance.json') err(`${o.case}: acceptance pointer is not the adjudication dir`)
        if (shaFile(normalize(join(UNIT, o.acceptance_service.path))) !== o.acceptance_service.sha256) err(`${o.case}: acceptance sha`)
        const acc = inp.acc[o.case]
        if (o.capture_form !== acc.capture_form || o.predicates !== acc.predicates.length || J(o.failed_predicates) !== J(acc.failed_predicates)) err(`${o.case}: capture form or predicates differ from acceptance`)
        if (J(o.analytic_branch_arms_targeted) !== J(inp.cases[o.case].branches_targeted.filter((b) => b.startsWith(c + '.')))) err(`${o.case}: targeted arms differ from branches_targeted`)
        const q = (o.qualifications ?? []).join(' ')
        if (CASE_QUAL_MARKERS[o.case] && !q.includes(CASE_QUAL_MARKERS[o.case])) err(`${o.case}: case qualification marker '${CASE_QUAL_MARKERS[o.case]}' missing`)
      }
      // braços observados e abertos
      const gotObs = Object.fromEntries((r.arms_observed_2b1 ?? []).map((a) => [a.arm_id, a.observed_by.map((x) => `${x.case}|${x.label}`)]))
      if (J(Object.keys(gotObs).sort()) !== J(Object.keys(e.observed).sort())) err(`${c}: observed arms differ`)
      for (const [id, list] of Object.entries(e.observed)) if (J(gotObs[id]) !== J(list)) err(`${c}: observed_by of ${id} differs`)
      const gotOpen = (r.arms_open_explicit_2b1 ?? []).map((a) => `${a.arm_id}|${a.case}|${a.arm}`)
      if (J(gotOpen) !== J(e.open)) err(`${c}: not discriminated arms differ`)
      for (const a of r.arms_open_explicit_2b1 ?? []) {
        if (a.reason !== 'NOT_DISCRIMINATED' || !(a.detail ?? '').includes('py/client.py:342')) err(`${c}: not discriminated arm ${a.arm_id} without reason`)
        for (const it of a.evidence_non_discriminating ?? []) { const v = verifyEvidence(a.case, it); if (v) err(`${c}: evidence of ${a.arm_id} (${a.case}, ${it.predicate}): ${v}`) }
      }
      for (const a of r.arms_observed_2b1 ?? []) {
        const key = `${c}|${a.arm_id}`
        for (const ob of a.observed_by) {
          if (ob.capture_form !== inp.acc[ob.case].capture_form) err(`${c}: capture form of ${a.arm_id}`)
          if (!(ob.evidence ?? []).length) err(`${c}: evidence of ${a.arm_id} (${ob.case}) empty`)
          for (const it of ob.evidence ?? []) {
            const v = verifyEvidence(ob.case, it)
            if (v) err(`${c}: evidence of ${a.arm_id} (${ob.case}, ${it.predicate}): ${v}`)
            ;(caseArmEvidence[ob.case] ??= new Set()).add(it.predicate)
          }
          if (!(ob.evidence ?? []).some((it) => it.class === 'client')) err(`${c}: no client-class evidence for ${a.arm_id} (${ob.case})`)
        }
        for (const m of QUAL_MARKERS[key] ?? []) if (!(a.qualification ?? '').includes(m)) err(`${c}: qualification marker '${m}' missing in ${a.arm_id}`)
        if (!QUAL_MARKERS[key] && a.qualification) err(`${c}: undeclared qualification in ${a.arm_id}`)
        const sr = SCOPE_RULES[key]
        if (sr && !((a.observed_scope ?? '').includes(sr.must) && !(a.observed_scope ?? '').includes(sr.mustNot) && (a.not_observed ?? '').includes(sr.notObserved))) err(`${c}: scope of ${a.arm_id} is not the observed one`)
        if (!sr && (a.observed_scope || a.not_observed)) err(`${c}: undeclared scope in ${a.arm_id}`)
      }
      if (r.remaining_after_2b1 !== e.p.remaining_after) err(`${c}: remaining differs from the plan`)
      if (MUST_REMAIN_ADD.has(c) && !(r.remaining_after_2b1_additions ?? '').length) err(`${c}: remaining additions missing`)
      const joined = r.remaining_after_2b1 + (r.remaining_after_2b1_additions ? ' ' + r.remaining_after_2b1_additions : '')
      if (v4.remaining_open_by_contract[c] !== joined) err(`${c}: remaining_open_by_contract`)
      if (!(r.residual_planned_profile_proposal ?? '').endsWith(': ' + e.p.remaining_after)) err(`${c}: residual proposal`)
    } else {
      if (r.observations_2b1 || r.arms_observed_2b1) err(`${c}: 2B-1 fields in a row outside the 2B-1 plan`)
      if (e.p) {
        if (J(r.cases_2b2_pending) !== J(e.p.cases)) err(`${c}: 2B-2 pending cases`)
        for (const k of ['planned_profile_proposal', 'residual_planned_profile_proposal']) if (r3[k] !== undefined && r3[k].startsWith('SOURCE_BATCH_2B') && !(r[k] ?? '').startsWith('SOURCE_BATCH_2B_2')) err(`${c}: 2B-2 proposal not relabelled`)
      } else {
        for (const k of ['planned_profile_proposal', 'residual_planned_profile_proposal']) if (r3[k] !== undefined && r[k] !== r3[k]) err(`${c}: proposal changed outside the plan`)
      }
    }
    if (r.whole_requirement_covered !== false || r.parity_credit !== 0 || r.functional_receipt !== null) err(`${c}: claims coverage, receipt or credit`)
  }
  // known_bad de cada caso coberto por alguma evidência de braço
  for (const cid of consumed) for (const jb of inp.acc[cid].known_bad.judged_by) if (!(caseArmEvidence[cid]?.has(jb))) err(`${cid}: known_bad predicate ${jb} not linked to any arm`)
  // topo
  const counts = { OBSERVED_CLIENT_PARTIAL: 0, OPEN_REQUIRED_FUTURE_PROFILE: 0 }
  for (const e of Object.values(exp)) counts[e.status] += 1
  if (J(v4.counts) !== J(counts)) err(`counts ${J(v4.counts)} != ${J(counts)}`)
  if (J(v4.counts_v3) !== J(v3.counts)) err('counts_v3 differs from the v3')
  const changes = v4.rows.filter((r) => r.status !== r.status_v3).map((r) => ({ contract: r.source_contract, from: r.status_v3, to: r.status }))
  if (J(changes) !== J(v4.status_changes)) err('status_changes differ from rows')
  if (J(v4.still_not_consumed_2a) !== J(v3.still_not_consumed_2a)) err('still_not_consumed_2a differs from the v3')
  const union = v4.rows.flatMap((r) => (r.arms_observed_2b1 ?? []).map((a) => ({ ...a, contract: r.source_contract })))
  if (J(union) !== J(v4.arms_observed_2b1)) err('top arms_observed_2b1 differs from the union of rows')
  const openU = [...v3.arms_open_explicit_v3, ...v4.rows.flatMap((r) => (r.arms_open_explicit_2b1 ?? []).map((a) => ({ ...a, contract: r.source_contract })))]
  if (J(openU) !== J(v4.arms_open_explicit_v4)) err('top arms_open_explicit_v4 differs from v3 plus rows')
  const closed = clone(v3.arms_closed_v3)
  const ts07 = v4.rows.find((r) => r.source_contract === 'TS07').carried_from_v3.arms_closed_v3.find((x) => armId(x.arm) === 'TS07.B03')
  for (const a of closed) if (armId(a.arm) === 'TS07.B03') Object.assign(a, F01.erratum_text, { evidence_v4: ts07.evidence_v4 })
  if (J(closed) !== J(v4.arms_closed_v3_carried)) err('TS07.B03 erratum: arms_closed_v3_carried differs from the v3 with the erratum')
  const ids = (v4.errata_applied ?? []).map((x) => x.id)
  if (J(ids) !== J(['2AC-BIND/R2-F01', '2AC-BIND/R2-F05'])) err('errata_applied list')
  const ea = Object.fromEntries((v4.errata_applied ?? []).map((x) => [x.id, x]))
  if (ea['2AC-BIND/R2-F01'] && (J(ea['2AC-BIND/R2-F01'].v4) !== J(F01.erratum_text) || J(ea['2AC-BIND/R2-F01'].v3) !== J(F01.wrong_text))) err('errata_applied R2-F01 texts')
  const ts12v3 = v3.rows.find((r) => r.source_contract === 'TS12').remaining_after_continuation_additions
  const ts12v4 = v4.rows.find((r) => r.source_contract === 'TS12').carried_from_v3.remaining_after_continuation_additions
  if (ea['2AC-BIND/R2-F05'] && (ea['2AC-BIND/R2-F05'].v3 !== ts12v3 || ea['2AC-BIND/R2-F05'].v4 !== ts12v4)) err('errata_applied R2-F05 texts')
  const rem = { ...v3.remaining_open_by_contract, TS12: v3.remaining_open_by_contract.TS12.replace(ts12v3, ts12v4) }
  for (const k of Object.keys(v4.remaining_open_by_contract)) if (!(k in rem) && !exp[k]?.in1) err(`remaining_open_by_contract: extra ${k}`)
  for (const [k, t] of Object.entries(rem)) if (v4.remaining_open_by_contract[k] !== t) err(`remaining_open_by_contract: ${k} differs from the v3 with the errata`)
  const voc = J(v4.status_vocabulary)
  if (!voc.includes('coverage-plan-2b.json') || !voc.includes('2B-2') || voc === J(v3.status_vocabulary)) err('status_vocabulary not updated for v4')
  for (const [k, s] of Object.entries(v4.sources)) if (shaFile(normalize(join(UNIT, s.path))) !== s.sha256) err(`source sha ${k}`)
  // lacunas
  const st = Object.fromEntries(Object.entries(exp).map(([k, v]) => [k, v.status]))
  const st3 = Object.fromEntries(v3.rows.map((r) => [r.source_contract, r.status]))
  const obsV4 = {}
  for (const g of inp.gaps3.rows) {
    const o4 = g.contracts.filter((c) => st[c] === 'OBSERVED_CLIENT_PARTIAL')
    const o3 = g.contracts.filter((c) => st3[c] === 'OBSERVED_CLIENT_PARTIAL')
    obsV4[g.id] = o4
    const gr = g4.rows.find((x) => x.id === g.id)
    if (!gr) { err(`${g.id}: gap row missing`); continue }
    if (J(gr.contracts_observed_client_partial_v4) !== J(o4) || J(gr.contracts_observed_client_partial_v3) !== J(o3)) err(`${g.id}: observed contracts`)
    if (J(gr.newly_observed_2b1) !== J(o4.filter((c) => !o3.includes(c)))) err(`${g.id}: newly observed`)
    if (gr.status_after !== 'OPEN' || gr.whole_requirement_covered !== false) err(`${g.id}: gap closed`)
    if (g4.summary.observed_v4[g.id] !== `${o4.length}/${g.contracts_total}`) err(`${g.id}: gap summary`)
    if (gr.observed_count_changed !== (o4.length !== o3.length)) err(`${g.id}: observed_count_changed`)
  }
  if (g4.summary.OPEN !== 6) err('gap summary: OPEN count')
  // projeção do 2B-2 (lacunas recalculadas como na variante do v3: gp)
  const rows2 = inp.plan.rows.filter((r) => J(r.sub_lots) === J(['2B-2']))
  for (const r of rows2) if (r.primary_cases.some((x) => inp.cases2[x]?.sub_lot !== '2B-2')) err(`projection: ${r.source_contract} primary case outside 2B-2`)
  const a2 = rows2.filter((r) => st[r.source_contract] === 'OPEN_REQUIRED_FUTURE_PROFILE').map((r) => r.source_contract)
  const pc = (n) => ({ OBSERVED_CLIENT_PARTIAL: counts.OBSERVED_CLIENT_PARTIAL + n, OPEN_REQUIRED_FUTURE_PROFILE: counts.OPEN_REQUIRED_FUTURE_PROFILE - n })
  const gp = (newly) => Object.fromEntries(inp.gaps3.rows.map((g) => [g.id, `${new Set([...obsV4[g.id], ...newly.filter((x) => g.contracts.includes(x))]).size}/${g.contracts_total}`]))
  if (J(pj.after_2B_2.rows_becoming_observed) !== J(a2)) err('projection rows')
  if (J(pj.after_2B_2.counts) !== J(pc(a2.length))) err('projection counts')
  if (J(pj.after_2B_2.gaps) !== J(gp(a2))) err('projection gaps')
  if (J(pj.base.counts_v4) !== J(counts)) err('projection base counts')
  const pv = pj.previous_projection_check
