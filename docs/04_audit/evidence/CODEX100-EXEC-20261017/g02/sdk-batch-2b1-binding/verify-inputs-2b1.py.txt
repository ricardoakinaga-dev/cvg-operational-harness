# DX100-G02:BIND-SDK-SOURCE-BATCH-2B-1 - passo 0 (somente leitura).
# Uso: python3 -I -B verify-inputs-2b1.py.txt <repo_root> <out_json>
# As pastas g02 estão não rastreadas no checkout da main. A conferência é por sha (manifestos, freezes, next-task) e por blob git
# (git hash-object local x git ls-tree do commit). Âncora durável (2AC-BIND/R2-F04): o commit do PR documental do Lead, que leva as
# pastas da especificação 2B, do BIND da continuação e da admissão 2B-1. Nada é escrito fora do <out_json>.
import hashlib
import json
import os
import subprocess
import sys
from datetime import datetime, timezone

ROOT, OUT = sys.argv[1], sys.argv[2]
E = "docs/04_audit/evidence/CODEX100-EXEC-20261017"
G02R = E + "/g02"
LEAD_PR_COMMIT = "1353e1bd750c028a2db66558ad9dbef72b5cfec1"  # commit do PR documental do Lead (worktree cvg-d9ef-docs-pr-20261019)
checks = []


def sha(p):
    with open(p, "rb") as f:
        return hashlib.sha256(f.read()).hexdigest()


def load(p):
    with open(p, encoding="utf-8") as f:
        return json.load(f)


def A(rel):
    return os.path.join(ROOT, rel)


def check(name, ok, detail=None):
    checks.append({"check": name, "ok": bool(ok), "detail": detail})


def git(*a):
    return subprocess.run(["git", "-C", ROOT] + list(a), capture_output=True, text=True)


def same_as_commit(rel, commit, exclude=()):
    """Todo arquivo da árvore do commit sob rel é igual ao arquivo local (blob git) e não há arquivo local a mais."""
    lt = git("ls-tree", "-r", commit, "--", rel)
    if lt.returncode != 0:
        return False, {"problems": ["ls-tree failed"]}
    tree = {}
    for line in lt.stdout.splitlines():
        meta, path = line.split("\t", 1)
        tree[path] = meta.split()[2]
    bad = []
    for path, blob in tree.items():
        if os.path.basename(path) in exclude:
            continue
        if not os.path.isfile(A(path)):
            bad.append("missing " + path)
        elif git("hash-object", A(path)).stdout.strip() != blob:
            bad.append("differs " + path)
    for dp, _, fs in os.walk(A(rel)):
        for f in fs:
            p = os.path.relpath(os.path.join(dp, f), ROOT)
            if p not in tree and f not in exclude:
                bad.append("extra " + p)
    return (not bad and len(tree) > 0), {"files_in_commit": len(tree), "problems": bad[:10]}

check("2B-1 source-freeze, identity-freeze and diag-admission sha as recorded in state.json", all(fr.values()), fr)
