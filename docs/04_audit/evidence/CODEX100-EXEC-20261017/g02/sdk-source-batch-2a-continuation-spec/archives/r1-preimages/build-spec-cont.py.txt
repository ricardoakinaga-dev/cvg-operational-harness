def sha_bytes(b):
    return hashlib.sha256(b).hexdigest()


def sha(p):
    with open(p, "rb") as f:
        return sha_bytes(f.read())

def anchor(key, text, count=1):
    """Linhas (1-based) em que o trecho literal aparece; falha se a contagem divergir."""
    lines = open(INPUT_FILES[key], encoding="utf-8").read().split("\n")

    return {"input": key, "sha256": INPUTS[key]["sha256"], "lines": found, "text": text}

def finish_case(c, slot, row):
    c["slot"] = slot
    c["case_input"] = row

def anchor_inputs_2a(src):
    """Anchors of a copied 2A case resolve through the 2A spec inputs (keys of cases-2a.json#/inputs), re-pathed from this folder."""
    keys = sorted({a["input"] for a in src["anchors"]} | set(src.get("inputs", [])))
    out = {}
    for k in keys:
        meta = SPEC2A_DOC["inputs"][k]
        p = os.path.normpath(os.path.join(SPEC2A, meta["path"]))
        assert sha(p) == meta["sha256"], ("2A anchor input drifted", k)
        out[k] = {"path": rel(p), "sha256": meta["sha256"]}
    return out

                      "unchanged_fields": ["contracts", "branches_targeted", "fixture", "sdk_calls", "expected", "expected_observation", "known_bad",
                                           "accepted_capture_forms", "inconclusive_on_deadline", "anchors", "inputs"],
                      "changed_fields": {"infra": "acrescenta F4-01", "predicates": "acrescenta capture.case_input_frozen_bytes"},
                      "why": "adjudication-r1.json da admissão: MISMATCH bruto não atribuível ao cliente (HARNESS_INPUT_DELIVERY, case_input entregue com chaves ordenadas)"}
    c["predicates"] = src["predicates"] + [CAPTURE_PRED]
    c["anchor_inputs"] = anchor_inputs_2a(src)
    c["infra"] = src["infra"] + ["F4-01 entrega do case_input pelos bytes congelados"]

        c.pop(k, None)
    finish_case(c, slot, {"sdk_calls": src["sdk_calls"], "fixture": src["fixture"]})
