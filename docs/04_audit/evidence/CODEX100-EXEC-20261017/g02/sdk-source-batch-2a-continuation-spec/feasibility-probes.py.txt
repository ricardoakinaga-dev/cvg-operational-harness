#!/usr/bin/env python3
# DX100-G02:SPECIFY-SDK-SOURCE-BATCH-2A-CONTINUATION - sondas de viabilidade, código próprio (sem SDK, sem serviço, sem rede).
# Uso: python3 -I -B feasibility-probes.py.txt <out_json> <scratch_dir>
# Fora do sandbox e só informativas: as provas que decidem são os DIAG da IMPLEMENT (DIAG-TS-EPIPE, DIAG-TS-PREABORT, DIAG-TS-ENOENT).
# 1. EPIPE: write de 1 MiB sem listener de erro no stdin de um filho que fecha o stdin sem ler.
# 2. inotify: IN_OPEN do executável observado durante um spawn com signal pré-abortado, sem privilégio; e zero eventos sem spawn.
# 3. Semântica da linguagem usada pelos valores esperados (buraco de array no join, 1e-7, Date sem entradas, typeof bigint, isFinite).
import ctypes, datetime, hashlib, json, os, select, struct, subprocess, sys

OUT, SCR = sys.argv[1], sys.argv[2]
NODE = "/home/ricardo/.nvm/versions/node/v24.20.0/bin/node"
NODE_SHA = "89af8424dd53e560b1933f87ba650d8bf57c83ca5a04600eefb31f416aabbae7"
os.makedirs(SCR, exist_ok=True)
EPIPE = r"""import { spawn } from 'node:child_process';
import fs from 'node:fs';
const log = (o) => fs.writeSync(1, JSON.stringify(o) + '\n');
process.on('uncaughtException', (e) => log({ uncaught: { name: e.name, code: e.code, syscall: e.syscall, message: e.message } }));
const child = spawn('/bin/sh', ['-c', 'exec 0<&-; sleep 0.5; exit 0']);
log({ parent_stdin_fd: fs.readlinkSync('/proc/self/fd/' + child.stdin._handle.fd).replace(/[0-9]+/, 'N') });
child.once('error', (e) => log({ child_error: e.code }));
child.stdin.write('x'.repeat(1048576));
child.stdin.end();
child.stdout.resume();
child.once('exit', (code, signal) => { log({ exit: { code, signal } }); });
"""
PREABORT = r"""import { spawn } from 'node:child_process';
const ac = new AbortController(); ac.abort();
const out = {};
const child = spawn(process.argv[2], ['x'], { signal: ac.signal });
child.on('error', (e) => { out.error = { name: e.name, code: e.code, message: e.message, cause: e.cause && e.cause.name }; });
child.on('exit', (code, signal) => { out.exit = { code, signal }; setTimeout(() => console.log(JSON.stringify(out)), 100); });
"""
SEMANTICS = r"""console.log(JSON.stringify({hole: [1,,3].map(x => String(x)).join(', '), exp: `${1e-7}`, date_entries: Object.entries(new Date(0)).length,
  bigint: typeof 1n, nan_finite: Number.isFinite(NaN), inf_finite: Number.isFinite(Infinity), undefined_type: typeof undefined,
  int_key_order: Object.keys({b: 1, 1: 2, a: 3})}))
"""
files = {"epipe.mjs": EPIPE, "preabort.mjs": PREABORT, "semantics.mjs": SEMANTICS, "target.sh": "#!/bin/sh\nsleep 2\n"}
for n, t in files.items():
    with open(os.path.join(SCR, n), "w") as f:
        f.write(t)
os.chmod(os.path.join(SCR, "target.sh"), 0o755)
libc = ctypes.CDLL("libc.so.6", use_errno=True)


def inotify_run(target, argv):
    fd = libc.inotify_init1(0o4000)
    libc.inotify_add_watch(fd, target.encode(), 0x1 | 0x20 | 0x10)
    def drain():
        ev = []
        while select.select([fd], [], [], 0.2)[0]:
            buf = os.read(fd, 4096)
            i = 0
            while i < len(buf):
                _w, mask, _c, ln = struct.unpack_from("iIII", buf, i)
                ev.append({0x1: "IN_ACCESS", 0x20: "IN_OPEN", 0x10: "IN_CLOSE_NOWRITE"}.get(mask, hex(mask)))
                i += 16 + ln
        return ev
    before = drain()
    p = subprocess.run(argv, capture_output=True, text=True, timeout=30)
    after = drain()
    os.close(fd)
    return before, after, p.stdout.strip()


results = {}
with open(NODE, "rb") as f:
    node_sha = hashlib.sha256(f.read()).hexdigest()
results["node"] = {"path": NODE, "sha256": node_sha, "pinned": node_sha == NODE_SHA}
results["epipe"] = []
for _ in range(3):
    p = subprocess.run([NODE, os.path.join(SCR, "epipe.mjs")], capture_output=True, text=True, timeout=30)
    results["epipe"].append({"returncode": p.returncode, "lines": [json.loads(x) for x in p.stdout.splitlines() if x.strip()]})
target = os.path.join(SCR, "target.sh")
results["preabort_inotify"] = []
for _ in range(3):
    b, a, out = inotify_run(target, [NODE, os.path.join(SCR, "preabort.mjs"), target])
    results["preabort_inotify"].append({"events_before": b, "events_during": a, "node": json.loads(out)})
b, a, out = inotify_run(target, [NODE, "-e", "0"])
results["no_spawn_inotify"] = {"events_before": b, "events_during": a}
p = subprocess.run([NODE, "--input-type=module", "-e", SEMANTICS], capture_output=True, text=True, timeout=30)
results["semantics"] = json.loads(p.stdout)
results["uid"] = os.getuid()
results["ptrace_scope"] = open("/proc/sys/kernel/yama/ptrace_scope").read().strip()
summary = {
    "epipe_uncaught_every_run": all(any("uncaught" in l and l["uncaught"]["code"] == "EPIPE" and l["uncaught"]["syscall"] == "write" for l in r["lines"]) for r in results["epipe"]),
    "epipe_child_error_never": all(not any("child_error" in l for l in r["lines"]) for r in results["epipe"]),
    "stdio_is_socketpair": all(any(l.get("parent_stdin_fd", "").startswith("socket:") for l in r["lines"]) for r in results["epipe"]),
    "preabort_in_open_every_run": all("IN_OPEN" in r["events_during"] for r in results["preabort_inotify"]),
    "preabort_abort_err_every_run": all(r["node"]["error"]["code"] == "ABORT_ERR" and r["node"]["exit"]["signal"] == "SIGTERM" for r in results["preabort_inotify"]),
    "no_spawn_zero_events": results["no_spawn_inotify"]["events_during"] == [],
    "unprivileged": results["uid"] != 0,
    "semantics_match_expected": results["semantics"]["hole"] == "1, , 3" and results["semantics"]["exp"] == "1e-7" and results["semantics"]["date_entries"] == 0
                                and results["semantics"]["bigint"] == "bigint" and results["semantics"]["nan_finite"] is False and results["semantics"]["inf_finite"] is False,
}
doc = {"kind": "FEASIBILITY_PROBES_INFORMATIONAL", "task": "DX100-G02:SPECIFY-SDK-SOURCE-BATCH-2A-CONTINUATION",
       "scope": "Código próprio fora do sandbox, sem SDK, sem serviço, sem rede; informativo. As provas que decidem são os DIAG da IMPLEMENT.",
       "probes": {n: hashlib.sha256(t.encode()).hexdigest() for n, t in files.items()}, "results": results, "summary": summary,
       "observed_at": datetime.datetime.now(datetime.timezone.utc).isoformat()}
with open(OUT, "w", encoding="utf-8") as f:
    f.write(json.dumps(doc, ensure_ascii=False, indent=2) + "\n")
print(json.dumps(summary))
