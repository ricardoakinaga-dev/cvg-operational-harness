def load(name):
    with open(os.path.join(SPEC, name), encoding="utf-8") as f:
        return json.load(f)


def sha(path):
    with open(path, "rb") as f:
        return hashlib.sha256(f.read()).hexdigest()


def ref(rel):
    return {"path": rel, "sha256": sha(os.path.normpath(os.path.join(SPEC, rel)))}


def dump(name, obj):
    with open(os.path.join(SPEC, name), "w", encoding="utf-8") as f:
        f.write(json.dumps(obj, ensure_ascii=False, indent=2) + "\n")

def used_by(cap):
    out = {"2B-1": [], "2B-2": []}
    for lot, doc in CASES.items():
        for c in doc["cases"]:
            if any(i.split(" ")[0] == cap for i in c["infra"]):
                out[lot].append(c["id"])
    return out
