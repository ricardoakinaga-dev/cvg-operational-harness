def A(key, text, nth=None, span=0):
    """Anchor: unique line (or the nth match) containing `text`; optional span of following lines."""
    ls = lines_of(key)
    hits = [i for i, l in enumerate(ls, 1) if text in l]
    if nth is None:
        if len(hits) != 1:
            ANCHOR_ERRORS.append((key, text, hits))
            return {"input": key, "error": "ambiguous_or_missing"}
        n = hits[0]
    else:
        if len(hits) < nth:
            ANCHOR_ERRORS.append((key, text, hits, nth))
            return {"input": key, "error": "missing"}
        n = hits[nth - 1]
    out = {"input": key, "sha256": INPUTS[key]["sha256"], "lines": [n] if not span else [n, n + span], "text": ls[n - 1].strip()}
    return out

# ------------------------------------------------------------------ retry formula (exact IEEE values from retry.py:37-41)


def retry_trace(errors_before_success, initial, maxd, script, jitter_ratio=0.2):
    """Return the (uniform args, uniform return, sleeps) the source formula produces for `errors_before_success` retryable failures."""
    delay = initial
    uni, sleeps = [], []
    for i in range(errors_before_success):
        jitter = delay * jitter_ratio
        a, b = -jitter, jitter
        u = {"a": a, "b": b}[script[i]]
        uni.append({"a": a, "b": b, "returned": u})
        sleep_for = min(maxd, delay) + u
        if sleep_for > 0:
            sleeps.append(sleep_for)
        delay = min(maxd, delay * 2)
    return uni, sleeps
