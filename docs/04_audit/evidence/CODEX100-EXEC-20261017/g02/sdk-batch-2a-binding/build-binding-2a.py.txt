def sha(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def load(path):
    with open(path, encoding="utf-8") as f:
        return json.load(f)


def ref(rel_from_out, abs_path):
    return {"path": rel_from_out, "sha256": sha(abs_path)}

def pred(cid, pid):
    return [p for p in acc31[cid]["predicates"] if p["id"] == pid][0]


def arms_of(cid, contract):
    """Branches of case cid attributed to contract (prefix 'TSxx.' or the single contract of the case)."""
    c = cases[cid]
    out = []
    for b in c["branches_targeted"]:
        m = re.match(r"^((?:TS|PY|PK)\d\d)\.", b)
        owner = m.group(1) if m else (c["contracts"][0] if len(c["contracts"]) == 1 else None)
        if owner is None:
            raise SystemExit("branch without owner: %s %s" % (cid, b))
        if owner == contract:
            out.append(b)
    return out


def split_conditional(cid, arms):
    """-> (observed arms, open arms with reason, condition records)."""
    observed, opened, cond = [], [], []
    if cid == "2A-TS10":
        form = pred(cid, "api.enoent_form_A_or_B_exclusive")["detail"]["form"]
        cond.append({"condition": "forma observada do 2A-TS10 (cases-2a.json expected.source_consistent_forms)", "observed_form": form})
        for a in arms:
            want = "A" if "forma A" in a else "B" if "forma B" in a else None
            if want is None or want == form:
                observed.append(a)
            else:
                opened.append({"arm": a, "reason": "forma %s não ocorreu (forma %s observada); braço sem observação" % (want, form)})
        return observed, opened, cond
    if cid == "2A-TS11":
        det = pred(cid, "wire.chunk_boundaries_recorded")["detail"]
        ok = det["chunk3_and_chunk4_in_separate_commits"] is True
        cond.append({"condition": "credit_condition do 2A-TS11: chunks 3 e 4 em FORWARD_COMMIT separados do adapter",
                     "observed": ok, "adapter_stdout_commits": det["adapter_stdout_commits"], "label": det["label"]})
        for a in arms:
            if "UTF-8 dividido" in a and not ok:
                observed.append(a.replace("UTF-8 dividido, ", ""))
                opened.append({"arm": "UTF-8 dividido (%s)" % a.split(" ")[0], "reason": "chunks 3 e 4 não ficaram em commits separados"})
            elif "UTF-8 dividido" in a:
                observed.append(a + " [UTF-8 dividido: fronteira do adapter observada; leitura do SDK não testemunhada]")
            else:
                observed.append(a)
        return observed, opened, cond
    return list(arms), [], cond


def observation(cid, contract):
    a = acc31[cid]
    r = adj_rows[cid]
    obs_arms, open_arms, cond = split_conditional(cid, arms_of(cid, contract))
    q = [x for x in (r.get("qualification"), EXTRA_QUAL.get(cid)) if x]
    return {
        "case": cid,
        "final_reading": r["final_reading"],
        "attribution": r["attribution"],
        "service_outcome_r3": r["service_outcome_r3"],
        "offline_outcome_r3_1": r["offline_outcome_r3_1"],
        "r3_1_composition_rule": a["composition_rule"],
        "capture_form": a["capture_form"],
        "predicates": len(a["predicates"]),
        "failed_predicates_r3_1": [p["id"] for p in a["predicates"] if not p["ok"]],
        "analytic_branch_arms_observed": obs_arms,
        "arms_open_from_conditions": open_arms,
        "conditions": cond,
        "qualifications": q,
        "acceptance_r3_1": ref(ADMR + r["dir"] + "/acceptance-r3_1.json", os.path.join(ADM, r["dir"], "acceptance-r3_1.json")),
        "acceptance_service": ref(ADMR + r["dir"] + "/acceptance.json", os.path.join(ADM, r["dir"], "acceptance.json")),
        "adjudication_pointer": ADMR + "adjudication-r1.json#/rows[case=%s]" % cid,
        "client_fact_only": True,
        "parity_receipt": None,
        "functional_receipts": 0,
    }
