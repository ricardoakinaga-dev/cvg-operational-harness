def sha(b):return hashlib.sha256(b).hexdigest()
def normalize(n):return re.sub(r'[-_.]+','-',n).lower()
def version(v):
    assert re.fullmatch(r'\d+(?:\.\d+)*',v),('unsupported finite version',v)
    return tuple(int(x) for x in v.split('.'))+(0,)*max(0,5-len(v.split('.')))
def satisfies(v,s):
    if not s:return True
    for term in s.strip('() ').split(','):
        m=re.fullmatch(r'\s*(==|!=|>=|<=|>|<)\s*(\d+(?:\.\d+)*)\s*',term)
        assert m,('unsupported finite specifier',s)
        assert {'==':operator.eq,'!=':operator.ne,'>=':operator.ge,'<=':operator.le,'>':operator.gt,'<':operator.lt}[m[1]](version(v),version(m[2])),(v,s)
    return True

def marker(text):
    if not text:return True
    root=ast.parse(text,mode='eval')
    allowed=(ast.Expression,ast.BoolOp,ast.And,ast.Or,ast.Compare,ast.Eq,ast.NotEq,ast.Lt,ast.LtE,ast.Gt,ast.GtE,ast.Name,ast.Load,ast.Constant)
    assert all(isinstance(n,allowed) for n in ast.walk(root)),('unsupported marker AST',text)
    def value(n):
        if isinstance(n,ast.Name):assert n.id in ENV;return ENV[n.id],n.id in ['python_version','python_full_version']
        assert isinstance(n,ast.Constant) and isinstance(n.value,str);return n.value,False
    def visit(n):
        if isinstance(n,ast.Expression):return visit(n.body)
        if isinstance(n,ast.BoolOp):
            values=[visit(x) for x in n.values];return all(values) if isinstance(n.op,ast.And) else any(values)
        assert isinstance(n,ast.Compare);lhs=value(n.left);results=[]
        for op,right in zip(n.ops,n.comparators):
            rhs=value(right);a,b=(version(lhs[0]),version(rhs[0])) if lhs[1] or rhs[1] else (lhs[0],rhs[0])
            results.append({ast.Eq:operator.eq,ast.NotEq:operator.ne,ast.Lt:operator.lt,ast.LtE:operator.le,ast.Gt:operator.gt,ast.GtE:operator.ge}[type(op)](a,b));lhs=rhs
        return all(results)
    return visit(root)
def safe_name(name,directory=False):
    assert name and not name.startswith(('/','\\')) and '\\' not in name and '\x00' not in name
    text=name[:-1] if directory and name.endswith('/') else name
    parts=text.split('/');assert all(x and x not in ['.','..'] and ':' not in x for x in parts),name
    assert all(not x.endswith('.pth') and x not in ['sitecustomize.py','usercustomize.py'] for x in parts),name
    return str(PurePosixPath(text))

def write_data(root,name,b,original_mode):
    p=root/name;p.parent.mkdir(parents=True,exist_ok=True,mode=0o700)
    assert not p.exists() and not p.is_symlink(),p
    with p.open('xb') as f:f.write(b)
    p.chmod(0o600)
    return {'path':name,'bytes':len(b),'sha256':sha(b),'archive_mode':original_mode,'private_mode':'0600','kind':'REGULAR_DATA_FILE','empty':len(b)==0}

def wheel(row,spec,root):
    payload={};modes={};seen=set();dirs=[]
    with zipfile.ZipFile(row['raw_path']) as z:
        assert len(z.infolist())<=5000
        for item in z.infolist():
            assert item.orig_filename == item.filename, 'NUL/truncated ZIP name'
            mode=item.external_attr>>16;kind=stat.S_IFMT(mode)
            assert kind in [0,stat.S_IFREG,stat.S_IFDIR] and not item.flag_bits&1,(item.filename,mode)
            isdir=item.is_dir();assert kind!=stat.S_IFDIR or isdir
            name=safe_name(item.filename,isdir);assert name.casefold() not in seen,name;seen.add(name.casefold())
            budget(name,item.file_size)
            if isdir:assert item.file_size==0;dirs.append(name);continue
            assert item.compress_type in [zipfile.ZIP_STORED,zipfile.ZIP_DEFLATED]
            assert item.file_size==0 or (item.compress_size>0 and item.file_size/item.compress_size<=500)
            with z.open(item) as f:b=f.read(PER_FILE+1)
            assert len(b)==item.file_size and len(b)<=PER_FILE
            payload[name]=b;modes[name]=mode
    meta_names=[x for x in payload if x.endswith('.dist-info/METADATA')];assert len(meta_names)==1
    base=meta_names[0].rsplit('/',1)[0];assert all('/' not in x or '.dist-info' not in x.split('/')[0] or x.startswith(base+'/') for x in payload)
    meta=email.parser.BytesParser().parsebytes(payload[base+'/METADATA']);wh=email.parser.BytesParser().parsebytes(payload[base+'/WHEEL'])
    assert normalize(meta['Name'])==spec['name'] and meta['Version']==spec['version']
    satisfies(host['version'],meta['Requires-Python']);tags=wh.get_all('Tag',[])
    expected=['cp312-cp312-manylinux_2_17_x86_64','cp312-cp312-manylinux2014_x86_64'] if spec['name']=='pydantic-core' else ['py3-none-any']
    assert set(tags)==set(expected),(spec['name'],tags)
    assert wh['Root-Is-Purelib']==('false' if spec['name']=='pydantic-core' else 'true')

    deps=[];active=set()

    assert active==set(spec['dependencies']),(spec['name'],active,spec['dependencies'])

def npm(row,spec,root):
    files=[];seen=set();manifest=None
    with tarfile.open(row['raw_path'],mode='r|gz') as t:
        for item in t:
            name=safe_name(item.name,item.isdir());assert name.startswith('package/') or name=='package',name
            assert item.isfile() or item.isdir(),('nonregular npm entry',name,item.type)
            assert not item.issparse();assert name.casefold() not in seen;seen.add(name.casefold());budget(name,item.size)
            if item.isdir():assert item.size==0;continue

            b=t.extractfile(item).read(PER_FILE+1);assert len(b)==item.size and len(b)<=PER_FILE

    for field in ['dependencies','optionalDependencies','peerDependencies']:assert not manifest.get(field),(field,manifest.get(field))
