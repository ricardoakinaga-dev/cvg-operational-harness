def rec(p):
 p=pathlib.Path(p);p=p if p.is_absolute() else ROOT/p
 b=p.read_bytes(); path=str(p.relative_to(ROOT)) if p.is_relative_to(ROOT) else str(p)
 if path not in records:records[path]={'path':path,'bytes':len(b),'sha256':hashlib.sha256(b).hexdigest(),'first_observed_at':datetime.datetime.now(datetime.timezone.utc).isoformat()}
 return b
def get(p):return json.loads(rec(p))
def checkfiles(rows,resolve=lambda x:ROOT/x['path']):
 bad=[]
 for r in rows:
  p=resolve(r)
  try:b=rec(p)
  except FileNotFoundError:bad.append({'path':str(p),'problem':'MISSING'});continue
  if hashlib.sha256(b).hexdigest()!=r['sha256'] or ('bytes'in r and len(b)!=r['bytes']):bad.append({'path':str(p),'expected':r,'actual_sha256':hashlib.sha256(b).hexdigest(),'actual_bytes':len(b)})
 return {'count':len(rows),'mismatches':bad}
