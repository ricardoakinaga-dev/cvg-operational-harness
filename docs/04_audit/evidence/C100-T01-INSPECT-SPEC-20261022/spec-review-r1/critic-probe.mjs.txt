// Implementação mínima e fiel do §7.4 (com toggles para mutantes).
function readSpec(p, cap, opts = {}, hooks = {}) {
  let st
  try { st = opts.useStat ? statSync(p) : lstatSync(p) } catch (e) { return { code: 'READ_FAILED', errno: e.code } }
  if (st.isSymbolicLink()) return { code: 'SYMLINK_REFUSED' }
  if (!st.isFile()) return { code: 'SPECIAL_FILE_REFUSED' }
  if (!opts.noNlink && st.nlink !== 1) return { code: 'HARDLINK_REFUSED' }
  if (st.size > cap) return { code: 'CAP_EXCEEDED' }
  hooks.afterLstat?.()
  const flags = C.O_RDONLY | (opts.noNofollow ? 0 : C.O_NOFOLLOW) | C.O_NONBLOCK
  let fd
  try { fd = openSync(p, flags) } catch (e) {
    return e.code === 'ELOOP' ? { code: 'SYMLINK_REFUSED' } : { code: 'READ_FAILED', errno: e.code }
  }
  try {
    const f = fstatSync(fd)
    if (f.dev !== st.dev || f.ino !== st.ino || !f.isFile() || (!opts.noNlink && f.nlink !== 1)) {
      return { code: 'FILE_CHANGED_DURING_READ' }
    }
    if (opts.procRoot) {
      const real = readlinkSync(`/proc/self/fd/${fd}`)
      if (!real.startsWith(opts.procRoot + '/')) return { code: 'OUTSIDE_ROOT_BY_FD_PATH' }
    }
    const buf = Buffer.alloc(Math.min(st.size, cap) + 1)
    let n = 0
    for (;;) { const r = readSync(fd, buf, n, buf.length - n, null); if (!r) break; n += r; if (n > cap) return { code: 'CAP_EXCEEDED' } }
    if (n !== f.size) return { code: 'FILE_CHANGED_DURING_READ' }
    return { code: 'OK', text: buf.subarray(0, n).toString('utf8'), bytes: n }
  } finally { closeSync(fd) }
}

// §7.5 mínimo: registra dev/ino do diretório da skill, lê SKILL.md, refaz lstat no fim.
function acquireSkill(root, name, opts = {}, hooks = {}) {
  const dir = join(root, name)
  const d0 = lstatSync(dir)
  if (!d0.isDirectory()) return { item: 'REFUSED' }
  const names = readdirSync(dir).sort()
  hooks.afterListing?.()
  const files = {}
  for (const n of names) files[n] = readSpec(join(dir, n), 1 << 20, opts)
  hooks.afterRead?.()
  const d1 = lstatSync(dir)
  const changed = d1.isSymbolicLink() || d1.dev !== d0.dev || d1.ino !== d0.ino
  return { coverage: changed ? 'INCOMPLETE(TREE_CHANGED_DURING_READ)' : 'COMPLETE', files }
}

// ---- A: corrida ABA no diretório da skill (troca por symlink e volta) ----
{
  const root = join(base, 'A-root'); const outside = join(base, 'A-outside', 'evil')
  mkdirSync(join(root, 'skill'), { recursive: true }); mkdirSync(outside, { recursive: true })
  writeFileSync(join(root, 'skill', 'SKILL.md'), '---\nname: skill\ndescription: Inside.\n---\nINSIDE_BODY\n')
  writeFileSync(join(outside, 'SKILL.md'), '---\nname: skill\ndescription: OUTSIDE_MARKER.\n---\nOUTSIDE_BODY\n')
  const swapOut = () => { renameSync(join(root, 'skill'), join(base, 'A-parked')); symlinkSync(outside, join(root, 'skill')) }
  const swapBack = () => { unlinkSync(join(root, 'skill')); renameSync(join(base, 'A-parked'), join(root, 'skill')) }
  const spec = acquireSkill(root, 'skill', {}, { afterListing: swapOut, afterRead: swapBack })
  const withProc = (() => { swapOut(); const r = readSpec(join(root, 'skill', 'SKILL.md'), 1 << 20, { procRoot: root }); swapBack(); return r })()
  // Emulação de openat: fd do diretório + /proc/self/fd/<dfd>/<nome>.
  const dfd = openSync(join(root, 'skill'), C.O_RDONLY | C.O_DIRECTORY | C.O_NOFOLLOW)
  swapOut()
  const pinnedList = readdirSync(`/proc/self/fd/${dfd}`)
  const pinned = readSpec(`/proc/self/fd/${dfd}/SKILL.md`, 1 << 20)
  swapBack(); closeSync(dfd)
  let dirSymlinkOpen
  try { const fd = openSync(join(base, 'A-dirlink'), 0); closeSync(fd) } catch {}
  symlinkSync(join(root, 'skill'), join(base, 'A-dirlink'))
  try { openSync(join(base, 'A-dirlink'), C.O_RDONLY | C.O_DIRECTORY | C.O_NOFOLLOW); dirSymlinkOpen = 'opened' } catch (e) { dirSymlinkOpen = e.code }
  out.A_abaRace = {
    spec74_75: { coverage: spec.coverage, bodyCode: spec.files['SKILL.md'].code, outsideMarkerRead: String(spec.files['SKILL.md'].text).includes('OUTSIDE_MARKER') },
    fdPathCheck: withProc.code,
    dirFdPinning: { listedNames: pinnedList, readCode: pinned.code, readInside: String(pinned.text).includes('INSIDE_BODY') },
    openDirSymlinkWithNofollow: dirSymlinkOpen,
    O_DIRECTORY: typeof C.O_DIRECTORY, O_NOFOLLOW: typeof C.O_NOFOLLOW
  }
}

// ---- B: mutante M01 contra os casos do SK05 (arquivo symlink, estático) ----
{
  const root = join(base, 'B-root'); const outsideFile = join(base, 'B-outside.md')
  mkdirSync(join(root, 's'), { recursive: true })
  writeFileSync(outsideFile, 'OUTSIDE_FILE_MARKER')
  symlinkSync(outsideFile, join(root, 's', 'SKILL.md'))
  const p = join(root, 's', 'SKILL.md')
  const correct = readSpec(p, 1 << 20)
  const statMutant = readSpec(p, 1 << 20, { useStat: true })
  const noNofollowMutant = readSpec(p, 1 << 20, { noNofollow: true })
  // troca arquivo -> symlink entre lstat e open (SK07b-like)
  const swapTarget = join(base, 'B-outside2.md'); writeFileSync(swapTarget, 'OUTSIDE2')
  const mk = () => { const q = join(root, 's', 'ref.md'); try { unlinkSync(q) } catch {} ; writeFileSync(q, 'inside'); return q }
  const raceCorrect = readSpec(mk(), 1 << 20, {}, { afterLstat: () => { const q = join(root, 's', 'ref.md'); unlinkSync(q); symlinkSync(swapTarget, q) } })
  const raceNoNofollow = readSpec(mk(), 1 << 20, { noNofollow: true }, { afterLstat: () => { const q = join(root, 's', 'ref.md'); unlinkSync(q); symlinkSync(swapTarget, q) } })
  // hardlink: remover só uma das duas checagens de nlink
  const hl = join(root, 's', 'hard.md'); linkSync(outsideFile, hl)
  out.B_mutantM01 = {
    sk05_staticSymlinkFile: { correct: correct.code, statInsteadOfLstat: statMutant.code, withoutONofollow: noNofollowMutant.code },
    swapFileToSymlinkBetweenLstatAndOpen: { correct: raceCorrect.code, withoutONofollow: raceNoNofollow.code },
    hardlink: { correct: readSpec(hl, 1 << 20).code, bothNlinkChecksRemoved: readSpec(hl, 1 << 20, { noNlink: true }).code }
  }
}
